BERK Programlama Dili

Döngüler

Tekrarlı işlemler for loop yapıları.

For Döngüsü (Range)

// 0'dan 4'e kadar (5 hariç)
loop i 0'dan 5'e {
    print(i)  // 0, 1, 2, 3, 4
}

// 0'dan 5'e kadar (5 dahil)
loop i 0'dan 6'ya kadar {
    print(i)  // 0, 1, 2, 3, 4, 5
}

// Tersden sayma
loop i 10'dan 0'a {
    print(i)  // 10, 9, 8, ..., 1
}

// Adım boyutu
loop i 0'dan 10'a adım 2 {
    print(i)  // 0, 2, 4, 6, 8
}

İngilizce syntax:

for i from 0 to 5 {
    print(i)
}

for i from 0 to 5 inclusive {
    print(i)
}

For-Each Döngüsü

variable sayılar = [1, 2, 3, 4, 5]

loop sayı sayılarda {
    print(sayı)
}

// İndeks ile
loop (indeks, değer) sayılarda {
    print(indeks, ": ", değer)
}

İngilizce:

for num in numbers {
    print(num)
}

for (index, value) in numbers {
    print(index, ": ", value)
}

While Döngüsü

variable sayaç = 0

while sayaç < 5 {
    print(sayaç)
    sayaç += 1
}

İngilizce:

var counter = 0

while counter < 5 {
    print(counter)
    counter += 1
}

Do-While Döngüsü

En az bir kez çalışır:

variable x = 0

do {
    print(x)
    x += 1
} while x < 5

Sonsuz loop

loop {
    // Sonsuz loop
    if koşul {
        break  // Çıkış
    }
}

Break and Continue

Break (break)

Döngüden çık:

loop i 0'dan 10'a {
    if i == 5 {
        break  // loop durur
    }
    print(i)  // 0, 1, 2, 3, 4
}

Continue (continue)

Sonraki iterasyona geç:

loop i 0'dan 10'a {
    if i % 2 == 0 {
        continue  // Çift sayıları atla
    }
    print(i)  // 1, 3, 5, 7, 9 (tek sayılar)
}

İç İçe Döngüler

loop i 1'den 4'e {
    loop j 1'den 4'e {
        print(i * j, " ", sonlandırma: "")
    }
    print()  // Yeni satır
}
// Çıktı:
// 1 2 3
// 2 4 6
// 3 6 9

Etiketli break:

dış: loop i 0'dan 5'e {
    iç: loop j 0'dan 5'e {
        if i * j > 10 {
            break dış  // Dış döngüden çık
        }
        print(i, "*", j, "=", i*j)
    }
}

loop ile Koleksiyonlar

Liste Üzerinde

variable meyveler = ["elma", "armut", "kiraz"]

loop meyve meyvelerde {
    print(meyve)
}

// Filtreleme
loop meyve meyvelerde {
    if meyve.uzunluk > 4 {
        print(meyve)
    }
}

Map Üzerinde

variable yaşlar = {
    "Ahmet": 25,
    "Mehmet": 30,
    "Ayşe": 28
}

loop (isim, yaş) yaşlarda {
    print(isim, ": ", yaş)
}

Range ile

// 1'den 10'a (10 dahil)
loop x [1..10]'da {
    print(x)
}

// Filtreli range
loop x [1..20]'de {
    if x % 3 == 0 {
        print(x)  // 3'e bölünenler
    }
}

Iterator Metodları

variable sayılar = [1, 2, 3, 4, 5]

// Map
variable kareler = sayılar.eşle(x => x * x)

// Filter
variable çiftler = sayılar.filtrele(x => x % 2 == 0)

// Reduce
variable toplam = sayılar.azalt((a, b) => a + b)

// ForEach
sayılar.herbiri(x => print(x))

Best Practices

✅ Yapılması Gerekenler

// Anlamlı variable isimleri
loop öğrenci öğrencilerde {
    işle(öğrenci)
}

// For-each tercih edin
variable liste = [1, 2, 3]
loop eleman listede {  // İyi
    print(eleman)
}

// Iterator metodları kullanın
variable sonuç = sayılar
    .filtrele(x => x > 0)
    .eşle(x => x * 2)
    .azalt((a, b) => a + b)

❌ Kaçınılması Gerekenler

// Manuel indeks yönetimi
loop i 0'dan liste.uzunluk'a {
    print(liste[i])  // Kötü
}
// Bunun yerine:
loop eleman listede {
    print(eleman)  // İyi
}

// Sonsuz loop tuzakları
while true {  // Tehlikeli
    // Çıkış koşulu olmadan
}

// Gereksiz iç içe döngüler
// O(n²) or daha kötü karmaşıklık

Examples

Fibonacci

işlev fibonacci(n: Sayı) -> Liste<Sayı> {
    variable sonuç = [0, 1]
    
    loop i 2'den n'ye {
        variable sonraki = sonuç[i-1] + sonuç[i-2]
        sonuç.ekle(sonraki)
    }
    
    geri return sonuç
}

Asal Sayılar

işlev asal_mi(n: Sayı) -> Boolean {
    if n < 2 {
        geri return false
    }
    
    loop i 2'den n'ye {
        if i * i > n {
            break
        }
        if n % i == 0 {
            geri return false
        }
    }
    
    geri return true
}
