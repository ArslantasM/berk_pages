🐕 상태::워치독

Watchdog Timer - 시스템 신뢰성 및 재설정 관리

~280줄 ~8가지 기능 STM32/GD32/ESP32

📖 개요

Watchdog 타이머는 소프트웨어 오류로 인한 시스템 정지를 자동으로 감지하고 시스템을 재설정합니다. 마이크로 컨트롤러가 예상치 못한 상태(무한 루프, 교착 상태)에 진입하면 감시 카운터가 재설정되지 않고 시스템이 다시 시작되기 때문에 타임아웃이 발생합니다.

감시 장치에는 두 가지 유형이 있습니다.IWDG(독립형 워치독 - LSI 클록과 함께 작동, 저전력 모드에서도 활성화됨)WWDG(Window Watchdog - 창 제어를 통해 보다 정확한 타이밍).

🔑 주요 기능

  • 독립 감시(IWDG) - LSI 클록(32kHz)
  • 윈도우 워치독(WWDG) - APB1 클록
  • 구성 가능한 시간 제한(ms - 초)
  • 조기 경고 중단(WWDG)
  • 소스 감지 재설정
  • 디버그 모드 동결 지원
  • 절전/정지 모드에서 작업
  • 프리스케일러 설정(4 - 256)

🚀 빠른 시작

import hal::watchdog, hal::core

// Independent watchdog (4 saniye timeout)
let wdg = watchdog.init_iwdg(do
    timeout_ms: 4000,
    window_mode: false
end)

// Ana for
for do
    // Normal işlemler
    core.delay_ms(500)
    
    // Watchdog'u besle (refresh)
    watchdog.refresh(wdg)
    
    // Eğer bu satır çalışmazsa sistem 4 saniye in resetlenir
end

📦 유형 및 열거형

// Watchdog Tipi
enum WatchdogType {
    IWDG,  // Independent watchdog
    WWDG   // Window watchdog
}

// Prescaler değerleri
enum Prescaler {
    DIV_4,
    DIV_8,
    DIV_16,
    DIV_32,
    DIV_64,
    DIV_128,
    DIV_256
}

// Reset Source
enum ResetSource {
    POWER_ON,      // Power-on reset
    PIN_RESET,     // External pin reset
    SOFTWARE,      // Software reset
    IWDG_RESET,    // Independent watchdog
    WWDG_RESET,    // Window watchdog
    LOW_POWER,     // Low-power reset
    BROWNOUT       // Brown-out reset
}

struct WatchdogConfig do
    type: WatchdogType,
    timeout_ms: sayı,       // Milliseconds
    window_mode: mantık,    // Window watchdog only
    prescaler: Prescaler,
    reload_value: sayı
end

💡 예 1: 기본 IWDG 사용

2초 감시 기능으로 시스템 보호

import hal::watchdog, hal::gpio, hal::core

let led_green = gpio.pin_init(gpio.PORT_C, 13, gpio.MODE_OUTPUT)
let led_red = gpio.pin_init(gpio.PORT_C, 14, gpio.MODE_OUTPUT)

function main() do
    // Check reset source
    let reset_cause = watchdog.get_reset_source()
    
    if reset_cause == watchdog.IWDG_RESET then do
        // Watchdog reset occurred!
        gpio.pin_write(led_red, gpio.HIGH)
        core.delay_ms(2000)
        gpio.pin_write(led_red, gpio.LOW)
    end
    
    // Initialize watchdog (2 second timeout)
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 2000,
        window_mode: false
    end)
    
    // Main loop
    for do
        // Normal operation
        gpio.pin_toggle(led_green)
        core.delay_ms(100)
        
        // Feed watchdog every 100ms (well before 2s timeout)
        watchdog.refresh(wdg)
    end
end

💡 예 2: 작업 모니터가 있는 워치독

다중 작업 모니터링 및 감시 장치 공급

import hal::watchdog, hal::time

let son_beslenme_zamani: Zaman
let watchdog_aktif = true

function kritik_gorev_1() do
    // Kritik işlemler
    core.delay_ms(50)
    
    // İşlem başarılı
    watchdog_besleme_istegi()
end

function kritik_gorev_2() do
    // Başka kritik işlemler
    core.delay_ms(75)
    
    // İşlem başarılı
    watchdog_besleme_istegi()
end

function watchdog_besleme_istegi() do
    son_beslenme_zamani = time.now()
end

function watchdog_yonetici(wdg: WatchdogHandle) do
    for do
        // Check if tasks are healthy
        let elapsed = time.now() - son_beslenme_zamani
        
        if elapsed < time.milliseconds(500) then do
            // Tasks are running OK, feed watchdog
            watchdog.refresh(wdg)
        yoksa do
            // Tasks hung! Let watchdog reset the system
            io.println("UYARI: Görevler askıda! Watchdog resetleyecek...")
        end
        
        core.delay_ms(100)
    end
end

function main() do
    // 1 saniye watchdog timeout
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 1000,
        window_mode: false
    end)
    
    son_beslenme_zamani = time.now()
    
    // Start tasks
    thread.spawn(|| kritik_gorev_1())
    thread.spawn(|| kritik_gorev_2())
    thread.spawn(|| watchdog_yonetici(wdg))
    
    thread.join_all()
end

💡 예 3: 윈도우 워치독(WWDG)

창 제어를 통한 정확한 타이밍

import hal::watchdog

// WWDG: Hem çok erken hem çok continue beslemeyi engeller
function wwdg_demo() do
    // Window: 50-100ms aralığında beslenmeli
    let wwdg = watchdog.init_wwdg(do
        timeout_ms: 100,
        window_ms: 50,  // Pencere başlangıcı
        early_wakeup_interrupt: true
    end)
    
    // Early warning interrupt handler
    watchdog.set_interrupt_handler(wwdg, || do
        io.println("Watchdog erken uyarı!")
        // Burası early wakeup interrupt (zamanında besle)
    end)
    
    let count_occurrences = 0
    for do
        // İşlemler
        core.delay_ms(30)
        
        // 30ms + 30ms = 60ms (pencere in: 50-100ms ✓)
        core.delay_ms(30)
        
        // Watchdog'u besle
        watchdog.refresh(wwdg)
        
        count_occurrences += 1
        
        // Simulate hang every 10 iterations
        if count_occurrences % 10 == 0 then do
            io.println("Simüle edilen gecikme...")
            core.delay_ms(150)  // Too late! WWDG will reset
        end
    end
end

💡 예 4: 워치독 + 백업 레지스터

재설정 사유 기록 및 분석

import hal::watchdog, hal::rtc

struct ResetInfo do
    count: sayı,
    last_reason: ResetSource,
    timestamp: sayı
end

function reset_bilgisi_kaydet() do
    let reason = watchdog.get_reset_source()
    
    // Backup register'from önceki bilgiyi input
    let reset_count = rtc.backup_read(0)
    reset_count += 1
    
    // Yeni bilgiyi kaydet
    rtc.backup_write(0, reset_count)
    rtc.backup_write(1, reason.to_int())
    rtc.backup_write(2, time.timestamp())
    
    io.println("Reset sayısı: " + reset_count.yazıya())
    io.println("Reset nedeni: " + reason.to_string())
end

function main() do
    // Initialize RTC for backup registers
    rtc.init()
    
    // Log reset information
    reset_bilgisi_kaydet()
    
    // Check for excessive resets
    let reset_count = rtc.backup_read(0)
    if reset_count > 10 then do
        io.println("HATA: Çok fazla reset! Sistem unstable.")
        // Enter safe mode or disable some features
        rtc.backup_write(0, 0)  // Reset counter
    end
    
    // Initialize watchdog
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 3000,
        window_mode: false
    end)
    
    // Normal operation
    for do
        // Your code here
        watchdog.refresh(wdg)
        core.delay_ms(500)
    end
end

💡 모범 사례

  • 시간 초과 선택:가장 긴 작업 기간의 2~3배여야 합니다.
  • 먹이는 빈도:타임아웃의 50~75% 간격으로 피드
  • 디버그 모드:개발 중 워치독 정지(디버거 중단점에서)
  • 중요 섹션:인터럽트가 비활성화된 장소에서 워치독 시간 초과 계산
  • 로깅 재설정:백업 레지스터에 재설정 이유 저장
  • IWDG 대 WWDG:프로덕션의 IWDG(LSI 독립), 중요한 타이밍을 위한 WWDG
  • 전원 모드:IWDG는 저전력 모드에서 작동하고 WWDG는 중지됩니다.

⚠️ 중요 사항

  • IWDG는 일단 시작되면 중지할 수 없습니다(하드웨어 제한).
  • 디버깅 중 워치독 재설정을 방지하려면 DBGMCU_APB1_FZ 레지스터를 설정하세요.
  • LSI 클록 허용 오차는 최대 40%까지 가능 - 타임아웃 계산 시 고려
  • 플래시 프로그래밍 중 워치독 비활성화(삭제는 시간이 오래 걸릴 수 있음)
