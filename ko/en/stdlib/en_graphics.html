📋개요

그래픽 도면(차트)

use graphics
use color

class ÇubukGrafik {
    değer canvas: graphics.Canvas
    değer genişlik: Tam
    değer yükseklik: Tam
    
    function new(genişlik: Tam, yükseklik: Tam) -> ÇubukGrafik {
        değer canvas = graphics.Canvas(genişlik, yükseklik)
        canvas.arkaplan_ayarla(color.BEYAZ)
        
        return ÇubukGrafik {
            canvas: canvas,
            genişlik: genişlik,
            yükseklik: yükseklik
        }
    }
    
    function çiz(veri: Liste<(Dize, Gerçek)>, başlık: Dize) {
        değer kenar_boşluk = 50
        değer grafik_genişlik = bu.genişlik - 2 * kenar_boşluk
        değer grafik_yükseklik = bu.yükseklik - 2 * kenar_boşluk
        
        // Başlık
        bu.canvas.font_ayarla("Arial", 24, kalın: true)
        bu.canvas.renk_ayarla(color.SİYAH)
        bu.canvas.metin_yaz(başlık, bu.genişlik / 2, 30, hizalama: "merkez")
        
        // Maksimum değeri bul
        değer max_değer = 0.0
        for (_, değer) in veri {
            if değer > max_değer {
                max_değer = değer
            }
        }
        
        // Eksenler
        bu.canvas.çizgi_kalınlığı(2)
        bu.canvas.renk_ayarla(color.SİYAH)
        
        // Y ekseni
        bu.canvas.çizgi(kenar_boşluk, kenar_boşluk, kenar_boşluk, bu.yükseklik - kenar_boşluk)
        
        // X ekseni
        bu.canvas.çizgi(kenar_boşluk, bu.yükseklik - kenar_boşluk, 
                       bu.genişlik - kenar_boşluk, bu.yükseklik - kenar_boşluk)
        
        // Çubukları çiz
        değer çubuk_genişlik = grafik_genişlik / (veri.len * 2)
        değer boşluk = çubuk_genişlik / 2
        değer renkler = [
            color.rgb(255, 99, 132),   // Kırmızı-pembe
            color.rgb(54, 162, 235),   // Mavi
            color.rgb(255, 206, 86),   // Sarı
            color.rgb(75, 192, 192),   // Turkuaz
            color.rgb(153, 102, 255),  // Mor
            color.rgb(255, 159, 64)    // Turuncu
        ]
        
        for (i, (etiket, değer)) in veri.numaralandır() {
            değer x = kenar_boşluk + boşluk + i * (çubuk_genişlik + boşluk)
            değer çubuk_yükseklik = (değer / max_değer) * grafik_yükseklik
            değer y = bu.yükseklik - kenar_boşluk - çubuk_yükseklik
            
            // Çubuğu çiz
            bu.canvas.dolgu_ayarla(renkler[i % renkler.len])
            bu.canvas.dikdörtgen(x, y, çubuk_genişlik, çubuk_yükseklik, dolgu: true)
            
            // Etiket
            bu.canvas.font_ayarla("Arial", 12)
            bu.canvas.renk_ayarla(color.SİYAH)
            bu.canvas.metin_yaz(etiket, x + çubuk_genişlik / 2, 
                              bu.yükseklik - kenar_boşluk + 20, hizalama: "merkez")
            
            // Değer
            bu.canvas.metin_yaz(değer.dizgeye(), x + çubuk_genişlik / 2, y - 10, 
                              hizalama: "merkez")
        }
    }
    
    function kaydet(dosya_adı: Dize) {
        bu.canvas.png_olarak_kaydet(dosya_adı)
    }
}

// Usage
değer satış_verileri = [
    ("Ocak", 12500.0),
    ("Şubat", 15800.0),
    ("Mart", 18200.0),
    ("Nisan", 14500.0),
    ("Mayıs", 21000.0),
    ("Haziran", 19500.0)
]

değer grafik = ÇubukGrafik.new(800, 500)
grafik.çiz(satış_verileri, "Aylık Satış Grafiği")
grafik.kaydet("satış_grafiği.png")

print("✓ Grafik oluşturuldu!")

애니메이션 및 파티클 시스템

use graphics
use color
use random
use math

class Partikül {
    değer x: Gerçek
    değer y: Gerçek
    değer hız_x: Gerçek
    değer hız_y: Gerçek
    değer yarıçap: Gerçek
    değer renk: color.Renk
    değer lifespan: Gerçek
    
    function güncelle(delta_zaman: Gerçek) {
        bu.x += bu.hız_x * delta_zaman
        bu.y += bu.hız_y * delta_zaman
        bu.hız_y += 0.5  // Yerçekimi
        bu.lifespan -= delta_zaman
    }
    
    function çiz(canvas: graphics.Canvas) {
        değer alfa = (bu.lifespan / 2.0).kısıtla(0.0, 1.0)
        değer renk_alfa = color.rgba(
            bu.renk.kırmızı(),
            bu.renk.yeşil(),
            bu.renk.mavi(),
            (alfa * 255).tam()
        )
        canvas.dolgu_ayarla(renk_alfa)
        canvas.daire(bu.x, bu.y, bu.yarıçap, dolgu: true)
    }
    
    function canlı_mı() -> boolean {
        return bu.lifespan > 0
    }
}

class PartikülSistemi {
    değer canvas: graphics.Canvas
    değer partiküller: Liste<Partikül>
    
    function new(genişlik: Tam, yükseklik: Tam) -> PartikülSistemi {
        return PartikülSistemi {
            canvas: graphics.Canvas(genişlik, yükseklik),
            partiküller: []
        }
    }
    
    function partikül_oluştur(x: Gerçek, y: Gerçek, sayı: Tam) {
        for _ in 0..sayı {
            değer açı = random.gerçek(0.0, 2.0 * math.PI)
            değer hız = random.gerçek(50.0, 150.0)
            
            değer partikül = Partikül {
                x: x,
                y: y,
                hız_x: math.cos(açı) * hız,
                hız_y: math.sin(açı) * hız,
                yarıçap: random.gerçek(2.0, 6.0),
                renk: color.rgb(
                    random.tam(100, 255),
                    random.tam(50, 150),
                    random.tam(0, 100)
                ),
                lifespan: random.gerçek(1.0, 3.0)
            }
            
            bu.partiküller.push(partikül)
        }
    }
    
    function güncelle(delta_zaman: Gerçek) {
        // Partikülleri güncelle
        for partikül in bu.partiküller {
            partikül.güncelle(delta_zaman)
        }
        
        // Ölü partikülleri temizle
        bu.partiküller = bu.partiküller.filtrele(|p| p.canlı_mı())
    }
    
    function çiz() {
        // Arkaplanı temizle (solma efekti for)
        bu.canvas.dolgu_ayarla(color.rgba(0, 0, 0, 10))
        bu.canvas.dikdörtgen(0, 0, bu.canvas.genişlik, bu.canvas.yükseklik, dolgu: true)
        
        // Partikülleri çiz
        for partikül in bu.partiküller {
            partikül.çiz(bu.canvas)
        }
    }
    
    function kare_kaydet(kare_no: Tam) {
        bu.canvas.png_olarak_kaydet(f"partikül_{kare_no:04}.png")
    }
}

// Animasyon oluştur (60 FPS, 5 saniye)
değer sistem = PartikülSistemi.new(800, 600)
değer fps = 60.0
değer toplam_kare = 300  // 5 saniye

print("Animasyon oluşturuluyor...")

for kare in 0..toplam_kare {
    // Her 10 karede bir partikül patlaması
    if kare % 10 == 0 {
        sistem.partikül_oluştur(400.0, 300.0, 30)
    }
    
    sistem.güncelle(1.0 / fps)
    sistem.çiz()
    sistem.kare_kaydet(kare)
    
    if kare % 60 == 0 {
        print(f"  {kare / 60 + 1}. saniye tamamlandı")
    }
}

print("✓ Animasyon kareleri oluşturuldu!")
print("ffmpeg ile video oluşturmak for:")
print("  ffmpeg -framerate 60 -i partikül_%04d.png -c:v libx264 output.mp4")
