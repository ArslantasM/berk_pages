    개요

    nn 모듈은 최신 신경망 아키텍처를 구축하는 데 필요한 모든 구성 요소를 제공합니다. PyTorch nn.Module API와 유사한 디자인으로 쉽게 사용할 수 있습니다.

    레이어 유형

    선형(완전 연결)

    완전 연결 레이어: y = xW^T + b

    use ai/nn;

// 784 giriş → 128 çıkış
let layer = Linear::new(784, 128);

// Forward pass
let x = Tensor::random([32, 784]);  // batch_size=32
let y = layer.ileri(x);               // [32, 128]

    Conv2d(2D 컨볼루션)

    2D 컨볼루션 레이어

    use ai/nn;

// 3 kanal → 64 kanal, 3×3 kernel
let conv = Conv2d::new(
    in_channels: 3,
    out_channels: 64,
    kernel_size: 3,
    stride: 1,
    padding: 1
);

let x = Tensor::random([32, 3, 224, 224]);  // NCHW format
let y = conv.ileri(x);  // [32, 64, 224, 224]

    LSTM(장단기 기억)

    순환 신경망 계층

    use ai/nn;

// 100 girdi → 256 hidden units
let lstm = LSTM::new(
    input_size: 100,
    hidden_size: 256,
    num_layers: 2,
    dropout: 0.2
);

// Sequence: [batch, seq_len, features]
let x = Tensor::random([32, 50, 100]);
let (output, (h_n, c_n)) = lstm.ileri(x);
// output: [32, 50, 256]

    순차 모델

    MNIST 분류기

    use ai/nn;

let model = Sequential::new([
    Linear::new(784, 256),
    ReLU::new(),
    Dropout::new(0.2),
    Linear::new(256, 128),
    ReLU::new(),
    Dropout::new(0.2),
    Linear::new(128, 10),
    Softmax::new()
]);

// Forward pass
let x = Tensor::random([32, 784]);
let logits = model.ileri(x);  // [32, 10]

    맞춤 모듈

    나만의 모듈 만들기

    use ai/nn;
use ai/tensor;

sınıf ResBlock {
    let conv1: Conv2d;
    let bn1: BatchNorm;
    let conv2: Conv2d;
    let bn2: BatchNorm;
    let relu: ReLU;
    
    function new(channels: int) -> ResBlock do
        ResBlock {
            conv1: Conv2d::new(channels, channels, 3, 1, 1),
            bn1: BatchNorm::new(channels),
            conv2: Conv2d::new(channels, channels, 3, 1, 1),
            bn2: BatchNorm::new(channels),
            relu: ReLU::new()
        }
    end
    
    function ileri(let kendi, x: Tensor) -> Tensor do
        let identity = x;
        
        let out = kendi.conv1.ileri(x);
        out = kendi.bn1.ileri(out);
        out = kendi.relu.ileri(out);
        
        out = kendi.conv2.ileri(out);
        out = kendi.bn2.ileri(out);
        
        out = out.topla(identity);  // Residual connection
        out = kendi.relu.ileri(out);
        
        return out;
    end
}

    손실 함수

    교차엔트로피손실

    use ai/nn;

let criterion = CrossEntropyLoss::new();

let logits = model.ileri(x);      // [32, 10]
let targets = Tensor::new([32]); // Class indices
let loss = criterion(logits, targets);

print("Loss: {:.4f}", loss.değer());

    MSE 손실

    use ai/nn;

let criterion = MSELoss::new();

let predictions = model.ileri(x);
let targets = Tensor::random([32, 1]);
let loss = criterion(predictions, targets);

    API 참조

    레이어

    • 선의:완전 연결 레이어
    • 2d 전환:2D 컨볼루션
    • 전환 3일:3D 컨볼루션
    • LSTM:장단기 기억
    • 그루:게이트 순환 단위
    • 변신 로봇:주의 레이어

    활성화

    • ReLU:최대(0,x)
    • LeakyReLU:최대(0.01x,x)
    • 시그모이드:1/(1+e^-x)
    • 탄:(e^x - e^-x)/(e^x + e^-x)
    • 소프트맥스:e^xi / Σe^xj
    • 겔루:가우스 오차 선형 단위
