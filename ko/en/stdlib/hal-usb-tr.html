🔌 상태::usb

USB 2.0 최고 속도 - 장치/호스트/OTG 컨트롤러

~420줄 ~25개 기능 STM32F4/F7/H7

📖 개요

USB 모듈은 USB 2.0 Full Speed(12Mbps) 통신을 제공합니다. CDC(Virtual COM), HID(키보드/마우스), MSC(Mass Storage) 장치 클래스와 PC 연결을 설정합니다. 호스트 모드에서는 USB 플래시 디스크 및 키보드 읽기가 가능합니다. 이는 IoT 데이터 로깅, 디버그 콘솔, 펌웨어 업데이트 및 사용자 인터페이스에 중요합니다.

🔑 주요 기능

  • USB 2.0 최고 속도:12Mbps (저속 1.5Mbps도 지원)
  • 장치 모드:PC에 연결하면 가상기기가 됩니다.
  • 호스트 모드:USB 주변기기 연결 가능(플래시, 키보드)
  • OTG 지원:장치와 호스트 모드 간의 동적 전환
  • 장치 클래스:CDC, HID, MSC, 오디오, 맞춤형
  • DMA 지원:고속 데이터 전송
  • VBUS 감지:USB 케이블 삽입 감지
  • 셀프/버스 전원:전력 관리 유연성

// USB Mode
enum Mode {
    DEVICE,    // Device mode (slave)
    HOST,      // Host mode (master)
    OTG        // On-The-Go (dynamic)
}

// Device Classes
enum DeviceClass {
    CDC,       // Communication Device Class (Virtual COM)
    HID,       // Human Interface Device (Keyboard/Mouse)
    MSC,       // Mass Storage Class (USB Disk)
    AUDIO,     // Audio Class
    CUSTOM     // Custom class
}

// USB Speed
enum Speed {
    LOW_SPEED,   // 1.5 Mbps
    FULL_SPEED,  // 12 Mbps
    HIGH_SPEED   // 480 Mbps (USB 2.0 HS only)
}

// Power Mode
enum PowerMode {
    SELF_POWERED,   // External power supply
    BUS_POWERED     // Powered from USB cable (max 500 mA)
}

struct USBConfig do
    mode: Mode = DEVICE,
    device_class: DeviceClass = CDC,
    speed: Speed = FULL_SPEED,
    power_mode: PowerMode = BUS_POWERED,
    max_power_ma: sayı = 100,
    vid: sayı16 = 0x0483,
    pid: sayı16 = 0x5740
end

💡 예 1: USB CDC 가상 COM 포트(디버그 콘솔)

문제: PC와의 직렬 통신에 UART 대신 USB를 사용합니다.

import hal::usb

let usb_rx_buffer: Dizi<sayı8> = Dizi.new(256)
let usb_rx_index: sayı = 0

function usb_console_init() do
    usb.init_cdc(do
        vid: 0x0483,
        pid: 0x5740,
        manufacturer: "BERK",
        product: "Debug Console",
        serial_number: "BERK-001"
    end)
    
    // Set receive callback
    usb.cdc_set_rx_callback(usb_console_rx_handler)
    
    usb.connect()
    
    // Wait for enumeration
    for do
        if usb.is_configured() do
            kır
        end
        delay_ms(100)
    end
    
    usb.cdc_write("\r\n=== BERK Embedded Debug Console ===\r\n")
    usb.cdc_write("Type 'help' for commands\r\n> ")
end

function usb_console_rx_handler(data: Dizi<sayı8>) do
    for byte in data do
        if byte == 13 do  // Enter key
            // Process command
            usb.cdc_write("\r\n")
            
            let cmd = Metin.from_bytes(usb_rx_buffer, usb_rx_index)
            usb_console_process_command(cmd)
            
            usb_rx_index = 0
            usb.cdc_write("> ")
        yoksa if byte == 8 or byte == 127 do  // Backspace
            if usb_rx_index > 0 do
                usb_rx_index = usb_rx_index - 1
                usb.cdc_write("\b \b")  // Erase character
            end
        yoksa do
            if usb_rx_index < 256 do
                usb_rx_buffer[usb_rx_index] = byte
                usb_rx_index = usb_rx_index + 1
                usb.cdc_write_byte(byte)  // Echo
            end
        end
    end
end

function usb_console_process_command(cmd: Metin) do
    eşleştir cmd do
        "help" => do
            usb.cdc_write("Commands:\r\n")
            usb.cdc_write("  help    - Show this help\r\n")
            usb.cdc_write("  status  - System status\r\n")
            usb.cdc_write("  reset   - Reset system\r\n")
        end,
        
        "status" => do
            usb.cdc_write("CPU: STM32F407\r\n")
            usb.cdc_write("Clock: 168 MHz\r\n")
            usb.cdc_write("Uptime: " + systick.get_ms().metne() + " ms\r\n")
        end,
        
        "reset" => do
            usb.cdc_write("Resetting...\r\n")
            delay_ms(100)
            nvic.system_reset()
        end,
        
        _ => do
            usb.cdc_write("Unknown command: " + cmd + "\r\n")
        end
    end
end

사용:PC에서 PuTTY/TeraTerm을 사용하여 /dev/ttyACM0 또는 COM3을 엽니다. 115200 보드 설정은 필요하지 않습니다(USB CDC).

💡 예 2: USB HID 키보드 에뮬레이션

문제: MCU를 USB 키보드처럼 보이게 만들기(자동 입력)

import hal::usb

function usb_keyboard_init() do
    usb.init_hid(do
        vid: 0x0483,
        pid: 0x5750,
        manufacturer: "BERK",
        product: "USB Keyboard",
        hid_type: usb.HID_KEYBOARD
    end)
    
    usb.connect()
    
    // Wait for enumeration
    for do
        if usb.is_configured() do
            kır
        end
        delay_ms(100)
    end
end

function keyboard_press(key: sayı8) do
    // HID keyboard report: [modifier, reserved, key1-6]
    let report: Dizi<sayı8> = [0, 0, key, 0, 0, 0, 0, 0]
    
    // Send key press
    usb.hid_send_report(report)
    
    delay_ms(50)
    
    // Send key release
    report[2] = 0
    usb.hid_send_report(report)
    
    delay_ms(50)
end

function keyboard_type_string(text: Metin) do
    for char_at in text do
        // Convert ASCII to HID keycode
        let keycode = ascii_to_hid_keycode(char_at)
        keyboard_press(keycode)
    end
end

function main() do
    usb_keyboard_init()
    
    io.yazdir_satır("USB Keyboard ready")
    
    // Wait for button press
    for do
        if button_pressed() do
            // Type "Hello World!" on host PC
            keyboard_type_string("Hello World!")
            keyboard_press(0x28)  // Enter key
            
            delay_ms(1000)
        end
    end
end

결론:PC에 연결하면 실제 키보드처럼 인식됩니다. 자동 쓰기, 매크로 시스템.

💡 예 3: USB MSC - 대용량 저장소(USB 디스크 에뮬레이션)

문제: SD 카드를 USB 플래시 디스크처럼 보이게 만들기

import hal::usb, hal::sdio

function usb_msc_init() do
    // Initialize SD card
    sdio.init()
    
    // Get SD card info
    let sd_info = sdio.get_card_info()
    
    io.yazdir_satır("SD Card: " + (sd_info.capacity / 1024 / 1024).metne() + " MB")
    
    // Initialize USB MSC
    usb.init_msc(do
        vid: 0x0483,
        pid: 0x5720,
        product: "BERK USB Disk",
        block_count: sd_info.block_count,
        block_size: sd_info.block_size,
        read_callback: msc_read_block,
        write_callback: msc_write_block
    end)
    
    usb.connect()
    
    io.yazdir_satır("USB Mass Storage ready")
end

function msc_read_block(block_num: sayı, buffer: Dizi<sayı8>) -> mantık do
    // Read block from SD card
    return sdio.read_blocks(block_num, 1, buffer)
end

function msc_write_block(block_num: sayı, data: Dizi<sayı8>) -> mantık do
    // Write block to SD card
    return sdio.write_blocks(block_num, 1, data)
end

function main() do
    usb_msc_init()
    
    for do
        // Handle USB events
        usb.poll()
        
        // LED blink when active
        if usb.is_active() do
            led_toggle()
        end
        
        delay_ms(100)
    end
end

사용:PC에 연결하면 USB 플래시 디스크처럼 보입니다. 파일 복사가 작동합니다.

⚠️ 중요 사항:
  • 48MHz 클록:USB에 필요한 전체 48MHz 클록(공차 ±0.25%)
  • VID/PID:프로덕션에서 자신만의 VID/PID 가져오기(USB-IF에서)
  • 열거 시간:PC 인식 시간은 1~3초 정도 소요될 수 있습니다.
  • 인터럽트 우선순위:USB 인터럽트는 우선순위가 높아야 합니다

✅ 모범 사례

  • ✅ 디버깅을 위한 CDC:UART 대신 USB CDC 사용(케이블 요구 사항은 동일)
  • ✅ UI용 HID:HID 키보드/마우스를 사용하여 간단한 제어
  • ✅ 데이터용 MSC:로그 파일에 MSC 사용(FAT32)
  • ✅ 재설정하기 전에 연결을 끊으세요.재설정하기 전에 usb.disconnect()를 호출하세요.
  • ❌ 부동 작업:USB 인터럽트에서 부동 소수점 사용
  • ❌ 긴 콜백:USB 콜백을 짧게 유지하세요(<1ms)
