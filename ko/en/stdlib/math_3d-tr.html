import math_3d

// Vector3
let v1 = math_3d.vec3(1.0, 2.0, 3.0)
let v2 = math_3d.vec3(4.0, 5.0, 6.0)
let sum = math_3d.vec3_add(v1, v2)
let nokta = math_3d.vec3_dot(v1, v2)
let çapraz = math_3d.vec3_cross(v1, v2)

// Matrix4 (4x4 transformation matrix)
let M = math_3d.mat4_identity()
M = math_3d.mat4_translate(M, [10.0, 0.0, 0.0])
M = math_3d.mat4_rotate_y(M, 45.0)
M = math_3d.mat4_scale(M, [2.0, 2.0, 2.0])

// Quaternion (rotation)
let q = math_3d.quat_from_axis_angle([0.0, 1.0, 0.0], 90.0)
let rotated = math_3d.quat_rotate_vec3(q, v1)

    let target: Vec3
    let up: Vec3
    
    function view_matrix() -> Mat4 do
        return math_3d.mat4_look_at(self.position, self.target, self.up)
    end
    
    function projection_matrix(fov: kesir, aspect: kesir, near: kesir, far: kesir) -> Mat4 do
        return math_3d.mat4_perspective(fov, aspect, near, far)
    end
end
