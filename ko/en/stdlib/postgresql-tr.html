📖 개요

PostgreSQL 모듈은 유선 프로토콜을 통해 모든 기능을 갖춘 PostgreSQL 데이터베이스 클라이언트를 제공합니다. 준비된 명령문, 트랜잭션, 연결 풀링 및 유형이 안전한 쿼리를 지원합니다.

🔑 주요 기능

  • 전체 SQL 지원(SELECT, INSERT, UPDATE, DELETE)
  • 매개변수가 포함된 준비된 문
  • 트랜잭션(BEGIN, COMMIT, ROLLBACK)
  • 연결 풀링
  • 유형 변환(int, bigint, float, bool, text, bytea)
  • NULL 처리
  • 결과 세트 반복
  • 비동기 쿼리(예정)

import postgresql olarakpg

💡 예: 사용자 관리 시스템

import postgresql olarakpg, crypto

struct User do
    id: sayı,
    username: yazı,
    email: yazı,
    password_hash: yazı,
    created_at: time.DateTime
end

function db_init() -> Sonuç[pg.Connection, Err] do
    let conn = pg.connect(
        "host=localhost port=5432 dbname=users user=app password=secret"
    )?
    
    // Create tables
    conn.exec("
        CREATE TABLE IF NOT EXISTS users (
            id SERIAL PRIMARY KEY,
            username VARCHAR(50) UNIQUE NOT NULL,
            email VARCHAR(100) UNIQUE NOT NULL,
            password_hash VARCHAR(255) NOT NULL,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    ")?
    
    return Ok(conn)
end

function create_user(conn: pg.Connection, username: yazı, email: yazı, password: yazı) 
    -> Sonuç[sayı, Err] do
    // Hash password
    let hash = crypto.argon2_hash(password)?
    
    // Insert with prepared statement
    let result = conn.query_one("
        INSERT INTO users (username, email, password_hash)
        VALUES ($1, $2, $3)
        RETURNING id
    ", [username, email, hash])?
    
    return result.get_int("id")
end

function get_user_by_username(conn: pg.Connection, username: yazı) 
    -> Sonuç[İsteğe_Bağlı[User], Err] do
    let result = conn.query_one("
        SELECT id, username, email, password_hash, created_at
        FROM users WHERE username = $1
    ", [username])
    
    eşle result do
        Ok(row) => return Ok(Some(User do
            id: row.get_int("id")?,
            username: row.get_string("username")?,
            email: row.get_string("email")?,
            password_hash: row.get_string("password_hash")?,
            created_at: row.get_timestamp("created_at")?
        end)),
        Err(_) => return Ok(None)
    end
end

function authenticate(conn: pg.Connection, username: yazı, password: yazı) 
    -> Sonuç[bool, Err] do
    let user = get_user_by_username(conn, username)?
    
    if user.is_none() then do
        return Ok(false)
    end
    
    let u = user.unwrap()
    return Ok(crypto.argon2_verify(password, u.password_hash)?)
end

function main() do
    let conn = db_init()?
    
    // Create user
    let user_id = create_user(conn, "alice", "alice@example.com", "secret123")?
    io.println("Created user: {}", user_id)
    
    // Authenticate
    if authenticate(conn, "alice", "secret123")? then do
        io.println("✓ Authentication successful")
    else do
        io.println("✗ Authentication failed")
    end
    
    conn.close()
end

💼 거래 관리

// Begin transaction
conn.begin()?

try do
    conn.exec("INSERT INTO accounts (user_id, balance) VALUES ($1, $2)", [1, 1000])?
    conn.exec("INSERT INTO transactions (account_id, amount) VALUES ($1, $2)", [1, 100])?
    conn.exec("UPDATE accounts SET balance = balance - $1 WHERE id = $2", [100, 1])?
    
    conn.commit()?
    io.println("Transaction completed")
catch e do
    conn.rollback()?
    io.eprintln("Transaction failed: {}", e)
end
