import result

function split(a: kesir, b: kesir) -> Result do
    if b == 0.0 then do
        return Err("Sıfıra bölme hatası")
    end else do
        return Ok(a / b)
    end
end

let sonuç = split(10.0, 2.0)

if sonuç.tamam_mı() then do
    print("Sonuç: " + sonuç.pop().yazıya())
end else do
    print("Err: " + sonuç.hata().mesaj())
end

// Pattern matching
eşleşme sonuç do
    Ok(değer) => print(değer),
    Err(hata) => print(hata)
end

철도 지향 프로그래밍

import result

function kullanıcı_oku(id: int) -> Result do
    // DB'from kullanıcı input
    if id < 0 then do
        return Err("Geçersiz ID")
    end
    return Ok(do "id": id, "name": "User " + id.yazıya() end)
end

function kullanıcı_validate(user: herhangi) -> Result do
    if user["name"].len() < 3 then do
        return Err("İsim çok kısa")
    end
    return Ok(user)
end

function kullanıcı_kaydet(user: herhangi) -> Result do
    // DB'ye kaydet
    print("Kullanıcı kaydedildi: " + user["name"])
    return Ok(user)
end

// Pipeline
let sonuç = kullanıcı_oku(1)
    .and_then(kullanıcı_validate)
    .and_then(kullanıcı_kaydet)

eşleşme sonuç do
    Ok(_) => print("Başarılı!"),
    Err(e) => print("Err: " + e)
end
