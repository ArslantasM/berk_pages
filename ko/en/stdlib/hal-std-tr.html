// Result type

    Ok(T),
    Err(HalError)
end

// Error codes

    HAL_OK,
    HAL_ERROR,
    HAL_BUSY,
    HAL_TIMEOUT,
    HAL_UNSUPPORTED
end

// Handle types

    port: Port,
    pin: sayı,
    config: PinConfig
end

    instance: Timer,
    channel: Channel,
    frequency: sayı
end
