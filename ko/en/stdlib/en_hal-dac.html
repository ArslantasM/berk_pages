📈 할::dac

디지털-아날로그 변환기 - 아날로그 신호 생성

391줄 ~10가지 기능 12비트

📖 개요

DAC 모듈은 디지털 값을 아날로그 전압으로 변환합니다. 파동 생성, 오디오, 센서 시뮬레이션에 사용됩니다.

🔑 주요 기능

  • 12비트 분해능(0-4095)
  • 2개의 출력 채널
  • DMA 지원
  • 삼각형/잡음파 생성
  • 타이머 트리거
  • 출력 버퍼

import hal::dac, hal::gpio

// DAC pin (PA4 = DAC_OUT1)

💡 예: 사인파 생성기

import hal::dac, hal::timer, math

constant SAMPLES = 100

function generate_sine_wave() do
    // Generate sine wave lookup table
    variable sine_table = [0u16; SAMPLES]
    her i in 0..SAMPLES for do
        variable angle = 2.0 * math.PI * i.kesir() / SAMPLES.kesir()
        variable value = (math.sin(angle) + 1.0) * 2047.5
        sine_table[i] = value.sayı()
    end
    
    // Setup DAC with DMA
    variable dac1 = dac.init(dac.DAC1, dac.CHANNEL_1)
    dac.enable_dma(dac1)
    
    // Timer trigger (1kHz = 1kHz sine with 100 samples)
    variable tim6 = timer.init_basic(timer.TIM6, do
        frequency: 100_000  // 100kHz update rate
    end)
    
    dac.set_trigger(dac1, dac.TRIGGER_TIM6)
    
    // Start DMA circular mode
    dma.circular_transfer(dac.get_dma_channel(), 
        sine_table.as_ptr(),
        dac.get_data_register(dac1),
        SAMPLES
    )
    
    timer.start(tim6)
end

💡 예: 오디오 톤 생성기

import hal::dac, math

function play_tone(freq: kesir, duration_ms: sayı) do
    variable dac1 = dac.init(dac.DAC1, dac.CHANNEL_1)
    
    variable sample_rate = 8000.0  // 8kHz
    variable samples = (sample_rate * duration_ms.kesir() / 1000.0).sayı()
    
    her i in 0..samples for do
        variable t = i.kesir() / sample_rate
        variable value = (math.sin(2.0 * math.PI * freq * t) + 1.0) * 2047.5
        dac.write(dac1, value.sayı())
        core.delay_us(125)  // 8kHz = 125us per sample
    end
end

function main() do
    // Play musical scale (C4 to C5)
    variable freqs = [261.63, 293.66, 329.63, 349.23, 392.00, 440.00, 493.88, 523.25]
    
    her freq in freqs for do
        play_tone(freq, 500)  // 500ms each note
        core.delay_ms(100)    // 100ms gap
    end
end

💡 예: 함수 생성기(삼각형, 톱니형, 정사각형)

import hal::dac, hal::timer, hal::dma

enum Waveform {
    SINE,
    TRIANGLE,
    SAWTOOTH,
    SQUARE
}

constant SAMPLES = 256

function generate_waveform(waveform: Waveform, freq: kesir) do
    variable wave_table = [0u16; SAMPLES]
    
    // Generate waveform lookup table
    her i in 0..SAMPLES for do
        variable phase = i.kesir() / SAMPLES.kesir()
        
        variable value = eşle waveform do
            Waveform::SINE => do
                variable angle = 2.0 * math.PI * phase
                (math.sin(angle) + 1.0) * 2047.5
            end,
            
            Waveform::TRIANGLE => do
                if phase < 0.5 then do
                    phase * 4.0 * 4095.0
                end elseif do
                    (1.0 - (phase - 0.5) * 2.0) * 4095.0
                end
            end,
            
            Waveform::SAWTOOTH => do
                phase * 4095.0
            end,
            
            Waveform::SQUARE => do
                if phase < 0.5 then 4095.0 elseif 0.0
            end
        end
        
        wave_table[i] = value.sayı()
    end
    
    // Setup DAC with DMA + Timer trigger
    variable dac1 = dac.init(dac.DAC1, dac.CHANNEL_1, do
        trigger: dac.TRIGGER_TIM6,
        dma_enable: true
    end)
    
    // Timer frequency = desired_freq * SAMPLES
    variable tim6 = timer.init_basic(timer.TIM6, do
        frequency: (freq * SAMPLES.kesir()).sayı()
    end)
    
    // DMA circular mode
    variable dma_ch = dma.init(dma.DMA1, dma.STREAM5, do
        direction: dma.M2P,
        src_address: wave_table.as_ptr(),
        dst_address: dac.get_data_register(dac1),
        data_count: SAMPLES,
        src_increment: true,
        dst_increment: false,
        circular: true,
        data_size: dma.SIZE_16BIT
    end)
    
    dma.start(dma_ch)
    timer.start(tim6)
    
    io.println("Generating {} waveform at {}Hz", waveform, freq)
end

function main() do
    // Generate 1kHz triangle wave
    generate_waveform(Waveform::TRIANGLE, 1000.0)
    
    loop do
        core.delay_ms(1000)
    end
end

💡 예: 듀얼 채널 DAC(스테레오 출력)

import hal::dac, hal::gpio

function dac_dual_init() -> (dac.Handle, dac.Handle) do
    // PA4 = DAC_OUT1 (left channel)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 4, do mode: gpio.MODE_ANALOG end)
    
    // PA5 = DAC_OUT2 (right channel)
    gpio.pin_init(gpio.PORT_A, 5, do mode: gpio.MODE_ANALOG end)
    
    // Initialize both channels
    variable dac1 = dac.init(dac.DAC1, dac.CHANNEL_1)
    variable dac2 = dac.init(dac.DAC1, dac.CHANNEL_2)
    
    return (dac1, dac2)
end

function stereo_tone(left_freq: kesir, right_freq: kesir) do
    variable (dac_left, dac_right) = dac_dual_init()
    
    variable sample_rate = 8000.0
    
    her i in 0..8000 for do  // 1 second
        variable t = i.kesir() / sample_rate
        
        // Left channel
        variable left_val = (math.sin(2.0 * math.PI * left_freq * t) + 1.0) * 2047.5
        dac.write(dac_left, left_val.sayı())
        
        // Right channel
        variable right_val = (math.sin(2.0 * math.PI * right_freq * t) + 1.0) * 2047.5
        dac.write(dac_right, right_val.sayı())
        
        core.delay_us(125)
    end
end

function main() do
    // Stereo test: 440Hz left, 880Hz right
    stereo_tone(440.0, 880.0)
end

💡 예: 전압 기준/센서 시뮬레이션

import hal::dac

function voltage_to_dac(voltage: kesir) -> u16 do
    // DAC: 0-3.3V = 0-4095
    variable dac_val = (voltage / 3.3) * 4095.0
    return dac_val.clamp(0.0, 4095.0).sayı()
end

function simulate_temperature_sensor() do
    // Simulate LM35 temperature sensor (10mV/°C)
    variable dac1 = dac.init(dac.DAC1, dac.CHANNEL_1)
    
    // Simulate 25°C room temperature
    variable temp_c = 25.0
    
    loop do
        // LM35: 10mV per degree Celsius
        variable voltage = temp_c * 0.01  // 25°C = 0.25V
        
        dac.write(dac1, voltage_to_dac(voltage))
        
        io.println("Simulating {}°C ({}V)", temp_c, voltage)
        
        // Slowly vary temperature
        temp_c += 0.5
        if temp_c > 40.0 then temp_c = 20.0
        
        core.delay_ms(1000)
    end
end

function calibration_voltages() do
    // Generate precise reference voltages for ADC calibration
    variable dac1 = dac.init(dac.DAC1, dac.CHANNEL_1)
    
    variable test_voltages = [0.0, 0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.3]
    
    her voltage in test_voltages for do
        dac.write(dac1, voltage_to_dac(voltage))
        io.println("Output: {}V (measure with ADC)", voltage)
        core.delay_ms(5000)  // Hold for 5 seconds
    end
end

function main() do
    simulate_temperature_sensor()
end

💡 예: DAC를 통한 PWM 방식 DC 모터 제어

import hal::dac

function motor_speed_control() do
    // DAC output -> Motor driver analog input
    variable dac1 = dac.init(dac.DAC1, dac.CHANNEL_1)
    
    // Speed ramp up
    io.println("Motor ramping up...")
    her speed in 0..=100 for do
        variable voltage = (speed.kesir() / 100.0) * 3.3
        dac.write(dac1, voltage_to_dac(voltage))
        core.delay_ms(50)
    end
    
    // Hold max speed
    io.println("Motor at max speed")
    core.delay_ms(2000)
    
    // Speed ramp down
    io.println("Motor ramping down...")
    her speed in (0..=100).rev() for do
        variable voltage = (speed.kesir() / 100.0) * 3.3
        dac.write(dac1, voltage_to_dac(voltage))
        core.delay_ms(50)
    end
    
    io.println("Motor stopped")
end

function voltage_to_dac(voltage: kesir) -> u16 do
    return ((voltage / 3.3) * 4095.0).clamp(0.0, 4095.0).sayı()
end

function main() do
    loop do
        motor_speed_control()
        core.delay_ms(1000)
    end
end

⚙️ DAC 구성

// DAC Configuration
struct DACConfig do
    channel: Channel,           // CHANNEL_1 or CHANNEL_2
    trigger: Trigger,           // Software, Timer, EXTI
    output_buffer: bool,        // Enable for external loads
    dma_enable: bool,           // DMA for waveform generation
    noise_wave: bool,           // Noise generation
    triangle_wave: bool,        // Triangle generation
    amplitude: sayı             // Wave amplitude (0-12)
end

// Trigger Sources
enum Trigger {
    SOFTWARE,    // Manual trigger
    TIM2_TRGO,   // Timer 2
    TIM4_TRGO,   // Timer 4
    TIM6_TRGO,   // Timer 6
    TIM7_TRGO,   // Timer 7
    EXTI9        // External line 9
}

// Channels
enum Channel {
    CHANNEL_1,  // PA4
    CHANNEL_2   // PA5
}

📊 DAC 사용 시나리오

애플리케이션 빈도 방법 설명
DC 전압 레퍼런스 공전 수동 쓰기 교정, 센서 시뮬레이션
오디오 톤(8kHz) 100Hz-4kHz 소프트웨어 루프 간단한 호출음, 알람
파형 발생기 1Hz-10kHz DMA + 타이머 함수 발생기, 테스트 장비
오디오 재생(44kHz) 20Hz-20kHz DMA + I2S/타이머 음악, 목소리
엔진 속도 제어 DC-100Hz 느린 업데이트 아날로그 모터 드라이버 입력

⚡ 성능 팁

  • DMA + 타이머: (CPU가 필요 없는) 파형 생성을 사용합니다.
  • 조회 테이블: 사인/삼각형 값을 미리 계산합니다.
  • 출력 버퍼: 외부 부하(>5kΩ)를 활성화합니다.
  • 듀얼 채널: 스테레오 또는 차동 출력을 사용합니다.
  • 트리거 소스: 이상적인 타이머 트리거 정확한 타이밍

🖥️ 플랫폼 지원

  • STM32F1: 2채널 12비트 DAC
  • STM32F4: 2채널 12비트 DAC, 최대 1MSPS
  • STM32L4: 2채널 12비트 DAC, 초저전력
  • STM32H7: 2채널 12비트 DAC, 최대 1MSPS
  • GD32: 호환 가능한 DAC

⚠️ 중요 사항

  • 전압 범위: 0V - 3.3V(VDDA 기준)
  • 출력 전류: 버퍼 없이 최대 ~5mA, 버퍼 포함 시 10-20mA
  • 출력 버퍼: 부하 < 5kΩ 임피던스에 대해 활성화
  • 핀 매핑: PA4 = DAC_OUT1, PA5 = DAC_OUT2(고정, 대체 없음)
  • 해결: 12비트(0-4095), 오른쪽 정렬 또는 왼쪽 정렬
  • 샘플링 속도: DMA + 타이머 트리거 사용 시 최대 1MSPS
  • 노이즈/삼각형: 하드웨어 파형 생성 모드(테스트에 적합)
  • 듀얼 모드: 두 채널 동시 업데이트(동기 출력)
