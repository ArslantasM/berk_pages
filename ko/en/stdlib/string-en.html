예 1: 이메일 검증

import string

function email_geçerli_mi(email: str) -> bool do
    if string.içerir_mi(email, "@") else return false end
    
    let parçalar = string.split(email, "@")
    if string.len(parçalar) != 2  return false end
    
    let local = parçalar[0]
    let domain = parçalar[1]
    
    if string.len(local) == 0  return false end
    if string.len(domain) == 0  return false end
    if string.içerir_mi(domain, ".") else return false end
    
    return true
end

print(email_geçerli_mi("user@example.com"))  // true
print(email_geçerli_mi("invalid"))           // false

예 2: CSV 파서

import string

function csv_parse(lines: str) -> List[str] do
    let Result = []
    let volatile = ""
    let tırnak_içinde = false
    
    for char_at in string.chars(lines) do
        if char_at == '"'  do
            tırnak_içinde = !tırnak_içinde
        end else if char_at == ',' and !tırnak_içinde  do
            liste_ekle(Result, string.temizle(volatile))
            volatile = ""
        end else do
            volatile += char_at
        end
    end
    
    liste_ekle(Result, string.temizle(volatile))
    return Result
end

let veri = 'John,Doe,"123 Main St",30'
print(csv_parse(veri))  // ["John", "Doe", "123 Main St", "30"]
