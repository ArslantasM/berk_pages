📚 프로토콜

  • TCP:안정적인 연결 지향 스트리밍

  • TLS/SSL:암호화된 TCP 연결
  • DNS:도메인 이름 확인
  • 웹소켓:전이중 통신

import network, thread

function handle_client(client: Socket) do
    try do
        for do
            let data = network.receive(client, 1024)
            
            if data.len() == 0 then do
                kır  // Client disconnected
            end
            
            print("Alındı: " + data)
            network.send(client, "Echo: " + data)
        end
    end catch hata do
        print("Client hatası: " + hata.mesaj())
    end sonunda do
        network.close(client)
    end
end

function sunucu_başlat() do
    let sunucu = network.tcp_listen("0.0.0.0", 8080)
    print("Server running on port 8080")
    
    for do
        let client = network.accept(sunucu)
        let client_addr = network.peer_addr(client)
        print("Yeni bağlantı: " + client_addr)
        
        // Her client for new thread
        thread.spawn(function() do
            handle_client(client)
        end)
    end
end

import network, crypto

function https_get(host: yazı, path: yazı) -> Sonuç do
    // DNS çözümleme
    let ip = network.dns_resolve(host)
    
    // TLS socket
    let socket = network.tcp_connect(ip, 443)
    let tls = network.tls_wrap(socket, host)
    
    // HTTP request
    let request = "GET " + path + " HTTP/1.1\r\n" +
                      "Host: " + host + "\r\n" +
                      "Connection: close\r\n\r\n"
    
    network.tls_send(tls, request)
    
    // Response okuma
    let yanıt = ""
    for do
        let chunk = network.tls_receive(tls, 4096)
        if chunk.len() == 0 then do
            kır
        end
        yanıt = yanıt + chunk
    end
    
    network.tls_close(tls)
    return Ok(yanıt)
end
