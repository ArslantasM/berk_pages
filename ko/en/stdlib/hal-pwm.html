📊 상태::pwm

펄스 폭 변조 - PWM 신호 생성

~320줄 ~12가지 기능 STM32/GD32/ESP32

📖 개요

PWM 모듈은 타이머 하드웨어를 사용하여 가변 폭 디지털 신호를 생성합니다. 모터 속도 제어, LED 밝기 조정, 서보 제어 및 아날로그 출력 시뮬레이션에 사용됩니다.

🔑 주요 기능

  • 구성 가능한 주파수(1Hz - 1MHz+)
  • 0-100% 듀티 사이클 제어
  • 다중 채널 지원(최대 4개 채널)
  • 보완 출력(데드타임 있음)
  • 하드웨어 동기화 및 트리거
  • DMA 지원 파형 생성
  • 원펄스 모드
  • 인코더 인터페이스 모드

🚀 빠른 시작

import hal::pwm, hal::gpio

// PWM pin setup (TIM3 CH1 -> PA6)
gpio.pin_init(gpio.PORT_A, 6, do
    mode: gpio.MODE_AF,
    af: gpio.AF2_TIM3
end)

// PWM initialization (1 kHz, 50% duty)
let pwm_handle = pwm.init(pwm.TIM3, pwm.CHANNEL_1, do
    frequency: 1000,      // 1 kHz
    duty_cycle: 50,       // 50%
    polarity: pwm.ACTIVE_HIGH,
    mode: pwm.MODE_PWM1
end)

// Start PWM
pwm.start(pwm_handle)

// Update duty cycle
pwm.set_duty_cycle(pwm_handle, 75)  // 75%

💡 예시 1: LED 디밍(밝기 조절)

PWM으로 LED 밝기 제어

import hal::pwm, hal::core

function led_dimmer() do
    // PWM setup (10 kHz)
    let led_pwm = pwm.init(pwm.TIM2, pwm.CHANNEL_1, do
        frequency: 10000,
        duty_cycle: 0,
        polarity: pwm.ACTIVE_HIGH,
        mode: pwm.MODE_PWM1
    end)
    
    pwm.start(led_pwm)
    
    // Fade in/out loop
    for do
        // Fade in (0% -> 100%)
        for brightness in 0..100 for do
            pwm.set_duty_cycle(led_pwm, brightness.kesir())
            core.delay_ms(10)
        end
        
        core.delay_ms(500)
        
        // Fade out (100% -> 0%)
        for brightness in (0..100).reverse() for do
            pwm.set_duty_cycle(led_pwm, brightness.kesir())
            core.delay_ms(10)
        end
        
        core.delay_ms(500)
    end
end

💡 예시 2: DC 모터 속도 제어

L298N 드라이버를 이용한 모터 제어

import hal::pwm, hal::gpio

struct Motor do
    pwm: PWMHandle,
    dir_pin1: GPIOPin,
    dir_pin2: GPIOPin
end

function motor_init() -> Motor do
    // Direction pins
    let in1 = gpio.pin_init(gpio.PORT_B, 0, gpio.MODE_OUTPUT)
    let in2 = gpio.pin_init(gpio.PORT_B, 1, gpio.MODE_OUTPUT)
    
    // PWM for speed (20 kHz)
    let motor_pwm = pwm.init(pwm.TIM1, pwm.CHANNEL_1, do
        frequency: 20000,
        duty_cycle: 0,
        polarity: pwm.ACTIVE_HIGH,
        mode: pwm.MODE_PWM1
    end)
    
    pwm.start(motor_pwm)
    
    return Motor do
        pwm: motor_pwm,
        dir_pin1: in1,
        dir_pin2: in2
    end
end

function motor_set_speed(motor: Motor, speed: kesir) do
    // speed: -100 to +100 (negative = reverse)
    if speed > 0 then do
        // Forward
        gpio.pin_write(motor.dir_pin1, gpio.HIGH)
        gpio.pin_write(motor.dir_pin2, gpio.LOW)
        pwm.set_duty_cycle(motor.pwm, speed)
    yoksa if speed < 0 then do
        // Reverse
        gpio.pin_write(motor.dir_pin1, gpio.LOW)
        gpio.pin_write(motor.dir_pin2, gpio.HIGH)
        pwm.set_duty_cycle(motor.pwm, -speed)
    yoksa do
        // Stop
        gpio.pin_write(motor.dir_pin1, gpio.LOW)
        gpio.pin_write(motor.dir_pin2, gpio.LOW)
        pwm.set_duty_cycle(motor.pwm, 0.0)
    end
end

function main() do
    let motor = motor_init()
    
    // Acceleration test
    for speed in 0..100 for do
        motor_set_speed(motor, speed.kesir())
        core.delay_ms(50)
    end
    
    core.delay_ms(2000)
    
    // Deceleration
    for speed in (0..100).reverse() for do
        motor_set_speed(motor, speed.kesir())
        core.delay_ms(50)
    end
    
    motor_set_speed(motor, 0.0)  // Stop
end

💡 예시 3: 서보 모터 제어(SG90)

50Hz PWM을 이용한 서보 각도 제어

import hal::pwm

struct Servo do
    pwm: PWMHandle,
    min_pulse: kesir,  // ms
    max_pulse: kesir   // ms
end

function servo_init(timer: Timer, channel: Channel) -> Servo do
    // Servo: 50Hz, 1-2ms pulse width
    let servo_pwm = pwm.init(timer, channel, do
        frequency: 50,      // 50 Hz (20ms period)
        duty_cycle: 7.5,    // 1.5ms pulse (90°)
        polarity: pwm.ACTIVE_HIGH,
        mode: pwm.MODE_PWM1
    end)
    
    pwm.start(servo_pwm)
    
    return Servo do
        pwm: servo_pwm,
        min_pulse: 1.0,  // 1ms = 0°
        max_pulse: 2.0   // 2ms = 180°
    end
end

function servo_set_angle(servo: Servo, angle: kesir) do
    // angle: 0-180 degrees
    let clamped = angle.clamp(0.0, 180.0)
    
    // Map angle to pulse width (1-2ms)
    let pulse_ms = servo.min_pulse + 
        (clamped / 180.0) * (servo.max_pulse - servo.min_pulse)
    
    // Convert to duty cycle (20ms period)
    let duty = (pulse_ms / 20.0) * 100.0
    
    pwm.set_duty_cycle(servo.pwm, duty)
end

function main() do
    let servo = servo_init(pwm.TIM2, pwm.CHANNEL_1)
    
    // Sweep 0° to 180°
    for do
        for angle in 0..180 for do
            servo_set_angle(servo, angle.kesir())
            core.delay_ms(15)
        end
        
        for angle in (0..180).reverse() for do
            servo_set_angle(servo, angle.kesir())
            core.delay_ms(15)
        end
    end
end

💡 예 4: 버저 멜로디 연주

PWM 주파수를 변경하여 노트 생성

import hal::pwm, hal::core

// Note frequencies (Hz)
const DO = 262
const RE = 294
const MI = 330
const FA = 349
const SOL = 392
const LA = 440
const SI = 494
const DO_HIGH = 523

function buzzer_init() -> PWMHandle do
    let buzzer = pwm.init(pwm.TIM3, pwm.CHANNEL_2, do
        frequency: 440,  // Default: LA (440 Hz)
        duty_cycle: 50,
        polarity: pwm.ACTIVE_HIGH,
        mode: pwm.MODE_PWM1
    end)
    
    return buzzer
end

function play_tone(buzzer: PWMHandle, freq: sayı, duration_ms: sayı) do
    pwm.set_frequency(buzzer, freq)
    pwm.start(buzzer)
    core.delay_ms(duration_ms)
    pwm.stop(buzzer)
end

function play_melody(buzzer: PWMHandle) do
    // "Twinkle Twinkle Little Star"
    let melody = [
        (DO, 500), (DO, 500), (SOL, 500), (SOL, 500),
        (LA, 500), (LA, 500), (SOL, 1000),
        (FA, 500), (FA, 500), (MI, 500), (MI, 500),
        (RE, 500), (RE, 500), (DO, 1000)
    ]
    
    her (note, duration) in melody for do
        play_tone(buzzer, note, duration)
        core.delay_ms(50)  // Short pause between notes
    end
end

function main() do
    let buzzer = buzzer_init()
    
    for do
        play_melody(buzzer)
        core.delay_ms(2000)
    end
end

💡 모범 사례

  • 주파수 선택:모터 제어용 20-25kHz(사람의 가청 범위 밖)
  • 서보 제어:50Hz 고정 주파수, 1~2ms 펄스 폭
  • LED 디밍:100+Hz(눈이 깜박임을 감지하지 못함)
  • 타이머 충돌:동일한 타이머의 다른 채널 사용
  • 데드타임:H-bridge 제어에서 슛스루를 방지하기 위해 필요
