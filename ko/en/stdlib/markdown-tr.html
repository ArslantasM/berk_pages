📖 개요

Markdown 모듈은 CommonMark 표준을 준수하는 Markdown 파서 및 HTML 렌더러를 제공합니다. AST 조작, 구문 강조 및 목차 생성을 지원합니다.

🔑 주요 기능

  • CommonMark 파서(표준 준수)
  • 사용자 정의를 통한 HTML 렌더링
  • AST(추상 구문 트리) 액세스
  • 제목, 단락, 목록, 인용문
  • 구문 강조가 포함된 코드 블록
  • 표, 각주, 작업 목록
  • 인라인: 굵게, 기울임꼴, 코드, 링크, 이미지
  • 목차 생성
  • 단어 수 및 독서 시간

💡 예 1: 블로그 생성기

import markdown, fs, time, regex

struct BlogPost do
    title: yazı,
    date: time.Date,
    author: yazı,
    content_html: yazı,
    excerpt: yazı,
    reading_time: sayı
end

function parse_blog_post(md_file: yazı) -> Sonuç[BlogPost, Err] do
    // Markdown dosyasını input
    let content = fs.read_to_string(md_file)?
    
    // Front matter parse et (YAML)
    let (frontmatter, body) = split_frontmatter(content)?
    
    let title = frontmatter.get("title")?
    let date = time.parse(frontmatter.get("date")?, "%Y-%m-%d")?
    let author = frontmatter.get("author")?
    
    // Markdown to HTML
    let doc = markdown.parse(body)
    
    // Table of contents push
    let toc = markdown.generate_toc(doc)
    
    // Syntax highlighting
    markdown.highlight_code_blocks(doc, do
        theme: "github",
        line_numbers: true
    end)
    
    let html = markdown.to_html(doc)
    
    // Excerpt (first paragraph)
    let excerpt = markdown.extract_text(doc, max_length: 150)
    
    // Reading time (avg 200 kelime/dakika)
    let word_count = markdown.word_count(doc)
    let reading_time = (word_count / 200).max(1)
    
    return Ok(BlogPost do
        title: title,
        date: date,
        author: author,
        content_html: html,
        excerpt: excerpt,
        reading_time: reading_time
    end)
end

function generate_blog_html(post: BlogPost) -> yazı do
    return "
        <article>
            <header>
                <h1>{}</h1>
                <div class='meta'>
                    <span>By {}</span>
                    <span>{}</span>
                    <span>{} min read</span>
                </div>
            </header>
            <div class='excerpt'>{}</div>
            <div class='content'>
                {}
            </div>
        </article>
    ".formatla(
        post.title,
        post.author,
        post.date.format("%B %d, %Y"),
        post.reading_time,
        post.excerpt,
        post.content_html
    )
end

function main() do
    let posts = []
    
    // Tüm blog postlarını parse et
    her file in fs.read_dir("posts")? for do
        if file.ends_with(".md") then do
            eşle parse_blog_post("posts/" + file) do
                Ok(post) => posts.push(post),
                Err(e) => io.eprintln("Error parsing {}: {}", file, e)
            end
        end
    end
    
    // Tarihe göre sort
    posts.sort_by(|a, b| b.date.cmp(&a.date))
    
    // HTML oluştur
    her post in posts for do
        let html = generate_blog_html(post)
        let slug = post.title.to_lowercase().replace(" ", "-")
        fs.write("output/{}.html".formatla(slug), html)?
    end
    
    io.println("Generated {} blog posts", posts.len())
end

💡 예 2: 문서 생성기

import markdown, fs

function generate_docs(input_dir: yazı, output_dir: yazı) -> Sonuç[None, Err] do
    // Template
    let template = fs.read_to_string("template.html")?
    
    // Sidebar navigation
    let nav_items = []
    
    // Process all markdown files
    her file in fs.glob("{}/**/*.md".formatla(input_dir))? for do
        let md_text = fs.read_to_string(file)?
        let doc = markdown.parse(md_text)
        
        // Extract title (first h1)
        let title = markdown.find_first_heading(doc, level: 1)
            .unwrap_or("Untitled")
        
        // Generate TOC
        let toc_html = markdown.generate_toc_html(doc, do
            min_level: 2,
            max_level: 4,
            id_prefix: "toc-"
        end)
        
        // Add IDs to headings
        markdown.add_heading_ids(doc)
        
        // Convert to HTML
        let content_html = markdown.to_html(doc)
        
        // Apply template
        let html = template
            .replace("{{title}}", title)
            .replace("{{toc}}", toc_html)
            .replace("{{content}}", content_html)
        
        // Output path
        let rel_path = file.strip_prefix(input_dir)?
        let out_file = "{}/{}.html".formatla(output_dir, 
            rel_path.strip_suffix(".md")?)
        
        fs.create_dir_all(fs.parent(out_file)?)?
        fs.write(out_file, html)?
        
        // Add to navigation
        nav_items.push(do
            title: title,
            path: rel_path.replace(".md", ".html")
        end)
        
        io.println("✓ Generated {}", out_file)
    end
    
    // Generate index with navigation
    generate_index(output_dir, nav_items)?
    
    return Ok(None)
end

📦AST 조작

// Parse to AST
let doc = markdown.parse("# Title\nContent")

// Traverse nodes
let root = doc.root()
her node in root.children() for do
    eşle node.type() do
        NodeType.Heading(level) => do
            io.println("H{}: {}", level, node.text())
        end,
        NodeType.Paragraph => do
            io.println("Paragraph: {}", node.text())
        end,
        _ => do end
    end
end

// Modify AST
let heading = markdown.Node.new_heading(2)
heading.append_text("New Section")
root.append_child(heading)

// Convert back to HTML
let html = markdown.to_html(doc)
