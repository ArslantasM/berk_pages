💡 예

절차적 메시 생성기

import mesh, math

function terrain_generate(width: int, height: int) -> Mesh do
    let m = mesh.new()
    
    // Generate vertices
    her y in range(0, height) for do
        her x in range(0, width) for do
            let z = math.sin(x.kesir() * 0.1) * math.cos(y.kesir() * 0.1) * 5.0
            mesh.add_vertex(m, [x.kesir(), z, y.kesir()])
        end
    end
    
    // Generate faces
    her y in range(0, height-1) for do
        her x in range(0, width-1) for do
            let i0 = y * width + x
            let i1 = i0 + 1
            let i2 = (y + 1) * width + x
            let i3 = i2 + 1
            
            mesh.add_triangle(m, i0, i1, i2)
            mesh.add_triangle(m, i1, i3, i2)
        end
    end
    
    mesh.compute_normals(m)
    return m
end

let terrain = terrain_generate(100, 100)
mesh.save_obj(terrain, "terrain.obj")
