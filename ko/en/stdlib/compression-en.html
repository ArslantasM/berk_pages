import compression, fs, io

function dosya_sıkıştır(kaynak: yazı, hedef: yazı) do
    let veri = io.dosya_oku_bytes(kaynak)
    let sıkıştırılmış = compression.zstd_compress(veri, 9)
    io.dosya_yaz_bytes(hedef, sıkıştırılmış)
    
    let orijinal_boyut = veri.len()
    let sıkışık_boyut = sıkıştırılmış.len()
    let oran = (1.0 - sıkışık_boyut.kesir() / orijinal_boyut.kesir()) * 100.0
    
    print("Sıkıştırma: " + orijinal_boyut.yazıya() + " -> " + sıkışık_boyut.yazıya())
    print("Oran: " + oran.yazıya() + "%")
end

dosya_sıkıştır("large_file.txt", "large_file.txt.zst")
