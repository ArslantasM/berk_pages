BERK Programlama Dili

루프

반복 작업을 위한 루프 구조.

For 루프(범위)

// 0'from 4'e to (5 hariç)
for i 0'from 5'e {
    print(i)  // 0, 1, 2, 3, 4
}

// 0'from 5'e to (5 dahil)
for i 0'from 6'ya to {
    print(i)  // 0, 1, 2, 3, 4, 5
}

// Tersden sayma
for i 10'from 0'a {
    print(i)  // 10, 9, 8, ..., 1
}

// Adım boyutu
for i 0'from 10'a adım 2 {
    print(i)  // 0, 2, 4, 6, 8
}

영어 구문:

for i from 0 to 5 {
    print(i)
}

for i from 0 to 5 inclusive {
    print(i)
}

For-Each 루프

let sayılar = [1, 2, 3, 4, 5]

for sayı sayılarda {
    print(sayı)
}

// İndeks ile
for (indeks, değer) sayılarda {
    print(indeks, ": ", değer)
}

영어:

for num in numbers {
    print(num)
}

for (index, value) in numbers {
    print(index, ": ", value)
}

while 루프

let count_occurrences = 0

while count_occurrences < 5 {
    print(count_occurrences)
    count_occurrences += 1
}

영어:

var counter = 0

while counter < 5 {
    print(counter)
    counter += 1
}

Do-While 루프

적어도 한 번은 작동합니다:

let x = 0

do {
    print(x)
    x += 1
} while x < 5

for i 1'from 4'e {
    for j 1'from 4'e {

    }

반복자 메서드

let sayılar = [1, 2, 3, 4, 5]

// Map

예

피보나치

function fibonacci(n: number) -> Liste<number> {
    let sonuç = [0, 1]
    
    for i 2'from n'ye {
        let sonraki = sonuç[i-1] + sonuç[i-2]
        sonuç.push(sonraki)
    }
    
    return sonuç
}
