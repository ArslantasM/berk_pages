BERK Programlama Dili

문자열 연산

BERK의 문자열 연산신뢰할 수 있는, 빠른맹세유용하다. UTF-8 지원, 풍부한 stdlib API 및 제로 복사 작업.

문자열 만들기

리터럴 문자열

let isim = "Ali"
let mesaj = "Merhaba Dünya!"
let çok_satırlı = """
    Bu bir
    çok satırlı
    string'dir
"""

이스케이프 문자

let yeni_satır = "Satır 1\nSatır 2"
let tab = "Kolon\tDeğer"
let tırnak = "O dedi: \"Merhaba\""
let unicode = "\u{1F600}"  // 😀

기본 작업

연쇄

let ad = "Ali"
let soyad = "Yılmaz"

// + operatörü
let tam_ad = ad + " " + soyad

// format! makrosu
let mesaj = format!("Merhaba {}, yaşınız {}.", isim, yaş)

길이 및 블랭크 제어

let s = "Merhaba"
print(s.len())      // 7
print(s.is_empty())       // false
print(s.karakter_sayısı()) // UTF-8 char_at sayısı

문자열 모듈 기능

케이스 변환

use string;

let String = "Merhaba Dünya"
print(string::büyüt(String))         // "MERHABA DÜNYA"
print(string::küçült(String))        // "merhaba dünya"
print(string::baş_harfi_büyüt(String)) // "Merhaba dünya"

트림 및 패드

let s = "  merhaba  "
print(string::trim(s))         // "merhaba"
print(string::trim_start(s))   // "merhaba  "
print(string::trim_end(s))     // "  merhaba"

let x = "42"
print(string::pad_left(x, 5, '0'))  // "00042"

나뉘다

let String = "elma,armut,muz"
let parçalar = string::split(String, ",")
// ["elma", "armut", "muz"]

// Satırlara split
let çok_satır = "Satır 1\nSatır 2\nSatır 3"
let satırlar = string::satırlara_böl(çok_satır)

바꾸다

let String = "Merhaba Dünya"
let new = string::borrow_mut(String, "Dünya", "BERK")
print(new)  // "Merhaba BERK"

// Tüm oluşumları borrow_mut
let s = "test test test"
print(string::değiştir_tümü(s, "test", "deneme"))

검색 및 제어

컨테이너

let String = "Merhaba Dünya"
print(string::contains(String, "Dünya"))  // true
print(string::contains(String, "Java"))   // false

시작과 끝

let dosya = "resim.png"
print(string::starts_with(dosya, "resim"))  // true
print(string::ends_with(dosya, ".png"))    // true

찾다

let String = "Merhaba Dünya"
let index = string::bul(String, "Dünya")

eşleştir index do
    Bir(i) => print("Bulundu: " + i),
    Hiçbiri => print("Bulunamadı")
end

형식 문자열

// Basit interpolasyon
let isim = "Ali"
let yaş = 25
let mesaj = format!("İsim: {}, Yaş: {}", isim, yaş)

// İsimlendirilmiş parametreler
let m = format!("x: {x}, y: {y}", x: 10, y: 20)

// number formatlama
let pi = 3.14159
print(format!("Pi: {:.2}", pi))  // "Pi: 3.14"

유니코드 및 UTF-8

캐릭터 반복

let String = "Merhaba 🌍"

for char_at in string::chars(String) do
    print(char_at)
end

// Byte iterasyonu
for byte in string::byte_lar(String) do
    print(byte)
end

터키어 문자 지원

let türkçe = "çalışma ığğöüş"
print(string::büyüt(türkçe))  // "ÇALIŞMA IĞĞÖÜŞ"
print(türkçe.len())       // Doğru byte sayısı
print(türkçe.karakter_sayısı()) // Görünen char_at sayısı

정규식 지원

use regex::Regex;

let re = Regex::new(r"\d{3}-\d{4}")
let telefon = "555-1234"

if re.eşleşir(telefon) do
    print("Geçerli telefon")
end

// Tüm eşleşmeleri bul
let String = "Email: ali@test.com and ayse@test.com"
let re2 = Regex::new(r"[a-z]+@[a-z]+\.[a-z]+")
for eşleşme in re2.tüm_eşleşmeler(String) do
    print(eşleşme)
end

스트링 빌더

효율적인 문자열 연결을 위해:

use string::StringBuilder;

let sb = StringBuilder::new()
sb.push("Merhaba")
sb.push(" ")
sb.push("Dünya")
sb.ekle_satır("!")

let sonuç = sb.inşa_et()
print(sonuç)  // "Merhaba Dünya!\n"

성능 팁

  1. ✅ StringBuilder 사용(여러 조인의 경우)

    // ❌ Yavaş
let s = ""
for i in 0..1000 do
    s += "x"
end

// ✅ Hızlı
let sb = StringBuilder::new()
for i in 0..1000 do
    sb.push("x")
end

  2. ✅ 스트링 슬라이스 사용(복사하지 않고)

    let s = "Merhaba Dünya"
let alt = s[0..7]  // "Merhaba" (zero-copy)

  3. ✅ 사이즈를 미리 예약하세요

    let sb = StringBuilder::with_capacity(1000)
