    mode: "min"            // Loss for 'min', acc for 'max'

    맞춤형 훈련 루프

    use ai/nn;
use ai/optim;

function özel_eğitim(
    model: Model,
    train_loader: DataLoader,
    epochs: int
) do
    let optimizer = Adam::new(model.parametreler(), lr: 0.001);
    
    her epoch in range(epochs) for do
        let epoch_loss = 0.0;
        
        her (x, y) in train_loader for do
            // Forward pass
            let pred = model.ileri(x);
            let loss = cross_entropy(pred, y);
            
            // Backward pass
            optimizer.sıfırla();
            loss.geri();
            optimizer.adım();
            
            epoch_loss += loss.değer();
        end
        
        print("Epoch {}: Loss = {:.4f}", epoch, epoch_loss);
    end
end
