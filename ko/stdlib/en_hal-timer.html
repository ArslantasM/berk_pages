하드웨어 타이머/카운터 - PWM 및 타이밍 제어

📖 개요

타이머 모듈은 하드웨어 타이머, PWM 생성, 입력 캡처 및 출력 비교 기능을 제공합니다. 모터 제어, 서보 제어, 주파수 측정에 사용됩니다.

🔑 주요 기능

  • PWM 생성(4채널)
  • 입력 캡처
  • 출력 비교
  • 인코더 모드
  • 원펄스 모드
  • 주기적 중단
  • DMA 버스트 모드
  • 16/32비트 카운터

💡 예시 1: LED 호흡 효과

import hal::timer, hal::gpio, hal::core

function main() do
    core.system_init()
    
    // Setup PWM for LED (PA0)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_AF,
        speed: gpio.SPEED_HIGH,
        af: 1
    end)
    
    // Initialize PWM (1kHz)
    variable pwm = timer.pwm_init(timer.TIM2, timer.CH1, do
        frequency: 1000,
        duty_cycle: 0
    end)
    
    timer.pwm_start(pwm)
    
    // Breathing effect
    loop do
        // Fade in
        her duty in 0..101 for do
            timer.pwm_set_duty(pwm, duty)
            core.delay_ms(10)
        end
        
        // Fade out
        her duty in (0..101).reverse() for do
            timer.pwm_set_duty(pwm, duty)
            core.delay_ms(10)
        end
    end
end

💡 예 2: 서보 모터 제어

import hal::timer

struct Servo do
    pwm: timer.PWMHandle,
    min_pulse: sayı,  // microseconds
    max_pulse: sayı
end

function servo_init(tim: timer.Timer, channel: timer.Channel) -> Servo do
    // Servo PWM: 50Hz (20ms period), pulse: 1-2ms
    variable pwm = timer.pwm_init(tim, channel, do
        frequency: 50,  // 50Hz
        duty_cycle: 0
    end)
    
    timer.pwm_start(pwm)
    
    return Servo do
        pwm: pwm,
        min_pulse: 1000,  // 1ms = 0°
        max_pulse: 2000   // 2ms = 180°
    end
end

function servo_set_angle(servo: Servo, angle: kesir) do
    // Map angle (0-180°) to pulse width (1-2ms)
    variable pulse = servo.min_pulse.kesir() + 
                     (servo.max_pulse - servo.min_pulse).kesir() * 
                     (angle / 180.0)
    
    // Convert pulse width to duty cycle (20ms period)
    variable duty = (pulse / 20000.0) * 100.0
    
    timer.pwm_set_duty(servo.pwm, duty.sayı())
end

function main() do
    // Setup servo on PA0 (TIM2_CH1)
    variable servo = servo_init(timer.TIM2, timer.CH1)
    
    loop do
        // Sweep 0° to 180°
        her angle in 0..181 for do
            servo_set_angle(servo, angle.kesir())
            core.delay_ms(15)
        end
        
        // Sweep 180° to 0°
        her angle in (0..181).reverse() for do
            servo_set_angle(servo, angle.kesir())
            core.delay_ms(15)
        end
    end
end

💡 예시 3: DC 모터 속도 제어(L298N 드라이버)

import hal::timer, hal::gpio

struct DCMotor do
    pwm: timer.PWMHandle,
    dir1: gpio.Pin,
    dir2: gpio.Pin
end

function motor_init(tim: timer.Timer, channel: timer.Channel) -> DCMotor do
    // PWM setup (10 kHz for motor control)
    variable pwm = timer.pwm_init(tim, channel, do
        frequency: 10000,  // 10 kHz - reduces audible noise
        duty_cycle: 0
    end)
    
    // Direction pins
    gpio.clock_enable(gpio.PORT_B)
    variable dir1 = gpio.pin_init(gpio.PORT_B, 0, do mode: gpio.MODE_OUTPUT end)
    variable dir2 = gpio.pin_init(gpio.PORT_B, 1, do mode: gpio.MODE_OUTPUT end)
    
    timer.pwm_start(pwm)
    
    return DCMotor do
        pwm: pwm,
        dir1: dir1,
        dir2: dir2
    end
end

function motor_set_speed(motor: DCMotor, speed: i32) do
    // speed: -100 to +100 (negative = reverse)
    
    if speed > 0 then do
        // Forward
        gpio.pin_write(motor.dir1, gpio.HIGH)
        gpio.pin_write(motor.dir2, gpio.LOW)
        timer.pwm_set_duty(motor.pwm, speed.abs())
    end elseif if speed < 0 then do
        // Reverse
        gpio.pin_write(motor.dir1, gpio.LOW)
        gpio.pin_write(motor.dir2, gpio.HIGH)
        timer.pwm_set_duty(motor.pwm, speed.abs())
    end elseif do
        // Brake
        gpio.pin_write(motor.dir1, gpio.LOW)
        gpio.pin_write(motor.dir2, gpio.LOW)
        timer.pwm_set_duty(motor.pwm, 0)
    end
end

function main() do
    core.system_init()
    
    // Initialize motor on TIM3_CH1 (PA6)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 6, do mode: gpio.MODE_AF, af: 2 end)
    
    variable motor = motor_init(timer.TIM3, timer.CH1)
    
    loop do
        // Accelerate forward
        her speed in 0..101 for do
            motor_set_speed(motor, speed)
            core.delay_ms(20)
        end
        
        core.delay_ms(1000)
        
        // Brake
        motor_set_speed(motor, 0)
        core.delay_ms(500)
        
        // Accelerate reverse
        her speed in 0..101 for do
            motor_set_speed(motor, -speed)
            core.delay_ms(20)
        end
        
        core.delay_ms(1000)
        motor_set_speed(motor, 0)
        core.delay_ms(500)
    end
end

💡 예 4: 로터리 인코더 읽기(구적 모드)

import hal::timer, hal::gpio

function encoder_init() -> timer.Handle do
    // Encoder pins: PA0=A, PA1=B (TIM2_CH1, TIM2_CH2)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_AF,
        pull: gpio.PULL_UP,
        af: 1
    end)
    gpio.pin_init(gpio.PORT_A, 1, do
        mode: gpio.MODE_AF,
        pull: gpio.PULL_UP,
        af: 1
    end)
    
    // Configure timer in encoder mode
    variable encoder = timer.encoder_init(timer.TIM2, do
        mode: timer.ENCODER_MODE_TI12,  // Count on both edges
        polarity_ch1: timer.POLARITY_RISING,
        polarity_ch2: timer.POLARITY_RISING,
        filter: 5  // Noise filter
    end)
    
    // Set counter to mid-range (for bidirectional counting)
    timer.set_counter(encoder, 32768)
    
    timer.encoder_start(encoder)
    
    return encoder
end

function encoder_get_position(encoder: timer.Handle) -> i32 do
    variable count = timer.get_counter(encoder)
    // Convert to relative position (from mid-range)
    return (count as i32) - 32768
end

function encoder_get_speed(encoder: timer.Handle) -> i32 do
    // Get direction (up/down counting)
    if timer.is_counting_up(encoder) then do
        return 1
    end elseif do
        return -1
    end
end

function main() do
    core.system_init()
    variable encoder = encoder_init()
    
    variable son_pos = 0
    
    loop do
        variable pos = encoder_get_position(encoder)
        variable diff = pos - son_pos
        
        if diff != 0 then do
            variable direction = if diff > 0 then "CW" elseif "CCW"
            io.println("Position: {}  Delta: {}  Dir: {}".formatla(pos, diff, direction))
            son_pos = pos
        end
        
        core.delay_ms(50)
    end
end

💡 예시 5: 초음파 센서(HC-SR04)

import hal::timer, hal::gpio

variable capture_start: u32 = 0
variable capture_done: boolean = false
variable pulse_width: u32 = 0

function hcsr04_capture_callback(value: u32, edge: timer.Edge) do
    if edge == timer.EDGE_RISING then do
        capture_start = value
        capture_done = false
    end elseif do
        pulse_width = value - capture_start
        capture_done = true
    end
end

function hcsr04_init() -> (gpio.Pin, timer.ICHandle) do
    // Trigger pin (PB0)
    gpio.clock_enable(gpio.PORT_B)
    variable trigger = gpio.pin_init(gpio.PORT_B, 0, do
        mode: gpio.MODE_OUTPUT
    end)
    gpio.pin_write(trigger, gpio.LOW)
    
    // Echo pin (PA0 = TIM2_CH1)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_AF,
        af: 1
    end)
    
    // Input capture for echo (both edges)
    variable ic = timer.input_capture_init(timer.TIM2, timer.CH1, do
        edge: timer.EDGE_BOTH,  // Capture both rising and falling
        prescaler: 0,
        callback: hcsr04_capture_callback
    end)
    
    timer.input_capture_start(ic)
    
    return (trigger, ic)
end

function hcsr04_read_distance(trigger: gpio.Pin) -> İsteğe_Bağlı[kesir] do
    // Send 10us trigger pulse
    gpio.pin_write(trigger, gpio.HIGH)
    core.delay_us(10)
    gpio.pin_write(trigger, gpio.LOW)
    
    // Wait for measurement (max 38ms)
    capture_done = false
    her i in 0..380 for do
        if capture_done then break end
        core.delay_us(100)
    end
    
    if !capture_done then return None end
    
    // Calculate distance: time(pow) / 58 = distance(cm)
    // pulse_width in timer ticks, assuming 1 MHz timer (1us per tick)
    variable distance = pulse_width.kesir() / 58.0
    
    if distance > 2.0 and distance < 400.0 then do
        return Some(distance)
    end elseif do
        return None
    end
end

function main() do
    core.system_init()
    
    variable (trigger, _ic) = hcsr04_init()
    
    loop do
        eşle hcsr04_read_distance(trigger) do
            Some(dist) => io.println("Distance: {:.1f} cm".formatla(dist)),
            None => io.println("Out of range")
        end
        
        core.delay_ms(500)
    end
end

💡 예 6: WS2812 RGB LED 스트립(DMA + PWM)

import hal::timer, hal::dma, hal::gpio

constant LED_COUNT = 8
constant BIT_COUNT = LED_COUNT * 24  // 24 bits per LED (GRB)

// PWM values for WS2812: 0 = 0.4us, 1 = 0.8us (1.25MHz PWM)
constant PWM_0 = 17   // ~0.4us high
constant PWM_1 = 35   // ~0.8us high
constant PWM_PERIOD = 50  // 1.25us total

variable pwm_buffer: [u8; BIT_COUNT] = [0; BIT_COUNT]

function rgb_to_grb(r: u8, g: u8, b: u8) -> [u8; 24] do
    variable bits = [0u8; 24]
    
    // Green (bits 0-7)
    her i in 0..8 for do
        bits[i] = if (g & (1 << (7-i))) != 0 then PWM_1 elseif PWM_0
    end
    
    // Red (bits 8-15)
    her i in 0..8 for do
        bits[8+i] = if (r & (1 << (7-i))) != 0 then PWM_1 elseif PWM_0
    end
    
    // Blue (bits 16-23)
    her i in 0..8 for do
        bits[16+i] = if (b & (1 << (7-i))) != 0 then PWM_1 elseif PWM_0
    end
    
    return bits
end

function ws2812_set_pixel(index: sayı, r: u8, g: u8, b: u8) do
    variable bits = rgb_to_grb(r, g, b)
    variable offset = index * 24
    
    her i in 0..24 for do
        pwm_buffer[offset + i] = bits[i]
    end
end

function ws2812_show(pwm: timer.PWMHandle, dma: dma.Handle) do
    // Start DMA transfer
    dma.start_memory_to_periph(dma, do
        source: pwm_buffer.adres(),
        dest: timer.get_ccr_address(pwm),
        count: BIT_COUNT
    end)
    
    // Wait for transfer complete
    loop do
        if dma.is_complete(dma) then break end
    end
    
    // Reset time (>50us)
    core.delay_us(60)
end

function main() do
    core.system_init()
    
    // PWM setup on PA0 (TIM2_CH1)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_AF,
        af: 1,
        speed: gpio.SPEED_VERY_HIGH
    end)
    
    // 800kHz bit rate = 1.25us per bit
    variable pwm = timer.pwm_init(timer.TIM2, timer.CH1, do
        frequency: 800000,
        duty_cycle: 0
    end)
    
    // DMA for automatic PWM duty cycle updates
    variable dma_ch = dma.init(dma.DMA1_CH5, do
        direction: dma.MEMORY_TO_PERIPH,
        circular: false,
        memory_inc: true,
        periph_inc: false
    end)
    
    timer.pwm_start_dma(pwm, dma_ch)
    
    // Rainbow animation
    variable hue = 0
    
    loop do
        her i in 0..LED_COUNT for do
            // HSV to RGB conversion (simplified)
            variable h = (hue + i * 30) % 360
            variable r = if h < 120 then 255 elseif if h < 240 then 0 elseif 255
            variable g = if h < 60 or h >= 300 then 0 elseif 255
            variable b = if h < 180 then 0 elseif 255
            
            ws2812_set_pixel(i, r, g, b)
        end
        
        ws2812_show(pwm, dma_ch)
        
        hue = (hue + 10) % 360
        core.delay_ms(50)
    end
end

💡 예 7: 스테퍼 모터 제어(A4988 드라이버)

import hal::timer, hal::gpio

struct StepperMotor do
    pwm: timer.PWMHandle,  // Step pulses
    dir: gpio.Pin,
    enable: gpio.Pin,
    steps_per_rev: sayı
end

function stepper_init(tim: timer.Timer, channel: timer.Channel) -> StepperMotor do
    // Step pin (PWM)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_AF,
        af: 1
    end)
    
    // Direction and enable pins
    gpio.clock_enable(gpio.PORT_B)
    variable dir = gpio.pin_init(gpio.PORT_B, 0, do mode: gpio.MODE_OUTPUT end)
    variable enable = gpio.pin_init(gpio.PORT_B, 1, do mode: gpio.MODE_OUTPUT end)
    
    // Start with motor disabled
    gpio.pin_write(enable, gpio.HIGH)  // Active low
    
    // PWM for step pulses (default 1kHz)
    variable pwm = timer.pwm_init(tim, channel, do
        frequency: 1000,
        duty_cycle: 50  // 50% duty cycle for step pulses
    end)
    
    return StepperMotor do
        pwm: pwm,
        dir: dir,
        enable: enable,
        steps_per_rev: 200  // Typical NEMA17 (1.8° per step)
    end
end

function stepper_move(motor: StepperMotor, steps: i32, speed_hz: sayı) do
    // Set direction
    if steps >= 0 then do
        gpio.pin_write(motor.dir, gpio.HIGH)
    end elseif do
        gpio.pin_write(motor.dir, gpio.LOW)
    end
    
    variable abs_steps = steps.abs()
    
    // Set speed (frequency)
    timer.pwm_set_frequency(motor.pwm, speed_hz)
    
    // Enable motor
    gpio.pin_write(motor.enable, gpio.LOW)
    
    // Generate step pulses
    timer.pwm_start(motor.pwm)
    
    // Wait for movement to complete
    variable delay_ms = (abs_steps.kesir() / speed_hz.kesir()) * 1000.0
    core.delay_ms(delay_ms.sayı())
    
    // Stop
    timer.pwm_stop(motor.pwm)
    gpio.pin_write(motor.enable, gpio.HIGH)
end

function stepper_move_angle(motor: StepperMotor, degrees: kesir, speed_rpm: kesir) do
    // Convert angle to steps
    variable steps = ((degrees / 360.0) * motor.steps_per_rev.kesir()).round().sayı()
    
    // Convert RPM to Hz
    variable speed_hz = ((speed_rpm * motor.steps_per_rev.kesir()) / 60.0).sayı()
    
    stepper_move(motor, steps, speed_hz)
end

function main() do
    core.system_init()
    
    variable motor = stepper_init(timer.TIM2, timer.CH1)
    
    loop do
        // Rotate 360° clockwise at 60 RPM
        stepper_move_angle(motor, 360.0, 60.0)
        core.delay_ms(500)
        
        // Rotate 360° counter-clockwise at 120 RPM
        stepper_move_angle(motor, -360.0, 120.0)
        core.delay_ms(500)
    end
end

⚡ 성능 팁

  • PWM 주파수:10-20kHz용 모터 제어, 50Hz용 서보, 500-1000Hz용 LED
  • 해상도 대 주파수:고주파수 = 저해상도(PWM 단계)
  • DMA 사용법:DMA용 LED 스트립 및 오디오 사용(CPU 부하 90% 감소)
  • 인코더 모드:x4 모드는 가장 높은 감도를 제공합니다.
  • 입력 필터:잡음이 있는 신호에 대한 필터 활성화(5-15)
  • 데드타임:데드타임을 사용하여 엔진 드라이버의 슛스루 방지

📐 PWM 계산

// PWM Frequency calculation:
// f_PWM = f_Timer / ((Prescaler + 1) * (Period + 1))

// Example: 1 kHz PWM on 72 MHz timer
// Prescaler = 71 (72 MHz / 72 = 1 MHz)
// Period = 999 (1 MHz / 1000 = 1 kHz)
// Resolution = 1000 steps (0.1% duty cycle precision)

// Duty Cycle:
// Duty% = (CCR / Period) * 100
// CCR = (Duty% * Period) / 100

// Example: 75% duty @ 1kHz
// CCR = (75 * 999) / 100 = 749

⚠️ 중요 사항

  • 모든 타이머특정 GPIO 핀존재함(대체 기능 표 참조)
  • 공식 PWM 주파수 = Timer_Clock / (프리스케일러 * 기간)에 따라 계산됩니다.
  • 서보 모터는 일반적으로50Hz PWM필요(20ms 주기)
  • 다음에 대한 입력 캡처인터럽트 활성화되어야 한다
  • DMA를 사용하면 CPU 부하를 크게 줄일 수 있습니다.
  • 고급 타이머(TIM1, TIM8)중단 입력맹세죽은 시간기능을 제공합니다
  • 인코더 모드만 해당TIM1-TIM5, TIM8사용 가능
  • 32비트 타이머:TIM2, TIM5(STM32F4+)
  • 일반적으로 타이머 시계APB 시계 x2(RCC 구성 확인)
