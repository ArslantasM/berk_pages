📚 API 참조

매트릭스 생성

linalg_birim_mat(n: tam_sayı) -> [float]

n×n 크기의 단위 행렬을 생성합니다.

매개변수:

  • n: 매트릭스 크기

보고:단위 행렬(일반 배열 형식)

예:

let I3 = linalg_birim_mat(3)
// I3 = [1, 0, 0,
//       0, 1, 0,
//       0, 0, 1]

linalg_sıfır_mat(m: tam_sayı, n: tam_sayı) -> [float]

m×n 크기의 제로 행렬을 생성합니다.

linalg_birler_mat(m: tam_sayı, n: tam_sayı) -> [float]

모든 요소가 1인 m×n 크기의 행렬을 만듭니다.

linalg_diyagonal_mat(d: [float], n: tam_sayı) -> [float]

대각선 요소d배열인 n×n 행렬을 생성합니다.

예:

let d = [2.0, 3.0, 5.0]
let D = linalg_diyagonal_mat(d, 3)
// D = [2, 0, 0,
//      0, 3, 0,
//      0, 0, 5]

행렬 산술

linalg_mat_topla(A: [float], B: [float], m: tam_sayı, n: tam_sayı) -> [float]

두 개의 m×n 행렬을 추가합니다: C = A + B

linalg_mat_çıkar(A: [float], B: [float], m: tam_sayı, n: tam_sayı) -> [float]

두 개의 m×n 행렬을 뺍니다: C = A - B

linalg_mat_skaler(α: float, A: [float], m: tam_sayı, n: tam_sayı) -> [float]

행렬에 스칼라를 곱합니다. B = α · A

linalg_mat_çarp(A: [float], B: [float], m: tam_sayı, k: tam_sayı, n: tam_sayı) -> [float]

행렬 곱셈: C = A × B

  • A: m×k
  • B: k×n
  • C: m×n

예:

// 2×2 matris çarpımı
let A = [1.0, 2.0,
              3.0, 4.0]
let B = [5.0, 6.0,
              7.0, 8.0]
let C = linalg_mat_çarp(A, B, 2, 2, 2)
// C = [19, 22,
//      43, 50]

linalg_mat_hadamard(A: [float], B: [float], m: tam_sayı, n: tam_sayı) -> [float]

요소별 곱(아다마르 곱): C[i,j] = A[i,j] × B[i,j]

벡터 연산

linalg_iç_çarpım(u: [float], v: [float], n: tam_sayı) -> float

내적: u · v = Σ(u[i] × v[i])

예:

let u = [1.0, 2.0, 3.0]
let v = [4.0, 5.0, 6.0]
let dot = linalg_iç_çarpım(u, v, 3)  // 32.0

linalg_norm2(v: [float], n: tam_sayı) -> float

L2 노름(유클리드 노름): ||v|| = √(Σv[i]²)

linalg_norm1(v: [float], n: tam_sayı) -> float

L1 노름(맨해튼 노름): ||v||₁ = Σ|v[i]|

linalg_norm_inf(v: [float], n: tam_sayı) -> float

L결함수 노름(최대 노름): ||v||무한 = max(|v[i]|)

linalg_normalize(v: [float], n: tam_sayı) -> [float]

벡터를 정규화합니다: v̂ = v / ||v||

linalg_dış_çarpım(u: [float], v: [float], m: tam_sayı, n: tam_sayı) -> [float]

외부곱: A[i,j] = u[i] × v[j]

linalg_cross_3d(u: [ondalık], v: [ondalık]) -> [ondalık]

3D 벡터 외적: w = u × v

예:

let u = [1.0, 0.0, 0.0]
let v = [0.0, 1.0, 0.0]
let w = linalg_cross_3d(u, v)  // [0, 0, 1]

매트릭스 속성

linalg_iz(A: [float], n: tam_sayı) -> float

추적: tr(A) = Σ A[i,i]

linalg_frobenius_norm(A: [float], m: tam_sayı, n: tam_sayı) -> float

프로베니우스 노름: ||A||_F = √(Σ Σ A[i,j]²)

linalg_determinant_2x2(A: [ondalık]) -> ondalık

2×2 행렬식: det(A) = a₁₁a₂₂ - a₁₂a₂₁

linalg_determinant_3x3(A: [ondalık]) -> ondalık

3×3 행렬 행렬식(Sarrus 규칙)

역행렬

linalg_inverse_2x2(A: [ondalık]) -> [ondalık]

2×2 행렬 역행렬을 계산합니다: A⁻¹

예:

let A = [4.0, 7.0,
              2.0, 6.0]
let A_inv = linalg_inverse_2x2(A)
// det(A) = 10
// A_inv = [0.6, -0.7,
//         -0.2,  0.4]

linalg_inverse_3x3(A: [ondalık]) -> [ondalık]

3×3 역행렬을 계산합니다(보조 인자 확장).

선형 시스템

linalg_gauss_çöz(A: [float], b: [float], n: tam_sayı) -> [float]

가우스 제거를 통해 선형 시스템 Ax = b를 풉니다.

특징:

  • 부분 피버팅
  • 포워드 엘리미네이션
  • 역치환

예:

// 3x + 2y = 7
// 2x + 5y = 12
let A = [3.0, 2.0,
              2.0, 5.0]
let b = [7.0, 12.0]
let x = linalg_gauss_çöz(A, b, 2)
// x = [1, 2]

행렬 분해

linalg_lu_decompose(A: [float], n: tam_sayı) -> [float]

LU 분해: A = L × U

  • L: 하삼각행렬(대각선 1)
  • U: 상부삼각행렬

사용:선형 시스템을 신속하게 해결하려면

linalg_qr_gram_schmidt(A: [float], m: tam_sayı, n: tam_sayı) -> ([float], [float])

QR 분해: A = Q × R

  • Q: 직교행렬
  • R: 상부삼각행렬

연산:그람-슈미트 직교화

보고:(Q,R) 튜플

linalg_cholesky(A: [float], n: tam_sayı) -> [float]

촐레스키 분해: A = L × Lᵀ

필요:A는 양의 정부호 대칭 행렬이어야 합니다.

사용:공분산 행렬, 최적화

고유값/고유벡터

linalg_power_iteration(A: [float], n: tam_sayı, max_iter: tam_sayı, tolerans: float) -> ([float], float)

거듭제곱 반복을 통해 지배적인 고유벡터를 찾습니다.

보고:(고유벡터, 고유값) 튜플

예:

let A = [4.0, 1.0,
              1.0, 3.0]
let (v, λ) = linalg_power_iteration(A, 2, 100, 1e-6)
// λ ≈ 4.618 (dominant eigenvalue)
// v: karşılık gelen eigenvector

매트릭스 테스트

linalg_simetrik_mi(A: [float], n: tam_sayı, tolerans: float) -> bool

행렬이 대칭인지 확인: Aᵢⱼ = Aⱼᵢ

linalg_diagonal_mi(A: [float], n: tam_sayı, tolerans: float) -> bool

대각선 매트릭스 제어

linalg_üst_üçgen_mi(A: [float], n: tam_sayı, tolerans: float) -> bool

상부 삼각 행렬 제어

linalg_alt_üçgen_mi(A: [float], n: tam_sayı, tolerans: float) -> bool

하부 삼각 행렬 제어

매트릭스 순위

linalg_rank(A: [float], m: tam_sayı, n: tam_sayı, tolerans: float) -> tam_sayı

가우스 제거를 사용하여 행렬 순위를 계산합니다.

예:

let A = [1.0, 2.0, 3.0,
              2.0, 4.0, 6.0,
              3.0, 6.0, 9.0]
let r = linalg_rank(A, 3, 3, 1e-10)  // r = 1

예제 1: 선형 회귀

use linalg
use stats

// y = ax + b modelini fit et
function linear_fit(X: [float], y: [float], n: tam_sayı): (float, float) do
    // X matrix: [ones, x_values]
    let A = yeni_dizi(n * 2)
    let i = 0
    while (i < n) do
        A[i * 2] = 1.0
        A[i * 2 + 1] = X[i]
        i = i + 1
    end
    
    // Normal equations: (A^T A) β = A^T y
    let params = linalg_least_squares(A, y, n, 2)
    return (params[0], params[1])  // (b, a)
end

실시예 2: PCA(주성분 분석)

use linalg

// Veriyi merkeze receive
function center_data(data: [float], m: tam_sayı, n: tam_sayı): [float] do
    let centered = linalg_mat_kopyala(data, m, n)
    
    // Her sütun for ortalamayı pop
    let j = 0
    while (j < n) do
        let avg = 0.0
        let i = 0
        while (i < m) do
            avg = avg + linalg_mat_al(data, i, j, n)
            i = i + 1
        end
        avg = avg / m
        
        i = 0
        while (i < m) do
            let değer = linalg_mat_al(centered, i, j, n) - avg
            linalg_mat_yaz(centered, i, j, n, değer)
            i = i + 1
        end
        
        j = j + 1
    end
    
    return centered
end

// Kovaryans matrisini hesapla
function covariance_matrix(data: [float], m: tam_sayı, n: tam_sayı): [float] do
    let centered = center_data(data, m, n)
    let CT = linalg_transpoz(centered, m, n)
    let cov = linalg_mat_çarp(CT, centered, n, m, n)
    
    // (n-1) ile split
    let i = 0
    while (i < n * n) do
        cov[i] = cov[i] / (m - 1)
        i = i + 1
    end
    
    return cov
end

예제 3: 행렬 노름 및 조건수

use linalg

function analyze_matrix(A: [float], n: tam_sayı): hiçbir do
    // Frobenius norm
    let norm = linalg_frobenius_norm(A, n, n)
    print("||A||_F = " + norm)
    
    // Determinant
    if (n == 2) then do
        let det = linalg_determinant_2x2(A)
        print("det(A) = " + det)
        
        // Inverse and condition number
        if (math_mutlak(det) > 1e-10) then do
            let A_inv = linalg_inverse_2x2(A)
            let κ = linalg_condition_frobenius(A, A_inv, n)
            print("κ(A) = " + κ)
            
            if (κ > 100.0) then do
                print("UYARI: Matris kötü koşullu!")
            end
        end
    end
    
    // Simetri kontrolü
    if (linalg_simetrik_mi(A, n, 1e-10)) then do
        print("Matris simetriktir")
    end
end
