🧠 상태::코어

시스템 코어 - 클록, 재설정, 전원 관리

367라인 ~15가지 기능 RCC

📖 개요

코어 모듈은 시스템 초기화, 클럭 구성, 재설정 및 전원 관리 기능을 제공합니다.

🔑 주요 기능

  • 시스템 초기화
  • 클록 구성(최대 168MHz)
  • 지연 기능(ms/us)
  • 시스템 재설정
  • 전원 모드(절전/정지/대기)
  • 가동 시간 카운터

import hal::core

function main() do
    // System initialization
    core.system_init()
    
    // Configure system clock to 72MHz
    core.clock_config(72_000_000)
    
    // Delays
    core.delay_ms(1000)  // 1 second
    core.delay_us(500)   // 500 microseconds
    
    // Get uptime
    let uptime = core.uptime_ms()
    io.println("Uptime: {}ms", uptime)
    
    // System reset
    core.system_reset()
end

💡 예: 시스템 클럭 구성(STM32F4 168MHz)

import hal::core

function configure_168mhz() do
    // Configure PLL for 168MHz from 8MHz HSE
    core.pll_config(do
        source: core.PLL_HSE,        // External crystal
        pllm: 8,                     // VCO input: 8MHz / 8 = 1MHz
        plln: 336,                   // VCO output: 1MHz * 336 = 336MHz
        pllp: core.PLLP_DIV2,        // System clock: 336MHz / 2 = 168MHz
        pllq: 7                      // USB clock: 336MHz / 7 = 48MHz
    end)
    
    // Configure bus clocks
    core.bus_config(do
        ahb_prescaler: core.AHB_DIV1,    // AHB = 168MHz
        apb1_prescaler: core.APB_DIV4,   // APB1 = 42MHz (max 42MHz)
        apb2_prescaler: core.APB_DIV2    // APB2 = 84MHz (max 84MHz)
    end)
    
    // Flash latency for 168MHz @ 3.3V
    core.flash_latency(5)  // 5 wait states
    
    // Switch to PLL
    core.system_clock_source(core.SYSCLK_PLL)
    
    // Wait for PLL ready
    for core.get_clock_source() != core.SYSCLK_PLL else do
        // Wait
    end
    
    io.println("System clock: {}MHz", core.get_sysclk_freq() / 1_000_000)
end

function main() do
    configure_168mhz()
    
    // Verify clocks
    io.println("SYSCLK: {}MHz", core.get_sysclk_freq() / 1_000_000)
    io.println("AHB: {}MHz", core.get_hclk_freq() / 1_000_000)
    io.println("APB1: {}MHz", core.get_pclk1_freq() / 1_000_000)
    io.println("APB2: {}MHz", core.get_pclk2_freq() / 1_000_000)
end

💡 예: 저전력 절전 모드

import hal::core, hal::int, hal::gpio

let wakeup_flag = false

function button_interrupt() do
    wakeup_flag = true
end

function sleep_until_button() do
    // Configure button interrupt (PA0)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_INPUT,
        pull: gpio.PULL_DOWN,
        interrupt: gpio.INT_RISING
    end)
    
    gpio.set_callback(gpio.PORT_A, 0, button_interrupt)
    
    io.println("Going to sleep... (press button to wake)")
    core.delay_ms(100)  // Flush UART
    
    // Enter sleep mode
    core.sleep()
    
    // Wakeup on any interrupt
    io.println("Woke up from sleep!")
end

function main() do
    for do
        sleep_until_button()
        core.delay_ms(1000)
    end
end

💡 예: 워치독 타이머(IWDG)

import hal::core

function watchdog_test() do
    // Initialize Independent Watchdog (4 second timeout)
    core.iwdg_init(do
        prescaler: core.IWDG_PRESCALER_64,  // LSI/64
        reload: 2000  // (32kHz/64) * 2000 = 4 seconds
    end)
    
    io.println("Watchdog started (4s timeout)")
    
    // Normal operation: feed watchdog regularly
    her i in 0..10 for do
        io.println("Working... {}", i)
        core.delay_ms(500)
        
        core.iwdg_refresh()  // "Feed" watchdog
    end
    
    // Simulate hang (return't feed watchdog)
    io.println("Simulating hang...")
    for do
        core.delay_ms(1000)
        // Watchdog will reset system after 4s
    end
end

function main() do
    // Check if last reset was watchdog
    if core.get_reset_reason() == core.RESET_IWDG then do
        io.println("⚠️ System reset by watchdog!")
        core.clear_reset_flags()
    end
    
    watchdog_test()
end

💡 예: 브라운아웃 감지 및 재설정 처리

import hal::core

enum ResetReason {
    POWER_ON,
    RESET_PIN,
    IWDG,
    WWDG,
    SOFTWARE,
    BROWNOUT,
    UNKNOWN
}

function get_reset_reason_detailed() -> ResetReason do
    let flags = core.get_reset_flags()
    
    let reason = eşle flags do
        _ if (flags & core.FLAG_LPWR) != 0 => ResetReason::BROWNOUT,
        _ if (flags & core.FLAG_WWDG) != 0 => ResetReason::WWDG,
        _ if (flags & core.FLAG_IWDG) != 0 => ResetReason::IWDG,
        _ if (flags & core.FLAG_SFT) != 0 => ResetReason::SOFTWARE,
        _ if (flags & core.FLAG_PIN) != 0 => ResetReason::RESET_PIN,
        _ if (flags & core.FLAG_POR) != 0 => ResetReason::POWER_ON,
        _ => ResetReason::UNKNOWN
    end
    
    core.clear_reset_flags()
    return reason
end

function reset_handler() do
    let reason = get_reset_reason_detailed()
    
    io.println("\n=== System Boot ===")
    io.println("Reset reason: {}", reason)
    
    eşle reason do
        ResetReason::POWER_ON => do
            io.println("Fresh power-on, initializing...")
            // First boot initialization
        end,
        
        ResetReason::BROWNOUT => do
            io.println("⚠️ Brownout detected! Check power supply.")
            // Save diagnostics
        end,
        
        ResetReason::IWDG => do
            io.println("⚠️ Watchdog reset! System hang detected.")
            // Crash recovery
        end,
        
        ResetReason::SOFTWARE => do
            io.println("Software reset (firmware update?)")
        end,
        
        _ => do
            io.println("Unexpected reset")
        end
    end
end

function main() do
    reset_handler()
    
    // Normal program
    for do
        io.println("Running...")
        core.delay_ms(1000)
    end
end

💡 예: CPU 부하 모니터링

import hal::core

let idle_counter = 0
let total_counter = 0

function calculate_cpu_load() -> kesir do
    let idle = idle_counter
    let total = total_counter
    
    // Reset counters
    idle_counter = 0
    total_counter = 0
    
    let cpu_load = 100.0 - (idle.kesir() / total.kesir() * 100.0)
    return cpu_load
end

function idle_task() do
    // Count idle cycles
    idle_counter += 1
    total_counter += 1
end

function work_task() do
    // Simulate work
    let sum = 0
    her i in 0..1000 for do
        sum += i
    end
    total_counter += 1000
end

function main() do
    for do
        // Do some work
        work_task()
        
        // Idle time
        idle_task()
        
        // Report every second
        if core.uptime_ms() % 1000 == 0 then do
            let cpu_load = calculate_cpu_load()
            io.println("CPU load: {:.1f}%", cpu_load)
        end
    end
end

📊 전원 모드 비교

방법 힘 기상 시간 웨이크업 소스
달리다 ~30mA - -
잠 ~15mA 즉각적인 모든 중단
멈추다 ~10μA ~5μs EXTI, RTC, 워치독
대기 ~1μA ~50μs WKUP 핀, RTC 알람

⚡ 성능 팁

  • 플래시 지연 시간: 고주파수 대기 상태 추가(168MHz = 5WS)
  • PLL 구성: VCO를 192~432MHz 범위로 유지(최적)
  • APB1: 최대 42MHz(STM32F4), 타이머 클럭 x2
  • 절전 모드: 빠른 절전, 즉시 깨우기
  • 감시견: 시스템 안정성을 위해 사용(중지 복구)

⚠️ 중요 사항

  • 플래시 지연 시간: 고주파에 맞게 자동 조정됨(168MHz @ 3.3V = 5 WS)
  • PLL 잠금 시간: PLL 안정화를 위해 ~200μs를 기다립니다.
  • 대기 모드: 종료 시 시스템 재설정이 발생하며, 백업 도메인이 보존됩니다.
  • 워치독 시작: IWDG는 일단 시작되면 중지할 수 없습니다.
  • LSE 스타트업: 외부 32kHz 크리스털의 시동 시간 ~1초
  • HSE 우회: 외부 클럭 소스용 크리스탈 대신

🔗 관련 모듈

  • hal::rtc- RTC 클럭 구성, 웨이크업 타이머
  • hal::int- 인터럽트 우선순위, 슬립 모드 웨이크업
  • hal::gpio- WKUP 핀 구성
