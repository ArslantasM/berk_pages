📖 사용 방법?

1. 모듈 가져오기

// Single module
import math

// Multiple modules
import math, stats, linalg

// Using alias
import math as m
import http as h

2. 함수 호출

// Full name
let result = math.sqrt(16.0)

// With alias
import math as m
let result = m.sqrt(16.0)

3. 상수 사용

import math

print(math.PI)        // 3.14159...
print(math.E)         // 2.71828...
print(math.TAU)       // 6.28318... (2π)
