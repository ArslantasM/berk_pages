🚀 빠른 시작

import xml

// Parse XML
let xml_str = "Ali"
let doc = xml.parse(xml_str)

// Query
let name = xml.find_text(doc, "//user/name")
print(name)  // "Ali"

// Attribute
let id = xml.find_attr(doc, "//user", "id")
print(id)  // "1"

// Build XML
let builder = xml.builder_new()
xml.start_element(builder, "root")
xml.start_element(builder, "user")
xml.attr(builder, "id", "2")
xml.element_text(builder, "name", "Ayşe")
xml.end_element(builder)  // user
xml.end_element(builder)  // root

let yeni_xml = xml.builder_to_string(builder)
print(yeni_xml)

RSS 피드 파서

import xml, http

function rss_parse(url: yazı) -> Dizi do
    let yanıt = http.get(url)
    let doc = xml.parse(yanıt.gövde)
    
    let items = []
    let item_nodes = xml.find_all(doc, "//item")
    
    her item_node in item_nodes for do
        let item = do
            "title": xml.find_text(item_node, "title"),
            "link": xml.find_text(item_node, "link"),
            "description": xml.find_text(item_node, "description"),
            "pubDate": xml.find_text(item_node, "pubDate")
        end
        items.push(item)
    end
    
    return items
end

let haberler = rss_parse("https://example.com/rss")
her haber in haberler for do
    print(haber["title"] + " - " + haber["link"])
end
