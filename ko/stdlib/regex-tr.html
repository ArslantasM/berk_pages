🚀 빠른 시작

import regex

// Pattern compile
let re = regex.compile(r"\d+")

// Match
if regex.is_match(re, "42") then do
    print("Eşleşti!")
end

// Find
let match = regex.find(re, "Yaşım 25, senin 30")
print(match.text())  // "25"

// Find all
let tüm_eşleşmeler = regex.find_all(re, "10, 20, 30")
her m in tüm_eşleşmeler for do
    print(m.text())  // "10", "20", "30"
end

// Capture groups
let email_re = regex.compile(r"(\w+)@(\w+\.\w+)")
let cap = regex.captures(email_re, "ali@example.com")
print(cap.get(1))  // "ali"
print(cap.get(2))  // "example.com"

// Replace
let new = regex.replace(re, "Yaşım 25", "XX")
print(new)  // "Yaşım XX"

    let html = markdown

Header
)
    html = regex.replace_all(
        regex.compile(r"^### (.+)$", MULTILINE),
        html,
        "
$1
"
    )
    html = regex.replace_all(
        regex.compile(r"^## (.+)$", MULTILINE),
        html,
        "
$1
"
    )
    html = regex.replace_all(
        regex.compile(r"^# (.+)$", MULTILINE),
        html,
        "
$1
"
    )
    
    // Bold (**bold** -> bold)
    html = regex.replace_all(
        regex.compile(r"\*\*(.+?)\*\*"),
        html,
        "$1"
    )
    
    // Italic (*italic* -> italic)
    html = regex.replace_all(
        regex.compile(r"\*(.+?)\*"),
        html,
        "$1"
    )
    
    // Links ([text](url) -> text)
    html = regex.replace_all(
        regex.compile(r"\[([^\]]+)\]\(([^\)]+)\)"),
        html,
        "$1"
    )
    
    // Code (`code` -> code)
    html = regex.replace_all(
        regex.compile(r"`([^`]+)`"),
        html,
        "$1"
    )
    
    // Paragraphs
    html = regex.replace_all(
        regex.compile(r"\n\n"),
        html,
        "
"
    )
    html = "
" + html + "
"
    
    dön html
