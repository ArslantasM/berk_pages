🚀 빠른 시작

import json

// JSON parse
let veri = '{"name":"BERK","version":"1.0","active":true}'
let obj = json.parse(veri)
print(obj["name"])  // "BERK"

// JSON stringify
let kullanıcı = {"id": 1, "isim": "Ali", "yaş": 25}
let json_str = json.stringify(kullanıcı)
print(json_str)  // {"id":1,"isim":"Ali","yaş":25}

// Pretty print
let güzel = json.stringify_pretty(kullanıcı, 2)
print(güzel)
// {
//   "id": 1,
//   "isim": "Ali",
//   "yaş": 25
// }
