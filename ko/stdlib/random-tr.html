💡 예

몬테카를로 시뮬레이션(Pi 추정)

import random, math

function pi_tahmini(numune_sayısı: int) -> kesir do
    let daire_içi = 0
    
    her i in range(0, numune_sayısı) for do
        let x = random.kesir()
        let y = random.kesir()
        
        // Nokta birim daire in mi?
        if (x*x + y*y) < 1.0 then do
            daire_içi = daire_içi + 1
        end
    end
    
    // Pi ≈ 4 * (daire içi nokta sayısı / sum nokta sayısı)
    return 4.0 * daire_içi.kesir() / numune_sayısı.kesir()
end

let tahmin = pi_tahmini(1000000)
print("Pi tahmini: " + tahmin.yazıya())
print("Gerçek Pi: " + math.PI.yazıya())
print("Err: " + math.abs(tahmin - math.PI).yazıya())

import random

function weighted_choice(items: Dizi, weights: Dizi) -> herhangi do
    // Kümülatif ağırlıklar
    let total = 0.0
    let cumulative = []
    
    her w in weights for do
        total = total + w
        cumulative.push(total)
    end
    
    // Random seçim
    let r = random.kesir() * total
    
    her i in range(0, cumulative.len()) for do
        if r < cumulative[i] then do
            return items[i]
        end
    end
    
    return items[items.len() - 1]
end
