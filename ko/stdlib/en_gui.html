🖥️ gui - 그래픽 사용자 인터페이스

창, 버튼, 메뉴 및 기타 GUI 구성요소가 포함된 데스크탑 애플리케이션

📦 169개 기능 📝 1636라인 💾 53.6KB

📋개요

gui이 모듈은 최신 그래픽 사용자 인터페이스를 생성하기 위한 포괄적인 위젯 및 창 관리 도구를 제공합니다. 네이티브 모양의 크로스 플랫폼 데스크톱 애플리케이션을 개발합니다.

🪟 창 관리

  • GUI.창()- 메인 애플리케이션 창
  • gui.Dialog()- 대화 상자
  • gui.set_title()- 창 제목
  • gui.set_size()- 창 크기(너비, 높이)
  • gui.get_center()- 창을 화면 중앙으로 이동

🎛️ 기본 위젯

  • GUI.버튼()- 클릭 가능한 버튼
  • gui.라벨()- 텍스트 라벨
  • gui.소개상자()- 한 줄 텍스트 소개
  • gui.텍스트필드()- 여러 줄의 텍스트 영역
  • gui.체크박스()- 체크박스
  • gui.라디오버튼()-라디오 버튼

📋 목록 및 선택

  • GUI.목록()- 목록보기

💡 사용 예

간단한 계산기

use gui

class HesapMakinesi {
    değer pencere: gui.Pencere
    değer sonuç_etiketi: gui.Etiket
    değer işlem: Dize = ""
    değer ilk_sayı: Gerçek = 0.0
    değer operatör: Dize = ""
    
    function new() -> HesapMakinesi {
        değer pencere = gui.Pencere()
        pencere.başlık_ayarla("Hesap Makinesi")
        pencere.boyut_ayarla(300, 400)
        pencere.merkeze_al()
        
        değer sonuç_etiketi = gui.Etiket("0")
        sonuç_etiketi.yazı_boyutu_ayarla(24)
        sonuç_etiketi.hizalama_ayarla(gui.Hizalama.SAĞ)
        
        return HesapMakinesi {
            pencere: pencere,
            sonuç_etiketi: sonuç_etiketi,
            işlem: "",
            ilk_sayı: 0.0,
            operatör: ""
        }
    }
    
    function rakam_tıkla(rakam: Dize) {
        bu.işlem += rakam
        bu.sonuç_etiketi.metin_ayarla(bu.işlem)
    }
    
    function operatör_tıkla(op: Dize) {
        if bu.işlem != "" {
            bu.ilk_sayı = bu.işlem.gerçeğe()
            bu.operatör = op
            bu.işlem = ""
        }
    }
    
    function eşittir_tıkla() {
        if bu.işlem != "" and bu.operatör != "" {
            değer ikinci_sayı = bu.işlem.gerçeğe()
            değer sonuç = 0.0
            
            eşleştir bu.operatör {
                "+" => { sonuç = bu.ilk_sayı + ikinci_sayı }
                "-" => { sonuç = bu.ilk_sayı - ikinci_sayı }
                "*" => { sonuç = bu.ilk_sayı * ikinci_sayı }
                "/" => { sonuç = bu.ilk_sayı / ikinci_sayı }
            }
            
            bu.işlem = sonuç.dizgeye()
            bu.sonuç_etiketi.metin_ayarla(bu.işlem)
            bu.operatör = ""
        }
    }
    
    function temizle() {
        bu.işlem = ""
        bu.ilk_sayı = 0.0
        bu.operatör = ""
        bu.sonuç_etiketi.metin_ayarla("0")
    }
    
    function arayüz_oluştur() {
        değer düzen = gui.DikeySandık()
        düzen.push(bu.sonuç_etiketi)
        
        // Rakam butonları
        değer rakamlar = [
            ["7", "8", "9", "/"],
            ["4", "5", "6", "*"],
            ["1", "2", "3", "-"],
            ["C", "0", "=", "+"]
        ]
        
        for satır in rakamlar {
            değer yatay = gui.YataySandık()
            
            for değer in satır {
                değer buton = gui.Buton(değer)
                buton.min_boyut_ayarla(60, 50)
                
                if değer == "C" {
                    buton.tıklandı_bağla(|| bu.temizle())
                } elseif if değer == "=" {
                    buton.tıklandı_bağla(|| bu.eşittir_tıkla())
                } elseif if değer in ["+", "-", "*", "/"] {
                    buton.tıklandı_bağla(|| bu.operatör_tıkla(değer))
                } elseif {
                    buton.tıklandı_bağla(|| bu.rakam_tıkla(değer))
                }
                
                yatay.push(buton)
            }
            
            düzen.push(yatay)
        }
        
        bu.pencere.düzen_ayarla(düzen)
    }
    
    function göster() {
        bu.arayüz_oluştur()
        bu.pencere.göster()
    }
}

// Uygulamayı başlat
değer uygulama = gui.Uygulama()
değer hesap_makinesi = HesapMakinesi.new()
hesap_makinesi.göster()
uygulama.çalıştır()

텍스트 편집기

use gui
use fs

class MetinEditörü {
    değer pencere: gui.Pencere
    değer metin_alanı: gui.MetinAlanı
    değer geçerli_dosya: Opsiyonel<Dize> = null
    
    function new() -> MetinEditörü {
        değer pencere = gui.Pencere()
        pencere.başlık_ayarla("Metin Editörü")
        pencere.boyut_ayarla(800, 600)
        pencere.merkeze_al()
        
        değer metin_alanı = gui.MetinAlanı()
        metin_alanı.yazı_tipi_ayarla("Courier New", 12)
        
        return MetinEditörü {
            pencere: pencere,
            metin_alanı: metin_alanı,
            geçerli_dosya: null
        }
    }
    
    function yeni_dosya() {
        if bu.değişiklik_var() {
            değer cevap = gui.onay_kutusu(
                "Kaydetmeden önce",
                "Değişiklikler kaydedilsin mi?",
                gui.Butonlar.EVET_HAYIR_İPTAL
            )
            
            if cevap == gui.Yanıt.EVET {
                bu.kaydet()
            } elseif if cevap == gui.Yanıt.İPTAL {
                return
            }
        }
        
        bu.metin_alanı.temizle()
        bu.geçerli_dosya = null
        bu.pencere.başlık_ayarla("Metin Editörü - Yeni Dosya")
    }
    
    function aç() {
        değer dosya_yolu = gui.dosya_aç_dialog(
            başlık: "Dosya Aç",
            filtreler: [("Metin Dosyaları", "*.txt"), ("Tüm Dosyalar", "*.*")]
        )
        
        if dosya_yolu.var_mı() {
            değer Content = fs.dosya_oku_metin(dosya_yolu.aç())
            bu.metin_alanı.metin_ayarla(Content)
            bu.geçerli_dosya = dosya_yolu
            bu.pencere.başlık_ayarla(f"Metin Editörü - {dosya_yolu.aç()}")
        }
    }
    
    function kaydet() {
        if bu.geçerli_dosya.var_mı() {
            bu.dosyaya_kaydet(bu.geçerli_dosya.aç())
        } elseif {
            bu.farklı_kaydet()
        }
    }
    
    function farklı_kaydet() {
        değer dosya_yolu = gui.dosya_kaydet_dialog(
            başlık: "Farklı Kaydet",
            varsayılan_ad: "yeni_dosya.txt",
            filtreler: [("Metin Dosyaları", "*.txt")]
        )
        
        if dosya_yolu.var_mı() {
            bu.dosyaya_kaydet(dosya_yolu.aç())
            bu.geçerli_dosya = dosya_yolu
        }
    }
    
    function dosyaya_kaydet(dosya_yolu: Dize) {
        değer Content = bu.metin_alanı.metin_al()
        fs.dosya_yaz_metin(dosya_yolu, Content)
        gui.mesaj_kutusu("Başarılı", "Dosya kaydedildi!", gui.MesajTipi.BİLGİ)
        bu.pencere.başlık_ayarla(f"Metin Editörü - {dosya_yolu}")
    }
    
    function değişiklik_var() -> boolean {
        // Basitleştirilmiş - gerçek uygulamada değişiklik takibi yapılır
        return bu.metin_alanı.metin_al().len > 0
    }
    
    function menü_oluştur() {
        değer menü_çubuğu = gui.MenuÇubuğu()
        
        // Dosya menüsü
        değer dosya_menü = gui.Menü("Dosya")
        dosya_menü.öğe_ekle("Yeni", "Ctrl+N", || bu.yeni_dosya())
        dosya_menü.öğe_ekle("Aç...", "Ctrl+O", || bu.aç())
        dosya_menü.öğe_ekle("Kaydet", "Ctrl+S", || bu.kaydet())
        dosya_menü.öğe_ekle("Farklı Kaydet...", "Ctrl+Shift+S", || bu.farklı_kaydet())
        dosya_menü.ayırıcı_ekle()
        dosya_menü.öğe_ekle("Çıkış", "Ctrl+Q", || bu.pencere.kapat())
        
        menü_çubuğu.menü_ekle(dosya_menü)
        bu.pencere.menü_çubuğu_ayarla(menü_çubuğu)
    }
    
    function göster() {
        bu.menü_oluştur()
        
        değer düzen = gui.DikeySandık()
        düzen.push(bu.metin_alanı)
        bu.pencere.düzen_ayarla(düzen)
        
        bu.pencere.göster()
    }
}

// Uygulamayı başlat
değer uygulama = gui.Uygulama()
değer editör = MetinEditörü.new()
editör.göster()
uygulama.çalıştır()
