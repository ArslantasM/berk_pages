import random

// Seed ayarla (reproducible results)
random.seed(42)

// Temel RNG
variable x = random.rand()  // [0, 1) arası
variable y = random.randint(1, 100)  // [1, 100] arası integer
variable z = random.uniform(10.0, 20.0)  // [10, 20) arası

// Dağılımlar
variable normal = random.normal(0.0, 1.0)  // μ=0, σ=1
variable exp = random.exponential(1.5)  // λ=1.5
variable gauss = random.gaussian(100.0, 15.0)  // IQ dağılımı

// Diziden seçim
variable liste = ["elma", "armut", "kiraz", "üzüm"]
variable seçim = random.choice(liste)
variable Example = random.sample(liste, 2)  // 2 elemanlı Example

import random

function estimate_pi(n: integer) -> float do
    variable içerde = 0
    
    her i in range(0, n) for do
        variable x = random.uniform(-1.0, 1.0)
        variable y = random.uniform(-1.0, 1.0)
        
        // Çemberin in mi?
        if (x*x + y*y) <= 1.0 then do
            içerde = içerde + 1
        end
    end
    
    // π ≈ 4 * (içerdekiler / sum)
    return 4.0 * içerde.kesir() / n.kesir()
end

random.seed(42)
variable pi_tahmin = estimate_pi(1000000)
print("π tahmini: " + pi_tahmin.yazıya())  // ~3.141592
