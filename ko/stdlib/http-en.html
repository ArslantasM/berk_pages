예 1: REST API 클라이언트

import http, json

function kullanıcı_al(id: int) -> Sonuç[yazı, yazı] do
    let url = "https://api.example.com/users/" + id.yazıya()
    let yanıt = http.get(url)
    
    if yanıt.case.kod == 200 then do
        return Ok(yanıt.gövde)
    end else do
        return Err("Kullanıcı bulunamadı")
    end
end

let sonuç = kullanıcı_al(123)
eşleşme sonuç do
    Ok[veri] => print("Kullanıcı: " + veri)
    Err[mesaj] => print("Err: " + mesaj)
end

예제 2: RESTful API 서버

import http, json

let kullanıcılar = {
    "1": {"id": 1, "isim": "Ali", "yaş": 25},
    "2": {"id": 2, "isim": "Ayşe", "yaş": 30}
}

let sunucu = http.sunucu_oluştur("0.0.0.0", 3000)

// All users
sunucu.rotala("GET", "/api/users", function(istek, yanıt) do
    yanıt.json(kullanıcılar)
end)

// Single user
sunucu.rotala("GET", "/api/users/:id", function(istek, yanıt) do
    let id = istek.params["id"]
    
    if kullanıcılar.contains(id) then do
        yanıt.json(kullanıcılar[id])
    end else do
        yanıt.case(404).json({"hata": "Kullanıcı bulunamadı"})
    end
end)

// Add new user
sunucu.rotala("POST", "/api/users", function(istek, yanıt) do
    let veri = json.parse(istek.gövde)
    let id = (kullanıcılar.len() + 1).yazıya()
    
    kullanıcılar[id] = veri
    yanıt.case(201).json(veri)
end)

// Update user
sunucu.rotala("PUT", "/api/users/:id", function(istek, yanıt) do
    let id = istek.params["id"]
    let veri = json.parse(istek.gövde)
    
    if kullanıcılar.contains(id) then do
        kullanıcılar[id] = veri
        yanıt.json(veri)
    end else do
        yanıt.case(404).json({"hata": "Kullanıcı bulunamadı"})
    end
end)

// Delete user
sunucu.rotala("DELETE", "/api/users/:id", function(istek, yanıt) do
    let id = istek.params["id"]
    
    if kullanıcılar.contains(id) then do
        kullanıcılar.sil(id)
        yanıt.case(204).gönder()
    end else do
        yanıt.case(404).json({"hata": "Kullanıcı bulunamadı"})
    end
end)

print("RESTful API Server http://localhost:3000 çalışıyor...")
sunucu.başlat()

예시 3: 미들웨어 및 CORS

import http

let sunucu = http.sunucu_oluştur("0.0.0.0", 8080)

// CORS middleware
sunucu.middleware(function(istek, yanıt, ileri) do
    yanıt.başlık_ekle("Access-Control-Allow-Origin", "*")
    yanıt.başlık_ekle("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE")
    yanıt.başlık_ekle("Access-Control-Allow-Headers", "Content-Type")
    
    if istek.metod == "OPTIONS" then do
        yanıt.case(204).gönder()
    end else do
        ileri()
    end
end)

// Logging middleware
sunucu.middleware(function(istek, yanıt, ileri) do
    let başlangıç = time.now()
    
    ileri()
    
    let süre = time.now() - başlangıç
    print(istek.metod + " " + istek.yol + " - " + süre.yazıya() + "ms")
end)

// Routes
sunucu.rotala("GET", "/api/data", function(istek, yanıt) do
    yanıt.json({"data": "BERK HTTP Server"})
end)

sunucu.başlat()
