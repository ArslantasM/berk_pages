import optim

// f(x) = x² - 4 = 0 denkleminin kökünü bul
function f(x: float): float do
    return x * x - 4.0
end

// Bisection metodu
let kök = optim_bisection(f, 0.0, 3.0, 0.001, 100)
print("Kök: " + kök)  // ≈ 2.0

// Newton-Raphson (türev gerekli)
function df(x: float): float do
    return 2.0 * x
end

let kök2 = optim_newton_raphson(f, df, 1.0, 0.001, 100)

단일 변수 최소값

function parabola(x: float): float do

📚 API 카테고리

1. 근 찾기

  • optim_bisection- 분할 방법
  • optim_newton_raphson- 뉴턴-랩슨 방법
  • optim_secant- 시컨트 방식
  • optim_fixed_point- 고정점 반복

2. 라인 검색(단일 변수)

  • optim_golden_section- 황금비율 검색
  • optim_brent- 브렌트 방식
  • optim_armijo_line_search- 아미조의 상태

3. 기울기 기반 최적화

  • optim_gradient_descent- 기본 경사하강법
  • optim_gradient_descent_momentum- 기세 넘치는 GD
  • optim_adam- 아담 옵티마이저
  • optim_bfgs_simple- BFGS 준뉴턴

4. 그라데이션 없는 최적화

  • optim_nelder_mead_2d– Nelder-Mead 심플렉스
  • optim_coordinate_descent- 좌표 하강

5.제한된 최적화

  • optim_penalty_method- 페널티 방식
  • optim_augmented_lagrangian-증강 라그랑지안
  • optim_gradient_projection- 그라데이션 투영

6. 최소제곱

  • optim_least_squares- 선형 최소제곱
  • optim_gauss_newton- 가우스-뉴턴 방법
  • optim_levenberg_marquardt- LM 알고리즘

7. 메타휴리스틱

  • optim_simulated_annealing- 모의 어닐링
  • optim_genetic_algorithm_1d- 유전 알고리즘

8. 유틸리티

  • optim_numerical_gradient- 수치적 그라데이션
  • optim_numerical_hessian- 수치 헤세 행렬

예 2: 엔지니어링 - 구조 최적화

// Kiriş optimizasyonu: minimum ağırlık, maksimum dayanım
function beam_cost(dimensions: [float], n: tam_sayı): float do
    let width = dimensions[0]
    let height = dimensions[1]
    
    // Maliyet = malzeme × domain
    let cost = width * height
    
    // Kısıt: moment of inertia > I_min
    let I = (width * height * height * height) / 12.0
    let penalty = 0.0
    if (I < 1000.0) then do
        penalty = 1000.0 * (1000.0 - I)
    end
    
    return cost + penalty
end

let initial = [10.0, 20.0]
let optimal = optim_simulated_annealing(beam_cost, initial, 2, 100.0, 0.95, 1000, 42)

function exponential_model(params: [float], x: float): float do
    let a = params[0]
    let b = params[1]
    let c = params[2]
    return a * math_exp(-b * x) + c
end

function residual(params: [float], X: [float], y: [float], n: tam_sayı): [float] do
    let r = yeni_dizi(n)
    let i = 0
    while (i < n) do
        r[i] = y[i] - exponential_model(params, X[i])
        i = i + 1
    end
    return r

🎓 최적화 전략

문제 유형에 따른 알고리즘 선택

  1. 매끄럽고 볼록한 문제:

    • 그라데이션이 있나요? → BFGS 또는 경사하강법
    • 헤시안이 있나요? → 뉴턴법

  2. 매끄럽지 않고 볼록하지 않음:

    • 모의 어닐링
    • 유전 알고리즘

  3. 제한된 문제:

    • 페널티 방법
    • 증강 라그랑지안

  4. 대규모 문제:

    • 확률적 경사하강법
    • 좌표 하강

  5. 최소제곱:

    • 가우스-뉴턴
    • Levenberg-Marquardt
