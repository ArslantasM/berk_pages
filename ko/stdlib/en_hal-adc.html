📊 상태::adc

아날로그-디지털 변환기 - 아날로그 신호 변환

400줄 ~25개 기능 12비트 ADC

📖 개요

ADC(Analog-to-Digital Converter) 모듈은 아날로그 신호를 디지털 값으로 변환합니다. 센서 판독, 온도 측정, 배터리 전압 제어와 같은 애플리케이션에 사용됩니다.

🔑 주요 기능

  • 6-16비트 분해능(장치에 따라 다름)
  • 다중 ADC 채널(최대 19개)
  • 단일/연속 변환
  • DMA 지원
  • 내부 온도 센서
  • 내부 전압 기준
  • 아날로그 워치독
  • 구성 가능한 샘플링 시간

💡 예 1: 온도 센서

import hal::adc, hal::gpio, hal::core

function sıcaklık_oku() -> kesir do
    // ADC setup
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    // GPIO for analog input (PA0)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    // Configure channel
    adc.channel_config(adc1, adc.CHANNEL_0, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Read multiple samples and average
    variable sum = 0.0
    variable samples = 10
    
    her i in 0..samples for do
        adc.start(adc1)
        variable raw = adc.read(adc1)
        sum += raw.kesir()
        core.delay_ms(10)
    end
    
    variable avg = sum / samples.kesir()
    
    // Convert to voltage (3.3V reference, 12-bit)
    variable voltage = avg * 3.3 / 4095.0
    
    // LM35 sensor: 10mV/°C
    variable temp = voltage * 100.0
    
    return temp
end

function main() do
    core.system_init()
    core.clock_config(72_000_000)
    
    loop do
        variable temp = sıcaklık_oku()
        io.println("Sıcaklık: {:.1f}°C".formatla(temp))
        core.delay_ms(1000)
    end
end

💡 예 2: 배터리 전압 모니터

import hal::adc, hal::gpio

function batarya_voltaj_oku() -> kesir do
    // Voltage divider: Battery -> 10K -> ADC -> 10K -> GND
    // Max input: 6.6V (for 3.3V ADC)
    
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 1, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    adc.channel_config(adc1, adc.CHANNEL_1, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Read 100 samples for accuracy
    variable sum = 0
    her i in 0..100 for do
        adc.start(adc1)
        sum += adc.read(adc1)
    end
    
    variable avg = sum / 100
    variable voltage = (avg.kesir() / 4095.0) * 3.3
    
    // Apply voltage divider ratio (2:1)
    variable battery_voltage = voltage * 2.0
    
    return battery_voltage
end

function batarya_durumu(voltage: kesir) -> string do
    if voltage > 4.1 then return "Tam"
    elseif if voltage > 3.7 then return "İyi"
    elseif if voltage > 3.5 then return "Orta"
    elseif return "Düşük"
end

function main() do
    core.system_init()
    
    loop do
        variable v = batarya_voltaj_oku()
        variable case = batarya_durumu(v)
        io.println("Batarya: {:.2f}V [{}]".formatla(v, case))
        core.delay_ms(5000)
    end
end

💡 예 3: DMA를 갖춘 다중 채널 ADC

import hal::adc, hal::dma, hal::gpio

constant CHANNELS = 4

function adc_dma_init() -> adc.Handle do
    // GPIO setup for 4 analog inputs (PA0-PA3)
    gpio.clock_enable(gpio.PORT_A)
    her i in 0..CHANNELS for do
        gpio.pin_init(gpio.PORT_A, i, do
            mode: gpio.MODE_ANALOG,
            pull: gpio.PULL_NONE
        end)
    end
    
    // ADC setup
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT,
        scan_mode: true,
        continuous: true
    end)
    
    // Configure channels
    adc.channel_config(adc1, adc.CHANNEL_0, do rank: 1 end)
    adc.channel_config(adc1, adc.CHANNEL_1, do rank: 2 end)
    adc.channel_config(adc1, adc.CHANNEL_2, do rank: 3 end)
    adc.channel_config(adc1, adc.CHANNEL_3, do rank: 4 end)
    
    return adc1
end

variable adc_buffer: [sayı; CHANNELS] = [0, 0, 0, 0]

function main() do
    core.system_init()
    variable adc1 = adc_dma_init()
    
    // DMA setup for continuous ADC transfer
    dma.clock_enable()
    variable dma_ch = dma.init(dma.DMA1_CH1, do
        direction: dma.PERIPH_TO_MEMORY,
        periph_addr: adc.data_register(adc1),
        memory_addr: adc_buffer.adres(),
        buffer_size: CHANNELS,
        periph_inc: false,
        memory_inc: true,
        periph_size: dma.SIZE_16BIT,
        memory_size: dma.SIZE_16BIT,
        circular: true,
        priority: dma.PRIORITY_HIGH
    end)
    
    // Start DMA and ADC
    dma.start(dma_ch)
    adc.start_dma(adc1)
    
    // Main loop - data is automatically updated by DMA
    loop do
        io.println("Kanal 0: {}  Kanal 1: {}  Kanal 2: {}  Kanal 3: {}".formatla(
            adc_buffer[0], adc_buffer[1], adc_buffer[2], adc_buffer[3]
        ))
        core.delay_ms(500)
    end
end

💡 예 4: 아날로그 워치독(전압 모니터)

import hal::adc, hal::gpio, hal::int

constant THRESHOLD_LOW = 1500   // ~1.2V
constant THRESHOLD_HIGH = 3000  // ~2.4V

variable alarm_triggered = false

function adc_watchdog_handler() do
    alarm_triggered = true
    io.println("⚠️ Warning: Voltaj threshold dışında!")
end

function ana() do
    core.system_init()
    
    // GPIO setup
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    // ADC setup
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    adc.channel_config(adc1, adc.CHANNEL_0, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Configure analog watchdog
    adc.watchdog_enable(adc1, do
        channel: adc.CHANNEL_0,
        low_threshold: THRESHOLD_LOW,
        high_threshold: THRESHOLD_HIGH
    end)
    
    // Enable interrupt
    adc.interrupt_enable(adc1, adc.WATCHDOG_INT)
    int.register_handler(int.ADC1_IRQ, adc_watchdog_handler)
    int.enable_irq(int.ADC1_IRQ)
    
    // Start continuous conversion
    adc.start_continuous(adc1)
    
    loop do
        if alarm_triggered ise do
            alarm_triggered = false
            variable value = adc.read(adc1)
            variable voltage = (value.kesir() / 4095.0) * 3.3
            io.println("Voltaj: {:.2f}V [ALARM]".formatla(voltage))
        end
        core.delay_ms(100)
    end
end

💡 예 5: 내부 온도 센서

import hal::adc

function cpu_sicaklik_oku() -> kesir do
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    // Enable internal temperature sensor
    adc.enable_temp_sensor()
    
    // Configure internal channel (usually CH16)
    adc.channel_config(adc1, adc.CHANNEL_TEMP, do
        sampling_time: adc.SAMPLETIME_480CYCLES  // Long sampling for accuracy
    end)
    
    // Read multiple samples
    variable sum = 0
    her i in 0..10 for do
        adc.start(adc1)
        sum += adc.read(adc1)
        core.delay_ms(5)
    end
    
    variable avg = sum / 10
    
    // STM32 formula: Temp = (V25 - Vsense) / Avg_Slope + 25
    // Typical: V25 = 1.43V (@ 25°C), Avg_Slope = 4.3mV/°C
    variable voltage = (avg.kesir() / 4095.0) * 3.3
    variable temp = ((1.43 - voltage) / 0.0043) + 25.0
    
    return temp
end

function main() do
    core.system_init()
    
    loop do
        variable temp = cpu_sicaklik_oku()
        io.println("CPU Sıcaklığı: {:.1f}°C".formatla(temp))
        
        // Thermal management
        if temp > 80.0 then do
            io.println("⚠️ Warning: Yüksek sıcaklık!")
            // Reduce clock speed or take action
        end
        
        core.delay_ms(2000)
    end
end

enum Resolution {
    RESOLUTION_6BIT,   // 6-bit (fastest)
    RESOLUTION_8BIT,   // 8-bit
    RESOLUTION_10BIT,  // 10-bit
    RESOLUTION_12BIT   // 12-bit (highest accuracy)

// Data Alignment
enum Alignment {
    ALIGN_RIGHT,  // Right-aligned (default)

}

// Sampling Time
enum SamplingTime {
    SAMPLETIME_3CYCLES,
    SAMPLETIME_15CYCLES,
    SAMPLETIME_28CYCLES,
    SAMPLETIME_56CYCLES,
    SAMPLETIME_84CYCLES,
    SAMPLETIME_112CYCLES,
    SAMPLETIME_144CYCLES,
    SAMPLETIME_480CYCLES  // Longest (most accurate)
}

// ADC Channels
constant CHANNEL_0 = 0
constant CHANNEL_1 = 1
// ... up to CHANNEL_18

    scan_mode: boolean,      // Multi-channel scan

⚡ 성능 팁

  • 샘플링 시간:샘플링 시간이 길수록 감도는 높아지지만 속도는 느려집니다.
  • 구경 측정:시작할 때마다 Calibrate()를 실행합니다.
  • 평균화:노이즈를 줄이기 위한 평균 10-100개 샘플
  • DMA 사용법:다중 채널 읽기에서 CPU 부하 감소
  • 연속 모드:고속 예제 사용
  • 입력 임피던스:고임피던스 소스의 샘플링 시간 늘리기

⚠️ 중요 사항

  • ADC를 사용하기 전에clock_enable()부르다
  • GPIO용 아날로그 핀은 MODE_ANALOG를 사용합니다.
  • 최대 입력 전압: VREF+(보통 3.3V)
  • 입력 임피던스: <50kΩ 권장
  • 변환 시간 = 샘플링 시간 + 12사이클(12비트)
  • 내부 온도 센서 ±2°C 감도
  • DMA 사용 시 순환 버퍼 사용
  • Watchdog은 하나의 채널만 시청할 수 있습니다.
