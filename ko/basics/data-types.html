BERK Programlama Dili

데이터 유형

BERK는 정적으로 유형이 지정된 언어이며 다양한 내장 데이터 유형을 제공합니다.

기본 유형

숫자

정수 및 십진수의 경우:

let tam = 42
let float = 3.14
let negatif = -100
let bilimsel = 1.5e10  // 15000000000

텍스트(문자열)

텍스트 값:

let isim = "Ahmet"
let mesaj = 'Merhaba Dünya!'
let çok_satır = """
    Bu
    çok satırlı
    bir metindir
"""

문자열 보간:

let ad = "Ali"
let yaş = 25
print("Benim adım {ad} and {yaş} yaşındayım")
// Çıktı: Benim adım Ali and 25 yaşındayım

부울(논리)

참/거짓 값:

let doğru_mu = true  // true
let yanlış_mı = false  // false

// İngilizce
var is_valid = true
var is_empty = false

null

가치 부족:

let değer = null
let sonuç: number? = null  // Nullable type

컬렉션 유형

목록 (목록)

순서가 지정된 요소 모음:

let sayılar = [1, 2, 3, 4, 5]
let isimler: Liste<string> = ["Ahmet", "Mehmet", "Ayşe"]
let karışık = [1, "iki", 3.0, true]  // Mixed types

// Erişim
print(sayılar[0])   // 1
print(isimler[2])   // Ayşe

// Değiştirme
sayılar[0] = 10

// Uzunluk
print(sayılar.len)  // 5

목록 방법:

let List = [1, 2, 3]

List.push(4)           // [1, 2, 3, 4]
List.pop(1)          // [1, 3, 4]
List.contains(3)         // true
List.temizle()         // []

튜플

고정된 크기, 다양한 유형의 요소:

let index_of = (41.0082, 28.9784)  // (lat, lon)
let kişi = ("Ahmet", 30, "İstanbul")

// Destructuring
let (lat, lon) = index_of
let (ad, yaş, şehir) = kişi

// İndeks erişimi
print(index_of.0)  // 41.0082
print(kişi.1)   // 30

지도(지도/사전)

키-값 쌍:

let sözlük = {
    "ad": "Ahmet",
    "yaş": 30,
    "şehir": "İstanbul"
}

// Erişim
print(sözlük["ad"])  // Ahmet

// Ekleme/Güncelleme
sözlük["meslek"] = "Mühendis"
sözlük["yaş"] = 31

// Kontrol
if sözlük.contains("ad") {
    print("Ad bulundu")
}

// Tip belirtimi
let notlar: Harita<string, number> = {
    "Matematik": 85,
    "Fizik": 90,
    "Kimya": 78
}

세트

독특한 요소 컬렉션:

let kume = {1, 2, 3, 3, 2, 1}  // {1, 2, 3}

kume.push(4)      // {1, 2, 3, 4}
kume.pop(2)     // {1, 3, 4}
kume.contains(3)    // true

// Set işlemleri
let a = {1, 2, 3}
let b = {2, 3, 4}

let birleşim = a | b     // {1, 2, 3, 4}
let kesişim = a & b      // {2, 3}
let fark = a - b         // {1}

구조체 유형

사용자 정의 데이터 구조 정의:

type Kişi = struct {
    ad: string,
    yaş: number,
    email: string
}

// Instance oluşturma
let kişi = Kişi {
    ad: "Ahmet",
    yaş: 30,
    email: "ahmet@example.com"
}

// Alan erişimi
print(kişi.ad)      // Ahmet
print(kişi.yaş)     // 30

// Güncelleme
kişi.yaş = 31

열거형 유형

상수 값 세트:

let case = HttpDurum.Ok

결과 유형

오류 처리의 경우:

type Sonuç<T, E> = enum {
    Ok(T),
    Err(E)
}

function split(a: number, b: number) -> Sonuç<number, string> {
    if b == 0 {
        return Err "Sıfıra bölme!"
    }
    return Ok a / b
}

let sonuç = split(10, 2)
match sonuç {
    Ok(değer) => print("Sonuç: ", değer),
    Err(mesaj) => print("HATA: ", mesaj)
}

옵션 종류(옵션)

널 입력 가능 대안:

type Seçenek<T> = enum {
    Some(T),
    Hiç
}

function bul(List: Liste<number>, hedef: number) -> Seçenek<number> {
    for i 0'from List.len'a to {
        if List[i] == hedef {
            return Some(i)
        }
    }
    return Hiç
}

let indeks = bul([1, 2, 3, 4], 3)
match indeks {
    Some(i) => print("Bulundu: ", i),
    Hiç => print("Bulunamadı")
}

유형 추론

BERK는 강력한 유형 추론을 수행합니다.

// Tip otomatik belirlenir
let sayı = 42            // number
let pi = 3.14            // Ondalık
let isim = "Ahmet"       // string
let doğru_mu = true     // Boolean
let List = [1, 2, 3]    // Liste<number>

// Karmaşık çıkarımlar
let harita = {           // Harita<string, number>
    "a": 1,
    "b": 2
}

let function = (x) => x * 2  // (number) -> number

모범 사례

✅ 즐길거리

// Açık type belirtimi karmaşık durumlarda
let kullanıcılar: Liste<Kullanıcı> = []

// Immutable veri for const use
const API_ANAHTARI = "gizli_anahtar"

// Result tipi hata yönetimi for
function fs::read(yol: string) -> Sonuç<string, Err> { ... }

❌ 피해야 할 것

// Gereksiz type belirtimi
let x: number = 10  // "let x = 10" yeterli

// Any tipini fazla kullanmak
let karışık: Any = ...  // Mümkünse spesifik type use

// Null yerine Option use
let değer = null  // Option<T> tercih et

예: 모든 유형

function main() {
    // Temel tipler
    let sayı: number = 42
    let float: Ondalık = 3.14
    let String: string = "Hello"
    let bool: Boolean = true
    
    // Koleksiyonlar
    let List: Liste<number> = [1, 2, 3]
    let Tuple: (number, string) = (1, "bir")
    let harita: Harita<string, number> = {"a": 1}
    let küme: Küme<number> = {1, 2, 3}
    
    // Özel tipler
    type Nokta = struct { x: number, y: number }
    let p = Nokta { x: 10, y: 20 }
    
    // Sonuç tipi
    let sonuç: Sonuç<number, string> = Ok(42)
}
