HAL - Hardware Abstraction Layer

BERK HAL (Hardware Abstraction Layer) is a unified hardware abstraction layer for embedded systems.33 moduleswith7 different microcontroller platformssupports and provides 100% market coverage.

  • 100% market coverage- Full coverage in microcontroller segment

🎯 Supported Platforms

1. ESP32 (Espressive)

  • Processor: Dual-core Xtensa LX6 @ 240 MHz
  • RAM: 520 KB SRAM
  • Flash: 4-16MB
  • Features: WiFi 802.11 b/g/n, Bluetooth 4.2/BLE
  • Areas of Use: IoT, smart home, wearables
  • Popularity: ⭐⭐⭐⭐⭐ (Most popular IoT platform)

2. STM32F4 (STMicroelectronics)

  • Processor: ARM Cortex-M4F @ 168 MHz
  • RAM: 192 KB SRAM
  • Flash: 512 KB - 1 MB
  • Features: FPU, DSP instructions, Ethernet, USB OTG
  • Areas of Use: Industrial automation, robotics, motor control
  • Popularity: ⭐⭐⭐⭐⭐ (Professional standard)

3. RP2040 (Raspberry Pi)

  • Processor: Dual-core ARM Cortex-M0+ @ 133 MHz
  • RAM: 264 KB SRAM
  • Flash: 2-16 MB (external QSPI)
  • Features: PIO (Programmable I/O), USB 1.1
  • Areas of Use: Maker projects, training, prototypes
  • Popularity: ⭐⭐⭐⭐⭐ (Raspberry Pi ecosystem)

4. nRF52 (Nordic Semiconductor)

  • Processor: ARM Cortex-M4F @ 64MHz
  • RAM: 256KB
  • Flash: 1MB
  • Features: Bluetooth 5.x/BLE, ultra-low power
  • Areas of Use: Wearables, beacons, wireless sensors
  • Popularity: ⭐⭐⭐⭐ (BLE industry standard)

5. GD32 (Gigadevice)

  • Processor: ARM Cortex-M4 @ 120MHz
  • RAM: 128 KB SRAM
  • Flash: 512KB
  • Features: STM32 pin-compatible, low cost
  • Areas of Use: Consumer electronics, industrial
  • Popularity: ⭐⭐⭐ (Cost-effective alternative)

6. Nuvoton

  • Processor: ARM Cortex-M23/M4 @ 72-192 MHz
  • RAM: 64-512 KB
  • Flash: 256 KB - 1 MB
  • Features: Security features, CAN-FD
  • Areas of Use: Automotive, industrial security
  • Popularity: ⭐⭐⭐ (Special applications)

7. Arduino (AVR/ARM)

  • Processor: ATmega328P @ 16 MHz (AVR), ARM Cortex-M0+ @ 48 MHz
  • RAM: 2-32 KB
  • Flash: 32-256 KB
  • Features: Wide ecosystem, easy to use
  • Areas of Use: Education, hobby projects, prototyping
  • Popularity: ⭐⭐⭐⭐⭐ (Most common maker platform)

🚀 Quick Start

GPIO - LED Blink

// ESP32 - GPIO LED kontrolü
import hal_gpio

// GPIO2'yi çıkış as ayarla (ESP32 yerleşik LED)
hal_gpio_init(2, GPIO_MODE_OUTPUT)

// Sonsuz for
while true do
    hal_gpio_write(2, 1)  // LED aç
    hal_delay_ms(1000)     // 1 saniye sleep
    
    hal_gpio_write(2, 0)  // LED kapat
    hal_delay_ms(1000)
end

UART - Serial Communication

// STM32F4 - UART ile veri gönderme
import hal_uart

// UART1'i 115200 baud rate ile başlat
hal_uart_init(UART1, 115200)

let mesaj = "Merhaba BERK!\n"
hal_uart_write_string(UART1, mesaj)

// Veri okuma
let buffer = yeni_dizi(64)
let okunan = hal_uart_read(UART1, buffer, 64, 1000)  // 1sn timeout

if okunan > 0 then do
    print("Alınan: " + buffer)
end

I2C - Sensor Reading

// RP2040 - BME280 sıcaklık/nem sensörü
import hal_i2c

let BME280_ADDR = 0x76
let REG_TEMP = 0xFA

// I2C0'ı 100 kHz ile başlat
hal_i2c_init(I2C0, 100000)

// Sıcaklık register'ını input (3 byte)
let veri = yeni_dizi(3)
hal_i2c_read_reg(I2C0, BME280_ADDR, REG_TEMP, veri, 3)

// Ham veriyi işle
let adc_T = (veri[0] << 12) | (veri[1] << 4) | (veri[2] >> 4)
let sıcaklık = adc_T / 5120.0  // Basitleştirilmiş
print("Sıcaklık: " + sıcaklık + "°C")

ADC - Analog Reading

// ESP32 - Analog potansiyometre okuma
import hal_adc

// ADC1 Channel 0 (GPIO36)
hal_adc_init(ADC1, CHANNEL_0, ADC_WIDTH_12BIT, ADC_ATTEN_11DB)

while true do
    // 0-4095 arası değer input (12-bit)
    let ham = hal_adc_read(ADC1, CHANNEL_0)
    
    // Voltaja çevir (0-3.3V)
    let voltaj = (ham / 4095.0) * 3.3
    
    print("ADC: " + ham + " -> " + voltaj + "V")
    hal_delay_ms(500)
end

PWM - Motor Control

// STM32F4 - DC motor PWM kontrolü
import hal_pwm

// Timer2, Channel 1, Pin PA5
// 1 kHz frekans, %50 duty cycle
hal_pwm_init(TIMER2, CHANNEL_1, 1000, 50)

// PWM'i başlat
hal_pwm_start(TIMER2, CHANNEL_1)

// Yavaşça hızlan (0% -> 100%)
let i = 0
while i <= 100 do
    hal_pwm_set_duty(TIMER2, CHANNEL_1, i)
    hal_delay_ms(50)
    i = i + 1
end

// PWM'i durdur
hal_pwm_stop(TIMER2, CHANNEL_1)

🎓 Usage Scenarios

IoT Smart Home Device (ESP32)

// Akıllı termostat - WiFi + sensör
import hal_wifi, hal_adc, hal_gpio, hal_http

function akıllı_termostat() do
    // WiFi bağlan
    hal_wifi_connect("EvWiFi", "sifre123", 10000)
    
    // Sıcaklık sensörü (LM35 - ADC)
    hal_adc_init(ADC1, CHANNEL_0, ADC_WIDTH_12BIT, ADC_ATTEN_11DB)
    
    // Röle çıkışı (ısıtıcı kontrolü)
    hal_gpio_init(GPIO_ROLE, GPIO_MODE_OUTPUT)
    
    while true do
        // Sıcaklık input
        let adc = hal_adc_read(ADC1, CHANNEL_0)
        let sıcaklık = (adc / 4095.0) * 330.0  // LM35: 10mV/°C
        
        // Buluta gönder
        let url = "http://api.iot.com/temp?value=" + sıcaklık
        hal_http_post(url, "")
        
        // Isıtıcı kontrolü (20°C altı)
        if sıcaklık < 20.0 then do
            hal_gpio_write(GPIO_ROLE, 1)  // Isıtıcı aç
        else do
            hal_gpio_write(GPIO_ROLE, 0)  // Isıtıcı kapat
        end
        
        hal_delay_ms(60000)  // 1 dakika sleep
    end
end

Engine Control System (STM32F4)

// BLDC motor kontrolü - PWM + encoder + PID
import hal_pwm, hal_timer, hal_gpio

let hedef_hız = 1000  // RPM
let encoder_sayac = 0

// Encoder interrupt callback
function encoder_callback() do
    encoder_sayac = encoder_sayac + 1
end

function motor_kontrol() do
    // PWM başlat (20 kHz)
    hal_pwm_init(TIMER1, CHANNEL_1, 20000, 0)
    hal_pwm_start(TIMER1, CHANNEL_1)
    
    // Encoder interrupt
    hal_gpio_init_interrupt(GPIO_ENC_A, GPIO_MODE_INPUT_PULLUP, 
                           GPIO_INT_RISING, encoder_callback)
    
    // PID değişkenleri
    let Kp = 0.5, Ki = 0.1, Kd = 0.01
    let integral = 0.0, son_hata = 0.0
    
    while true do
        // Hız hesapla (encoder sayacından)
        let anlık_hız = encoder_sayac * 60  // RPM
        encoder_sayac = 0
        
        // PID kontrol
        let hata = hedef_hız - anlık_hız
        integral = integral + hata
        let türev = hata - son_hata
        
        let çıkış = Kp*hata + Ki*integral + Kd*türev
        son_hata = hata
        
        // PWM duty cycle ayarla (0-100%)
        let duty = math_clamp(çıkış, 0.0, 100.0)
        hal_pwm_set_duty(TIMER1, CHANNEL_1, duty)
        
        hal_delay_ms(100)  // 10 Hz kontrol döngüsü
    end
end

Low Power Sensor Node (nRF52)

// BLE beacon + sensör - Ultra-low power
import hal_ble, hal_adc, hal_power

function ble_sensör_beacon() do
    // BLE başlat
    hal_ble_init("BERK_Sensör")
    hal_ble_set_advertising(100)  // 100ms interval
    
    // ADC başlat (sıcaklık sensörü)
    hal_adc_init(ADC1, CHANNEL_0, ADC_WIDTH_12BIT, ADC_GAIN_1_6)
    
    while true do
        // Sensör input
        let sıcaklık = hal_adc_read_temp()  // Internal temp
        let batarya = hal_adc_read_vbat()   // Battery voltage
        
        // BLE advertising paketine push
        hal_ble_update_data("temp", sıcaklık)
        hal_ble_update_data("vbat", batarya)
        
        // Deep sleep (1 dakika)
        hal_power_deep_sleep(60000)  // 60sn uyku, ~3µA
        // Wake-up sonrası buradan devam eder
    end
end
