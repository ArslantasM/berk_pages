Error Types

Result Type

Functions that return success or error:

type Sonuç<T, E> = enum {
    Ok(T),
    Err(E)
}

function split(a: number, b: number) -> Sonuç<number, string> {
    if b == 0 {
        return Err "Sıfıra bölme"
    }
    return Ok a / b
}

match split(10, 2) {
    Ok(x) => print("Sonuç: ", x),
    Err(e) => print("Err: ", e)
}

Option Type

May or may not have value:

type Seçenek<T> = enum {
    Some(T),
    Hiç
}

function bul(List: Liste<number>, değer: number) -> Seçenek<number> {
    for (i, x) listede {
        if x == değer {
            return Some(i)
        }
    }
    return Hiç
}
