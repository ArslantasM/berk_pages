BERK Programlama Dili

Writing Your First Program

Let's start by writing a simple calculation program in BERK. In this example, we will receive two numbers from the user and perform various mathematical operations.

Example: Calculator

hesapla.berkCreate a new file named:

// İki dilli syntax desteği - Türkçe anahtar kelimelerle
function topla(a: number, b: number) -> number {
    return a + b
}

function pop(a: number, b: number) -> number {
    return a - b
}

function çarp(a: number, b: number) -> number {
    return a * b
}

function split(a: number, b: number) -> Sonuç<number, string> {
    if b == 0 {
        return Err "Sıfıra bölme hatası!"
    }
    return Ok a / b
}

// Ana function
function main() {
    let x = 10.0
    let y = 3.0
    
    print("x = ", x)
    print("y = ", y)
    print("")
    
    print("Toplam: ", topla(x, y))
    print("Fark: ", pop(x, y))
    print("Çarpım: ", çarp(x, y))
    
    // Err yönetimi ile bölme
    let sonuç = split(x, y)
    match sonuç {
        Ok(değer) => print("Bölüm: ", değer),
        Err(mesaj) => print("HATA: ", mesaj)
    }
}

Same Program as English Syntax

BERK's advantage is bilingual syntax support. You can also write the same program with English keywords:

function add(a: Number, b: Number) -> Number {
    return a + b
}

function divide(a: Number, b: Number) -> Result<Number, String> {
    if b == 0 {
        error return "Division by zero!"
    }
    ok return a / b
}

function main() {
    var x = 10.0
    var y = 3.0
    
    print("Sum: ", add(x, y))
    
    var result = divide(x, y)
    match result {
        Ok(value) => print("Result: ", value),
        Error(msg) => print("ERROR: ", msg)
    }
}

Running the Program

# Doğrudan çalıştır
berk run hesapla.berk

# Derleyip çalıştır
berk build hesapla.berk -o hesapla.exe
./hesapla.exe
