BERK,strong static typingwith safe andtype inferenceIt offers a practical language. A modern type system inspired by Rust and TypeScript.

Type Inference

BERK compiler can automatically extract types:

// Tip belirtilmemiş ama derleyici çıkarıyor
let x = 42          // sayı
let y = 3.14        // kesir
let isim = "Ali"    // String
let List = [1,2,3] // Dizi<sayı>

Collection Types

Arrays

// Sabit boyutlu Array
let sayılar: [sayı; 5] = [1, 2, 3, 4, 5]

// Dinamik Array
let List: Dizi<sayı> = Dizi::new()
List.push(42)

Vectors

let vec = Vec::from([1, 2, 3, 4])
vec.push(5)
print(vec.len())  // 5

hash map

use koleksiyonlar::HashMap;

let yaşlar = HashMap::new()
yaşlar.push("Ali", 25)
yaşlar.push("Ayşe", 30)

Structures

struct Nokta do
    x: kesir,
    y: kesir
end

// Kullanım
let p = Nokta { x: 1.0, y: 2.0 }
print(p.x)  // 1.0

Tuple Structures

struct Renkli(sayı, sayı, sayı);  // RGB

let kırmızı = Renkli(255, 0, 0)
print(kırmızı.0)  // 255

Option and Result Types

Choice(Option)

function split(a: kesir, b: kesir) -> Seçenek<kesir> do
    if b == 0.0 do
        return Hiçbiri
    end
    return Bir(a / b)
end

Conclusion(Result)

function fs::read(yol: String) -> Sonuç<String, Err> do
    if yol == "" do
        return Err("Boş yol")
    end
    return Ok(içerik)
end

Generic Types

struct Kutu<T> do
    değer: T
end

function yeni_kutu<T>(değer: T) -> Kutu<T> do
    return Kutu { değer: değer }
end

// Kullanım
let sayı_kutusu = yeni_kutu(42)
let metin_kutusu = yeni_kutu("merhaba")

Trait System (Interfaces)

trait Yazdırılabilir do
    function print(self)
end

struct Kişi do
    isim: String
end

uygula Yazdırılabilir for Kişi do
    function print(self) do
        print("Kişi: " + self.isim)
    end
end

Type Conversions

Open Casting

let x: sayı = 42
let y: kesir = kayıt(kesir, x)
let z: sayı8 = kayıt(sayı8, x)

Safe Conversion

let metin1 = "42"
let sayı1 = metin1.sayıya_çevir()  // Sonuç<sayı, Err>

eşleştir sayı1 do
    Ok(n) => print(n),
    Err(e) => print("Err: " + e)
end

Type Aliases

type Sonuç<T> = Sonuç<T, String>
type Nokta2D = (kesir, kesir)
type Matris3x3 = [[kesir; 3]; 3]

Pattern Matching

eşleştir değer do
    0 => print("Sıfır"),
    1..10 => print("Bir ile dokuz arası"),
    x if x % 2 == 0 => print("Çift sayı"),
    _ => print("Diğer")
end
