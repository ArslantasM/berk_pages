🚀 Quick Start

import fmt

// printf style
fmt.printf("Merhaba %s, yaşın %d\n", "Ali", 25)

// Rust style (format!)
let msg = fmt.format("Sonuç: {}, Durum: {}", 42, "OK")

// Formatting options
fmt.printf("Float: %.2f\n", 3.14159)  // 3.14
fmt.printf("Hex: %#x\n", 255)  // 0xff
fmt.printf("Binary: %b\n", 42)  // 101010

Table Formatter

import fmt

function print_table(headers: Dizi, rows: Dizi) do
    // Column widths
    let widths = headers.map(function(h) do return h.len() end)
    
    // Header
    her (i, h) in headers.enumerate() for do
        fmt.printf("%-*s ", widths[i], h)
    end
    fmt.printf("\n")
    
    // Separator
    her w in widths for do
        fmt.printf("%s ", "-".repeat(w))
    end
    fmt.printf("\n")
    
    // Rows
    her row in rows for do
        her (i, cell) in row.enumerate() for do
            fmt.printf("%-*s ", widths[i], cell)
        end
        fmt.printf("\n")
    end
end

print_table(
    ["Name", "Age", "City"],
    [
        ["Ali", "25", "Istanbul"],
        ["Ayşe", "30", "Ankara"]
    ]
)
