💡 Usage Examples

Parsing and Reading XML

use xml

// XML metnini ayrıştırma
değer xml_metin = """
<kitaplar>
    <kitap id="1">
        <başlık>BERK Programlama</başlık>
        <yazar>Ahmet Yılmaz</yazar>
        <fiyat>49.90</fiyat>
    </kitap>
    <kitap id="2">
        <başlık>Web Geliştirme</başlık>
        <yazar>Ayşe Demir</yazar>
        <fiyat>59.90</fiyat>
    </kitap>
</kitaplar>
"""

değer belge = xml.ayrıştır(xml_metin)

// XPath ile sorgulama
değer kitaplar = belge.xpath("//kitap")
for kitap in kitaplar {
    değer başlık = kitap.bul("başlık").String
    değer yazar = kitap.bul("yazar").String
    değer fiyat = kitap.bul("fiyat").String
    değer id = kitap.öznitelik_al("id")
    
    print(f"Kitap #{id}: {başlık} - {yazar} ({fiyat} TL)")
}

// Çıktı:
// Kitap #1: BERK Programlama - Ahmet Yılmaz (49.90 TL)
// Kitap #2: Web Geliştirme - Ayşe Demir (59.90 TL)

Creating XML

use xml

// Yeni XML belgesi oluşturma
değer belge = xml.Belge("1.0", "UTF-8")
değer kök = belge.kök_element_oluştur("katalog")

// Ürünler ekleme
değer ürünler = [
    {"id": "P001", "ad": "Laptop", "fiyat": 5500, "stok": 15},
    {"id": "P002", "ad": "Mouse", "fiyat": 75, "stok": 120},
    {"id": "P003", "ad": "Klavye", "fiyat": 250, "stok": 45}
]

for ürün in ürünler {
    değer elem = xml.Element("ürün")
    elem.öznitelik_ayarla("id", ürün["id"])
    elem.öznitelik_ayarla("stok", ürün["stok"].dizgeye())
    
    değer ad = xml.Element("ad")
    ad.metin_ayarla(ürün["ad"])
    elem.alt_ekle(ad)
    
    değer fiyat = xml.Element("fiyat")
    fiyat.metin_ayarla(ürün["fiyat"].dizgeye())
    fiyat.öznitelik_ayarla("para_birimi", "TRY")
    elem.alt_ekle(fiyat)
    
    kök.alt_ekle(elem)
}

// XML'i dosyaya kaydetme
xml.dosyaya_yaz(belge, "katalog.xml", girintili: true)
print("XML belgesi oluşturuldu!")

Configuration File Manager

use xml
use fs

class YapılandırmaYöneticisi {
    değer dosya_yolu: Dize
    değer belge: xml.Belge
    
    function new(dosya_yolu: Dize) -> YapılandırmaYöneticisi {
        değer Example = YapılandırmaYöneticisi {
            dosya_yolu: dosya_yolu,
            belge: null
        }
        
        if fs.dosya_var_mı(dosya_yolu) {
            Example.belge = xml.dosyadan_ayrıştır(dosya_yolu)
        } elseif {
            Example.belge = xml.Belge("1.0", "UTF-8")
            Example.belge.kök_element_oluştur("yapılandırma")
        }
        
        return Example
    }
    
    function ayar_al(Section: Dize, anahtar: Dize) -> Sonuç<Dize, Err> {
        değer xpath = f"//Section[@ad='{Section}']/{anahtar}"
        değer bulunan = bu.belge.xpath(xpath)
        
        if bulunan.len > 0 {
            return Ok(bulunan[0].String)
        }
        return Err(f"Ayar bulunamadı: {Section}.{anahtar}")
    }
    
    function ayar_ata(Section: Dize, anahtar: Dize, değer: Dize) {
        değer kök = bu.belge.kök()
        değer Section_xpath = f"//Section[@ad='{Section}']"
        değer Section_elem = kök.xpath(Section_xpath)
        
        if Section_elem.len == 0 {
            Section_elem = xml.Element("Section")
            Section_elem.öznitelik_ayarla("ad", Section)
            kök.alt_ekle(Section_elem)
        } elseif {
            Section_elem = Section_elem[0]
        }
        
        değer ayar_elem = xml.Element(anahtar)
        ayar_elem.metin_ayarla(değer)
        Section_elem.alt_ekle(ayar_elem)
    }
    
    function kaydet() {
        xml.dosyaya_yaz(bu.belge, bu.dosya_yolu, girintili: true)
    }
}

// Usage
değer yapılandırma = YapılandırmaYöneticisi.new("uygulama.xml")

yapılandırma.ayar_ata("veritabanı", "sunucu", "localhost")
yapılandırma.ayar_ata("veritabanı", "port", "5432")
yapılandırma.ayar_ata("uygulama", "dil", "tr")
yapılandırma.ayar_ata("uygulama", "tema", "koyu")

yapılandırma.kaydet()

// Ayarları okuma
değer sunucu = yapılandırma.ayar_al("veritabanı", "sunucu").aç()
değer dil = yapılandırma.ayar_al("uygulama", "dil").aç()
print(f"Sunucu: {sunucu}, Dil: {dil}")
