🐕 state::watchdog

Watchdog Timer - System Reliability and Reset Management

~280 lines ~8 functions STM32/GD32/ESP32

📖 Overview

Watchdog timer automatically detects system freezes caused by software errors and resets the system. If the microcontroller enters an unexpected state (infinite loop, deadlock), timeout occurs because the watchdog counter is not reset and the system is restarted.

There are two types of watchdogs:IWDG(Independent Watchdog - works with LSI clock, active even in low power modes) andWWDG(Window Watchdog - more precise timing with window control).

🔑 Key Features

  • Independent watchdog (IWDG) - LSI clock (32 kHz)
  • Window watchdog (WWDG) - APB1 clock
  • Configurable timeout (ms - seconds)
  • Early warning interrupt (WWDG)
  • Reset source detection
  • Debug mode freeze support
  • Working in Sleep/Stop mode
  • Prescaler settings (4 - 256)

🚀 Quick Start

import hal::watchdog, hal::core

// Independent watchdog (4 saniye timeout)
let wdg = watchdog.init_iwdg(do
    timeout_ms: 4000,
    window_mode: false
end)

// Ana for
for do
    // Normal işlemler
    core.delay_ms(500)
    
    // Watchdog'u besle (refresh)
    watchdog.refresh(wdg)
    
    // Eğer bu satır çalışmazsa sistem 4 saniye in resetlenir
end

💡 Example 1: Using Basic IWDG

System protection with 2-second watchdog

import hal::watchdog, hal::gpio, hal::core

let led_green = gpio.pin_init(gpio.PORT_C, 13, gpio.MODE_OUTPUT)
let led_red = gpio.pin_init(gpio.PORT_C, 14, gpio.MODE_OUTPUT)

function main() do
    // Check reset source
    let reset_cause = watchdog.get_reset_source()
    
    if reset_cause == watchdog.IWDG_RESET then do
        // Watchdog reset occurred!
        gpio.pin_write(led_red, gpio.HIGH)
        core.delay_ms(2000)
        gpio.pin_write(led_red, gpio.LOW)
    end
    
    // Initialize watchdog (2 second timeout)
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 2000,
        window_mode: false
    end)
    
    // Main loop
    for do
        // Normal operation
        gpio.pin_toggle(led_green)
        core.delay_ms(100)
        
        // Feed watchdog every 100ms (well before 2s timeout)
        watchdog.refresh(wdg)
    end
end

💡 Example 2: Watchdog with Task Monitor

Multi-task monitoring and watchdog feeding

import hal::watchdog, hal::time

let son_beslenme_zamani: Zaman
let watchdog_aktif = true

function kritik_gorev_1() do
    // Kritik işlemler
    core.delay_ms(50)
    
    // İşlem başarılı
    watchdog_besleme_istegi()
end

function kritik_gorev_2() do
    // Başka kritik işlemler
    core.delay_ms(75)
    
    // İşlem başarılı
    watchdog_besleme_istegi()
end

function watchdog_besleme_istegi() do
    son_beslenme_zamani = time.now()
end

function watchdog_yonetici(wdg: WatchdogHandle) do
    for do
        // Check if tasks are healthy
        let elapsed = time.now() - son_beslenme_zamani
        
        if elapsed < time.milliseconds(500) then do
            // Tasks are running OK, feed watchdog
            watchdog.refresh(wdg)
        yoksa do
            // Tasks hung! Let watchdog reset the system
            io.println("UYARI: Görevler askıda! Watchdog resetleyecek...")
        end
        
        core.delay_ms(100)
    end
end

function main() do
    // 1 saniye watchdog timeout
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 1000,
        window_mode: false
    end)
    
    son_beslenme_zamani = time.now()
    
    // Start tasks
    thread.spawn(|| kritik_gorev_1())
    thread.spawn(|| kritik_gorev_2())
    thread.spawn(|| watchdog_yonetici(wdg))
    
    thread.join_all()
end

💡 Example 3: Window Watchdog (WWDG)

Precise timing with window control

import hal::watchdog

// WWDG: Hem çok erken hem çok continue beslemeyi engeller
function wwdg_demo() do
    // Window: 50-100ms aralığında beslenmeli
    let wwdg = watchdog.init_wwdg(do
        timeout_ms: 100,
        window_ms: 50,  // Pencere başlangıcı
        early_wakeup_interrupt: true
    end)
    
    // Early warning interrupt handler
    watchdog.set_interrupt_handler(wwdg, || do
        io.println("Watchdog erken uyarı!")
        // Burası early wakeup interrupt (zamanında besle)
    end)
    
    let count_occurrences = 0
    for do
        // İşlemler
        core.delay_ms(30)
        
        // 30ms + 30ms = 60ms (pencere in: 50-100ms ✓)
        core.delay_ms(30)
        
        // Watchdog'u besle
        watchdog.refresh(wwdg)
        
        count_occurrences += 1
        
        // Simulate hang every 10 iterations
        if count_occurrences % 10 == 0 then do
            io.println("Simüle edilen gecikme...")
            core.delay_ms(150)  // Too late! WWDG will reset
        end
    end
end

💡 Example 4: Watchdog + Backup Register

Reset reason recording and analysis

import hal::watchdog, hal::rtc

struct ResetInfo do
    count: sayı,
    last_reason: ResetSource,
    timestamp: sayı
end

function reset_bilgisi_kaydet() do
    let reason = watchdog.get_reset_source()
    
    // Backup register'from önceki bilgiyi input
    let reset_count = rtc.backup_read(0)
    reset_count += 1
    
    // Yeni bilgiyi kaydet
    rtc.backup_write(0, reset_count)
    rtc.backup_write(1, reason.to_int())
    rtc.backup_write(2, time.timestamp())
    
    io.println("Reset sayısı: " + reset_count.yazıya())
    io.println("Reset nedeni: " + reason.to_string())
end

function main() do
    // Initialize RTC for backup registers
    rtc.init()
    
    // Log reset information
    reset_bilgisi_kaydet()
    
    // Check for excessive resets
    let reset_count = rtc.backup_read(0)
    if reset_count > 10 then do
        io.println("HATA: Çok fazla reset! Sistem unstable.")
        // Enter safe mode or disable some features
        rtc.backup_write(0, 0)  // Reset counter
    end
    
    // Initialize watchdog
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 3000,
        window_mode: false
    end)
    
    // Normal operation
    for do
        // Your code here
        watchdog.refresh(wdg)
        core.delay_ms(500)
    end
end

💡 Best Practices

  • Timeout selection:It should be 2-3 times the longest task duration
  • Feeding frequency:Feed at intervals of 50-75% of timeout
  • Debug mode:Freeze watchdog during development (at debugger breakpoints)
  • Critical sections:Calculate watchdog timeout in places where interrupt is disabled
  • Reset logging:Store reset reasons in the backup register
  • IWDG vs WWDG:IWDG (LSI independent) in Production, WWDG for critical timing
  • Power modes:IWDG works in low-power modes, WWDG stops

⚠️ Important Notes

  • Once started, IWDG cannot be stopped (hardware restriction)
  • Set DBGMCU_APB1_FZ register to prevent watchdog resets during debugging
  • LSI clock tolerance can be up to 40% - take into account in timeout calculations
  • Disable watchdog during flash programming (erase may take a long time)
