📖 Overview

Device management, resource allocation and device tree functions.

🚀 Usage

import hal::dev

// Register device
let led_dev = dev.register(do
    name: "led0",
    type: dev.TYPE_GPIO,
    resource: gpio_handle
end)

// Get device by name
let dev_handle = dev.get("led0")?

// List all devices
let devices = dev.list()
her d in devices for do
    io.println("Device: {}", d.name)
end

// Unregister
dev.unregister(led_dev)
