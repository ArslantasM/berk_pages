🚀 Quick Start

import complex, math

// Complex number (a + bi)
let z1 = complex.new(3.0, 4.0)  // 3 + 4i
let z2 = complex.new(1.0, -2.0)  // 1 - 2i

// Arithmetic
let sum = complex.add(z1, z2)
let product = complex.mul(z1, z2)

// Properties
let mod = complex.abs(z1)  // |z| = sqrt(3²+4²) = 5
let açı = complex.arg(z1)  // θ = atan2(4,3)
let eşlenik = complex.conj(z1)  // 3 - 4i

// Polar form
let z_polar = complex.from_polar(5.0, math.PI/4)

print(complex.to_string(z1))  // "3+4i"

Mandelbrot Set

import complex, graphics

function mandelbrot(c: Complex, max_iter: int) -> int do
    let z = complex.new(0.0, 0.0)
    
    her i in range(0, max_iter) for do
        z = complex.add(complex.mul(z, z), c)
        
        if complex.abs(z) > 2.0 then do
            return i
        end
    end
    
    return max_iter
end

// Render Mandelbrot
her y in range(0, 600) for do
    her x in range(0, 800) for do
        let re = (x.kesir() - 400.0) / 200.0
        let im = (y.kesir() - 300.0) / 200.0
        
        let c = complex.new(re, im)
        let iterations = mandelbrot(c, 100)
        
        let renk = iterations.kesir() / 100.0
        graphics.set_pixel(x, y, renk, renk, renk)
    end
end
