📊 state::adc

Analog-to-Digital Converter - Analog Signal Conversion

400 lines ~25 functions 12-bit ADC

📖 Overview

ADC (Analog-to-Digital Converter) module converts analog signals into digital values. It is used in applications such as sensor reading, temperature measurement, battery voltage control.

🔑 Key Features

  • 6-16 bit resolution (device dependent)
  • Multiple ADC channels (up to 19)
  • Single/continuous conversion
  • DMA support
  • Internal temperature sensor
  • Internal voltage reference
  • analog watchdog
  • Configurable sampling time

💡 Example 1: Temperature Sensor

import hal::adc, hal::gpio, hal::core

function sıcaklık_oku() -> kesir do
    // ADC setup
    adc.clock_enable()
    let adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    // GPIO for analog input (PA0)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    // Configure channel
    adc.channel_config(adc1, adc.CHANNEL_0, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Read multiple samples and average
    let sum = 0.0
    let samples = 10
    
    her i in 0..samples for do
        adc.start(adc1)
        let raw = adc.read(adc1)
        sum += raw.kesir()
        core.delay_ms(10)
    end
    
    let avg = sum / samples.kesir()
    
    // Convert to voltage (3.3V reference, 12-bit)
    let voltage = avg * 3.3 / 4095.0
    
    // LM35 sensor: 10mV/°C
    let temp = voltage * 100.0
    
    return temp
end

function main() do
    core.system_init()
    core.clock_config(72_000_000)
    
    for do
        let temp = sıcaklık_oku()
        io.println("Sıcaklık: {:.1f}°C".formatla(temp))
        core.delay_ms(1000)
    end
end

💡 Example 2: Battery Voltage Monitor

import hal::adc, hal::gpio

function batarya_voltaj_oku() -> kesir do
    // Voltage divider: Battery -> 10K -> ADC -> 10K -> GND
    // Max input: 6.6V (for 3.3V ADC)
    
    adc.clock_enable()
    let adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 1, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    adc.channel_config(adc1, adc.CHANNEL_1, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Read 100 samples for accuracy
    let sum = 0
    her i in 0..100 for do
        adc.start(adc1)
        sum += adc.read(adc1)
    end
    
    let avg = sum / 100
    let voltage = (avg.kesir() / 4095.0) * 3.3
    
    // Apply voltage divider ratio (2:1)
    let battery_voltage = voltage * 2.0
    
    return battery_voltage
end

function batarya_durumu(voltage: kesir) -> yazı do
    if voltage > 4.1 then return "Tam"
    yoksa if voltage > 3.7 then return "İyi"
    yoksa if voltage > 3.5 then return "Orta"
    yoksa return "Düşük"
end

function main() do
    core.system_init()
    
    for do
        let v = batarya_voltaj_oku()
        let case = batarya_durumu(v)
        io.println("Batarya: {:.2f}V [{}]".formatla(v, case))
        core.delay_ms(5000)
    end
end

💡 Example 3: Multi-Channel ADC with DMA

import hal::adc, hal::dma, hal::gpio

const CHANNELS = 4

function adc_dma_init() -> adc.Handle do
    // GPIO setup for 4 analog inputs (PA0-PA3)
    gpio.clock_enable(gpio.PORT_A)
    her i in 0..CHANNELS for do
        gpio.pin_init(gpio.PORT_A, i, do
            mode: gpio.MODE_ANALOG,
            pull: gpio.PULL_NONE
        end)
    end
    
    // ADC setup
    adc.clock_enable()
    let adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT,
        scan_mode: true,
        continuous: true
    end)
    
    // Configure channels
    adc.channel_config(adc1, adc.CHANNEL_0, do rank: 1 end)
    adc.channel_config(adc1, adc.CHANNEL_1, do rank: 2 end)
    adc.channel_config(adc1, adc.CHANNEL_2, do rank: 3 end)
    adc.channel_config(adc1, adc.CHANNEL_3, do rank: 4 end)
    
    return adc1
end

let adc_buffer: [sayı; CHANNELS] = [0, 0, 0, 0]

function main() do
    core.system_init()
    let adc1 = adc_dma_init()
    
    // DMA setup for continuous ADC transfer
    dma.clock_enable()
    let dma_ch = dma.init(dma.DMA1_CH1, do
        direction: dma.PERIPH_TO_MEMORY,
        periph_addr: adc.data_register(adc1),
        memory_addr: adc_buffer.adres(),
        buffer_size: CHANNELS,
        periph_inc: false,
        memory_inc: true,
        periph_size: dma.SIZE_16BIT,
        memory_size: dma.SIZE_16BIT,
        circular: true,
        priority: dma.PRIORITY_HIGH
    end)
    
    // Start DMA and ADC
    dma.start(dma_ch)
    adc.start_dma(adc1)
    
    // Main loop - data is automatically updated by DMA
    for do
        io.println("Kanal 0: {}  Kanal 1: {}  Kanal 2: {}  Kanal 3: {}".formatla(
            adc_buffer[0], adc_buffer[1], adc_buffer[2], adc_buffer[3]
        ))
        core.delay_ms(500)
    end
end

💡 Example 4: Analog Watchdog (Voltage Monitor)

import hal::adc, hal::gpio, hal::int

const THRESHOLD_LOW = 1500   // ~1.2V
const THRESHOLD_HIGH = 3000  // ~2.4V

let alarm_triggered = false

function adc_watchdog_handler() do
    alarm_triggered = true
    io.println("⚠️ UYARI: Voltaj threshold dışında!")
end

function main() do
    core.system_init()
    
    // GPIO setup
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    // ADC setup
    adc.clock_enable()
    let adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    adc.channel_config(adc1, adc.CHANNEL_0, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Configure analog watchdog
    adc.watchdog_enable(adc1, do
        channel: adc.CHANNEL_0,
        low_threshold: THRESHOLD_LOW,
        high_threshold: THRESHOLD_HIGH
    end)
    
    // Enable interrupt
    adc.interrupt_enable(adc1, adc.WATCHDOG_INT)
    int.register_handler(int.ADC1_IRQ, adc_watchdog_handler)
    int.enable_irq(int.ADC1_IRQ)
    
    // Start continuous conversion
    adc.start_continuous(adc1)
    
    for do
        if alarm_triggered then do
            alarm_triggered = false
            let value = adc.read(adc1)
            let voltage = (value.kesir() / 4095.0) * 3.3
            io.println("Voltaj: {:.2f}V [ALARM]".formatla(voltage))
        end
        core.delay_ms(100)
    end
end

💡 Example 5: Internal Temperature Sensor

import hal::adc

function cpu_sicaklik_oku() -> kesir do
    adc.clock_enable()
    let adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    // Enable internal temperature sensor
    adc.enable_temp_sensor()
    
    // Configure internal channel (usually CH16)
    adc.channel_config(adc1, adc.CHANNEL_TEMP, do
        sampling_time: adc.SAMPLETIME_480CYCLES  // Long sampling for accuracy
    end)
    
    // Read multiple samples
    let sum = 0
    her i in 0..10 for do
        adc.start(adc1)
        sum += adc.read(adc1)
        core.delay_ms(5)
    end
    
    let avg = sum / 10
    
    // STM32 formula: Temp = (V25 - Vsense) / Avg_Slope + 25
    // Typical: V25 = 1.43V (@ 25°C), Avg_Slope = 4.3mV/°C
    let voltage = (avg.kesir() / 4095.0) * 3.3
    let temp = ((1.43 - voltage) / 0.0043) + 25.0
    
    return temp
end

function main() do
    core.system_init()
    
    for do
        let temp = cpu_sicaklik_oku()
        io.println("CPU Sıcaklığı: {:.1f}°C".formatla(temp))
        
        // Thermal management
        if temp > 80.0 then do
            io.println("⚠️ UYARI: Yüksek sıcaklık!")
            // Reduce clock speed or take action
        end
        
        core.delay_ms(2000)
    end
end

⚡ Performance Tips

  • Sampling Time:Longer sampling time = higher sensitivity, but slower
  • Calibration:Run calibrate() at every startup
  • Averaging:Average 10-100 samples to reduce noise
  • DMA Usage:Reduces CPU load in multi-channel reading
  • Continuous Mode:Use for high speed sampling
  • Input Impedance:Increase sampling time for high impedance sources

⚠️ Important Notes

  • Before using ADCclock_enable()call
  • For analog pins use GPIO MODE_ANALOG
  • Maximum input voltage: VREF+ (usually 3.3V)
  • Input impedance: <50kΩ recommended
  • Conversion time = sampling time + 12 cycles (12-bit)
  • Internal temperature sensor ±2°C sensitivity
  • Use circular buffer when using DMA
  • Watchdog can only watch one channel

💡 Örnek 3: Multi-Channel Data Logger

import hal::adc, hal::gpio, hal::time

struct SensorData do
    timestamp: time.Instant,
    channel0: kesir,  // Temperature
    channel1: kesir,  // Light sensor
    channel2: kesir,  // Humidity
    channel3: kesir   // Pressure
end

function adc_init_multi() -> adc.Handle do
    adc.clock_enable()
    
    let adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT,
        scan_mode: true,  // Enable scan mode
        continuous: true   // Continuous conversion
    end)
    
    // Setup GPIO pins for 4 channels
    gpio.clock_enable(gpio.PORT_A)
    her i in 0..4 for do
        gpio.pin_init(gpio.PORT_A, i, do
            mode: gpio.MODE_ANALOG,
            pull: gpio.PULL_NONE
        end)
    end
    
    // Configure channels
    her i in 0..4 for do
        adc.channel_config(adc1, i, do
            sampling_time: adc.SAMPLETIME_480CYCLES,
            rank: i + 1  // Scan sequence
        end)
    end
    
    return adc1
end

function sensör_oku(adc1: adc.Handle) -> SensorData do
    // Start conversion
    adc.start(adc1)
    
    // Read all channels
    let readings = []
    her i in 0..4 for do
        readings.push(adc.read(adc1))
    end
    
    // Convert to physical values
    return SensorData do
        timestamp: time.now(),
        channel0: readings[0].kesir() * 3.3 / 4095.0 * 100.0,  // Temp (°C)
        channel1: readings[1].kesir() * 100.0 / 4095.0,         // Light (%)
        channel2: readings[2].kesir() * 100.0 / 4095.0,         // Humidity (%)
        channel3: readings[3].kesir() * 3.3 / 4095.0 * 50.0     // Pressure (kPa)
    end
end

function main() do
    let adc1 = adc_init_multi()
    let log_dosyası = fs.aç("sensor_log.csv", "w")?
    
    log_dosyası.print("timestamp,temp,light,humidity,pressure\n")?
    
    for do
        let data = sensör_oku(adc1)
        
        let line = "{},{:.1f},{:.1f},{:.1f},{:.1f}\n".formatla(
            data.timestamp.yazıya(),
            data.channel0,
            data.channel1,
            data.channel2,
            data.channel3
        )
        
        log_dosyası.print(line)?
        log_dosyası.sync()?  // Flush to disk
        
        io.println("Logged: Temp={:.1f}°C Light={:.0f}%".formatla(
            data.channel0, data.channel1
        ))
        
        core.delay_ms(10000)  // Log every 10 seconds
    end
end
