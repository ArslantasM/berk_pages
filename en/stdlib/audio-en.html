🚀 Quick Start

import audio

// Load audio file
let ses = audio.load("music.wav")
audio.play(ses)

// Generate sound (synthesizer)
let osc = audio.oscillator_create(SINE, 440.0)  // A4 note
audio.oscillator_play(osc, 1.0)  // 1 second

// Microphone input
let mic = audio.input_open()
let buffer = audio.input_read(mic, 1024)

// Real-time processing
let output = audio.output_open(44100, 2)  // 44.1kHz stereo

for do
    let samples = audio.input_read(mic, 512)
    
    // Apply effect
    let processed = audio.reverb(samples, 0.5)
    
    audio.output_write(output, processed)
end

Synthesizer (ADSR Envelope)

import audio, math

function synthesizer_note(frekans: kesir, süre: kesir) -> AudioBuffer do
    let sample_rate = 44100
    let num_samples = (sample_rate * süre).int()
    
    // ADSR parameters
    let attack = 0.05    // 50ms attack
    let decay = 0.1      // 100ms decay
    let sustain = 0.7    // 70% sustain level
    let release = 0.2    // 200ms release
    
    let buffer = audio.buffer_create(num_samples, 1)
    
    her i in range(0, num_samples) for do
        let t = i.kesir() / sample_rate
        
        // Oscillator
        let sample = math.sin(2.0 * math.PI * frekans * t)
        
        // ADSR envelope
        let amplitude = 0.0
        
        if t < attack then do
            amplitude = t / attack
        end else if t < (attack + decay) then do
            let decay_t = (t - attack) / decay
            amplitude = 1.0 - decay_t * (1.0 - sustain)
        end else if t < (süre - release) then do
            amplitude = sustain
        end else do
            let release_t = (t - (süre - release)) / release
            amplitude = sustain * (1.0 - release_t)
        end
        
        audio.buffer_set(buffer, i, sample * amplitude * 0.3)
    end
    
    return buffer
end

// C major chord
let C4 = synthesizer_note(261.63, 2.0)  // C
let E4 = synthesizer_note(329.63, 2.0)  // E
let G4 = synthesizer_note(392.00, 2.0)  // G

// Mix
let chord = audio.mix([C4, E4, G4])
audio.play(chord)

// Reverb effect
sınıf Reverb do
    let decay: kesir
    let delay_buffer: AudioBuffer
    
    function new(decay_factor: kesir) do
        self.decay = decay_factor
        self.delay_buffer = audio.buffer_create(44100, 1)  // 1 second buffer
    end
    
    function process(input: AudioBuffer) -> AudioBuffer do
        let output = audio.buffer_create(input.len(), 1)
        
        her i in range(0, input.len()) for do
            let sample = audio.buffer_get(input, i)
            let delayed = audio.buffer_get(self.delay_buffer, i % self.delay_buffer.len())
            
            let mixed = sample + delayed * self.decay
            audio.buffer_set(output, i, mixed)
            audio.buffer_set(self.delay_buffer, i % self.delay_buffer.len(), mixed)
        end
        
        return output
    end
end

// Effect chain setup
let chain = EffectChain.new()
chain.push(Reverb.new(0.5))
chain.push(audio.compressor_create(-20.0, 4.0))  // Threshold, ratio

// Real-time processing
let input = audio.input_open()
let output = audio.output_open(44100, 1)

for do
    let buffer = audio.input_read(input, 512)
    let processed = chain.process(buffer)
    audio.output_write(output, processed)
end
