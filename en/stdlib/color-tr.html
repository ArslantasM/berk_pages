🚀 Quick Start

import color

// RGB color
let red = color.rgb(255, 0, 0)

// Hex color
let blue = color.from_hex("#0000FF")

// HSV (Hue, Saturation, Value)
let hsv = color.rgb_to_hsv(red)
print("H: " + hsv.h.yazıya())  // 0°

// HSL (Hue, Saturation, Lightness)
let hsl = color.rgb_to_hsl(red)

// Color blending
let blended = color.blend(red, blue, 0.5)  // Purple

// Lighten/Darken
let lighter = color.lighten(red, 0.2)
let darker = color.darken(red, 0.2)

Color Palette Generator

import color

function complementary_palette(base_color: Color) -> Dizi do
    let hsv = color.rgb_to_hsv(base_color)
    
    // Complementary (180° hue shift)
    let comp_hsv = do h: (hsv.h + 180) % 360, s: hsv.s, v: hsv.v end
    let complementary = color.hsv_to_rgb(comp_hsv)
    
    return [base_color, complementary]
end

function triadic_palette(base_color: Color) -> Dizi do
    let hsv = color.rgb_to_hsv(base_color)
    
    let c1 = color.hsv_to_rgb(do h: (hsv.h + 120) % 360, s: hsv.s, v: hsv.v end)
    let c2 = color.hsv_to_rgb(do h: (hsv.h + 240) % 360, s: hsv.s, v: hsv.v end)
    
    return [base_color, c1, c2]
end

let base = color.from_hex("#FF5733")
let palette = triadic_palette(base)

her c in palette for do
    print(color.to_hex(c))
end
