🚀 Quick Start

import physics

// 3D World
let world = physics.world_create([0.0, -9.81, 0.0])

// Ground plane
let ground = physics.body_create_static()
physics.body_add_shape(ground, physics.plane_shape([0.0, 1.0, 0.0], 0.0))
physics.world_add_body(world, ground)

// Sphere
let sphere = physics.body_create_dynamic(1.0)
physics.body_add_shape(sphere, physics.sphere_shape(0.5))
physics.body_set_position(sphere, [0.0, 10.0, 0.0])
physics.world_add_body(world, sphere)

// Simulate
her i in range(0, 100) for do
    physics.world_step(world, 1.0/60.0)
    let pos = physics.body_get_position(sphere)
    print("t=" + i.yazıya() + ": y=" + pos[1].yazıya())
end
