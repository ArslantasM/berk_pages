🚗 state::can

Controller Area Network - Automotive Communication

383 lines ~14 functions Automotive

📖 Overview

CAN (Controller Area Network) is a reliable communication protocol used in automotive and industrial applications. It provides communication between ECU (Engine Control Unit), sensors and actuators.

🔑 Key Features

  • CAN 2.0A (11-bit ID) / 2.0B (29-bit ID)
  • Up to 1 Mbps bitrate
  • Error detection/correction
  • Message filtering (28 filters)
  • 3TX mailboxes
  • 2 RX FIFOs
  • Priority-based arbitration
  • automatic retransmission

🚀 Quick Start

import hal::can, hal::gpio

// CAN pins: PB8=RX, PB9=TX
gpio.clock_enable(gpio.PORT_B)
gpio.pin_init(gpio.PORT_B, 8, do mode: gpio.MODE_AF, af: 9 end)
gpio.pin_init(gpio.PORT_B, 9, do mode: gpio.MODE_AF, af: 9 end)

// Initialize CAN (500 kbps)
let can1 = can.init(can.CAN1, do
    bitrate: 500_000,
    mode: can.MODE_NORMAL
end)

// Send message
can.send(can1, do
    id: 0x123,
    data: [0x01, 0x02, 0x03],
    extended_id: false
end)

// Receive message
if msg = can.receive(can1, 100) then do
    io.println("Received ID: 0x{:X}".formatla(msg.id))
end

💡 Example: ECU Simulator

import hal::can, hal::core

const ENGINE_RPM = 0x100
const VEHICLE_SPEED = 0x101
const COOLANT_TEMP = 0x102
const THROTTLE_POS = 0x103

function send_rpm(can1: can.Handle, rpm: sayı) do
    let data = [
        (rpm >> 8) & 0xFF,
        rpm & 0xFF
    ]
    
    can.send(can1, do
        id: ENGINE_RPM,
        data: data,
        extended_id: false
    end)
end

function send_speed(can1: can.Handle, speed: sayı) do
    can.send(can1, do
        id: VEHICLE_SPEED,
        data: [speed & 0xFF],
        extended_id: false
    end)
end

function send_temp(can1: can.Handle, temp: sayı) do
    // Temperature in Celsius + 40
    let temp_byte = (temp + 40) & 0xFF
    
    can.send(can1, do
        id: COOLANT_TEMP,
        data: [temp_byte],
        extended_id: false
    end)
end

function main() do
    core.system_init()
    
    // Setup CAN
    let can1 = can.init(can.CAN1, do
        bitrate: 500_000,
        mode: can.MODE_NORMAL
    end)
    
    // Configure filter (accept all)
    can.filter_config(can1, do
        filter_id: 0,
        id: 0x000,
        mask: 0x000,  // Accept all IDs
        fifo: can.FIFO0
    end)
    
    // Simulate ECU data
    let rpm = 800  // Idle RPM
    let speed = 0
    let temp = 20
    
    for do
        // Send telemetry at 10Hz
        send_rpm(can1, rpm)
        send_speed(can1, speed)
        send_temp(can1, temp)
        
        // Simulate acceleration
        if rpm < 6000 then do
            rpm += 100
            speed = (rpm / 100)  // Simplified
            temp += 1  // Engine heats up
        else do
            rpm = 800  // Back to idle
            speed = 0
            temp = 20
        end
        
        core.delay_ms(100)
    end
end

💡 Example: CAN Bus Monitor

import hal::can

function message_name(id: u32) -> yazı do
    return eşle id do
        0x100 => "Engine RPM",
        0x101 => "Vehicle Speed",
        0x102 => "Coolant Temp",
        0x103 => "Throttle Pos",
        0x200 => "Brake Status",
        0x300 => "Door Status",
        _ => "Unknown"
    end
end

function parse_data(msg: can.Message) do
    io.print("0x{:03X} [{}] ".formatla(msg.id, msg.dlc))
    
    // Print data bytes
    her byte in msg.data[0..msg.dlc] for do
        io.print("{:02X} ".formatla(byte))
    end
    
    // Interpret known messages
    let name = message_name(msg.id)
    io.print(" // {}", name)
    
    eşle msg.id do
        0x100 => do  // RPM
            let rpm = (msg.data[0] << 8) | msg.data[1]
            io.print(" = {} RPM", rpm)
        end,
        0x101 => do  // Speed
            io.print(" = {} km/h", msg.data[0])
        end,
        0x102 => do  // Temp
            let temp = msg.data[0].sayı() - 40
            io.print(" = {}°C", temp)
        end,
        _ => do end
    end
    
    io.println()
end

function main() do
    let can1 = can.init(can.CAN1, do
        bitrate: 500_000,
        mode: can.MODE_NORMAL
    end)
    
    // Accept all messages
    can.filter_config(can1, do
        filter_id: 0,
        id: 0x000,
        mask: 0x000,
        fifo: can.FIFO0
    end)
    
    io.println("CAN Bus Monitor Started (500 kbps)")
    io.println("Press Ctrl+C to exit\n")
    
    for do
        // Check for messages (non-blocking)
        if msg = can.receive(can1, 10) then do
            parse_data(msg)
        end
    end
end

💡 Example 3: OBD-II Diagnostics

import hal::can

const OBD_REQUEST = 0x7DF
const OBD_RESPONSE = 0x7E8
const PID_ENGINE_RPM = 0x0C
const PID_VEHICLE_SPEED = 0x0D

function obd_request(can1: can.Handle, pid: u8) -> Sonuç[sayı, Err] do
    can.send(can1, do
        id: OBD_REQUEST,
        data: [0x02, 0x01, pid, 0, 0, 0, 0, 0],
        dlc: 8
    end)?
    
    if response = can.receive(can1, 100) then do
        if response.id == OBD_RESPONSE then do
            return Ok(eşle pid do
                PID_ENGINE_RPM => ((response.data[3] << 8) | response.data[4]) / 4,
                PID_VEHICLE_SPEED => response.data[3].sayı(),
                _ => 0
            end)
        end
    end
    
    return Err("Timeout")
end

function main() do
    let can1 = can.init(can.CAN1, do bitrate: 500_000 end)
    
    for do
        if rpm = obd_request(can1, PID_ENGINE_RPM)? then do
            io.println("RPM: {}", rpm)
        end
        core.delay_ms(500)
    end
end

⚠️ Important Notes

  • CAN transceiver (TJA1050, MCP2551) required
  • 120Ω termination resistors required (at both ends of the bus)
  • Max bus length: 40m (1 Mbps), 500m (125 kbps)
  • Use twisted-pair cable (CAN_H, CAN_L)
  • Filter configuration is important (in high traffic)
  • ISO 11898 standard is used

🎯 Application Scenarios

  • Automotive ECU communication
  • OBD-II diagnostics
  • Industrial automation (CANopen)
  • Maritime (NMEA 2000)
