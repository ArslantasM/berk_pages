🔌 state::usb

USB 2.0 Full Speed ​​- Device/Host/OTG Controller

~420 lines ~25 functions STM32F4/F7/H7

📖 Overview

The USB module provides USB 2.0 Full Speed ​​(12 Mbps) communication. It establishes a PC connection with CDC (Virtual COM), HID (Keyboard/Mouse), MSC (Mass Storage) device classes. With host mode, USB flash disk and keyboard reading can be done. It is critical for IoT data logging, debug console, firmware update and user interface.

🔑 Key Features

  • USB 2.0 Full Speed:12 Mbps (Low Speed ​​1.5 Mbps is also supported)
  • Device Mode:It becomes a virtual device by connecting to a PC
  • Host Mode:USB peripherals can be connected (flash, keyboard)
  • OTG Support:Dynamic switching between Device and Host modes
  • Device Classes:CDC, HID, MSC, Audio, Custom
  • DMA Support:High-speed data transfer
  • VBUS Detection:USB cable insertion detection
  • Self/Bus Powered:Power management flexibility

function usb_console_rx_handler(data: Dizi<sayı8>) do
    for byte in data do
        if byte == 13 do  // Enter key
            // Process command
            usb.cdc_write("\r\n")
            
            let cmd = Metin.from_bytes(usb_rx_buffer, usb_rx_index)
            usb_console_process_command(cmd)
            
            usb_rx_index = 0
            usb.cdc_write("> ")
        yoksa if byte == 8 or byte == 127 do  // Backspace
            if usb_rx_index > 0 do
                usb_rx_index = usb_rx_index - 1
                usb.cdc_write("\b \b")  // Erase character
            end
        yoksa do
            if usb_rx_index < 256 do
                usb_rx_buffer[usb_rx_index] = byte
                usb_rx_index = usb_rx_index + 1
                usb.cdc_write_byte(byte)  // Echo
            end
        end
    end
end

function usb_console_process_command(cmd: Metin) do
    eşleştir cmd do

            usb.cdc_write("Commands:\r\n")
            usb.cdc_write("  help    - Show this help\r\n")
            usb.cdc_write("  status  - System status\r\n")
            usb.cdc_write("  reset   - Reset system\r\n")

function keyboard_press(key: sayı8) do
    // HID keyboard report: [modifier, reserved, key1-6]
    let report: Dizi<sayı8> = [0, 0, key, 0, 0, 0, 0, 0]
    
    // Send key press

function usb_msc_init() do

        vid: 0x0483,
        pid: 0x5720,
        product: "BERK USB Disk",
        block_count: sd_info.block_count,
        block_size: sd_info.block_size,
        read_callback: msc_read_block,
        write_callback: msc_write_block
    end)
    
    usb.connect()
    
    io.yazdir_satır("USB Mass Storage ready")
end

function msc_read_block(block_num: sayı, buffer: Dizi<sayı8>) -> mantık do

⚠️ Important Notes:
  • 48MHz Clock:Full 48 MHz clock required for USB (tolerance ±0.25%)
  • VID/PID:Get your own VID/PID in Production (from USB-IF)
  • Enumeration Time:PC recognition time may take 1-3 seconds
  • Interrupt Priority:USB interrupt must be high priority

✅ Best Practices

  • ✅ CDC for debugging:Use USB CDC instead of UART (cable requirement is the same)
  • ✅ HID for UI:Use HID keyboard/mouse for simple control
  • ✅ MSC for data:Use MSC for log files (FAT32)
  • ✅ Disconnect before reset:Call usb.disconnect() before reset
  • ❌ Float operations:Using floating-point in USB interrupt
  • ❌ Long callbacks:Keep USB callbacks short (<1 ms)
