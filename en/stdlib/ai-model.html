    Saving a Model

    Simple Registration

    use ai/model;
use ai/nn;

let model = Sequential::new([
    Linear::new(784, 256),
    ReLU::new(),
    Linear::new(256, 10)
]);

// Model eğitimi...

// LZ4 sıkıştırma ile kaydet
model.kaydet("model.berk")?;
print("Model kaydedildi!");

    Compression Levels

    use ai/model;

// Hızlı kaydetme (Level 1 - varsayılan)
model.kaydet_sıkıştırma("model.berk", seviye: 1)?;

// Maksimum sıkıştırma (Level 9)
model.kaydet_sıkıştırma("model_compressed.berk", seviye: 9)?;

// Sıkıştırmasız (debug for)
model.kaydet_ham("model_raw.berk")?;

    Model Loading

    Simple Installation

    use ai/model;
use ai/nn;

// Model yapısını tanımla
let model = Sequential::new([
    Linear::new(784, 256),
    ReLU::new(),
    Linear::new(256, 10)
]);

// Ağırlıkları yükle
model.yükle("model.berk")?;

// Inference
let x = Tensor::random([1, 784]);
let pred = model.ileri(x);
print("Tahmin: {}", pred.argmax());

    Partial Loading

    use ai/model;

// Sadece belirli katmanları yükle
model.yükle_kısmi("model.berk", katmanlar: ["conv1", "conv2", "fc"])?;

// Transfer learning for
model.dondur_katmanlar(["conv1", "conv2"]); // Bu katmanları eğitme
model.çöz_katmanlar(["fc"]);                 // Sadece fc'yi eğit

    Binary Format

    File Structure

    
┌─────────────────────────────────────┐
│ Magic Number (4 bytes): "BERK"     │
├─────────────────────────────────────┤
│ Version (2 bytes): 1.0              │
├─────────────────────────────────────┤
│ Compression Type (1 byte): LZ4=1   │
├─────────────────────────────────────┤
│ Metadata Length (4 bytes)           │
├─────────────────────────────────────┤
│ Metadata (JSON):                    │
│   - model_type                      │
│   - layer_count                     │
│   - param_count                     │
│   - created_at                      │
├─────────────────────────────────────┤
│ Compressed Tensor Data (LZ4)        │
│   Layer 1 weights + biases          │
│   Layer 2 weights + biases          │
│   ...                               │
└─────────────────────────────────────┘

    Model Metadata

    Reading Metadata

    use ai/model;

let metadata = Model::oku_metadata("model.berk")?;

print("Model tipi: {}", metadata.model_type);
print("Parametre sayısı: {}", metadata.param_count);
print("Oluşturulma: {}", metadata.created_at);
print("Berk versiyonu: {}", metadata.berk_version);

// Dosya boyutu
print("Dosya boyutu: {} MB", metadata.file_size / 1024.0 / 1024.0);

    Checkpoint Management

    Training Checkpoints

    use ai/model;
use ai/train;

// Her 10 epoch'ta kaydet
her epoch in range(100) for do
    trainer.eğit_epoch();
    
    if epoch % 10 == 0 then do
        model.kaydet(f"checkpoints/epoch_{epoch}.berk")?;
        print("Checkpoint kaydedildi: epoch {}", epoch);
    end
end

// En iyi modeli kaydet
if val_loss < best_loss then do
    model.kaydet("checkpoints/best_model.berk")?;
    best_loss = val_loss;
end

    Performance Optimizations

    LZ4 Compression

    • Level 1:Fast compression (default) - 28.8x faster
    • Dictionary:Optimized for repetitive weight patterns
    • parallel:Multi-core compression support
    • Streaming:Memory-efficient for large models

    API Reference

    Model Class

    • kaydet(path: dizgi) -> Sonuç- Save model (LZ4 Level 1)
    • yükle(path: dizgi) -> Sonuç- Upload model
    • kaydet_sıkıştırma(path: dizgi, seviye: int) -> Sonuç- Custom compression level
    • yükle_kısmi(path: dizgi, katmanlar: [dizgi]) -> Sonuç- Partial loading
    • oku_metadata(path: dizgi) -> Metadata- Just read metadata
    • onnx_export(path: dizgi, ...) -> Sonuç- ONNX export

    Checkpoint Class

    • kaydet_checkpoint(epoch, model, optimizer, metrics)
    • yükle_checkpoint(path) -> (epoch, model, optimizer)
    • en_iyi_checkpoint() -> dizgi- Best checkpoint path
    • checkpoint_listesi() -> [dizgi]- All checkpoints
