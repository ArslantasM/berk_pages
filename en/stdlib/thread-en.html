🚀 Quick Start

import thread

// Thread spawn
let handle = thread.spawn(function() do
    print("Thread'from merhaba!")
    return 42
end)

let Result = thread.join(handle)
print("Result: " + Result.yazıya())

// Mutex (mutual exclusion)
let count_occurrences = thread.mutex_new(0)

let threads = []
each i in range(0, 10) for do
    threads.push(thread.spawn(function() do
        each j in range(0, 100) for do
            thread.mutex_lock(count_occurrences, function(değer) do
                return değer + 1
            end)
        end
    end))
end

each t in threads for do
    thread.join(t)
end

let sum = thread.mutex_get(count_occurrences)
print("Toplam: " + sum.yazıya())  // 1000

// Channel (message passing)
let (tx, rx) = thread.channel_create()

thread.spawn(function() do
    thread.channel_send(tx, "Mesaj 1")
    thread.channel_send(tx, "Mesaj 2")
end)

print(thread.channel_receive(rx))  // "Mesaj 1"
print(thread.channel_receive(rx))  // "Mesaj 2"

📚 Sync

  • Mutex:Mutual exclusion lock
  • RwLock:Read-write lock (multiple readers, single writer)
  • Channel:Message passing (mpsc, mpmc)
  • Atomic:Lock-free atomic operations
  • Arc:Atomic reference counting
  • Barrier:Wait for all threads
  • Semaphore:resource counting

Parallel Map-Reduce

import thread

function parallel_map(veri: Dizi, f: function, thread_sayısı: int) -> Dizi do
    let chunk_boyutu = (veri.len() + thread_sayısı - 1) / thread_sayısı
    let handles = []
    
    each i in range(0, thread_sayısı) for do
        let başlangıç = i * chunk_boyutu
        let bitiş = math.min(başlangıç + chunk_boyutu, veri.len())
        
        if başlangıç >= veri.len()  do
            devam
        end
        
        let chunk = veri.dilimle(başlangıç, bitiş)
        
        let handle = thread.spawn(function() do
            return chunk.map(f)
        end)
        
        handles.push(handle)
    end
    
    // Resultları join
    let Result = []
    each handle in handles for do
        let chunk_result = thread.join(handle)
        Result = Result + chunk_result
    end
    
    return Result
end

function parallel_reduce(veri: Dizi, f: function, başlangıç: herhangi) -> herhangi do
    let partial_results = parallel_map(veri, function(x) do return x end, 4)
    
    // Local reduce'lar
    let handles = []
    let chunk_boyutu = (partial_results.len() + 4 - 1) / 4
    
    each i in range(0, 4) for do
        let başlangıç_idx = i * chunk_boyutu
        let bitiş_idx = math.min(başlangıç_idx + chunk_boyutu, partial_results.len())
        
        if başlangıç_idx >= partial_results.len()  do
            devam
        end
        
        let chunk = partial_results.dilimle(başlangıç_idx, bitiş_idx)
        
        handles.push(thread.spawn(function() do
            let acc = başlangıç
            each x in chunk for do
                acc = f(acc, x)
            end
            return acc
        end))
    end
    
    // Final reduce
    let Result = başlangıç
    each handle in handles for do
        Result = f(Result, thread.join(handle))
    end
    
    return Result
end

// useım: 1'from 1_000_000'a to topla
let sayılar = range(1, 1_000_001).collect()
let sum = parallel_reduce(sayılar, function(a, b) do return a + b end, 0)
print("Toplam: " + sum.yazıya())  // 500000500000
