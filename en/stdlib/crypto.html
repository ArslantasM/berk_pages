🔒 crypto - Cryptography and Encryption

Comprehensive cryptography tools for encryption, hashing and digital signature operations

📦 101 Functions 📝 1381 Lines 💾 49KB

📋Overview

cryptoThe module provides comprehensive tools for data encryption, hash functions, digital signatures and secure key management with modern cryptography algorithms. Supports AES, RSA, SHA, HMAC and more.

🔐 Symmetric Encryption

  • crypto.aes_encrypt()- AES encryption (128/192/256 bit)
  • crypto.aes_solve()- AES decryption
  • crypto.chacha20()- ChaCha20 stream encryption
  • crypto.aes_gcm()- AES-GCM (authenticated encryption)

🔑 Asymmetric Encryption

  • crypto.rsa_create_key_pair()- RSA key pair generation
  • crypto.rsa_encrypt()- Encryption with RSA
  • crypto.rsa_solve()- Decryption with RSA
  • crypto.ecc_key_generate()- Elliptic curve switches

#️⃣ Hash Functions

  • crypto.sha256()- SHA-256 hash
  • crypto.sha512()- SHA-512 hash
  • crypto.sha3()- SHA-3 hash
  • crypto.blake3()- BLAKE3 fast hash
  • crypto.hmac()- HMAC message authentication

✍️ Digital Signatures

  • crypto.rsa_sign()- RSA digital signature
  • crypto.rsa_verify()- RSA signature verification
  • crypto.ecdsa_sign()- ECDSA signing
  • crypto.ed25519_sign()- Signing Ed25519

🔐 Password Security

  • crypto.password_hash()- Secure password hash (Argon2id)
  • crypto.password_verify()- Password verification
  • crypto.pbkdf2()- PBKDF2 key derivation
  • crypto.bcrypt()- bcrypt password hash

🎲 Safe Randomness

  • crypto.random_byte()- Cryptographic random bytes
  • crypto.random_full()- Secure random integer
  • crypto.uuid()- Random UUID generation

💡 Usage Examples

AES Encryption and Decryption

use crypto
use encoding

// AES-256-GCM şifreleme (kimlik doğrulamalı)
değer mesaj = "Bu çok gizli bir mesajdır!"
değer anahtar = crypto.rastgele_bayt(32)  // 256-bit anahtar
değer nonce = crypto.rastgele_bayt(12)    // 96-bit nonce

print("Orijinal mesaj: " + mesaj)

// Şifreleme
değer şifreli = crypto.aes_gcm_şifrele(
    veri: mesaj.bayt(),
    anahtar: anahtar,
    nonce: nonce
)

değer şifreli_base64 = encoding.base64_kodla(şifreli)
print(f"\nŞifreli (Base64): {şifreli_base64}")

// Şifre çözme
değer çözülmüş = crypto.aes_gcm_çöz(
    veri: şifreli,
    anahtar: anahtar,
    nonce: nonce
).aç()

değer çözülmüş_metin = Dize.bayttan(çözülmüş)
print(f"Çözülmüş mesaj: {çözülmüş_metin}")

// Anahtar and nonce'ı güvenli sakla
print(f"\nAnahtar (hex): {encoding.hex_kodla(anahtar)}")
print(f"Nonce (hex): {encoding.hex_kodla(nonce)}")

Secure Password Management

use crypto
use time

sınıf ParolaYöneticisi {
    değer parola_karmaları: Sözlük<Dize, Dize>
    
    function new() -> ParolaYöneticisi {
        return ParolaYöneticisi {
            parola_karmaları: {}
        }
    }
    
    function kullanıcı_ekle(kullanıcı_adı: Dize, parola: Dize) -> Sonuç<Boş, Err> {
        // Parola gücünü kontrol et
        if parola.len < 8 {
            return Err("Parola en az 8 char_at olmalı")
        }
        
        // Argon2id ile güvenli karma oluştur
        değer başlangıç = time.now()
        değer karma = crypto.parola_karma(parola, maliyet: 15)
        değer süre = time.now() - başlangıç
        
        bu.parola_karmaları[kullanıcı_adı] = karma
        print(f"Parola karması oluşturuldu ({süre.milisaniye}ms)")
        
        return Ok(null)
    }
    
    function giriş_yap(kullanıcı_adı: Dize, parola: Dize) -> Mantıksal {
        if kullanıcı_adı !in bu.parola_karmaları {
            return false
        }
        
        değer karma = bu.parola_karmaları[kullanıcı_adı]
        değer başlangıç = time.now()
        değer geçerli = crypto.parola_doğrula(parola, karma)
        değer süre = time.now() - başlangıç
        
        print(f"Parola doğrulama: {süre.milisaniye}ms")
        return geçerli
    }
}

// Kullanım
değer yönetici = ParolaYöneticisi.new()

// Kullanıcı ekleme
yönetici.kullanıcı_ekle("ahmet", "Güçlü_P@rol@_123").aç()
yönetici.kullanıcı_ekle("ayşe", "Başka_Güçlü_P@ss").aç()

// Giriş denemeleri
print("\n--- Giriş Denemeleri ---")

if yönetici.giriş_yap("ahmet", "Güçlü_P@rol@_123") {
    print("✓ Ahmet başarıyla giriş yaptı")
} yoksa {
    print("✗ Giriş başarısız")
}

if yönetici.giriş_yap("ahmet", "yanlış_parola") {
    print("✓ Giriş başarılı")
} yoksa {
    print("✗ Yanlış parola!")
}

File Encryption System

use crypto
use fs
use encoding

sınıf DosyaŞifreleyici {
    function dosya_şifrele(
        kaynak: Dize,
        hedef: Dize,
        parola: Dize
    ) -> Sonuç<Boş, Err> {
        // Dosyayı input
        değer içerik = fs.fs::read(kaynak)?
        
        // Parola'from anahtar türet
        değer tuz = crypto.rastgele_bayt(16)
        değer anahtar = crypto.pbkdf2(
            parola.bayt(),
            tuz,
            iterasyon: 100000,
            len: 32
        )
        
        // Şifreleme for nonce oluştur
        değer nonce = crypto.rastgele_bayt(12)
        
        // AES-GCM ile şifrele
        değer şifreli = crypto.aes_gcm_şifrele(
            veri: içerik,
            anahtar: anahtar,
            nonce: nonce
        )
        
        // Dosya formatı: [TUZ:16][NONCE:12][ŞİFRELİ_VERİ]
        değer çıktı = tuz + nonce + şifreli
        fs.dosya_yaz(hedef, çıktı)?
        
        print(f"✓ Dosya şifrelendi: {kaynak} -> {hedef}")
        print(f"  Boyut: {içerik.len} -> {çıktı.len} bayt")
        
        return Ok(null)
    }
    
    function dosya_çöz(
        kaynak: Dize,
        hedef: Dize,
        parola: Dize
    ) -> Sonuç<Boş, Err> {
        // Şifreli dosyayı input
        değer veri = fs.fs::read(kaynak)?
        
        // Tuz and nonce'ı ayıkla
        değer tuz = veri[0..16]
        değer nonce = veri[16..28]
        değer şifreli = veri[28..]
        
        // Parola'from anahtar türet
        değer anahtar = crypto.pbkdf2(
            parola.bayt(),
            tuz,
            iterasyon: 100000,
            len: 32
        )
        
        // Şifreyi çöz
        değer içerik = crypto.aes_gcm_çöz(
            veri: şifreli,
            anahtar: anahtar,
            nonce: nonce
        )?
        
        // Dosyaya print
        fs.dosya_yaz(hedef, içerik)?
        
        print(f"✓ Dosya çözüldü: {kaynak} -> {hedef}")
        print(f"  Boyut: {veri.len} -> {içerik.len} bayt")
        
        return Ok(null)
    }
}

// Kullanım
değer şifreleyici = DosyaŞifreleyici {}

// Dosya şifreleme
şifreleyici.dosya_şifrele(
    "gizli_belgeler.pdf",
    "gizli_belgeler.pdf.enc",
    "çok_güçlü_parola_123"
).aç()

// Dosya çözme
şifreleyici.dosya_çöz(
    "gizli_belgeler.pdf.enc",
    "gizli_belgeler_çözülmüş.pdf",
    "çok_güçlü_parola_123"
).aç()

🎯 Good Practices

  • Never hardcode private keys or passwords
  • Use AES-GCM for authenticated encryption
  • Prefer Argon2id or bcrypt for password hashing
  • For random number generationcrypto.rastgele_bayt()use
  • Use at least 2048-bit key size for RSA
  • Store keys in secure key management systems
  • Rotate keys regularly
  • Avoid outdated cryptography algorithms (MD5, SHA1, DES)

⚠️ Safety Warnings

  • 🔴 Never share private keys or embed them in source code
  • 🔴 Standard in password or key generationrandomdo not use
  • 🔴 Use HMAC or GCM to protect against encrypted data tampering attacks
  • 🔴 Be careful when keeping sensitive data in memory (memory dumps)
