🚀 Quick Start

import mesh

// Load mesh
let model = mesh.load_obj("model.obj")

// Primitives
let cube = mesh.cube(1.0)
let sphere = mesh.sphere(0.5, 32, 32)
let cylinder = mesh.cylinder(0.5, 2.0, 32)

// Transform
mesh.translate(cube, [1.0, 0.0, 0.0])
mesh.rotate(cube, [0.0, 1.0, 0.0], 45.0)
mesh.scale(cube, [2.0, 1.0, 1.0])

// Normals
mesh.compute_normals(model)

// Save
mesh.save_obj(model, "output.obj")

Procedural Mesh Generator

import mesh, math

function terrain_generate(width: int, height: int) -> Mesh do
    let m = mesh.new()
    
    // Generate vertices
    her y in range(0, height) for do
        her x in range(0, width) for do
            let z = math.sin(x.kesir() * 0.1) * math.cos(y.kesir() * 0.1) * 5.0
            mesh.add_vertex(m, [x.kesir(), z, y.kesir()])
        end
    end
    
    // Generate faces
    her y in range(0, height-1) for do
        her x in range(0, width-1) for do
            let i0 = y * width + x
            let i1 = i0 + 1
            let i2 = (y + 1) * width + x
            let i3 = i2 + 1
            
            mesh.add_triangle(m, i0, i1, i2)
            mesh.add_triangle(m, i1, i3, i2)
        end
    end
    
    mesh.compute_normals(m)
    return m
end

let terrain = terrain_generate(100, 100)
mesh.save_obj(terrain, "terrain.obj")
