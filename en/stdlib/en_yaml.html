📝 yaml - YAML Operations

Reading and writing YAML (YAML Ain't Markup Language) files

📦 59 functions 📝 595 Lines 💾 15.7 KB

📋Overview

yamlThe module provides tools for reading, writing and editing configuration files in YAML format. It offers easy work with YAML, a human-readable data serialization format.

📖 YAML Parsing

  • yaml.parse()- Convert YAML text to BERK data structure
  • load_from yaml.file()- Reading YAML file
  • yaml.secure_load()- Secure YAML parsing
  • yaml.multi_load()- Upload multiple YAML documents

✏️ Creating YAML

  • yaml.yaz()- Converting BERK data to YAML format
  • yaml.write_to_file()- Writing to YAML file
  • yaml.multi_write()- Writing multiple documents

🔧 Configuration Options

  • yaml.indent()- Indent level adjustment
  • yaml.width()- Adjust line width
  • yaml.flow_style()- Stream or block style selection
  • yaml.default_stream()- Set default streaming style

💡 Usage Examples

Reading a YAML Configuration File

use yaml
use fs

// config.yaml dosyası:
// veritabanı:
//   sunucu: localhost
//   port: 5432
//   kullanıcı: admin
//   parola: güvenli123
//   veritabanları:
//     - üretim
//     - test
//     - geliştirme
//
// uygulama:
//   ad: BerkApp
//   sürüm: 1.0.0
//   ortam: üretim
//   özellikler:
//     cache_aktif: true
//     log_seviye: info
//     max_bağlantı: 100

değer yapılandırma = yaml.dosyadan_yükle("config.yaml")

// Yapılandırma değerlerini okuma
değer vt_sunucu = yapılandırma["veritabanı"]["sunucu"]
değer vt_port = yapılandırma["veritabanı"]["port"]
değer uygulama_ad = yapılandırma["uygulama"]["ad"]

print(f"Uygulama: {uygulama_ad}")
print(f"Veritabanı: {vt_sunucu}:{vt_port}")

// Veritabanı listesi
değer veritabanları = yapılandırma["veritabanı"]["veritabanları"]
print("\nMevcut veritabanları:")
for vt in veritabanları {
    print(f"  - {vt}")
}

// Özellikler
değer özellikler = yapılandırma["uygulama"]["özellikler"]
if özellikler["cache_aktif"] {
    print("\nÖnbellek aktif")
}
print(f"Log seviyesi: {özellikler['log_seviye']}")
print(f"Maksimum bağlantı: {özellikler['max_bağlantı']}")

Creating a YAML Configuration

use yaml

// Uygulama yapılandırması oluşturma
değer yapılandırma = {
    "sunucu": {
        "host": "0.0.0.0",
        "port": 8080,
        "ssl": {
            "aktif": true,
            "sertifika": "/path/to/cert.pem",
            "anahtar": "/path/to/key.pem"
        }
    },
    "veritabanı": {
        "sürücü": "postgresql",
        "bağlantı": {
            "host": "localhost",
            "port": 5432,
            "kullanıcı": "app_user",
            "parola": "güvenli_parola"
        },
        "havuz": {
            "min_boyut": 5,
            "max_boyut": 20,
            "timeout": 30
        }
    },
    "özellikler": {
        "hata_ayıklama": false,
        "log_dosyası": "/var/log/app.log",
        "log_seviye": "info",
        "izin_verilen_originler": [
            "https://example.com",
            "https://api.example.com",
            "https://admin.example.com"
        ]
    },
    "cache": {
        "type": "redis",
        "sunucu": "localhost:6379",
        "ttl": 3600,
        "önbellek_anahtarları": [
            "kullanıcı_oturumları",
            "ürün_listesi",
            "yapılandırma"
        ]
    }
}

// YAML dosyasına yazma
yaml.dosyaya_yaz(yapılandırma, "app_config.yaml")

// YAML metnine dönüştürme
değer yaml_metin = yaml.print(yapılandırma)
print("\nYAML içeriği:")
print(yaml_metin)

Kubernetes Manifest Manager

use yaml

class K8sManifest {
    function deployment_oluştur(ad: Dize, imaj: Dize, port: Tam, replika: Tam) -> Sözlük {
        return {
            "apiVersion": "apps/v1",
            "kind": "Deployment",
            "metadata": {
                "name": ad,
                "labels": {
                    "app": ad
                }
            },
            "spec": {
                "replicas": replika,
                "selector": {
                    "matchLabels": {
                        "app": ad
                    }
                },
                "template": {
                    "metadata": {
                        "labels": {
                            "app": ad
                        }
                    },
                    "spec": {
                        "containers": [{
                            "name": ad,
                            "image": imaj,
                            "ports": [{
                                "containerPort": port
                            }]
                        }]
                    }
                }
            }
        }
    }
    
    function service_oluştur(ad: Dize, port: Tam, hedef_port: Tam) -> Sözlük {
        return {
            "apiVersion": "v1",
            "kind": "Service",
            "metadata": {
                "name": f"{ad}-service"
            },
            "spec": {
                "type": "LoadBalancer",
                "selector": {
                    "app": ad
                },
                "ports": [{
                    "protocol": "TCP",
                    "port": port,
                    "targetPort": hedef_port
                }]
            }
        }
    }
}

// Kubernetes manifestleri oluşturma
değer k8s = K8sManifest {}

değer deployment = k8s.deployment_oluştur(
    "web-app",
    "nginx:1.21",
    80,
    3
)

değer service = k8s.service_oluştur("web-app", 80, 80)

// Tek dosyaya birden fazla manifest yazma
değer manifestler = [deployment, service]
yaml.dosyaya_yaz(manifestler, "k8s-deployment.yaml")

print("Kubernetes manifestleri oluşturuldu!")
print(f"Deployment: {deployment['metadata']['name']}")
print(f"Replika sayısı: {deployment['spec']['replicas']}")
print(f"Service: {service['metadata']['name']}")
