    Overview

    The rl module provides popular algorithms (DQN, PPO, SAC) and gym-compatible environments for reinforcement learning.

    DQN

    Deep Q-Network

    • value-based
    • Discrete actions
    • Experience replay

    PPO

    Proximal Policy Optimization

    • policy gradient
    • clipped surrogate
    • On-policy

    HAIR

    Soft Actor-Critic

    • off-policy
    • Continuous actions
    • maximum entropy

    A2C/A3C

    Actor-Critic

    • Advantage estimation
    • Asynchronous training
    • fast learning

    DQN - Deep Q-Network

    CartPole-v1 Training

    use ai/rl;
use ai/nn;

// Q-Network tanımla
let q_network = Sequential::new([
    Linear::new(4, 128),   // State dimension
    ReLU::new(),
    Linear::new(128, 128),
    ReLU::new(),
    Linear::new(128, 2)    // Action dimension
]);

// DQN agent
let agent = DQN::new(
    q_network: q_network,
    learning_rate: 0.001,
    gamma: 0.99,
    epsilon_start: 1.0,
    epsilon_end: 0.01,
    epsilon_decay: 0.995,
    buffer_size: 10000,
    batch_size: 64
);

// Ortam
let env = gym::make("CartPole-v1")?;

// Training loop
her episode in range(500) for do
    let state = env.reset()?;
    let total_reward = 0.0;
    
    çevrim do
        // Action match (epsilon-greedy)
        let action = agent.act(state);
        
        // Step
        let (next_state, reward, done, _) = env.step(action)?;
        
        // Deneyimi kaydet
        agent.remember(state, action, reward, next_state, done);
        
        // Replay and öğren
        if agent.buffer.len() > batch_size then do
            agent.replay()?;
        end
        
        state = next_state;
        total_reward += reward;
        
        if done then do kır; end
    end
    
    print("Episode {}: Reward = {:.2f}, Epsilon = {:.3f}", 
           episode, total_reward, agent.epsilon);
end

    Continuous Control

    use ai/rl;
use ai/nn;

// Actor-Critic network
sınıf ActorCritic {
    let shared: Sequential;
    let actor: Linear;
    let critic: Linear;
    
    function new(state_dim: int, action_dim: int) -> ActorCritic do
        ActorCritic {
            shared: Sequential::new([
                Linear::new(state_dim, 256),
                ReLU::new(),
                Linear::new(256, 256),
                ReLU::new()
            ]),
            actor: Linear::new(256, action_dim),
            critic: Linear::new(256, 1)
        }
    end
    
    function ileri(let kendi, state: Tensor) -> (Tensor, Tensor) do
        let features = kendi.shared.ileri(state);
        let action_mean = kendi.actor.ileri(features);
        let value = kendi.critic.ileri(features);
        return (action_mean, value);
    end
}

// PPO agent
let agent = PPO::new(
    policy: ActorCritic::new(state_dim: 24, action_dim: 4),
    learning_rate: 3e-4,
    gamma: 0.99,
    gae_lambda: 0.95,
    clip_epsilon: 0.2,
    epochs: 10,
    batch_size: 64
);

// Training
let env = gym::make("BipedalWalker-v3")?;

her iteration in range(1000) for do
    // Rollout topla
    let trajectories = agent.collect_rollouts(env, steps: 2048)?;
    
    // PPO update
    let loss = agent.update(trajectories)?;
    
    // Değerlendirme
    let eval_reward = agent.evaluate(env, episodes: 5)?;
    print("Iteration {}: Loss = {:.4f}, Reward = {:.2f}", 
           iteration, loss, eval_reward);
end

let agent = SAC::new(
    state_dim: 11,
    action_dim: 3,
    hidden_dims: [256, 256],
    learning_rate: 3e-4,
    gamma: 0.99,
    tau: 0.005,          // Soft target update

    auto_tune_alpha: true
);

let env = gym::make("Pendulum-v1")?;

her step in range(100000) for do
    let state = env.reset()?;
    
    çevrim do
        // Action sample (stochastic policy)
        let action = agent.select_action(state, evaluate: false);
        
        let (next_state, reward, done, _) = env.step(action)?;
        
        // Replay buffer'a push
        agent.buffer.push(state, action, reward, next_state, done);
        
        // Update
        if step > 1000 then do  // Warm-up
            agent.update(batch_size: 256)?;
        end
        
        state = next_state;
        if done then do kır; end
    end
end

    Custom Environment

    Create Your Own Environment

    use ai/rl;

trait Environment {
    function reset(let kendi) -> Tensor;
    function step(let kendi, action: Tensor) -> (Tensor, float, bool, Metadata);
    function render(let kendi);
    function close(let kendi);
}

sınıf GridWorld {
    let grid_size: int;
    let agent_pos: (int, int);
    let goal_pos: (int, int);
}

uygula Environment for GridWorld do
    function reset(let kendi) -> Tensor do
        kendi.agent_pos = (0, 0);
        return kendi.get_state();
    end
    
    function step(let kendi, action: Tensor) -> (Tensor, float, bool, Metadata) do
        // Action: 0=up, 1=right, 2=down, 3=left
        let (x, y) = kendi.agent_pos;
        
        match action.item() do
            0 => y -= 1,  // Up
            1 => x += 1,  // Right
            2 => y += 1,  // Down
            3 => x -= 1,  // Left
            _ => {}
        end
        
        // Sınırları kontrol et
        x = x.clamp(0, kendi.grid_size - 1);
        y = y.clamp(0, kendi.grid_size - 1);
        kendi.agent_pos = (x, y);
        
        // Reward hesapla
        let reward = if kendi.agent_pos == kendi.goal_pos then do
            1.0
        end yoksa do
            -0.01  // Small penalty for each step
        end;
        
        let done = kendi.agent_pos == kendi.goal_pos;
        
        return (kendi.get_state(), reward, done, Metadata::new());
    end
end

    Agent Evaluation

    use ai/rl;

function evaluate_policy(
    agent: Agent,
    env: Environment,
    num_episodes: int
) -> float do
    let total_rewards = [];
    
    her episode in range(num_episodes) for do
        let state = env.reset();
        let episode_reward = 0.0;
        
        çevrim do
            // Deterministic action (no exploration)
            let action = agent.select_action(state, evaluate: true);
            let (next_state, reward, done, _) = env.step(action)?;
            
            episode_reward += reward;
            state = next_state;
            
            if done then do kır; end
        end
        
        total_rewards.push(episode_reward);
    end
    
    return total_rewards.iter().sum::() / num_episodes as float;
end

// Kullanım
let mean_reward = evaluate_policy(agent, env, num_episodes: 10);
print("Ortalama reward: {:.2f}", mean_reward);

    API Reference

    DQN Class

    • yeni(q_network: Network, ...) -> DQN
    • act(kendi, state: Tensor) -> Tensor- Select action
    • remember(kendi, ...) - Save experience
    • replay(kendi) -> Sonuç- Batch update

    PPO Class

    • yeni(policy: ActorCritic, ...) -> PPO
    • collect_rollouts(kendi, env: Environment, ...) -> Trajectories
    • update(kendi, trajectories: Trajectories) -> ondalık
    • evaluate(kendi, env: Environment, ...) -> ondalık

    ReplayBuffer Class

    • new(capacity: int, ...) -> ReplayBuffer
    • push(kendi, state, action, reward, next_state, done)
    • sample(kendi, batch_size: int) -> Batch
    • len(kendi) -> int
