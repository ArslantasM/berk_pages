📖 Overview

The Redis module provides a full-featured Redis client over RESP (REdis Serialization Protocol). It offers string, hash, list, set operations and pub/sub support.

🔑 Key Features

  • String operations: GET, SET, INCR, DECR
  • Hash operations: HSET, HGET, HGETALL
  • List operations: LPUSH, RPUSH, LRANGE
  • Set operations: SADD, SMEMBERS, SISMEMBER
  • Key expiration: EXPIRE, TTL
  • Pub/Sub messaging
  • Pipeline & transactions
  • connection pooling

🚀 Quick Start

import redis

// Connect
let client = redis.connect("127.0.0.1:6379")?

// String operations
client.set("user:1:name", "Alice")?
let name = client.get("user:1:name")?
io.println("Name: {}", name)

// With expiration (5 minutes)
client.setex("session:abc", 300, "user_data")?

client.close()

💡 Example: Session Management

import redis, crypto, time, json

struct SessionStore do
    redis: redis.Client
end

struct Session do
    user_id: sayı,
    username: yazı,
    created_at: time.Instant,
    data: Map[yazı, yazı]
end

function session_store_new(redis_url: yazı) -> Sonuç[SessionStore, Err] do
    return Ok(SessionStore do
        redis: redis.connect(redis_url)?
    end)
end

function create_session(store: SessionStore, user_id: sayı, username: yazı) 
    -> Sonuç[yazı, Err] do
    // Generate session ID
    let session_id = crypto.random_hex(32)
    
    // Create session
    let session = Session do
        user_id: user_id,
        username: username,
        created_at: time.now(),
        data: Map.new()
    end
    
    // Store in Redis (1 hour TTL)
    let key = "session:{}".formatla(session_id)
    let value = json.encode(session)
    store.redis.setex(key, 3600, value)?
    
    return Ok(session_id)
end

function get_session(store: SessionStore, session_id: yazı) 
    -> Sonuç[İsteğe_Bağlı[Session], Err] do
    let key = "session:{}".formatla(session_id)
    let value = store.redis.get(key)?
    
    if value.is_none() then do
        return Ok(None)
    end
    
    let session: Session = json.decode(value.unwrap())?
    return Ok(Some(session))
end

function destroy_session(store: SessionStore, session_id: yazı) -> Sonuç[None, Err] do
    let key = "session:{}".formatla(session_id)
    store.redis.del(key)?
    return Ok(None)
end

// Kullanım
function main() do
    let store = session_store_new("127.0.0.1:6379")?
    
    // Login - create session
    let session_id = create_session(store, 123, "alice")?
    io.println("Session created: {}", session_id)
    
    // Verify session
    if session = get_session(store, session_id)? then do
        io.println("User: {} (ID: {})", session.username, session.user_id)
    end
    
    // Logout - destroy session
    destroy_session(store, session_id)?
end

💡 Example: Rate Limiting

import redis, time

function rate_limit_check(client: redis.Client, user_id: sayı, max_requests: sayı, window_secs: sayı) 
    -> Sonuç[bool, Err] do
    let key = "rate_limit:{}".formatla(user_id)
    
    // Increment counter
    let count = client.incr(key)?
    
    // Set expiry on first request
    if count == 1 then do
        client.expire(key, window_secs)?
    end
    
    // Check limit
    return Ok(count <= max_requests)
end

// API endpoint
function handle_request(client: redis.Client, user_id: sayı) -> Sonuç[None, Err] do
    // 100 requests per minute
    if rate_limit_check(client, user_id, 100, 60)? then do
        // Process request
        io.println("Request processed")
    else do
        io.println("Rate limit exceeded")
    end
    return Ok(None)
end
