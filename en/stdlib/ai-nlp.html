    tokenization

    BPE Tokenizer

    use ai/nlp;

// Pre-trained tokenizer yükle
let tokenizer = Tokenizer::yükle("bert-base-uncased")?;

let text = "BERK programlama dili çok güçlü!";
let tokens = tokenizer.encode(text);

print("Tokens: {:?}", tokens.ids);
print("Token texts: {:?}", tokens.tokens);

// Decode
let decoded = tokenizer.decode(tokens.ids);
print("Decoded: {}", decoded);

    BERT Embeddings

    Sentence Embeddings

    use ai/nlp;

// BERT model yükle
let bert = BERT::yükle(
    model: "bert-base-multilingual-cased",
    device: "cuda"
)?;

let sentences = [
    "BERK programlama dili",
    "Rust benzeri syntax",
    "Modern and güvenli"
];

// Embeddings receive
let embeddings = bert.encode(sentences)?;
print("Shape: {:?}", embeddings.shape());  // [3, 768]

// Similarity hesapla
let sim_01 = cosine_similarity(embeddings[0], embeddings[1]);
print("Benzerlik: {:.4f}", sim_01);

    Fine-tuning BERT

    use ai/nlp;
use ai/nn;
use ai/optim;

// BERT + Classification head
sınıf SentimentClassifier {
    let bert: BERT;
    let classifier: Linear;
    
    function new(num_classes: int) -> SentimentClassifier do
        let bert = BERT::yükle("bert-base-uncased")?;
        let classifier = Linear::new(768, num_classes);
        
        return SentimentClassifier { bert, classifier };
    end
    
    function ileri(let kendi, input_ids: Tensor) -> Tensor do
        let bert_output = kendi.bert.ileri(input_ids);
        let cls_token = bert_output[:, 0, :];  // [CLS] token
        return kendi.classifier.ileri(cls_token);
    end
}

// Eğitim
let model = SentimentClassifier::new(num_classes: 3);
let optimizer = AdamW::new(model.parametreler(), lr: 2e-5);

her epoch in range(3) for do
    her batch in train_loader for do
        let logits = model.ileri(batch.input_ids);
        let loss = cross_entropy(logits, batch.labels);
        
        optimizer.sıfırla();
        loss.geri();
        optimizer.adım();
    end
end

    Named Entity Recognition (NER)

    Entity Detection

    use ai/nlp;

// Pre-trained NER model
let ner = NER::yükle("bert-base-ner")?;

let text = "Ahmet Ankara'da Microsoft'ta çalışıyor.";
let entities = ner.predict(text)?;

her entity in entities for do
    print!("{}: {} ({})", 
            entity.text, 
            entity.label, 
            entity.confidence);
end

// Çıktı:
// Ahmet: PERSON (0.98)
// Ankara: LOCATION (0.95)
// Microsoft: ORGANIZATION (0.97)

    Multi-label Classification

    use ai/nlp;

let classifier = TextClassifier::new(
    base_model: "bert-base-uncased",
    labels: ["teknoloji", "spor", "politika", "ekonomi"],
    multi_label: true
);

let text = "Apple new iPhone modelini tanıttı. Hisse fiyatları yükseldi.";

her pred in predictions for do
    if pred.score > 0.5 then do

    end

// ekonomi: 0.78

    use ai/nlp;

// Pre-trained embeddings yükle
let w2v = Word2Vec::yükle("fasttext-tr-300d")?;

// Kelime vektörü receive
let vec = w2v.get_vector("bilgisayar")?;

    print!("{}: {:.4f}", word, similarity);
end

// Analoji: kral - erkek + kadın = ?

print!("Analoji sonucu: {}", result);  // "kraliçe"

    API Reference

    Tokenizer Class

    • yükle(model: dizgi) -> Tokenizer- Pre-trained tokenizer
    • encode(text: dizgi) -> Encoding- Tokenize text
    • decode(ids: [int]) -> dizgi- Decode Token IDs
    • encode_batch(texts: [dizgi]) -> [Encoding]- Batch encoding

    BERT Class

    • yükle(model: dizgi, ...) -> BERT- Load BERT model
    • encode(texts: [dizgi]) -> Tensor- Sentence embeddings
    • ileri(input_ids: Tensor) -> BertOutput- Forward pass

    NER Class

    • yükle(model: dizgi) -> NER- Load NER model
    • predict(text: dizgi) -> [Entity]- Entity detection
    • predict_batch(texts: [dizgi]) -> [[Entity]]
