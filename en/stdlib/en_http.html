REST API Client

import http, json

function kullanıcı_getir(id: integer) -> Sonuç do
    variable url = "https://jsonplaceholder.typicode.com/users/" + id.yazıya()
    variable yanıt = http.get(url)
    
    if yanıt.case != 200 then do
        return Err("HTTP " + yanıt.case.yazıya())
    end
    
    variable veri = json.parse(yanıt.gövde)
    return Ok(veri)
end

// Usage
variable sonuç = kullanıcı_getir(1)
eşleştir sonuç do
    Ok(user) => print("Kullanıcı: " + user["name"]),
    Err(e) => print("Err: " + e)
end

REST API Server with Middleware

import http, time, collections

variable sunucu = http.Sunucu_new(3000)
variable veritabanı = collections.HashMap_new()

// CORS Middleware
sunucu.middleware(function(istek, yanıt, ileri) do
    yanıt.header_ekle("Access-Control-Allow-Origin", "*")
    yanıt.header_ekle("Access-Control-Allow-Methods", "GET,POST,PUT,DELETE")
    
    if istek.metod == "OPTIONS" then do
        yanıt.case(204).gönder()
    elseif do
        ileri()
    end
end)

// Logging middleware
sunucu.middleware(function(istek, yanıt, ileri) do
    variable Getting Started = time.now()
    ileri()
    variable süre = time.now() - Getting Started
    print(istek.metod + " " + istek.yol + " - " + süre.yazıya() + "ms")
end)

// Routes
sunucu.rotala("GET", "/api/users", function(istek, yanıt) do
    yanıt.json(collections.HashMap_values(veritabanı))
end)

sunucu.rotala("POST", "/api/users", function(istek, yanıt) do
    variable user = json.parse(istek.gövde)
    variable id = collections.HashMap_size(veritabanı) + 1
    user["id"] = id
    collections.HashMap_set(veritabanı, id.yazıya(), user)
    yanıt.case(201).json(user)
end)

print("Sunucu http://localhost:3000 adresinde çalışıyor")
sunucu.başlat()
