🚀 Quick Start

import network

// TCP Server
let sunucu = network.TcpListener_bind("127.0.0.1:8080")
print("TCP sunucu 8080 portunda dinliyor...")

for do
    let (soket, adres) = sunucu.kabul_et()
    print("Bağlantı: " + adres.yazıya())
    
    let veri = soket.input(1024)
    soket.print("Echo: " + veri)
    soket.kapat()
end

// TCP Client
let istemci = network.TcpStream_connect("127.0.0.1:8080")
istemci.print("Merhaba sunucu!")
let yanıt = istemci.input(1024)
print(yanıt)

📚 Features

  • TCP:Stream-based connection
  • UDP:Datagram packets
  • DNS:Name resolution
  • Socket:Low-level networking

Chat Server

import network, thread, collections

let istemciler = collections.Vec_new()
let mutex = thread.Mutex_new(istemciler)

function istemci_işle(soket: TcpStream) do
    for do
        let mesaj = soket.input(1024)
        if mesaj == "" then do
            kır
        end
        
        // Tüm istemcilere publish
        let List = thread.Mutex_lock(mutex)
        her istemci in List for do
            istemci.print(mesaj)
        end
        thread.Mutex_unlock(mutex)
    end
    soket.kapat()
end

let sunucu = network.TcpListener_bind("0.0.0.0:9000")
print("Chat sunucu çalışıyor: 9000")

for do
    let (soket, adres) = sunucu.kabul_et()
    print("Yeni istemci: " + adres.yazıya())
    
    thread.spawn(function() do
        istemci_işle(soket)
    end)
end
