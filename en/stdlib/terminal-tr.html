🚀 Quick Start

import terminal

// Colors
terminal.println(terminal.red("Err!"))
terminal.println(terminal.green("Başarılı!"))
terminal.println(terminal.bold("Kalın yazı"))

// Cursor movement
terminal.clear()
terminal.cursor_move(10, 5)
terminal.print("(10, 5) konumunda")

// Progress bar
let progress = terminal.progress_bar_new(100)
her i in range(0, 101) for do
    terminal.progress_bar_set(progress, i)
    time.sleep(50)
end
