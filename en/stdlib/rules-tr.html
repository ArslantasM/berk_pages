🚀 Quick Start

import rules

// Rule engine
let engine = rules.engine_new()

// Add rules
rules.add_rule(engine, "discount_vip", function(facts) do
    if facts["customer_type"] == "VIP" then do
        facts["discount"] = 20.0
        return true
    end
    return false
end)

rules.add_rule(engine, "discount_bulk", function(facts) do
    if facts["quantity"] > 100 then do
        facts["discount"] = 15.0
        return true
    end
    return false
end)

// Execute
let facts = do
    "customer_type": "VIP",
    "quantity": 50,
    "price": 100.0
end

rules.execute(engine, facts)
print("İndirim: %" + facts["discount"].yazıya())

Loan Approval System

import rules

let loan_engine = rules.engine_new()

rules.add_rule(loan_engine, "high_income", function(facts) do
    if facts["annual_income"] > 100000 then do
        facts["score"] = facts.get("score", 0) + 30
        return true
    end
    return false
end)

rules.add_rule(loan_engine, "good_credit", function(facts) do
    if facts["credit_score"] > 700 then do
        facts["score"] = facts.get("score", 0) + 40
        return true
    end
    return false
end)

rules.add_rule(loan_engine, "low_debt", function(facts) do
    if facts["debt_ratio"] < 0.3 then do
        facts["score"] = facts.get("score", 0) + 30
        return true
    end
    return false
end)

rules.add_rule(loan_engine, "approve", function(facts) do
    if facts.get("score", 0) >= 70 then do
        facts["approved"] = true
    end else do
        facts["approved"] = false
    end
    return true
end)

// Test
let application = do
    "annual_income": 120000,
    "credit_score": 750,
    "debt_ratio": 0.25
end

rules.execute(loan_engine, application)
print("Onaylandı: " + application["approved"].yazıya())
print("Skor: " + application["score"].yazıya())
