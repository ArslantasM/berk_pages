🚀 Quick Start

import time

// Şu an
let now = time.now()
print(now.yazıya())  // "2025-01-15 14:30:45"

// Unix timestamp
let timestamp = time.unix()
print(timestamp.yazıya())  // 1705328445

// Formatting
let formatted = time.format(now, "%Y-%m-%d %H:%M:%S")
print(formatted)  // "2025-01-15 14:30:45"

// Parsing
let tarih = time.parse("2025-01-15", "%Y-%m-%d")

// Duration
let başlangıç = time.now()
// ... işlemler ...
let bitiş = time.now()
let süre = time.duration(başlangıç, bitiş)
print("Geçen süre: " + süre.saniye().yazıya() + "s")

// Sleep
time.sleep(1000)  // 1000ms = 1 saniye

import time

sınıf Stopwatch do
    let başlangıç: DateTime
    let tur_zamanları: Dizi
    
    function başlat() do
        self.başlangıç = time.now()
        self.tur_zamanları = []
    end
    
    function type() do
        self.tur_zamanları.push(time.now())
    end
    
    function durdur() -> Duration do
        return time.duration(self.başlangıç, time.now())
    end
    
    function tur_süresi(index: int) -> Duration do
        if index == 0  do
            return time.duration(self.başlangıç, self.tur_zamanları[0])
        end else do
            return time.duration(self.tur_zamanları[index-1], self.tur_zamanları[index])
        end
    end
end

import time, thread

sınıf ScheduledTask do
    let function: function
    let çalışma_zamanı: DateTime
    
    function new(f: function, çalışma_zamanı: DateTime) do
        self.function = f
        self.çalışma_zamanı = çalışma_zamanı
    end
end

sınıf Scheduler do
    let görevler: Dizi
    
    function new() do
        self.görevler = []
    end
    
    function zamanla(f: function, gecikme_ms: int) do
        let çalışma_zamanı = time.add_ms(time.now(), gecikme_ms)
        self.görevler.push(ScheduledTask.new(f, çalışma_zamanı))
    end
    
    function çalıştır() do
        loop do
            let now = time.now()
            let çalıştırılan = 0
            
            each task in self.görevler for do
                if time.before(task.çalışma_zamanı, now)  do
                    task.function()
                    çalıştırılan = çalıştırılan + 1
                end
            end
            
            // Çalıştırılan görevleri kaldır
            self.görevler = self.görevler.filtrele(function(g) do
                return time.after(g.çalışma_zamanı, now)
            end)
            
            if self.görevler.len() == 0  do
                kır
            end
            
            time.sleep(100)  // 100ms check interval
        end
    end
end

let scheduler = Scheduler.new()

scheduler.zamanla(function() do
    print("Görev 1 çalıştı: " + time.now().yazıya())
end, 1000)

scheduler.zamanla(function() do
    print("Görev 2 çalıştı: " + time.now().yazıya())
