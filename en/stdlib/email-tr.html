📧 email

SMTP/IMAP Email Client - Sending and Receiving Email

~450 lines ~30 functions SMTP/IMAP

📖 Overview

The Email module provides functions for sending and receiving emails via SMTP and IMAP protocols. It offers HTML content, file attachments, SSL/TLS security and multi-recipient support.

🔑 Key Features

  • SMTP client (sending e-mail)
  • IMAP client (reading e-mail)
  • HTML and plain text emails
  • File attachments
  • SSL/TLS and STARTTLS support
  • Multiple recipients
  • CC and BCC support
  • MIME encoding
  • Base64 attachment encoding
  • Authentication

import email

function send_welcome_email(user_email: yazı, user_name: yazı) -> Sonuç[None, Err] do
    // HTML içerik
    let html_body = "
        <html>
        <head>
            <style>
                body { font-family: Arial, sans-serif; }
                .header { background: #4CAF50; color: white; padding: 20px; }
                .content { padding: 20px; }
                .button { background: #008CBA; color: white; padding: 10px 20px;
                         text-decoration: none; border-radius: 5px; }
            </style>
        </head>
        <body>
            <div class='header'>
                <h1>Hoş Geldiniz, {}!</h1>
            </div>
            <div class='content'>
                <p>BERK platformuna katıldığınız for teşekkürler.</p>
                <p>Hesabınızı aktive etmek for aşağıdaki butona tıklayın:</p>
                <p>
                    <a href='https://example.com/activate' class='button'>
                        Hesabı Aktive Et
                    </a>
                </p>
            </div>
        </body>
        </html>
    ".formatla(user_name)
    
    // SMTP setup
    let smtp = email.smtp_new(do
        server: "smtp.gmail.com",
        port: 587,
        username: "noreply@example.com",
        password: "app-password",
        use_tls: true
    end)
    
    // Email oluştur
    let mail = email.new()
    mail.set_from("noreply@example.com")
    mail.add_to(user_email)
    mail.set_subject("Hoş Geldiniz - Hesabınızı Aktive Edin")

    // Alternative plain text
    mail.set_text_body("Hoş geldiniz! Hesabınızı aktive etmek for: https://example.com/activate")
    
    // Gönder
    smtp.send(mail)?
    smtp.disconnect()
    
    return Ok(None)
end

function main() do

💡 Example 2: Sending Bulk Email (Newsletter)

import email, fs, csv

function send_newsletter(subscribers_file: yazı, template_file: yazı) -> Sonuç[sayı, Err] do
    // Email template input
    let template = fs.read_to_string(template_file)?
    
    // SMTP bağlantısı
    let smtp = email.smtp_new(do
        server: "smtp.mailgun.org",
        port: 587,
        username: "postmaster@example.com",
        password: "mailgun-api-key",
        use_tls: true
    end)
    
    smtp.connect()?
    
    // Abone listesi input
    let subscribers = csv.read_file(subscribers_file)?
    let sent_count = 0
    
    her subscriber in subscribers for do
        let email_addr = subscriber["email"]
        let name = subscriber["name"]
        
        // Template'i kişiselleştir
        let body = template
            .replace("{name}", name)
            .replace("{email}", email_addr)
        
        // Email gönder
        let mail = email.new()
        mail.set_from("newsletter@example.com")
        mail.add_to(email_addr)
        mail.set_subject("BERK v1.0.0 Yayınlandı! 🎉")
        mail.set_html_body(body)
        
        try do
            smtp.send(mail)?
            sent_count += 1
            io.println("✓ Gönderildi: {}", email_addr)
        catch e do
            io.eprintln("✗ Başarısız: {} - {}", email_addr, e)
        end
        
        // Rate limiting (saniyede 5 email)
        thread.sleep_ms(200)
    end
    
    smtp.disconnect()
    return Ok(sent_count)
end

function main() do
    eşle send_newsletter("subscribers.csv", "newsletter_template.html") do
        Ok(count) => io.println("Newsletter gönderildi: {} subscriber", count),
        Err(e) => io.eprintln("Err: {}", e)
    end
end

💡 Example 3: Reading IMAP Email

import email

function check_new_orders() -> Sonuç[Vektör[Order], Err] do
    // IMAP bağlantısı
    let imap = email.imap_new(do
        server: "imap.gmail.com",
        port: 993,
        username: "orders@example.com",
        password: "app-password",
        use_ssl: true
    end)
    
    imap.connect()?
    
    // INBOX match
    imap.select_folder("INBOX")?
    
    // Okunmamış mesajları getir
    let messages = imap.search("UNSEEN")?
    
    let orders = []
    
    her msg_id in messages for do
        let msg = imap.fetch_message(msg_id)?
        
        // Subject'te "Order #" var mı kontrol et
        if msg.subject.contains("Order #") then do
            // Email body'from sipariş detaylarını parse et
            let order = parse_order_email(msg)?
            orders.push(order)
            
            // Mesajı okundu işaretle
            imap.mark_read(msg_id)?
            
            // Siparişler klasörüne taşı
            imap.move_message(msg_id, "Orders")?
            
            io.println("Yeni sipariş: #{}", order.order_id)
        end
    end
    
    imap.disconnect()
    return Ok(orders)
end

struct Order do
    order_id: yazı,
    customer_email: yazı,
    total_amount: kesir,
    items: Vektör[yazı]
end

function parse_order_email(msg: email.Message) -> Sonuç[Order, Err] do
    // Email body'from sipariş bilgilerini parse et
    let body = msg.text_body
    
    // Basitleştirilmiş parsing örneği
    let order_id = regex.find(body, r"Order #(\d+)")?.group(1)
    let amount_str = regex.find(body, r"Total: \$([0-9.]+)")?.group(1)
    
    return Ok(Order do
        order_id: order_id,
        customer_email: msg.from,
        total_amount: amount_str.parse_float()?,
        items: []
    end)
end

function main() do
    // Her 5 dakikada new siparişleri kontrol et
    for do
        eşle check_new_orders() do
            Ok(orders) => do
                if orders.len() > 0 then do
                    io.println("✓ {} new sipariş bulundu", orders.len())
                    // Process orders...
                end
            end,
            Err(e) => io.eprintln("Email kontrolü başarısız: {}", e)
        end
        
        thread.sleep(300_000)  // 5 dakika
    end
end

💡 Example 4: Automatic Report Email (with Attachment)

import email, time, fs

function generate_daily_report() -> Sonuç[yazı, Err] do
    // Rapor oluştur
    let report = "
        Günlük Sistem Raporu - {}
        ========================
        
        Sistem Durumu: ✓ Normal
        İşlem Sayısı: 1,234
        Kullanıcı Sayısı: 567
        Err Sayısı: 3
        
        Detaylar for ek dosyaya bakınız.
    ".formatla(time.now().format("%Y-%m-%d"))
    
    // Raporu dosyaya kaydet
    let filename = "report_{}.txt".formatla(time.now().format("%Y%m%d"))
    fs.write(filename, report)?
    
    return Ok(filename)
end

function send_daily_report() -> Sonuç[None, Err] do
    // Rapor oluştur
    let report_file = generate_daily_report()?
    
    // Email hazırla
    let mail = email.new()
    mail.set_from("system@example.com")
    mail.add_to("admin@example.com")
    mail.add_to("manager@example.com")
    mail.set_subject("Günlük Sistem Raporu - {}", time.now().format("%Y-%m-%d"))
    
    let body = "
        Merhaba,
        
        Günlük sistem raporunu ekte bulabilirsiniz.
        
        Herhangi bir sorun for lütfen sistem ekibiyle iletişime geçiniz.
        
        İyi çalışmalar,
        Otomatik Rapor Sistemi
    "
    mail.set_text_body(body)
    
    // Raporu ek as push
    mail.add_attachment("daily_report.txt", report_file)?
    
    // CSV formatında detaylı log push
    mail.add_attachment("transaction_log.csv", "logs/transactions.csv")?
    
    // SMTP ile gönder
    email.quick_send_with_attachments(
        smtp_server: "smtp.office365.com",
        port: 587,
        username: "system@example.com",
        password: "system-password",
        mail: mail
    )?
    
    io.println("Günlük rapor emaili gönderildi")
    return Ok(None)
end

function main() do
    // Her gün saat 09:00'da rapor gönder
    for do
        let now = time.now()
        
        if now.hour() == 9 and now.minute() == 0 then do
            send_daily_report()?
            thread.sleep(60_000)  // 1 dakika sleep (tekrar gönderme)
        end
        
        thread.sleep(30_000)  // 30 saniye
    end
end

  • HTML Email:Use inline CSS, external CSS is not supported
