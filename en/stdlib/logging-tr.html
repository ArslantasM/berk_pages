🚀 Quick Start

import logging

// Basit kullanım
logging.debug("Debug mesajı")
logging.info("Bilgi mesajı")
logging.warning("Uyarı!")
logging.error("Err oluştu")
logging.critical("Kritik hata!")

// Logger oluştur
let logger = logging.logger_new("app")
logging.set_level(logger, logging.INFO)

// File handler
let handler = logging.file_handler("app.log")
logging.add_handler(logger, handler)

logging.info(logger, "Uygulama başladı")
logging.error(logger, "Bağlantı hatası")

Structured Logging

import logging, json

let logger = logging.logger_new("api")

function api_request_log(method: yazı, path: yazı, status: int, duration_ms: kesir) do
    let data = do
        "method": method,
        "path": path,
        "status": status,
        "duration_ms": duration_ms,
        "timestamp": time.now().yazıya()
    end
    
    logging.info(logger, json.stringify(data))
end

api_request_log("GET", "/api/users", 200, 45.3)
