💾 sqlite - SQLite Database

Lightweight, file-based SQL database management

📋Overview

sqliteThe module provides lightweight, serverless SQL database management using the SQLite database engine. It offers comprehensive tools for creating, querying and managing file-based databases.

🔌 Connection Management

  • sqlite.connect()- Connecting to the database
  • sqlite.memory_vtı()- In-memory database
  • sqlite.close()- Close the connection
  • sqlite.connection_status()- Connection control

📝 SQL Queries

  • sqlite.query()- Running SQL query
  • sqlite.run()- Execute SQL command
  • sqlite.prepared_query()- Prepared statement
  • sqlite.batch_run()- Batch SQL execution

📊 Data Transactions

  • sqlite.add()- Adding data (INSERT)
  • sqlite.update()- Data update (UPDATE)
  • sqlite.delete()- Deleting data (DELETE)
  • sqlite.find()- Data query (SELECT)
  • sqlite.find_all()- Fetch all records

🔄 Transaction Management

  • sqlite.start_transaction()- Transaction initiation

  • sqlite.add_column()- Add columns

💡 Usage Examples

Basic Database Operations

use sqlite

// Veritabanına bağlan (dosya elseif oluşturulur)
değer vt = sqlite.bağlan("kullanıcılar.db")

// Tablo oluştur
vt.çalıştır("""
    CREATE TABLE IF NOT EXISTS kullanıcılar (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        ad TEXT NOT NULL,
        email TEXT UNIQUE NOT NULL,
        yaş INTEGER,
        kayıt_tarihi TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )
""")

print("✓ Tablo oluşturuldu")

// Veri ekleme
vt.çalıştır(
    "INSERT INTO kullanıcılar (ad, email, yaş) VALUES (?, ?, ?)",
    ["Ahmet Yılmaz", "ahmet@example.com", 28]
)

vt.çalıştır(
    "INSERT INTO kullanıcılar (ad, email, yaş) VALUES (?, ?, ?)",
    ["Ayşe Demir", "ayse@example.com", 32]
)

vt.çalıştır(
    "INSERT INTO kullanıcılar (ad, email, yaş) VALUES (?, ?, ?)",
    ["Mehmet Kaya", "mehmet@example.com", 25]
)

print("✓ Kullanıcılar eklendi")

// Tüm kullanıcıları sorgula
değer kullanıcılar = vt.sorgu("SELECT * FROM kullanıcılar")

print("\n--- Tüm Kullanıcılar ---")
for kullanıcı in kullanıcılar {
    print(f"ID: {kullanıcı['id']}, Ad: {kullanıcı['ad']}, " +
           f"Email: {kullanıcı['email']}, Yaş: {kullanıcı['yaş']}")
}

// Filtreleme
değer genç_kullanıcılar = vt.sorgu(
    "SELECT * FROM kullanıcılar WHERE yaş < ?",
    [30]
)

print(f"\n30 yaşından küçük {genç_kullanıcılar.len} kullanıcı bulundu")

// Bağlantıyı kapat
vt.kapat()
print("\n✓ Veritabanı bağlantısı kapatıldı")

Transaction Management

use sqlite

değer vt = sqlite.bağlan("banka.db")

// Hesaplar tablosu oluştur
vt.çalıştır("""
    CREATE TABLE IF NOT EXISTS hesaplar (
        hesap_no TEXT PRIMARY KEY,
        sahip TEXT NOT NULL,
        bakiye REAL NOT NULL DEFAULT 0
    )
""")

// İlk verileri push
vt.çalıştır(
    "INSERT OR REPLACE INTO hesaplar VALUES (?, ?, ?)",
    ["TR001", "Ahmet", 1000.0]
)
vt.çalıştır(
    "INSERT OR REPLACE INTO hesaplar VALUES (?, ?, ?)",
    ["TR002", "Ayşe", 500.0]
)

print("İlk bakiyeler:")
for hesap in vt.sorgu("SELECT * FROM hesaplar") {
    print(f"  {hesap['hesap_no']}: {hesap['bakiye']} TL")
}

// Para transferi (transaction ile güvenli)
function para_transfer(vt: sqlite.Bağlantı, kaynak: Dize, hedef: Dize, miktar: Gerçek) -> Sonuç<null, Err> {
    try {
        // Transaction başlat
        vt.işlem_başlat()
        
        // Kaynak hesabı kontrol et
        değer kaynak_bakiye = vt.sorgu(
            "SELECT bakiye FROM hesaplar WHERE hesap_no = ?",
            [kaynak]
        )[0]["bakiye"]
        
        if kaynak_bakiye < miktar {
            vt.geri_al()
            return Err("Yetersiz bakiye!")
        }
        
        // Kaynaktan take
        vt.çalıştır(
            "UPDATE hesaplar SET bakiye = bakiye - ? WHERE hesap_no = ?",
            [miktar, kaynak]
        )
        
        // Hedefe push
        vt.çalıştır(
            "UPDATE hesaplar SET bakiye = bakiye + ? WHERE hesap_no = ?",
            [miktar, hedef]
        )
        
        // Transaction'ı kaydet
        vt.kaydet()
        print(f"✓ Transfer başarılı: {miktar} TL ({kaynak} -> {hedef})")
        return Ok(null)
        
    } catch hata {
        vt.geri_al()
        return Err(f"Transfer başarısız: {hata}")
    }
}

// Transfer do
para_transfer(vt, "TR001", "TR002", 300.0).aç()

print("\nSon bakiyeler:")
for hesap in vt.sorgu("SELECT * FROM hesaplar") {
    print(f"  {hesap['hesap_no']}: {hesap['bakiye']} TL")
}

vt.kapat()

E-Commerce Product Management

use sqlite
use time

class ÜrünYöneticisi {
    değer vt: sqlite.Bağlantı
    
    function new(vt_dosyası: Dize) -> ÜrünYöneticisi {
        değer vt = sqlite.bağlan(vt_dosyası)
        
        // Tabloları oluştur
        vt.çalıştır("""
            CREATE TABLE IF NOT EXISTS kategoriler (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                ad TEXT UNIQUE NOT NULL
            )
        """)
        
        vt.çalıştır("""
            CREATE TABLE IF NOT EXISTS ürünler (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                ad TEXT NOT NULL,
                Description TEXT,
                fiyat REAL NOT NULL,
                stok INTEGER NOT NULL DEFAULT 0,
                kategori_id INTEGER,
                oluşturma_tarihi TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                FOREIGN KEY (kategori_id) REFERENCES kategoriler(id)
            )
        """)
        
        // İndeksler
        vt.çalıştır("CREATE INDEX IF NOT EXISTS idx_fiyat ON ürünler(fiyat)")
        vt.çalıştır("CREATE INDEX IF NOT EXISTS idx_kategori ON ürünler(kategori_id)")
        
        return ÜrünYöneticisi { vt: vt }
    }
    
    function kategori_ekle(ad: Dize) -> Tam {
        bu.vt.çalıştır(
            "INSERT OR IGNORE INTO kategoriler (ad) VALUES (?)",
            [ad]
        )
        değer sonuç = bu.vt.sorgu(
            "SELECT id FROM kategoriler WHERE ad = ?",
            [ad]
        )
        return sonuç[0]["id"]
    }
    
    function ürün_ekle(ad: Dize, Description: Dize, fiyat: Gerçek, stok: Tam, kategori_ad: Dize) {
        değer kategori_id = bu.kategori_ekle(kategori_ad)
        
        bu.vt.çalıştır(
            "INSERT INTO ürünler (ad, Description, fiyat, stok, kategori_id) VALUES (?, ?, ?, ?, ?)",
            [ad, Description, fiyat, stok, kategori_id]
        )
        
        print(f"✓ Ürün eklendi: {ad}")
    }
    
    function ürün_ara(arama_terimi: Dize) -> Liste {
        return bu.vt.sorgu(
            "SELECT ü.*, k.ad as kategori FROM ürünler ü " +
            "JOIN kategoriler k ON ü.kategori_id = k.id " +
            "WHERE ü.ad LIKE ? OR ü.Description LIKE ?",
            [f"%{arama_terimi}%", f"%{arama_terimi}%"]
        )
    }
    
    function fiyat_aralığında_ürünler(min_fiyat: Gerçek, max_fiyat: Gerçek) -> Liste {
        return bu.vt.sorgu(
            "SELECT * FROM ürünler WHERE fiyat BETWEEN ? AND ? ORDER BY fiyat",
            [min_fiyat, max_fiyat]
        )
    }
    
    function stok_güncelle(ürün_id: Tam, yeni_stok: Tam) {
        bu.vt.çalıştır(
            "UPDATE ürünler SET stok = ? WHERE id = ?",
            [yeni_stok, ürün_id]
        )
    }
    
    function düşük_stok_raporu(eşik: Tam = 10) -> Liste {
        return bu.vt.sorgu(
            "SELECT * FROM ürünler WHERE stok < ? ORDER BY stok",
            [eşik]
        )
    }
    
    function kapat() {
        bu.vt.kapat()
    }
}

// Usage
değer yönetici = ÜrünYöneticisi.new("eticaret.db")

// Ürünler push
yönetici.ürün_ekle("Laptop", "15.6 inç, 16GB RAM, 512GB SSD", 8500.0, 15, "Elektronik")
yönetici.ürün_ekle("Klavye", "Mekanik RGB Klavye", 450.0, 45, "Elektronik")
yönetici.ürün_ekle("Mouse", "Kablosuz Optik Mouse", 120.0, 80, "Elektronik")
yönetici.ürün_ekle("Kitap", "Python Programlama", 75.0, 5, "Kitap")
yönetici.ürün_ekle("Kulaklık", "Bluetooth Kulaklık", 350.0, 8, "Elektronik")

// Arama
print("\n--- 'Klavye' Araması ---")
for ürün in yönetici.ürün_ara("Klavye") {
    print(f"{ürün['ad']}: {ürün['fiyat']} TL (Stok: {ürün['stok']})")
}

// Fiyat aralığı
print("\n--- 100-500 TL Arası Ürünler ---")
for ürün in yönetici.fiyat_aralığında_ürünler(100.0, 500.0) {
    print(f"{ürün['ad']}: {ürün['fiyat']} TL")
}

// Düşük stok Warningsı
print("\n--- Düşük Stok Warningsı (< 10) ---")
for ürün in yönetici.düşük_stok_raporu(10) {
    print(f"⚠️  {ürün['ad']}: {ürün['stok']} adet kaldı!")
}

yönetici.kapat()
