🖼️ image - Image Processing

Image loading, saving, conversion and filtering operations

📦 77 Functions 📝 1770 Lines 💾 61.7 KB

📋Overview

imageThe module provides comprehensive tools for loading, saving, resizing, filtering and converting image files. Supports PNG, JPEG, GIF, BMP and WebP formats.

📂 Upload and Save Images

  • image.load()- Upload image file
  • image.png_save()- Save as PNG
  • image.jpeg_save()- Save as JPEG
  • image.webp_save()- Save as WebP
  • image.new()- Create blank image

💡 Usage Examples

Image Resizing and Thumbnail

use image
use fs

// Görüntü yükleme
değer resim = image.yükle("fotoğraf.jpg")

print(f"Orijinal len: {resim.genişlik()}x{resim.yükseklik()}")

// Yüksek kaliteli yeniden boyutlandırma
değer orta_boy = resim.yeniden_boyutlandır(
    genişlik: 1024,
    yükseklik: 768,
    kalite: image.Kalite.YÜKSEK
)

orta_boy.jpeg_kaydet("fotoğraf_orta.jpg", kalite: 90)
print("✓ Orta boy kaydedildi")

// Thumbnail oluştur (oran korunur)
değer thumbnail = resim.küçük_resim(200, 200)
thumbnail.png_kaydet("fotoğraf_thumb.png")
print("✓ Thumbnail oluşturuldu")

// Kare kırpma (profil fotoğrafı for)
değer merkez_x = resim.genişlik() / 2
değer merkez_y = resim.yükseklik() / 2
değer kare_boyut = 500

değer profil = resim.trim(
    x: merkez_x - kare_boyut / 2,
    y: merkez_y - kare_boyut / 2,
    genişlik: kare_boyut,
    yükseklik: kare_boyut
)

profil.png_kaydet("profil.png")
print("✓ Profil fotoğrafı oluşturuldu")

Image Filtering and Effects

use image

değer resim = image.yükle("manzara.jpg")

// Gri tonlama

gri.jpeg_kaydet("manzara_gri.jpg")
print("✓ Gri tonlama uygulandı")

// Sepia efekti (nostaljik görünüm)

sepia.jpeg_kaydet("manzara_sepia.jpg")
print("✓ Sepia efekti uygulandı")

// Parlaklık and kontrast ayarlama
değer parlak = resim
    .parlaklık(1.2)      // %20 daha parlak
    .kontrast(1.3)       // %30 daha fazla kontrast
    .doygunluk(1.1)      // %10 daha doygun renkler

parlak.jpeg_kaydet("manzara_parlak.jpg")
print("✓ Renk ayarlaması yapıldı")

// Bulanıklaştırma
değer bulanık = resim.bulanıklaştır(yarıçap: 5.0)
bulanık.jpeg_kaydet("manzara_bulanık.jpg")
print("✓ Blur efekti uygulandı")

// Keskinleştirme
değer keskin = resim.keskinleştir(miktar: 1.5)
keskin.jpeg_kaydet("manzara_keskin.jpg")

// Kenar algılama (edge detection)
değer kenarlar = resim.kenar_algıla(eşik: 100)

Batch Image Processing

use image
use fs

sınıf Görüntüİşlemci {
    function toplu_yeniden_boyutlandır(
        kaynak_klasör: Dize,
        hedef_klasör: Dize,
        max_genişlik: Tam,
        max_yükseklik: Tam
    ) -> Sonuç<Tam, Err> {
        // Hedef klasörü oluştur
        fs.klasör_oluştur(hedef_klasör)?
        
        // Tüm görüntü dosyalarını bul
        değer dosyalar = fs.dosyalar_listele(kaynak_klasör)?
        değer görüntü_uzantıları = [".jpg", ".jpeg", ".png", ".bmp", ".gif"]
        
        değer işlenen = 0
        
        for dosya in dosyalar {
            değer uzantı = fs.uzantı(dosya).küçük_harfe()
            
            if uzantı in görüntü_uzantıları {
                değer kaynak_yol = fs.join(kaynak_klasör, dosya)
                değer hedef_yol = fs.join(hedef_klasör, dosya)
                
                try {
                    değer resim = image.yükle(kaynak_yol)
                    
                    // Boyutu kontrol et
                    if resim.genişlik() > max_genişlik or resim.yükseklik() > max_yükseklik {
                        // Orantılı boyutlandırma
                        değer oran = (max_genişlik.gerçek() / resim.genişlik().gerçek())
                            .min(max_yükseklik.gerçek() / resim.yükseklik().gerçek())
                        
                        değer yeni_genişlik = (resim.genişlik().gerçek() * oran).tam()
                        değer yeni_yükseklik = (resim.yükseklik().gerçek() * oran).tam()
                        
                        değer yeni_resim = resim.yeniden_boyutlandır(
                            yeni_genişlik,
                            yeni_yükseklik,
                            kalite: image.Kalite.YÜKSEK
                        )
                        
                        yeni_resim.jpeg_kaydet(hedef_yol, kalite: 85)
                        print(f"✓ İşlendi: {dosya} ({resim.genişlik()}x{resim.yükseklik()} -> {yeni_genişlik}x{yeni_yükseklik})")
                        işlenen += 1
                    } yoksa {
                        // Boyut zaten uygun, sadece kopyala
                        fs.dosya_kopyala(kaynak_yol, hedef_yol)?
                        print(f"→ Kopyalandı: {dosya}")
                        işlenen += 1
                    }
                } catch hata {
                    print(f"❌ Err ({dosya}): {hata}")
                }
            }
        }
        
        return Ok(işlenen)
    }
    
    function watermark_ekle(
        kaynak: Dize,
        hedef: Dize,
        watermark_metin: Dize,
        index_of: Dize = "sağ_alt"
    ) -> Sonuç<Boş, Err> {
        değer resim = image.yükle(kaynak)?
        
        // Watermark konumunu hesapla
        değer (x, y) = eşleştir index_of {
            "sol_üst" => (10, 10),
            "sağ_üst" => (resim.genişlik() - 200, 10),
            "sol_alt" => (10, resim.yükseklik() - 40),
            "sağ_alt" => (resim.genişlik() - 200, resim.yükseklik() - 40),
            _ => (10, resim.yükseklik() - 40)
        }
        
        // Yarı şeffaf String push
        resim.metin_yaz(
            watermark_metin,
            x: x,
            y: y,
            font: "Arial",
            len: 24,
            renk: image.rgba(255, 255, 255, 180),  // Beyaz, %70 opaklık
            gölge: true
        )
        
        resim.jpeg_kaydet(hedef, kalite: 90)?
        return Ok(null)
    }
}

// Kullanım
değer işlemci = Görüntüİşlemci {}

// Toplu yeniden boyutlandırma
print("Görüntüler yeniden boyutlandırılıyor...")
değer sonuç = işlemci.toplu_yeniden_boyutlandır(
    "orijinal_resimler",
    "küçük_resimler",
    1920,
    1080
).aç()

print(f"\n✓ Toplam {sonuç} görüntü işlendi")

// Watermark push
işlemci.watermark_ekle(
    "fotoğraf.jpg",
    "fotoğraf_watermark.jpg",
    "© 2024 Benim Fotoğraflarım",
    index_of: "sağ_alt"
).aç()

print("✓ Watermark eklendi")

🎯 Good Practices

  • Use high quality filters when resizing
  • Quality value between 85-95 is optimal for JPEG recording
  • Consider WebP format to reduce file size
  • Use streaming processing on large images
  • Maintain aspect ratio for thumbnails
  • Use transparent PNG for Watermark
  • Combine filters rather than applying them repeatedly
