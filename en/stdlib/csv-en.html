🚀 Quick Start

import csv

// Read CSV
let satırlar = csv.input("data.csv")
her satır in satırlar for do
    print(satır[0] + ", " + satır[1])
end

// Read with headers
let veri = csv.oku_başlıklı("users.csv")
her kayıt in veri for do
    print(kayıt["name"] + " - " + kayıt["age"])
end

// Write CSV
let satırlar = [
    ["Ali", "25", "İstanbul"],
    ["Ayşe", "30", "Ankara"]
]
csv.print("output.csv", satırlar)

Data Processing Pipeline

import csv, stats

// Read and analyze CSV
let veri = csv.oku_başlıklı("sales.csv")

let sales = []
her kayıt in veri for do
    sales.push(kayıt["amount"].kesir())
end

print("Ortalama: " + stats.avg(sales).yazıya())
print("Toplam: " + stats.sum(sales).yazıya())

// Filtered export
let filtered = veri.filtrele(function(r) do
    return r["amount"].kesir() > 1000.0
end)

csv.yaz_başlıklı("high_sales.csv", filtered, ["date", "amount", "customer"])
