🚀 Quick Start

import sqlite

// Veritabanı aç
let db = sqlite.open("app.db")

// Tablo oluştur
sqlite.exec(db, "CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    email TEXT UNIQUE,
    age INTEGER
)")

// Insert
sqlite.exec(db, "INSERT INTO users (name, email, age) VALUES ('Ali', 'ali@test.com', 25)")

// Prepared statement (SQL injection koruması)
let stmt = sqlite.prepare(db, "INSERT INTO users (name, email, age) VALUES (?, ?, ?)")
sqlite.bind(stmt, ["Ayşe", "ayse@test.com", 30])
sqlite.step(stmt)
sqlite.finalize(stmt)

// Query
let sonuçlar = sqlite.query(db, "SELECT * FROM users WHERE age > 20")
her satır in sonuçlar for do
    print(satır["name"] + " - " + satır["email"])
end

// Transaction
sqlite.begin(db)
try do
    sqlite.exec(db, "UPDATE users SET age = age + 1")
    sqlite.commit(db)
end catch hata do
    sqlite.rollback(db)
    print("Err: " + hata.mesaj())
end

sqlite.close(db)

📚 Features

  • SQLite 3.x embedded database
  • Full SQL support (SELECT, INSERT, UPDATE, DELETE, JOIN, etc.)
  • Prepared statements (SQL injection prevention)
  • Transactions (ACID properties)
  • Type safety
  • connection pooling

User Management System

import sqlite

sınıf UserDB do
    let db: Database
    
    function new(dosya_yolu: yazı) do
        self.db = sqlite.open(dosya_yolu)
        self.init_schema()
    end
    
    function init_schema() do
        sqlite.exec(self.db, "
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                email TEXT UNIQUE NOT NULL,
                password_hash TEXT NOT NULL,
                created_at INTEGER DEFAULT (strftime('%s', 'now'))
            )
        ")
    end
    
    function kullanıcı_ekle(username: yazı, email: yazı, password_hash: yazı) -> Sonuç do
        let stmt = sqlite.prepare(self.db,
            "INSERT INTO users (username, email, password_hash) VALUES (?, ?, ?)")
        
        sqlite.bind(stmt, [username, email, password_hash])
        
        if sqlite.step(stmt) then do
            let id = sqlite.last_insert_id(self.db)
            sqlite.finalize(stmt)
            return Ok(id)
        end else do
            sqlite.finalize(stmt)
            return Err("Kullanıcı eklenemedi")
        end
    end
    
    function kullanıcı_bul(username: yazı) -> Seçenek do
        let stmt = sqlite.prepare(self.db,
            "SELECT * FROM users WHERE username = ?")
        sqlite.bind(stmt, [username])
        
        if sqlite.step(stmt) then do
            let user = do
                "id": sqlite.column_int(stmt, 0),
                "username": sqlite.column_text(stmt, 1),
                "email": sqlite.column_text(stmt, 2),
                "password_hash": sqlite.column_text(stmt, 3),
                "created_at": sqlite.column_int(stmt, 4)
            end
            sqlite.finalize(stmt)
            return Some(user)
        end else do
            sqlite.finalize(stmt)
            return Hiç
        end
    end
    
    function tüm_kullanıcılar() -> Dizi do
        return sqlite.query(self.db, "SELECT id, username, email FROM users")
    end
    
    function kullanıcı_sil(id: int) -> bool do
        let stmt = sqlite.prepare(self.db, "DELETE FROM users WHERE id = ?")
        sqlite.bind(stmt, [id])
        let başarılı = sqlite.step(stmt)
        sqlite.finalize(stmt)
        return başarılı
    end
end

// Kullanım
let user_db = UserDB.new("users.db")
user_db.kullanıcı_ekle("berk_dev", "berk@example.com", "hash_buraya")

let kullanıcı = user_db.kullanıcı_bul("berk_dev")
if kullanıcı then do
    print("Kullanıcı bulundu: " + kullanıcı.değer()["email"])
end

Batch Insert with Transaction

import sqlite, csv

function csv_import(db: Database, csv_dosya: yazı, tablo: yazı) do
    let satırlar = csv.input(csv_dosya)
    
    sqlite.begin(db)
    
    try do
        let stmt = sqlite.prepare(db,
            "INSERT INTO " + tablo + " VALUES (?, ?, ?)")
        
        her satır in satırlar for do
            sqlite.bind(stmt, satır)
            sqlite.step(stmt)
            sqlite.reset(stmt)
        end
        
        sqlite.finalize(stmt)
        sqlite.commit(db)
        print("Import başarılı: " + satırlar.len().yazıya() + " satır")
    end catch hata do
        sqlite.rollback(db)
        print("Import başarısız: " + hata.mesaj())
    end
end

let db = sqlite.open("data.db")
csv_import(db, "users.csv", "users")
