    Overview

    The metrics module provides commonly used metrics to evaluate the performance of machine learning models.

    Classification Metrics

    Accuracy

    use ai/metrics;

let y_true = [0, 1, 1, 0, 1, 0];
let y_pred = [0, 1, 0, 0, 1, 0];

let acc = accuracy(y_true, y_pred);
print("Doğruluk: {:.2f}%", acc * 100.0);
// Çıktı: Doğruluk: 83.33%

    Precision, Recall, F1

    use ai/metrics;

let precision = precision_score(y_true, y_pred);
let recall = recall_score(y_true, y_pred);
let f1 = f1_score(y_true, y_pred);

print("Precision: {:.4f}", precision);
print("Recall: {:.4f}", recall);
print("F1 Score: {:.4f}", f1);

    Confusion Matrix

    use ai/metrics;

let cm = confusion_matrix(y_true, y_pred, num_classes: 2);

print("Confusion Matrix:");
print("{:?}", cm);
// [[TP, FP],
//  [FN, TN]]

// Sınıf bazında metrikler
her i in range(num_classes) for do
    let precision_i = cm[i][i] / cm[i].topla();
    let recall_i = cm[i][i] / cm.column(i).topla();
    print("Sınıf {}: P={:.3f}, R={:.3f}", i, precision_i, recall_i);
end

    Multi-Class Metrics

    Macro/Micro/Weighted Averages

    use ai/metrics;

let y_true = [0, 1, 2, 0, 1, 2];
let y_pred = [0, 2, 1, 0, 0, 1];

// Macro average (her sınıf eşit ağırlık)
let f1_macro = f1_score(y_true, y_pred, average: "macro");

// Micro average (tüm örnekler eşit)
let f1_micro = f1_score(y_true, y_pred, average: "micro");

// Weighted (sınıf dağılımına göre)
let f1_weighted = f1_score(y_true, y_pred, average: "weighted");

print("F1 Macro: {:.4f}", f1_macro);
print("F1 Micro: {:.4f}", f1_micro);
print("F1 Weighted: {:.4f}", f1_weighted);

    Regression Metrics

    MSE, RMSE, MAE

    use ai/metrics;

let y_true = [3.0, -0.5, 2.0, 7.0];
let y_pred = [2.5, 0.0, 2.1, 7.8];

let mse = mean_squared_error(y_true, y_pred);
let rmse = mse.sqrt();
let mae = mean_absolute_error(y_true, y_pred);

print("MSE: {:.4f}", mse);
print("RMSE: {:.4f}", rmse);
print("MAE: {:.4f}", mae);

    Custom Metrics

    Create Your Own Metric

    use ai/metrics;

function custom_accuracy(y_true: Tensor, y_pred: Tensor) -> float do
    let threshold = 0.7;
    let high_conf_mask = y_pred.gt(threshold);
    let correct = y_true.eq(y_pred.round()).and(high_conf_mask);
    
    return correct.topla() as float / y_true.len() as float;
end

// Kullanım
let acc = custom_accuracy(y_true, y_pred);
print("High-confidence accuracy: {:.2f}%", acc * 100.0);

    Batch Evaluation

    Batch Evaluation with DataLoader

    use ai/metrics;
use ai/train;

function evaluate(model: Model, test_loader: DataLoader) -> Metrics do
    model.değerlendirme();
    
    let tüm_preds = [];
    let tüm_targets = [];
    
    her (x, y) in test_loader for do
        let pred = model.ileri(x);
        tüm_preds.extend(pred.argmax(dim: 1));
        tüm_targets.extend(y);
    end
    
    return Metrics {
        accuracy: accuracy(tüm_targets, tüm_preds),
        f1: f1_score(tüm_targets, tüm_preds, average: "weighted"),
        precision: precision_score(tüm_targets, tüm_preds, average: "weighted"),
        recall: recall_score(tüm_targets, tüm_preds, average: "weighted")
    };
end
