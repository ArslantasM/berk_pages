📚 Assertions

  • assert_eq:Equality check
  • assert_ne:not equal
  • assert_true/false:boolean control
  • assert_close:Approximate equality (float)
  • assert_panic:Error throw control
  • assert_ok/err:Result type control

Comprehensive Test Suite

import testing, collections

// Test edilen fonksiyonlar
function stack_new() -> Stack do
    return do içerik: [] end
end

function stack_push(s: Stack, value: herhangi) do
    s.içerik.push(value)
end

function stack_pop(s: Stack) -> Seçenek do
    if s.içerik.len() > 0 then do
        return Some(s.içerik.pop())
    end else do
        return Hiç
    end
end

// Test suite
test_grubu "Stack Veri Yapısı" do
    test "new stack null olmalı" do
        let s = stack_new()
        testing.assert_eq(s.içerik.len(), 0)
    end
    
    test "push eleman eklemeli" do
        let s = stack_new()
        stack_push(s, 10)
        stack_push(s, 20)
        testing.assert_eq(s.içerik.len(), 2)
    end
    
    test "pop end elemanı çıkarmalı" do
        let s = stack_new()
        stack_push(s, 10)
        stack_push(s, 20)
        
        let popped = stack_pop(s)
        testing.assert_some(popped)
        testing.assert_eq(popped.pop(), 20)
        testing.assert_eq(s.içerik.len(), 1)
    end
    
    test "null stack'ten pop Hiç dönmeli" do
        let s = stack_new()
        let popped = stack_pop(s)
        testing.assert_none(popped)
    end
    
    test "LIFO sırası korunmalı" do
        let s = stack_new()
        stack_push(s, 1)
        stack_push(s, 2)
        stack_push(s, 3)
        
        testing.assert_eq(stack_pop(s).pop(), 3)
        testing.assert_eq(stack_pop(s).pop(), 2)
        testing.assert_eq(stack_pop(s).pop(), 1)
    end
end

// Tüm testleri çalıştır
testing.run_all()

Property-Based Testing

import testing, random

// Property: reverse(reverse(list)) == list
test "reverse tersine çevirme idempotent" do
    her i in range(0, 100) for do
        // Random List oluştur
        let len = random.range(0, 50)
        let List = []
        
        her j in range(0, len) for do
            List.push(random.range(0, 1000))
        end
        
        // Property kontrolü
        let ters1 = List.reverse()
        let ters2 = ters1.reverse()
        
        testing.assert_eq(List, ters2)
    end
end

// Property: sort(list) sıralı olmalı
test "sort sıralama doğruluğu" do
    her i in range(0, 100) for do
        let len = random.range(1, 50)
        let List = []
        
        her j in range(0, len) for do
            List.push(random.range(0, 1000))
        end
        
        let sıralı = List.sort()
        
        // Sıralı mı kontrol
        her k in range(0, sıralı.len() - 1) for do
            testing.assert_true(sıralı[k] <= sıralı[k + 1])
        end
    end
end
