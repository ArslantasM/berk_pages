🚀 Quick Start

import crypto

// Hash functions
let veri = "Merhaba BERK"
let md5 = crypto.md5(veri)
let sha256 = crypto.sha256(veri)
let sha512 = crypto.sha512(veri)

// HMAC (keyed hash)
let anahtar = "secret_key"
let hmac = crypto.hmac_sha256(veri, anahtar)

// AES encryption (symmetric)
let aes_key = crypto.aes_key_generate(256)
let şifreli = crypto.aes_encrypt(veri, aes_key)
let çözülmüş = crypto.aes_decrypt(şifreli, aes_key)

// RSA encryption (asymmetric)
let (public_key, private_key) = crypto.rsa_keypair_generate(2048)
let encrypted = crypto.rsa_encrypt(veri, public_key)
let decrypted = crypto.rsa_decrypt(encrypted, private_key)

// Digital signature
let imza = crypto.rsa_sign(veri, private_key)
let geçerli = crypto.rsa_verify(veri, imza, public_key)

// Random bytes (secure)
let random = crypto.random_bytes(32)

📚 Cryptography Algorithms

  • Hash:MD5, SHA-1, SHA-256, SHA-512, SHA-3, BLAKE2
  • Symmetric:AES-128/192/256 (CBC, GCM), ChaCha20
  • Asymmetric:RSA-2048/4096, ECC (secp256k1, ed25519)
  • HMAC:HMAC-SHA256, HMAC-SHA512
  • Key Derivation:PBKDF2, Argon2, scrypt
  • Random:CSPRNG (cryptographically secure)

💡 Examples

Password Hashing (Argon2)

import crypto

function şifre_hash(şifre: yazı) -> yazı do
    // Argon2id (memory-hard, GPU resistant)
    let salt = crypto.random_bytes(16)
    let hash = crypto.argon2id(şifre, salt, do
        zaman_maliyeti: 3,
        bellek_maliyeti: 65536,  // 64 MB
        parallellik: 4
    end)
    
    // Combine salt + hash
    return crypto.encode_base64(salt) + "$" + crypto.encode_base64(hash)
end

function şifre_doğrula(şifre: yazı, hash: yazı) -> bool do
    let parçalar = hash.split("$")
    let salt = crypto.decode_base64(parçalar[0])
    let beklenen = crypto.decode_base64(parçalar[1])
    
    let hesaplanan = crypto.argon2id(şifre, salt, do
        zaman_maliyeti: 3,
        bellek_maliyeti: 65536,
        parallellik: 4
    end)
    
    return crypto.constant_time_compare(hesaplanan, beklenen)
end

// Usage
let hash = şifre_hash("güçlü_şifre_123")
print("Hash: " + hash)

let geçerli = şifre_doğrula("güçlü_şifre_123", hash)
print("Doğrulama: " + geçerli.yazıya())  // true

Secure Message Exchange (RSA + AES Hybrid)

import crypto, json

function şifreli_mesaj_gönder(mesaj: yazı, alıcı_public_key: RSAPublicKey) -> yazı do
    // 1. Generate random AES key
    let aes_key = crypto.aes_key_generate(256)
    
    // 2. Encrypt message with AES (fast)
    let şifreli_mesaj = crypto.aes_gcm_encrypt(mesaj, aes_key)
    
    // 3. Encrypt AES key with RSA (secure)
    let şifreli_key = crypto.rsa_encrypt(aes_key, alıcı_public_key)
    
    // 4. Create package
    let paket = do
        "encrypted_message": crypto.encode_base64(şifreli_mesaj),
        "encrypted_key": crypto.encode_base64(şifreli_key)
    end
    
    return json.stringify(paket)
end

function şifreli_mesaj_al(paket_json: yazı, private_key: RSAPrivateKey) -> yazı do
    // 1. Parse package
    let paket = json.parse(paket_json)
    let şifreli_mesaj = crypto.decode_base64(paket["encrypted_message"])
    let şifreli_key = crypto.decode_base64(paket["encrypted_key"])
    
    // 2. Decrypt AES key
    let aes_key = crypto.rsa_decrypt(şifreli_key, private_key)
    
    // 3. Decrypt message
    let mesaj = crypto.aes_gcm_decrypt(şifreli_mesaj, aes_key)
    
    return mesaj
end

// Usage
let (alice_pub, alice_priv) = crypto.rsa_keypair_generate(2048)
let (bob_pub, bob_priv) = crypto.rsa_keypair_generate(2048)

// Alice -> Bob
let paket = şifreli_mesaj_gönder("Gizli mesaj!", bob_pub)
let mesaj = şifreli_mesaj_al(paket, bob_priv)
print(mesaj)  // "Gizli mesaj!"
