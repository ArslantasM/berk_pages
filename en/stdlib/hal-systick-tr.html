⏱️ state::systick

System Tick Timer - Millisecond Timebase for RTOS and Delays

~240 lines ~10 functions ARM Cortex-M

📖 Overview

SysTick is a 24-bit system timer available on ARM Cortex-M architectures. It is generally used to generate 1ms tick interrupts and forms the basis of RTOS (Real-Time Operating System) schedulers. Also used for ms-based delay functions and uptime counters.

🔑 Key Features

  • 24-bit down counter (0xFFFFFF)
  • Configurable tick rate (usually 1ms)
  • Exception/interrupt generation
  • Sourced from CPU clock or external clock
  • Low overhead timing
  • RTOS scheduler timebase
  • Millisecond delay functions
  • uptime counter

🚀 Quick Start

import hal::systick, hal::core

let g_tick_count: u64 = 0

// SysTick interrupt handler
@interrupt
function SysTick_Handler() do
    g_tick_count += 1
end

function main() do
    // 72 MHz CPU, 1ms tick
    systick.init(72000000 / 1000)  // 1ms period
    systick.enable_interrupt()
    systick.enable()
    
    // Now delay_ms() works
    for do
        io.println("Uptime: " + g_tick_count.yazıya() + " ms")
        systick.delay_ms(1000)
    end
end

💡 Example 1: Millisecond Delay Implementation

Blocking delay and non-blocking timeout

import hal::systick

let g_ticks: sayacı64 = 0

@interrupt
function SysTick_Handler() do
    g_ticks += 1
end

// Blocking delay
function delay_ms(ms: sayı) do
    let start = g_ticks
    for (g_ticks - start) < ms.to_u64() for do
        // Wait
    end
end

// Non-blocking timeout
struct Timeout do
    start: sayı64,
    duration: sayı64
end

function timeout_start(ms: sayı) -> Timeout do
    return Timeout do
        start: g_ticks,
        duration: ms.to_u64()
    end
end

function timeout_expired(timeout: Timeout) -> mantık do
    return (g_ticks - timeout.start) >= timeout.duration
end

// Kullanım örneği
function main_loop() do
    let timeout = timeout_start(5000)  // 5 saniye
    
    for do
        // Non-blocking işlemler
        process_data()
        
        if timeout_expired(timeout) then do
            io.println("5 saniye geçti!")
            timeout = timeout_start(5000)  // Reset
        end
    end
end

💡 Example 2: Simple RTOS Scheduler

Cooperative multitasking with SysTick

import hal::systick

struct Task do
    function: Fonksiyon(),
    period_ms: sayı,
    son_calistirma: sayı64
end

let g_ticks: sayı64 = 0
let g_tasks: Dizi[Task] = []

@interrupt
function SysTick_Handler() do
    g_ticks += 1
end

function task_register(func: Fonksiyon(), period_ms: sayı) do
    g_tasks.push(Task do
        function: func,
        period_ms: period_ms,
        son_calistirma: 0
    end)
end

function scheduler_run() do
    for do
        her task in g_tasks for do
            let elapsed = g_ticks - task.son_calistirma
            
            if elapsed >= task.period_ms.to_u64() then do
                // Task zamanı geldi
                task.function()
                task.son_calistirma = g_ticks
            end
        end
    end
end

// Örnek tasklar
function led_blink_task() do
    gpio.pin_toggle(LED_PIN)
end

function sensor_read_task() do
    let temp = sensor.read_temperature()
    io.println("Sıcaklık: " + temp.yazıya())
end

function main() do
    systick.init(72000000 / 1000)  // 1ms
    systick.enable_interrupt()
    systick.enable()
    
    // Register tasks
    task_register(led_blink_task, 500)      // 500ms period
    task_register(sensor_read_task, 1000)   // 1000ms period
    
    // Run scheduler
    scheduler_run()
end

💡 Example 3: Performance Profiling

Function execution time measurement

import hal::systick

function micros() -> sayı64 do
    // SysTick counts down from reload value
    let reload = systick.get_reload()
    let current = systick.get_value()
    
    // Calculate microseconds
    let ticks_elapsed = reload - current
    let us_per_tick = 1000 / reload  // Assuming 1ms period
    
    return (g_ticks * 1000) + (ticks_elapsed * us_per_tick).to_u64()
end

function profile_start() -> sayı64 do
    return micros()
end

function profile_end(start: sayı64) -> sayı64 do
    return micros() - start
end

// Kullanım
function slow_function() do
    for i in 0..1000 for do
        math.sqrt(i.kesir())
    end
end

function main() do
    systick.init(72000)  // 1ms
    systick.enable()
    
    let start = profile_start()
    slow_function()
    let duration = profile_end(start)
    
    io.println("Süre: " + duration.yazıya() + " μs")
end

💡 Best Practices

  • Tick ​​rate:1ms industry standard for RTOS
  • Interrupt priority:SysTick is usually the lowest priority (15)
  • Handler duration:Interrupt handler should be kept minimal (<10μs)
  • Overflow handling:Prevent overflow using 64-bit counter
  • Power optimization:Consider tickless idle mode (RTOS)
  • Debug freeze:You can stop SysTick during debug

⚠️ Important Notes

  • 24-bit counter: Maximum reload value 0xFFFFFF (16,777,215)
  • reload = 72000 for 1ms at 72 MHz
  • Long processing in interrupt handler (context switch overhead)
  • Interrupt disable may be required for atomic operations
