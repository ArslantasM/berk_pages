🚀 Fast Getting Started

Simple Usage

// Mathematics
use math

variable sonuç = math_sin(math_PI / 2.0)  // 1.0
variable faktöriyel = math_factorial(5)     // 120

Iterators

use iter

variable sayılar = [1, 2, 3, 4, 5]
variable kareler = iter_map(sayılar, function(x) do return x * x end)
variable sum = iter_sum(kareler)  // 55

Linear Algebra

kullan linalg

// 3×3 birim matris
variable I = linalg_birim_mat(3)

// Matris çarpımı
variable A = [1.0, 2.0, 3.0, 4.0]  // 2×2
variable B = [5.0, 6.0, 7.0, 8.0]
variable C = linalg_mat_çarp(A, B, 2, 2, 2)

Statistics

kullan stats

variable data = [1.5, 2.3, 3.1, 4.8, 5.2]
variable ortalama = stats_ortalama(data, 5)       // 3.38
variable std_dev = stats_standart_sapma(data, 5)  // 1.52
variable medyan = stats_medyan(data, 5)           // 3.1

optimization

use optim

// f(x) = x² - 4x + 4 fonksiyonunun minimumunu bul
function f(x: float): float do
    return x * x - 4.0 * x + 4.0
end

variable minimum = optim_golden_section(f, 0.0, 5.0, 0.001)
// minimum ≈ 2.0

Numerical Simulation

// Monte Carlo π tahmini
kullan random

variable içerde = 0
variable toplam = 1000000

variable rng = random_xoshiro_seed(12345)

variable i = 0
while (i < toplam) do
    variable x = random_xoshiro_next_float(rng)
    variable y = random_xoshiro_next_float(rng)
    
    if (x*x + y*y < 1.0) ise do
        içerde = içerde + 1
    end
    
    i = i + 1
end

variable pi_tahmini = 4.0 * içerde / toplam
print("π ≈ " + pi_tahmini)  // ~3.14159

Optimization Problems

// Gradyan inişi ile function minimizasyonu
use optim
use linalg

// f(x, y) = x² + y² - 2x - 4y + 5
function gradyan(pos: [float], n: tam_sayı): [float] do
    variable grad = [2.0 * pos[0] - 2.0, 2.0 * pos[1] - 4.0]
    return grad
end

variable Getting Started = [0.0, 0.0]
variable minimum = optim_gradient_descent(gradyan, Getting Started, 2, 0.1, 100, 0.001)

print("Minimum nokta: (" + minimum[0] + ", " + minimum[1] + ")")
// Beklenen: (1.0, 2.0)
