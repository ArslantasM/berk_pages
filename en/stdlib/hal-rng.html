🎲 state::rng

Hardware True Random Number Generator

~380 lines ~14 functions STM32F4/F7/H7

📖 Overview

The RNG module generates hardware-based true random numbers. It provides cryptographic quality random values ​​using an analog noise source. IoT security is critical for UUID generation, cryptographic key generation and Monte Carlo simulations.

🔑 Key Features

  • True Random:Analog noise source, not pseudo-random
  • 32-bit Output:Full 32-bit random value on each read
  • FIPS 140-2 Compliant:Compliant with cryptographic standards
  • Single Cycle:New random value in 40 CPU cycles
  • Clock Error Detection:Clock problem is detected
  • Seed Error Detection:Entropy error is detected
  • Interrupt Support:Data ready interrupt
  • DMA Compatible:Continuous random stream

Problem: Creating unique identifier (IoT device ID)

import hal::rng

struct UUID do
    data: Dizi<sayı8> = Dizi.new(16)
end

function uuid_v4_generate() -> UUID do
    let uuid = UUID do end
    
    // Generate 128 bits (16 bytes) of random data
    for i in 0..4 do
        let rand32 = rng.get_random()
        
        // Split into 4 bytes
        uuid.data[i*4 + 0] = (rand32 >> 24) & 0xFF kayıt sayı8
        uuid.data[i*4 + 1] = (rand32 >> 16) & 0xFF kayıt sayı8
        uuid.data[i*4 + 2] = (rand32 >> 8) & 0xFF kayıt sayı8
        uuid.data[i*4 + 3] = rand32 & 0xFF kayıt sayı8
    end
    
    // Set version (4) and variant (RFC 4122)
    uuid.data[6] = (uuid.data[6] & 0x0F) | 0x40  // Version 4
    uuid.data[8] = (uuid.data[8] & 0x3F) | 0x80  // Variant RFC4122
    
    return uuid
end

function uuid_to_string(uuid: UUID) -> Metin do
    // Format: xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx
    let hex = ""
    for i in 0..16 do
        hex = hex + uuid.data[i].onaltılıya(2)
        if (i == 3 or i == 5 or i == 7 or i == 9) do
            hex = hex + "-"
        end
    end
    return hex
end

function main() do
    rng.init()
    
    for i in 0..5 do
        let uuid = uuid_v4_generate()
        io.yazdir_satır("UUID: " + uuid_to_string(uuid))
    end
end

💡 Example 2: Cryptographic Key Generation

Problem: Generating 256-bit random key for AES-256

import hal::rng, hal::crypto

const AES256_KEY_SIZE: sayı = 32  // 256 bit = 32 bytes

struct CryptoKey do
    data: Dizi<sayı8>,
    size: sayı
end

function generate_aes256_key() -> CryptoKey do
    let key = CryptoKey do
        data: Dizi.new(AES256_KEY_SIZE),
        size: AES256_KEY_SIZE
    end
    
    // Generate 256 bits of cryptographic random data
    for i in 0..8 do  // 8 * 32 bits = 256 bits
        let rand = rng.get_random()
        
        if rng.get_flag(rng.FLAG_CECS) or rng.get_flag(rng.FLAG_SECS) do
            io.yazdir_satır("RNG ERROR: Clock or seed error!")
            return key  // Invalid key
        end
        
        key.data[i*4 + 0] = (rand >> 24) & 0xFF kayıt sayı8
        key.data[i*4 + 1] = (rand >> 16) & 0xFF kayıt sayı8
        key.data[i*4 + 2] = (rand >> 8) & 0xFF kayıt sayı8
        key.data[i*4 + 3] = rand & 0xFF kayıt sayı8
    end
    
    return key
end

function key_to_hex(key: CryptoKey) -> Metin do
    let hex = ""
    for byte in key.data do
        hex = hex + byte.onaltılıya(2)
    end
    return hex
end

function main() do
    rng.init()
    
    io.yazdir_satır("Generating AES-256 key...")
    let key = generate_aes256_key()
    io.yazdir_satır("Key: " + key_to_hex(key))
    
    // Use key for AES encryption
    crypto.aes_init(crypto.AES256, key.data)
end

💡 Example 3: Random Number Distribution (Dice Roll)

Problem: Dice roll simulation with uniform distribution

import hal::rng

// Random in range [min, max]
function random_range(min: sayı, max: sayı) -> sayı do
    let range = max - min + 1
    let rand = rng.get_random()
    
    // Avoid modulo bias using rejection sampling
    let limit = 0xFFFF_FFFF - (0xFFFF_FFFF % range)
    
    for do
        if rand < limit do
            return min + (rand % range)
        end
        rand = rng.get_random()
    end
end

function dice_roll() -> sayı do
    return random_range(1, 6)
end

function test_distribution() do
    let histogram: Dizi<sayı> = Dizi.new(6, 0)
    const ROLLS: sayı = 10000
    
    for i in 0..ROLLS do
        let roll = dice_roll()
        histogram[roll - 1] = histogram[roll - 1] + 1
    end
    
    io.yazdir_satır("Dice roll distribution (" + ROLLS.metne() + " rolls):")
    for i in 0..6 do
        let face = i + 1
        let count = histogram[i]
        let percent = (count kayıt kesir / ROLLS kayıt kesir) * 100.0
        
        io.yazdir_satır("  [" + face.metne() + "]: " + count.metne() +
                        " (" + percent.metne() + "%)")
    end
end

function main() do
    rng.init()
    test_distribution()
end

Conclusion:Uniform distribution ~16.67% per face (ideal: 16.666...%).

⚠️ Important Notes:
  • Clock Required:48 MHz clock required for RNG (from PLL)
  • Error Handling:CECS/SECS flags should be checked
  • First Read:~40 cycle waiting for the first random value
  • Modulo Bias:Use rejection sampling in range operations

✅ Best Practices

  • ✅ Error check:Check CECS/SECS flags
  • ✅ Seeding:Use RNG for PRNG seed
  • ✅ Key generation:RNG is required for crypto keys
  • ✅ Rejection sampling:Avoid modulo bias
  • ❌ Not predictable:RNG is not predictable even in debug
  • ❌ Clock dependency:It won't work without a 48 MHz clock
