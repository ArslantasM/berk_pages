🚀 Quick Start

import image

// Load image
let img = image.load("photo.jpg")

// Resize
let thumbnail = image.resize(img, 200, 200)

// Filters
let blurred = image.gaussian_blur(img, 5.0)
let edges = image.sobel_edge_detect(img)
let grayscale = image.to_grayscale(img)

// Save

Batch Image Processor

import image, fs

function batch_resize(dizin: yazı, genişlik: int, yükseklik: int) do
    let dosyalar = fs.dizin_listele(dizin)
    
    her dosya in dosyalar for do
        if dosya.bitiyor_mu(".jpg") or dosya.bitiyor_mu(".png") then do
            let tam_yol = fs.path_join(dizin, dosya)
            let img = image.load(tam_yol)
            let resized = image.resize(img, genişlik, yükseklik)
            
            let yeni_dosya = "resized_" + dosya
            let yeni_yol = fs.path_join(dizin, yeni_dosya)
            image.save(resized, yeni_yol)
            
            print("İşlendi: " + dosya)
        end
    end
end

batch_resize("./photos", 800, 600)
