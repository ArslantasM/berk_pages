🚀 Quick Start

import encoding

// Base64
let encoded = encoding.base64_encode("Merhaba BERK")
print(encoded)  // "TWVyaGFiYSBCRVJL"
let decoded = encoding.base64_decode(encoded)
print(decoded)  // "Merhaba BERK"

// Hex
let hex = encoding.hex_encode([0x42, 0x45, 0x52, 0x4B])
print(hex)  // "4245524B"

// URL encoding
let url = encoding.url_encode("merhaba dünya?")
print(url)  // "merhaba%20d%C3%BCnya%3F"

Data Transfer Encoding

import encoding, json, crypto

function güvenli_veri_gönder(veri: herhangi, anahtar: yazı) -> yazı do
    // 1. JSON serialize
    let json_str = json.stringify(veri)
    
    // 2. Encrypt
    let şifreli = crypto.aes_encrypt(json_str, anahtar)
    
    // 3. Base64 encode (binary -> text)
    let encoded = encoding.base64_encode(şifreli)
    
    return encoded
end

function güvenli_veri_al(encoded: yazı, anahtar: yazı) -> herhangi do
    // 1. Base64 decode
    let şifreli = encoding.base64_decode(encoded)
    
    // 2. Decrypt
    let json_str = crypto.aes_decrypt(şifreli, anahtar)
    
    // 3. JSON parse
    return json.parse(json_str)
end
