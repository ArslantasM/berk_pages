🚀 Quick Start

import iter

// map
let sayılar = [1, 2, 3, 4, 5]
let kareler = iter.map(sayılar, function(x) do return x * x end)
// [1, 4, 9, 16, 25]

// filter
let çiftler = iter.filter(sayılar, function(x) do return x % 2 == 0 end)
// [2, 4]

// reduce
let sum = iter.reduce(sayılar, 0, function(acc, x) do return acc + x end)
// 15

// chain
let combined = iter.chain([1, 2], [3, 4], [5, 6])
// [1, 2, 3, 4, 5, 6]

// zip
let isimler = ["Ali", "Ayşe", "Mehmet"]
let yaşlar = [25, 30, 35]
let paired = iter.zip(isimler, yaşlar)
// [("Ali", 25), ("Ayşe", 30), ("Mehmet", 35)]

💡 Examples

Functional Data Pipeline

import iter

// Word frequency (functional style)
let String = "the quick brown fox jumps over the lazy dog"

let kelime_frekansı = String
    .split(" ")
    |> iter.map(function(w) do return string.to_lowercase(w) end)
    |> iter.filter(function(w) do return w.len() > 3 end)
    |> iter.group_by(function(w) do return w end)
    |> iter.map(function((kelime, grup)) do
        return (kelime, grup.len())
    end)
    |> iter.collect()

print(kelime_frekansı)
