BERK Programlama Dili

operators

Various operators are used in BERK for mathematical operations, comparisons and logical operations.

Arithmetic Operators

Basic Operations

let a = 10
let b = 3

print(a + b)   // 13 - Toplama
print(a - b)   // 7  - Çıkarma
print(a * b)   // 30 - Çarpma
print(a / b)   // 3  - Bölme (int)
print(a % b)   // 1  - Mod (kalan)
print(a ** b)  // 1000 - Üs alma

Decimal Division

let x = 10.0
let y = 3.0

print(x / y)   // 3.333333 - Ondalık bölme

Unary Operators

let n = 5

print(+n)  // 5  - Pozitif işareti
print(-n)  // -5 - Negatif işareti
print(++n) // 6  - Ön artırma
print(n++) // 6  - Son artırma (7 olur)
print(--n) // 6  - Ön azaltma
print(n--) // 6  - Son azaltma (5 olur)

Comparison Operators

let a = 10
let b = 5

print(a == b)  // false - Eşit mi
print(a != b)  // true  - Eşit not mi
print(a > b)   // true  - Büyük mü
print(a < b)   // false - Küçük mü
print(a >= b)  // true  - Büyük eşit mi
print(a <= b)  // false - Küçük eşit mi

Logical Operators

let x = true
let y = false

print(x && y)  // false - VE (AND)
print(x || y)  // true  - VEYA (OR)
print(!x)      // false - DEĞİL (NOT)

Short-Circuit Evaluation

// İkinci ifade değerlendirilmez
if (x || çağır_fonksiyon()) {
    // x true then çağır_fonksiyon() çalıştırılmaz
}

// İkinci ifade değerlendirilir
if (x && çağır_fonksiyon()) {
    // x false then çağır_fonksiyon() çalıştırılmaz
}

Assignment Operators

Simple Assignment

let x = 10
x = 20  // Yeni değer atama

Compound Assignment

let n = 10

n += 5   // n = n + 5  (15)
n -= 3   // n = n - 3  (12)
n *= 2   // n = n * 2  (24)
n /= 4   // n = n / 4  (6)
n %= 5   // n = n % 5  (1)
n **= 3  // n = n ** 3 (1)

Multiple Assignment

let a, b, c = 1, 2, 3
print(a, b, c)  // 1 2 3

// Değer takası (swap)
a, b = b, a
print(a, b)  // 2 1

Bitwise Operators

let a = 0b1010  // 10
let b = 0b1100  // 12

print(a & b)   // 0b1000 (8)  - AND
print(a | b)   // 0b1110 (14) - OR
print(a ^ b)   // 0b0110 (6)  - XOR
print(~a)      // 0b0101 (-11) - NOT
print(a << 1)  // 0b10100 (20) - Sol kaydırma
print(a >> 1)  // 0b0101 (5)   - Sağ kaydırma

String Operators

merge

let ad = "Ahmet"
let soyad = "Yılmaz"

let tam_ad = ad + " " + soyad
print(tam_ad)  // Ahmet Yılmaz

// Bileşik atama
let mesaj = "Merhaba "
mesaj += "Dünya"
print(mesaj)  // Merhaba Dünya

Repeat

let yıldız = "*" * 10
print(yıldız)  // **********

To compare

print("abc" == "abc")   // true
print("abc" < "abd")    // true (leksikografik)
print("a" + "b" == "ab") // true

Range Operator

// Inclusive range (end dahil)
for i 1'from 5'e to {
    print(i)  // 1, 2, 3, 4, 5
}

// Exclusive range (end hariç)
for i 1'from 5'e {
    print(i)  // 1, 2, 3, 4
}

// Liste oluşturma
let sayılar = [1..10]  // [1, 2, 3, ..., 10]
let harfler = [to'..'z']

Elvis Operator (Null Coalescing)

let değer: number? = null

// Eğer null then varsayılan değer use
let sonuç = değer ?: 0
print(sonuç)  // 0

değer = 42
sonuç = değer ?: 0
print(sonuç)  // 42

Safe Navigation Operator

let kullanıcı: Kullanıcı? = null

// Null kontrolü olmadan güvenli erişim
let email = kullanıcı?.email
// Eğer kullanıcı null then email de null olur

// Zincirleme
let şehir = kullanıcı?.adres?.şehir?.ad

Pipeline Operator

// Fonksiyon zincirleme
let sonuç = [1, 2, 3, 4, 5]
    |> filtrele(x => x % 2 == 0)
    |> eşle(x => x * 2)
    |> azalt((a, b) => a + b)

print(sonuç)  // 12 (2*2 + 4*2)

Spread Operator

// Liste spread
let liste1 = [1, 2, 3]
let liste2 = [4, 5, 6]
let birleşik = [...liste1, ...liste2]
print(birleşik)  // [1, 2, 3, 4, 5, 6]

// Fonksiyon argümanı
function topla(a, b, c) -> number {
    return a + b + c
}

let args = [1, 2, 3]
print(topla(...args))  // 6

Ternary Operator

let yaş = 18
let case = yaş >= 18 ? "Yetişkin" : "Çocuk"
print(case)  // Yetişkin

// İç içe ternary
let not = 85
let harf = not >= 90 ? "A" :
                not >= 80 ? "B" :
                not >= 70 ? "C" : "F"
print(harf)  // B

Type Check Operator

let değer: Any = "Hello"

if değer is string {
    print("Bu bir string")
}

if değer !is number {
    print("Bu bir sayı not")
}

// Casting ile
let String = değer as string

In Operator

// Liste kontrol
let sayılar = [1, 2, 3, 4, 5]
if 3 in sayılar {
    print("3 listede var")
}

// String kontrol
let String = "Merhaba Dünya"
if "Dünya" in String {
    print("'Dünya' bulundu")
}

// Range kontrol
if 5 in 1..10 {
    print("5 aralıkta")
}

// Map kontrol
let sözlük = {"ad": "Ahmet", "yaş": 30}
if "ad" in sözlük {
    print("'ad' anahtarı var")
}

Changing Priority with Parentheses

let a = 2 + 3 * 4      // 14 (çarpma önce)
let b = (2 + 3) * 4    // 20 (toplama önce)

let c = 10 / 2 + 3     // 8
let d = 10 / (2 + 3)   // 2

Special Operators

Assert Operator

let x = 10
assert x > 0, "x pozitif olmalı"

// Eğer koşul false then program durur

Match Operator

let sayı = 2

let sonuç = sayı match {
    1 => "bir",
    2 => "iki",
    3 => "üç",
    _ => "diğer"
}

print(sonuç)  // iki

Best Practices

✅ Things to Do

// Karmaşık ifadelerde parantez use
let sonuç = (a + b) * (c - d)

// Short-circuit evaluation'from faydalanın
if kullanıcı != null && kullanıcı.aktif {
    // ...
}

// Null-safe operatörler use
let email = kullanıcı?.email ?: "Bilinmiyor"

// Operatör etrafında boşluk
let sum = a + b  // İyi
let sum = a+b    // Okunabilir not

❌ Things to Avoid

// Çok uzun ternary zincirleri
let x = a ? b ? c ? d : e : f : g  // Okunması zor

// Gereksiz karmaşık bitwise işlemler
let y = (a & b) | (~c ^ d)  // Ne yaptığı belli not

// Yan etki yaratan operatörler
let z = x++ + ++x + x++  // Kafa karıştırıcı

Example: Various Operators

function main() {
    // Aritmetik
    let sum = 10 + 20
    let fark = 100 - 50
    let product = 5 * 6
    let bölüm = 20 / 4
    
    // Karşılaştırma
    if sum > fark {
        print("Toplam daha büyük")
    }
    
    // Mantıksal
    let a = true
    let b = false
    if a && !b {
        print("Koşul sağlandı")
    }
    
    // String
    let mesaj = "Merhaba" + " " + "Dünya"
    
    // Null-safe
    let değer: number? = null
    let sonuç = değer ?: 0
    
    print("Tüm işlemler tamamlandı")
}
