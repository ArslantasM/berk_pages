🔌 hal::gpio

General Purpose Input/Output Control - GPIO Pin Yönetimi

381 satır ~30 function STM32/GD32/ESP32

📖 Genel Bakış

GPIO (General Purpose Input/Output) Moduleü, mikrokontrolcü pinlerinin dijital Introduction/çıkış olarak kontrolünü sağlar. LED kontrolü, buton okuma, sensör arayüzleri and dijital sinyaller for kullanılır.

🔑 Temel Özellikler

  • Digital input/output modes
  • Push-pull/Open-drain output
  • Pull-up/Pull-down resistors
  • Pin speed configuration
  • Interrupt support (EXTI)
  • Alternate function mapping
  • Atomic bit operations
  • Port locking

🚀 Hızlı Getting Started

import hal::gpio

// GPIO clock enable
gpio.clock_enable(gpio.PORT_C)

// LED pin setup (output)
variable led = gpio.pin_init(gpio.PORT_C, 13, do
    mode: gpio.MODE_OUTPUT,
    speed: gpio.SPEED_LOW,
    pull: gpio.PULL_NONE
end)

// LED control
gpio.pin_write(led, gpio.HIGH)  // Turn on
gpio.pin_toggle(led)  // Toggle
gpio.pin_write(led, gpio.LOW)  // Turn off

📦 Tipler and Enum'lar

// GPIO Portları
enum Port {
    A, B, C, D, E, F, G, H, I
}

// Pin Modları
enum Mode {
    INPUT,        // Digital input
    OUTPUT,       // Push-pull output
    OUTPUT_OD,    // Open-drain output
    AF,           // Alternate function push-pull
    AF_OD,        // Alternate function open-drain
    ANALOG        // Analog mode
}

// Hız Seviyesi
enum Speed {
    LOW,       // Low speed (2 MHz)
    MEDIUM,    // Medium speed (25 MHz)
    HIGH,      // High speed (50 MHz)
    VERY_HIGH  // Very high speed (100 MHz)
}

// Pull Direnci
enum Pull {
    NONE,  // No pull resistor
    UP,    // Pull-up
    DOWN   // Pull-down
}

// Logic Level
enum Level {
    LOW,   // Logic 0 (0V)
    HIGH   // Logic 1 (3.3V/5V)
}

// Interrupt Tetikleme
enum Trigger {
    RISING,   // Rising edge
    FALLING,  // Falling edge
    BOTH      // Both edges
}

💡 Example 1: LED Blink

import hal::gpio, hal::core

function ana() do
    // System init
    core.system_init()
    core.clock_config(72_000_000)  // 72MHz
    
    // LED setup (PC13 on STM32 Blue Pill)
    gpio.clock_enable(gpio.PORT_C)
    variable led = gpio.pin_init(gpio.PORT_C, 13, do
        mode: gpio.MODE_OUTPUT,
        speed: gpio.SPEED_LOW,
        pull: gpio.PULL_NONE
    end)
    
    // Blink loop
    loop do
        gpio.pin_toggle(led)
        core.delay_ms(500)
    end
end

💡 Example 2: Button Input with Interrupt

import hal::gpio, hal::int

variable button_pressed = false

function button_handler() do
    button_pressed = true
    io.println("Button pressed!")
end

function ana() do
    // LED setup (output)
    gpio.clock_enable(gpio.PORT_C)
    variable led = gpio.pin_init(gpio.PORT_C, 13, do
        mode: gpio.MODE_OUTPUT,
        speed: gpio.SPEED_LOW,
        pull: gpio.PULL_NONE
    end)
    
    // Button setup (input with pull-up)
    gpio.clock_enable(gpio.PORT_A)
    variable button = gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_INPUT,
        pull: gpio.PULL_UP,
        speed: gpio.SPEED_LOW
    end)
    
    // Enable button interrupt
    gpio.interrupt_enable(button, gpio.TRIGGER_FALLING, button_handler)
    
    // Main loop
    loop do
        if button_pressed ise do
            button_pressed = false
            gpio.pin_toggle(led)
        end
        core.delay_ms(10)
    end
end

💡 Example 3: Multi-LED Control (Port Operations)

import hal::gpio, hal::core

function ana() do
    core.system_init()
    
    // Setup 8 LEDs on Port B (PB0-PB7)
    gpio.clock_enable(gpio.PORT_B)
    
    // Initialize all 8 pins
    her i içinde 0..8 for do
        gpio.pin_init(gpio.PORT_B, i, do
            mode: gpio.MODE_OUTPUT,
            speed: gpio.SPEED_MEDIUM,
            pull: gpio.PULL_NONE
        end)
    end
    
    // LED patterns
    variable patterns = [
        0b00000001,  // Single LED
        0b00000011,  // 2 LEDs
        0b00000111,  // 3 LEDs
        0b00001111,  // 4 LEDs
        0b00011111,  // 5 LEDs
        0b00111111,  // 6 LEDs
        0b01111111,  // 7 LEDs
        0b11111111   // All LEDs
    ]
    
    loop do
        her pattern içinde patterns for do
            gpio.port_write(gpio.PORT_B, pattern)
            core.delay_ms(200)
        end
        
        // Reverse
        her pattern içinde patterns.ters() for do
            gpio.port_write(gpio.PORT_B, pattern)
            core.delay_ms(200)
        end
    end
end

💡 Example 4: Traffic Light Controller

import hal::gpio, hal::core

struct TrafficLight do
    red: gpio.Pin,
    yellow: gpio.Pin,
    green: gpio.Pin
end

function traffic_light_init() -> TrafficLight do
    gpio.clock_enable(gpio.PORT_A)
    
    variable tl = TrafficLight do
        red: gpio.pin_init(gpio.PORT_A, 0, do
            mode: gpio.MODE_OUTPUT,
            speed: gpio.SPEED_LOW,
            pull: gpio.PULL_NONE
        end),
        yellow: gpio.pin_init(gpio.PORT_A, 1, do
            mode: gpio.MODE_OUTPUT,
            speed: gpio.SPEED_LOW,
            pull: gpio.PULL_NONE
        end),
        green: gpio.pin_init(gpio.PORT_A, 2, do
            mode: gpio.MODE_OUTPUT,
            speed: gpio.SPEED_LOW,
            pull: gpio.PULL_NONE
        end)
    end
    
    return tl
end

function set_state(tl: TrafficLight, red: boolean, yellow: boolean, green: boolean) do
    gpio.pin_write(tl.red, if red ise gpio.HIGH elseif gpio.LOW)
    gpio.pin_write(tl.yellow, if yellow ise gpio.HIGH elseif gpio.LOW)
    gpio.pin_write(tl.green, if green ise gpio.HIGH elseif gpio.LOW)
end

function ana() do
    core.system_init()
    variable tl = traffic_light_init()
    
    loop do
        // Green light
        set_state(tl, false, false, true)
        core.delay_ms(5000)
        
        // Yellow light
        set_state(tl, false, true, false)
        core.delay_ms(2000)
        
        // Red light
        set_state(tl, true, false, false)
        core.delay_ms(5000)
    end
end

⚠️ Important Notes

  • Pin kullanmadan önce port clock'unu enable etmelisiniz
  • Open-drain çıkışlar external pull-up gerektirir
  • 5V tolerant pinler for datasheet kontrol edin
  • Interrupt kullanırken NVIC konfigürasyonu gerekir
  • Maximum pin current sınırlarına Attention edin (~25mA)
  • Analog mode kullanırken dijital bufferleri disable edilir

📚 Platform Desteği

  • STM32: Tüm F/L/H/G serisi
  • GD32: GD32VF103 (RISC-V)
  • ESP32: GPIO 0-39
  • Nordic: nRF52/53 GPIO
