📊 hal::adc

Analog-to-Digital Converter - Analog Sinyal Dönüşümü

400 satır ~25 function 12-bit ADC

📖 Genel Bakış

ADC (Analog-to-Digital Converter) Moduleü, analog sinyalleri dijital değerlere dönüştürür. Sensör okuma, sıcaklık ölçümü, batarya voltajı kontrolü gibi uygulamalarda kullanılır.

🔑 Temel Özellikler

  • 6-16 bit resolution (device dependent)
  • Multiple ADC channels (up to 19)
  • Single/continuous conversion
  • DMA support
  • Internal temperature sensor
  • Internal voltage reference
  • Analog watchdog
  • Configurable sampling time

🚀 Hızlı Getting Started

import hal::adc, hal::gpio

// ADC initialization
adc.clock_enable()
variable adc1 = adc.init(adc.ADC1, do
    resolution: adc.RESOLUTION_12BIT,
    align: adc.ALIGN_RIGHT
end)

// GPIO setup for analog input
variable ain = gpio.pin_init(gpio.PORT_A, 0, do
    mode: gpio.MODE_ANALOG,
    pull: gpio.PULL_NONE
end)

// Configure channel
adc.channel_config(adc1, adc.CHANNEL_0, do
    sampling_time: adc.SAMPLETIME_480CYCLES
end)

// Read value
adc.start(adc1)
variable value = adc.read(adc1)
io.println("ADC: " + value.yazıya())

💡 Example 1: Temperature Sensor

import hal::adc, hal::gpio, hal::core

function sıcaklık_oku() -> kesir do
    // ADC setup
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    // GPIO for analog input (PA0)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    // Configure channel
    adc.channel_config(adc1, adc.CHANNEL_0, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Read multiple samples and average
    variable toplam = 0.0
    variable samples = 10
    
    her i içinde 0..samples for do
        adc.start(adc1)
        variable raw = adc.read(adc1)
        toplam += raw.kesir()
        core.delay_ms(10)
    end
    
    variable ortalama = toplam / samples.kesir()
    
    // Convert to voltage (3.3V reference, 12-bit)
    variable voltage = ortalama * 3.3 / 4095.0
    
    // LM35 sensor: 10mV/°C
    variable temp = voltage * 100.0
    
    return temp
end

function ana() do
    core.system_init()
    core.clock_config(72_000_000)
    
    loop do
        variable temp = sıcaklık_oku()
        io.println("Sıcaklık: {:.1f}°C".formatla(temp))
        core.delay_ms(1000)
    end
end

💡 Example 2: Battery Voltage Monitor

import hal::adc, hal::gpio

function batarya_voltaj_oku() -> kesir do
    // Voltage divider: Battery -> 10K -> ADC -> 10K -> GND
    // Max input: 6.6V (for 3.3V ADC)
    
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 1, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    adc.channel_config(adc1, adc.CHANNEL_1, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Read 100 samples for accuracy
    variable toplam = 0
    her i içinde 0..100 for do
        adc.start(adc1)
        toplam += adc.read(adc1)
    end
    
    variable ortalama = toplam / 100
    variable voltage = (ortalama.kesir() / 4095.0) * 3.3
    
    // Apply voltage divider ratio (2:1)
    variable battery_voltage = voltage * 2.0
    
    return battery_voltage
end

function batarya_durumu(voltage: kesir) -> string do
    if voltage > 4.1 ise return "Tam"
    elseif if voltage > 3.7 ise return "İyi"
    elseif if voltage > 3.5 ise return "Orta"
    elseif return "Düşük"
end

function ana() do
    core.system_init()
    
    loop do
        variable v = batarya_voltaj_oku()
        variable durum = batarya_durumu(v)
        io.println("Batarya: {:.2f}V [{}]".formatla(v, durum))
        core.delay_ms(5000)
    end
end

💡 Example 3: Multi-Channel ADC with DMA

import hal::adc, hal::dma, hal::gpio

constant CHANNELS = 4

function adc_dma_init() -> adc.Handle do
    // GPIO setup for 4 analog inputs (PA0-PA3)
    gpio.clock_enable(gpio.PORT_A)
    her i içinde 0..CHANNELS for do
        gpio.pin_init(gpio.PORT_A, i, do
            mode: gpio.MODE_ANALOG,
            pull: gpio.PULL_NONE
        end)
    end
    
    // ADC setup
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT,
        scan_mode: true,
        continuous: true
    end)
    
    // Configure channels
    adc.channel_config(adc1, adc.CHANNEL_0, do rank: 1 end)
    adc.channel_config(adc1, adc.CHANNEL_1, do rank: 2 end)
    adc.channel_config(adc1, adc.CHANNEL_2, do rank: 3 end)
    adc.channel_config(adc1, adc.CHANNEL_3, do rank: 4 end)
    
    return adc1
end

variable adc_buffer: [sayı; CHANNELS] = [0, 0, 0, 0]

function ana() do
    core.system_init()
    variable adc1 = adc_dma_init()
    
    // DMA setup for continuous ADC transfer
    dma.clock_enable()
    variable dma_ch = dma.init(dma.DMA1_CH1, do
        direction: dma.PERIPH_TO_MEMORY,
        periph_addr: adc.data_register(adc1),
        memory_addr: adc_buffer.adres(),
        buffer_size: CHANNELS,
        periph_inc: false,
        memory_inc: true,
        periph_size: dma.SIZE_16BIT,
        memory_size: dma.SIZE_16BIT,
        circular: true,
        priority: dma.PRIORITY_HIGH
    end)
    
    // Start DMA and ADC
    dma.start(dma_ch)
    adc.start_dma(adc1)
    
    // Main loop - data is automatically updated by DMA
    loop do
        io.println("Kanal 0: {}  Kanal 1: {}  Kanal 2: {}  Kanal 3: {}".formatla(
            adc_buffer[0], adc_buffer[1], adc_buffer[2], adc_buffer[3]
        ))
        core.delay_ms(500)
    end
end

💡 Example 4: Analog Watchdog (Voltage Monitor)

import hal::adc, hal::gpio, hal::int

constant THRESHOLD_LOW = 1500   // ~1.2V
constant THRESHOLD_HIGH = 3000  // ~2.4V

variable alarm_triggered = false

function adc_watchdog_handler() do
    alarm_triggered = true
    io.println("⚠️ Warning: Voltaj threshold dışında!")
end

function ana() do
    core.system_init()
    
    // GPIO setup
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_ANALOG,
        pull: gpio.PULL_NONE
    end)
    
    // ADC setup
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    adc.channel_config(adc1, adc.CHANNEL_0, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Configure analog watchdog
    adc.watchdog_enable(adc1, do
        channel: adc.CHANNEL_0,
        low_threshold: THRESHOLD_LOW,
        high_threshold: THRESHOLD_HIGH
    end)
    
    // Enable interrupt
    adc.interrupt_enable(adc1, adc.WATCHDOG_INT)
    int.register_handler(int.ADC1_IRQ, adc_watchdog_handler)
    int.enable_irq(int.ADC1_IRQ)
    
    // Start continuous conversion
    adc.start_continuous(adc1)
    
    loop do
        if alarm_triggered ise do
            alarm_triggered = false
            variable value = adc.read(adc1)
            variable voltage = (value.kesir() / 4095.0) * 3.3
            io.println("Voltaj: {:.2f}V [ALARM]".formatla(voltage))
        end
        core.delay_ms(100)
    end
end

💡 Example 5: Internal Temperature Sensor

import hal::adc

function cpu_sicaklik_oku() -> kesir do
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    // Enable internal temperature sensor
    adc.enable_temp_sensor()
    
    // Configure internal channel (usually CH16)
    adc.channel_config(adc1, adc.CHANNEL_TEMP, do
        sampling_time: adc.SAMPLETIME_480CYCLES  // Long sampling for accuracy
    end)
    
    // Read multiple samples
    variable toplam = 0
    her i içinde 0..10 for do
        adc.start(adc1)
        toplam += adc.read(adc1)
        core.delay_ms(5)
    end
    
    variable ortalama = toplam / 10
    
    // STM32 formula: Temp = (V25 - Vsense) / Avg_Slope + 25
    // Typical: V25 = 1.43V (@ 25°C), Avg_Slope = 4.3mV/°C
    variable voltage = (ortalama.kesir() / 4095.0) * 3.3
    variable temp = ((1.43 - voltage) / 0.0043) + 25.0
    
    return temp
end

function ana() do
    core.system_init()
    
    loop do
        variable temp = cpu_sicaklik_oku()
        io.println("CPU Sıcaklığı: {:.1f}°C".formatla(temp))
        
        // Thermal management
        if temp > 80.0 ise do
            io.println("⚠️ Warning: Yüksek sıcaklık!")
            // Reduce clock speed or take action
        end
        
        core.delay_ms(2000)
    end
end

⚡ Performans İpuçları

  • Sampling Time: Daha uzun sampling time = daha yüksek hassasiyet, ancak daha yavaş
  • Kalibrasyon: Her açılışta calibrate() çalıştırın
  • Ortalama Alma: Gürültüyü azaltmak for 10-100 sample ortalaması alın
  • DMA Usageı: Multi-channel okumada CPU yükünü azaltır
  • Continuous Mode: Yüksek hızlı Exampleleme for kullanın
  • Input Impedance: Yüksek empedanslı kaynaklarda sampling time arttırın

⚠️ Important Notes

  • ADC kullanmadan önce clock_enable() çağırın
  • Analog pinler for GPIO MODE_ANALOG kullanın
  • Maximum input voltage: VREF+ (genellikle 3.3V)
  • Input impedance: <50kΩ önerilir
  • Conversion time = sampling time + 12 cycles (12-bit)
  • Internal temperature sensor ±2°C hassasiyet
  • DMA kullanırken circular buffer kullanın
  • Watchdog sadece tek kanal izleyebilir
