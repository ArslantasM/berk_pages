BERK Programming Language

📡 hal::usart

Universal Synchronous/Asynchronous Receiver/Transmitter

407 satır ~20 function UART/USART

📖 Genel Bakış

USART/UART Moduleü, seri haberleşme protokolüdür. GPS, GSM Moduleler, sensörler and PC ile haberleşme for kullanılır.

🔑 Temel Özellikler

  • Asynchronous (UART) mode
  • Synchronous (USART) mode
  • Configurable baud rates (1200-12Mbps)
  • 7/8/9 bit data
  • Parity: None/Even/Odd
  • 1/2 stop bits
  • Hardware flow control (RTS/CTS)
  • DMA support
  • Interrupt-driven TX/RX

🚀 Hızlı Getting Started

import hal::usart, hal::gpio

// GPIO setup for USART1
gpio.clock_enable(gpio.PORT_A)
gpio.pin_init(gpio.PORT_A, 9, do   // TX
    mode: gpio.MODE_AF,
    af: 7  // AF7 = USART1
end)
gpio.pin_init(gpio.PORT_A, 10, do  // RX
    mode: gpio.MODE_AF,
    pull: gpio.PULL_UP,
    af: 7
end)

// USART init
variable uart = usart.init(usart.USART1, do
    baud_rate: 115200,
    data_bits: usart.DATA_8BIT,
    parity: usart.PARITY_NONE,
    stop_bits: usart.STOP_1BIT
end)

// Send data
usart.write(uart, "Hello World!\n")

// Receive data
variable data = usart.read(uart, 100)  // Read up to 100 bytes

💡 Example 1: Serial Console

import hal::usart, hal::gpio, hal::core

function uart_print(uart: usart.Handle, msg: string) do
    usart.write(uart, msg)
end

function uart_println(uart: usart.Handle, msg: string) do
    usart.write(uart, msg + "\r\n")
end

function ana() do
    core.system_init()
    core.clock_config(72_000_000)
    
    // Setup USART1 GPIO (PA9=TX, PA10=RX)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 9, do
        mode: gpio.MODE_AF,
        speed: gpio.SPEED_HIGH,
        af: 7
    end)
    gpio.pin_init(gpio.PORT_A, 10, do
        mode: gpio.MODE_AF,
        pull: gpio.PULL_UP,
        af: 7
    end)
    
    // Initialize USART
    usart.clock_enable(usart.USART1)
    variable uart = usart.init(usart.USART1, do
        baud_rate: 115200,
        data_bits: usart.DATA_8BIT,
        parity: usart.PARITY_NONE,
        stop_bits: usart.STOP_1BIT,
        flow_control: usart.FLOW_NONE
    end)
    
    uart_println(uart, "BERK v1.0.0 Console")
    uart_println(uart, "Type 'help' for commands")
    
    variable buffer = ""
    
    loop do
        // Check if data available
        if usart.available(uart) > 0 ise do
            variable ch = usart.read_byte(uart)
            
            if ch == '\r' or ch == '\n' ise do
                if buffer.uzunluk() > 0 ise do
                    uart_println(uart, "")
                    process_command(uart, buffer)
                    buffer = ""
                    uart_print(uart, "> ")
                end
            else if ch == '\b' ise do  // Backspace
                if buffer.uzunluk() > 0 ise do
                    buffer = buffer[0..buffer.uzunluk()-1]
                    uart_print(uart, "\b \b")  // Erase character
                end
            else do
                buffer += ch
                usart.write_byte(uart, ch)  // Echo
            end
        end
    end
end

function process_command(uart: usart.Handle, cmd: string) do
    if cmd == "help" ise do
        uart_println(uart, "Available commands:")
        uart_println(uart, "  help    - Show this message")
        uart_println(uart, "  status  - System status")
        uart_println(uart, "  reset   - Reset system")
        uart_println(uart, "  info    - System info")
    else if cmd == "status" ise do
        variable uptime = core.uptime_ms() / 1000
        uart_println(uart, "System Status:")
        uart_println(uart, "  Uptime: " + uptime.yazıya() + "s")
        uart_println(uart, "  Clock: 72 MHz")
        uart_println(uart, "  Free RAM: 64 KB")
    else if cmd == "reset" ise do
        uart_println(uart, "Resetting...")
        core.delay_ms(100)
        core.system_reset()
    else if cmd == "info" ise do
        uart_println(uart, "BERK Embedded System")
        uart_println(uart, "Version: 1.0.0")
        uart_println(uart, "Platform: STM32F103")
    else do
        uart_println(uart, "Unknown command: " + cmd)
    end
end

💡 Example 2: GPS Parser (NMEA)

import hal::usart, string

struct GPSData do
    latitude: kesir,
    longitude: kesir,
    altitude: kesir,
    speed: kesir,
    satellites: sayı,
    valid: boolean
end

function parse_nmea(sentence: string) -> İsteğe_Bağlı[GPSData] do
    // Parse GPGGA sentence: $GPGGA,time,lat,N,lon,E,quality,sats,hdop,alt,M,...
    if sentence.starts_with("$GPGGA") else do
        return Hiçbir
    end
    
    variable parts = sentence.split(",")
    if parts.uzunluk() < 15 ise do
        return Hiçbir
    end
    
    variable quality = parts[6].parse_int()?
    if quality == 0 ise do  // Invalid fix
        return Hiçbir
    end
    
    // Parse latitude (DDMM.MMMM format)
    variable lat_str = parts[2]
    variable lat_deg = lat_str[0..2].parse_int()?
    variable lat_min = lat_str[2..].parse_float()?
    variable latitude = lat_deg.kesir() + lat_min / 60.0
    if parts[3] == "S" ise do
        latitude = -latitude
    end
    
    // Parse longitude (DDDMM.MMMM format)
    variable lon_str = parts[4]
    variable lon_deg = lon_str[0..3].parse_int()?
    variable lon_min = lon_str[3..].parse_float()?
    variable longitude = lon_deg.kesir() + lon_min / 60.0
    if parts[5] == "W" ise do
        longitude = -longitude
    end
    
    return Bazı(GPSData do
        latitude: latitude,
        longitude: longitude,
        altitude: parts[9].parse_float()?,
        speed: 0.0,
        satellites: parts[7].parse_int()?,
        valid: true
    end)
end

function ana() do
    // Setup USART for GPS (9600 baud, typical for GPS)
    variable gps_uart = usart.init(usart.USART2, do
        baud_rate: 9600,
        data_bits: usart.DATA_8BIT,
        parity: usart.PARITY_NONE,
        stop_bits: usart.STOP_1BIT
    end)
    
    variable buffer = ""
    
    loop do
        if usart.available(gps_uart) > 0 ise do
            variable ch = usart.read_byte(gps_uart)
            
            if ch == '\n' ise do
                // Process complete sentence
                if gps_data = parse_nmea(buffer) ise do
                    io.println("GPS Fix:")
                    io.println("  Lat: {:.6f}".formatla(gps_data.latitude))
                    io.println("  Lon: {:.6f}".formatla(gps_data.longitude))
                    io.println("  Alt: {:.1f}m".formatla(gps_data.altitude))
                    io.println("  Sats: {}".formatla(gps_data.satellites))
                end
                buffer = ""
            else if ch != '\r' ise do
                buffer += ch
            end
        end
        
        core.delay_ms(10)
    end
end

💡 Example 3: AT Command Interface (GSM)

import hal::usart, hal::core

struct GSMModem do
    uart: usart.Handle
end

function send_at_command(modem: GSMModem, cmd: string, timeout_ms: sayı) -> Sonuç[string, Hata] do
    // Send command
    usart.write(modem.uart, cmd + "\r\n")
    
    // Wait for response
    variable response = ""
    variable start_time = core.millis()
    
    loop do
        if usart.available(modem.uart) > 0 ise do
            variable ch = usart.read_byte(modem.uart)
            response += ch
            
            // Check for termination
            if response.ends_with("OK\r\n") or response.ends_with("ERROR\r\n") ise do
                return Tamam(response)
            end
        end
        
        // Timeout check
        if core.millis() - start_time > timeout_ms ise do
            return Hata("AT command timeout")
        end
        
        core.delay_ms(10)
    end
end

function gsm_init() -> Sonuç[GSMModem, Hata] do
    variable uart = usart.init(usart.USART3, do
        baud_rate: 115200,
        data_bits: usart.DATA_8BIT,
        parity: usart.PARITY_NONE,
        stop_bits: usart.STOP_1BIT
    end)
    
    variable modem = GSMModem do uart: uart end
    
    // Test communication
    send_at_command(modem, "AT", 1000)?
    
    // Disable echo
    send_at_command(modem, "ATE0", 1000)?
    
    // Check SIM card
    variable sim_status = send_at_command(modem, "AT+CPIN?", 1000)?
    if sim_status.contains("READY") else do
        return Hata("SIM card not ready")
    end
    
    return Tamam(modem)
end

function send_sms(modem: GSMModem, phone: string, message: string) -> Sonuç[Hiçbir, Hata] do
    // Set text mode
    send_at_command(modem, "AT+CMGF=1", 1000)?
    
    // Set phone number
    send_at_command(modem, "AT+CMGS=\"" + phone + "\"", 1000)?
    
    // Send message
    usart.write(modem.uart, message)
    usart.write_byte(modem.uart, 0x1A)  // Ctrl+Z
    
    variable response = send_at_command(modem, "", 10000)?
    if response.contains("OK") else do
        return Hata("Failed to send SMS")
    end
    
    return Tamam(Hiçbir)
end

function ana() do
    variable modem = gsm_init()?
    
    io.println("GSM Modem ready!")
    
    // Send alert SMS
    send_sms(modem, "+905551234567", "BERK System Alert: Temperature exceeded 50°C")?
    
    io.println("SMS sent successfully!")
end

⚠️ Important Notes

  • TX/RX pinlerini true alternatif fonksiyona ayarlayın
  • Baud rate hesabı clock frequency'ye bağlıdır
  • Uzun kablolarda düşük baud rate kullanın
  • RX buffer overflow'dan kaçının (interrupt/DMA kullanın)
  • 3.3V-5V level shifting gerekebilir
