🚀 Hızlı Getting Started

import collections

// Vec (dinamik dizi)
variable v = collections.vec_new()
collections.vec_push(v, 10)
collections.vec_push(v, 20)
variable eleman = collections.vec_get(v, 0)

// HashMap
variable map = collections.hashmap_new()
collections.hashmap_insert(map, "name", "BERK")
collections.hashmap_insert(map, "version", "1.0")
variable isim = collections.hashmap_get(map, "name")

// HashSet
variable set = collections.hashset_new()
collections.hashset_insert(set, "apple")
collections.hashset_insert(set, "banana")
variable var_mı = collections.hashset_contains(set, "apple")

// LinkedList
variable list = collections.linkedlist_new()
collections.linkedlist_push_back(list, 1)
collections.linkedlist_push_front(list, 0)

// BinaryHeap (öncelik kuyruğu)
variable heap = collections.heap_new()
collections.heap_push(heap, 5)
collections.heap_push(heap, 3)
variable en_büyük = collections.heap_pop(heap)  // 5

📚 Veri Yapıları

  • Vec: Dinamik dizi, O(1) erişim, O(1) amortized push
  • HashMap: Hash tablosu, O(1) average insert/get/delete
  • HashSet: Unique elemanlar, O(1) average contains
  • LinkedList: Çift bağlı liste, O(1) push/pop başta/sonda
  • BTreeMap: Sıralı map, O(log n) operations
  • BinaryHeap: Max-heap, O(log n) push/pop, O(1) peek

Word Frequency Counter

import collections, string, io

function word_frequency(text: string) -> HashMap do
    variable freq = collections.hashmap_new()
    variable kelimeler = string.böl(text, " ")
    
    her kelime içinde kelimeler for do
        variable küçük = string.küçük_harf(kelime)
        
        if collections.hashmap_contains(freq, küçük) ise do
            variable count = collections.hashmap_get(freq, küçük)
            collections.hashmap_insert(freq, küçük, count + 1)
        end else do
            collections.hashmap_insert(freq, küçük, 1)
        end
    end
    
    return freq
end

variable metin = io.dosya_oku("document.txt")
variable frekans = word_frequency(metin)

// En çok geçen 10 kelimeyi göster
variable kelimeler = collections.hashmap_keys(frekans)
collections.vec_sort_by(kelimeler, function(a, b) do
    return collections.hashmap_get(frekans, b) - collections.hashmap_get(frekans, a)
end)

her i içinde aralık(0, 10) for do
    variable k = collections.vec_get(kelimeler, i)
    print(k + ": " + collections.hashmap_get(frekans, k).yazıya())
end

LRU Cache Implementation

import collections

class LRUCache do
    variable kapasite: integer
    variable cache: HashMap
    variable sıra: LinkedList
    
    function yeni(kap: integer) do
        self.kapasite = kap
        self.cache = collections.hashmap_new()
        self.sıra = collections.linkedlist_new()
    end
    
    function get(anahtar: string) -> herhangi do
        if collections.hashmap_contains(self.cache, anahtar) else do
            return yok
        end
        
        // Sıraya taşı (LRU güncelle)
        collections.linkedlist_remove(self.sıra, anahtar)
        collections.linkedlist_push_back(self.sıra, anahtar)
        
        return collections.hashmap_get(self.cache, anahtar)
    end
    
    function put(anahtar: string, değer: herhangi) do
        if collections.hashmap_contains(self.cache, anahtar) ise do
            collections.linkedlist_remove(self.sıra, anahtar)
        end
        
        collections.hashmap_insert(self.cache, anahtar, değer)
        collections.linkedlist_push_back(self.sıra, anahtar)
        
        if collections.hashmap_size(self.cache) > self.kapasite ise do
            // En eski elemanı çıkar
            variable eski = collections.linkedlist_pop_front(self.sıra)
            collections.hashmap_remove(self.cache, eski)
        end
    end
end

variable cache = LRUCache.yeni(3)
cache.put("a", 1)
cache.put("b", 2)
cache.put("c", 3)
print(cache.get("a"))  // 1
cache.put("d", 4)  // "b" çıkarılır
print(cache.get("b"))  // yok
