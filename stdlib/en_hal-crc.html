🔐 hal::crc

Hardware CRC32 Calculator - Fast Data Integrity Check

~360 satır ~12 function STM32/GD32

📖 Genel Bakış

CRC Moduleü, hardware accelerated CRC32 hesaplaması sağlar. Polynomial 0x04C11DB7 kullanır (Ethernet standardı). Firmware integrity check, flash verification, communication protocols and data corruption detection for kritiktir. Software CRC'den 10-100x daha hızlıdır.

🔑 Temel Özellikler

  • Hardware CRC32: Single-cycle 32-bit CRC calculation
  • Standard Polynomial: 0x04C11DB7 (Ethernet, ZIP, PNG)
  • Fast Performance: 1 clock cycle per 32-bit word
  • Initial Value: Configurable (default 0xFFFFFFFF)
  • DMA Compatible: Can be used with DMA transfers
  • Accumulation: Sequential CRC calculation support
  • Reset Capability: Quick reset to initial value
  • Low Power: Minimal power consumption

🚀 Hızlı Getting Started

import hal::crc, hal::rcc

// Enable CRC clock

💡 Example 1: Firmware Integrity Check

Problem: Flash'taki firmware'in bozulmadığını doğrulamak

import hal::crc, hal::flash

constant FIRMWARE_START: sayı = 0x0800_8000  // Application start
constant FIRMWARE_SIZE: sayı = 128 * 1024     // 128 KB
constant CRC_STORAGE: sayı = 0x0807_FFFC      // Last 4 bytes of app region

function firmware_calculate_crc() -> sayı do
    crc.reset()
    
    // Calculate CRC of firmware (32-bit words)
    variable word_count = FIRMWARE_SIZE / 4
    variable addr = FIRMWARE_START
    
    for i içinde 0..word_count do
        variable word = *(addr kayıt *sayı)
        crc.accumulate(word)
        addr = addr + 4
    end
    
    return crc.get_value()
end

function firmware_store_crc(checksum: sayı) do
    // Unlock flash
    flash.unlock()
    
    // Erase last page
    flash.erase_page(CRC_STORAGE)
    
    // Write CRC
    flash.program_word(CRC_STORAGE, checksum)
    
    flash.lock()
    
    io.yazdir_satır("CRC stored: 0x" + checksum.onaltılıya())
end

function firmware_verify() -> mantık do
    // Read stored CRC
    variable stored_crc = *(CRC_STORAGE kayıt *sayı)
    
    // Calculate current CRC
    variable calculated_crc = firmware_calculate_crc()
    
    io.yazdir_satır("Stored CRC:     0x" + stored_crc.onaltılıya())
    io.yazdir_satır("Calculated CRC: 0x" + calculated_crc.onaltılıya())
    
    if stored_crc == calculated_crc do
        io.yazdir_satır("✅ Firmware integrity OK")
        return true
    elseif do
        io.yazdir_satır("❌ Firmware corrupted!")
        return false
    end
end

function ana() do
    crc.init()
    
    // On firmware update: store CRC
    // firmware_store_crc(firmware_calculate_crc())
    
    // On boot: verify firmware
    if not firmware_verify() do
        io.yazdir_satır("ERROR: Cannot start corrupted firmware")
        // Enter safe mode or bootloader
        loop do end  // Halt
    end
    
    io.yazdir_satır("Starting application...")
end

Sonuç: Boot sırasında firmware corruption detect edilir. Güvenli başlatma sağlanır.

💡 Example 2: UART Packet Verification

Problem: Serial communication'da packet integrity kontrolü

import hal::crc, hal::usart

struct Packet do
    header: sayı8 = 0xAA,
    length: sayı8,
    data: Dizi<sayı8>,
    crc32: sayı
end

function packet_calculate_crc(pkt: Packet) -> sayı do
    crc.reset()
    
    // CRC over header + length + data
    crc.accumulate_byte(pkt.header)
    crc.accumulate_byte(pkt.length)
    
    for byte içinde pkt.data do
        crc.accumulate_byte(byte)
    end
    
    return crc.get_value()
end

function packet_send(pkt: Packet) do
    // Calculate CRC
    pkt.crc32 = packet_calculate_crc(pkt)
    
    // Send packet
    usart.write_byte(usart.USART1, pkt.header)
    usart.write_byte(usart.USART1, pkt.length)
    
    for byte içinde pkt.data do
        usart.write_byte(usart.USART1, byte)
    end
    
    // Send CRC (big-endian)
    usart.write_byte(usart.USART1, (pkt.crc32 >> 24) & 0xFF kayıt sayı8)
    usart.write_byte(usart.USART1, (pkt.crc32 >> 16) & 0xFF kayıt sayı8)
    usart.write_byte(usart.USART1, (pkt.crc32 >> 8) & 0xFF kayıt sayı8)
    usart.write_byte(usart.USART1, pkt.crc32 & 0xFF kayıt sayı8)
end

function packet_receive() -> Seçenek<Packet> do
    // Read header
    variable header = usart.read_byte(usart.USART1)
    if header != 0xAA do
        return Seçenek.Hiçbiri
    end
    
    // Read length
    variable length = usart.read_byte(usart.USART1)
    
    // Read data
    variable data = Dizi.yeni(length)
    for i içinde 0..length do
        data[i] = usart.read_byte(usart.USART1)
    end
    
    // Read CRC
    variable crc_recv = 0 kayıt sayı
    crc_recv = (crc_recv << 8) | usart.read_byte(usart.USART1) kayıt sayı
    crc_recv = (crc_recv << 8) | usart.read_byte(usart.USART1) kayıt sayı
    crc_recv = (crc_recv << 8) | usart.read_byte(usart.USART1) kayıt sayı
    crc_recv = (crc_recv << 8) | usart.read_byte(usart.USART1) kayıt sayı
    
    // Create packet
    variable pkt = Packet do
        header: header,
        length: length,
        data: data,
        crc32: crc_recv
    end
    
    // Verify CRC
    variable crc_calc = packet_calculate_crc(pkt)
    
    if crc_calc == crc_recv do
        io.yazdir_satır("✅ Packet CRC OK")
        return Seçenek.Bazı(pkt)
    elseif do
        io.yazdir_satır("❌ Packet CRC mismatch!")
        return Seçenek.Hiçbiri
    end
end

Sonuç: UART üzerinden gelen corrupt paketler reddedilir. Data integrity garanti edilir.

💡 Example 3: SD Card File Verification

Problem: SD card'dan okunan dosyanın bozulmadığını kontrol etmek

import hal::crc, hal::sdio

function file_calculate_crc(filename: Metin) -> sayı do
    // Open file
    variable file = sdio.file_open(filename, sdio.MODE_READ)
    
    if not file.is_valid() do
        io.yazdir_satır("ERROR: Cannot open file")
        return 0
    end
    
    crc.reset()
    
    // Read file in 512-byte blocks
    variable buffer: Dizi<sayı8> = Dizi.yeni(512)
    
    loop do
        variable bytes_read = sdio.file_read(file, buffer)
        
        if bytes_read == 0 do
            break  // EOF
        end
        
        // Process full 32-bit words
        for i içinde 0..(bytes_read/4) do
            variable word = (buffer[i*4] kayıt sayı) |
                           ((buffer[i*4+1] kayıt sayı) << 8) |
                           ((buffer[i*4+2] kayıt sayı) << 16) |
                           ((buffer[i*4+3] kayıt sayı) << 24)
            crc.accumulate(word)
        end
        
        // Handle remaining bytes
        variable remainder = bytes_read % 4
        if remainder > 0 do
            for i içinde (bytes_read - remainder)..bytes_read do
                crc.accumulate_byte(buffer[i])
            end
        end
    end
    
    sdio.file_close(file)
    
    return crc.get_value()
end

function file_verify(filename: Metin, expected_crc: sayı) -> mantık do
    io.yazdir_satır("Verifying " + filename + "...")
    
    variable calculated = file_calculate_crc(filename)
    
    io.yazdir_satır("Expected:   0x" + expected_crc.onaltılıya())
    io.yazdir_satır("Calculated: 0x" + calculated.onaltılıya())
    
    return calculated == expected_crc
end

Usage: Firmware update dosyalarının SD card'dan okunurken bozulmadığını garanti eder.

✅ Best Practices

  • ✅ Firmware check: Boot sırasında firmware CRC kontrolü do
  • ✅ Communication: Paketlerde CRC kullan (especially wireless)
  • ✅ Storage: Flash/SD card'dan okunan veriler for CRC
  • ✅ Reset before use: Her hesaplama öncesi crc.reset() çağır
  • ❌ Not for crypto: CRC security for not, error detection for
  • ❌ Byte alignment: Non-aligned erişim performans kaybı yaratır
