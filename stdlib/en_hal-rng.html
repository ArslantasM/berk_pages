🎲 hal::rng

Hardware True Random Number Generator

~380 satır ~14 function STM32F4/F7/H7

📖 Genel Bakış

RNG Moduleü, hardware tabanlı gerçek rastgele sayı üretir. Analog noise source kullanarak kriptografik kalitede random değerler sağlar. IoT güvenliği, UUID generation, cryptographic key generation and Monte Carlo simülasyonları for kritiktir.

🔑 Temel Özellikler

  • True Random: Pseudo-random not, analog gürültü kaynağı
  • 32-bit Output: Her okumada tam 32-bit random değer
  • FIPS 140-2 Compliant: Kriptografik standartlara uygun
  • Single Cycle: 40 CPU cycle'da yeni random değer
  • Clock Error Detection: Clock problemi tespit edilir
  • Seed Error Detection: Entropi hatası algılanır
  • Interrupt Support: Data ready interrupt
  • DMA Compatible: Continuous random stream

🚀 Hızlı Getting Started

import hal::rng, hal::rcc

// Enable RNG clock
rcc.periph_clock_enable(rcc.PERIPH_RNG)

// Initialize RNG
rng.init()

// Get random 32-bit value
variable rastgele = rng.get_random()
io.yazdir_satır("Random: 0x" + rastgele.onaltılıya())

// Random range [0, max)
function random_range(max: sayı) -> sayı do
    return (rng.get_random() % max)
end

💡 Example 1: UUID Generation (RFC 4122)

Problem: Unique identifier oluşturmak (IoT device ID)

import hal::rng

struct UUID do
    data: Dizi<sayı8> = Dizi.yeni(16)
end

function uuid_v4_generate() -> UUID do
    variable uuid = UUID do end
    
    // Generate 128 bits (16 bytes) of random data
    for i içinde 0..4 do
        variable rand32 = rng.get_random()
        
        // Split into 4 bytes
        uuid.data[i*4 + 0] = (rand32 >> 24) & 0xFF kayıt sayı8
        uuid.data[i*4 + 1] = (rand32 >> 16) & 0xFF kayıt sayı8
        uuid.data[i*4 + 2] = (rand32 >> 8) & 0xFF kayıt sayı8
        uuid.data[i*4 + 3] = rand32 & 0xFF kayıt sayı8
    end
    
    // Set version (4) and variant (RFC 4122)
    uuid.data[6] = (uuid.data[6] & 0x0F) | 0x40  // Version 4
    uuid.data[8] = (uuid.data[8] & 0x3F) | 0x80  // Variant RFC4122
    
    return uuid
end

function uuid_to_string(uuid: UUID) -> Metin do
    // Format: xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx
    variable hex = ""
    for i içinde 0..16 do
        hex = hex + uuid.data[i].onaltılıya(2)
        if (i == 3 or i == 5 or i == 7 or i == 9) do
            hex = hex + "-"
        end
    end
    return hex
end

function ana() do
    rng.init()
    
    for i içinde 0..5 do
        variable uuid = uuid_v4_generate()
        io.yazdir_satır("UUID: " + uuid_to_string(uuid))
    end
end

💡 Example 2: Cryptographic Key Generation

Problem: AES-256 for 256-bit random key oluşturmak

import hal::rng, hal::crypto

constant AES256_KEY_SIZE: sayı = 32  // 256 bit = 32 bytes

struct CryptoKey do
    data: Dizi<sayı8>,
    size: sayı
end

function generate_aes256_key() -> CryptoKey do
    variable key = CryptoKey do
        data: Dizi.yeni(AES256_KEY_SIZE),
        size: AES256_KEY_SIZE
    end
    
    // Generate 256 bits of cryptographic random data
    for i içinde 0..8 do  // 8 * 32 bits = 256 bits
        variable rand = rng.get_random()
        
        if rng.get_flag(rng.FLAG_CECS) or rng.get_flag(rng.FLAG_SECS) do
            io.yazdir_satır("RNG ERROR: Clock or seed error!")
            return key  // Invalid key
        end
        
        key.data[i*4 + 0] = (rand >> 24) & 0xFF kayıt sayı8
        key.data[i*4 + 1] = (rand >> 16) & 0xFF kayıt sayı8
        key.data[i*4 + 2] = (rand >> 8) & 0xFF kayıt sayı8
        key.data[i*4 + 3] = rand & 0xFF kayıt sayı8
    end
    
    return key
end

function key_to_hex(key: CryptoKey) -> Metin do
    variable hex = ""
    for byte içinde key.data do
        hex = hex + byte.onaltılıya(2)
    end
    return hex
end

function ana() do
    rng.init()
    
    io.yazdir_satır("Generating AES-256 key...")
    variable key = generate_aes256_key()
    io.yazdir_satır("Key: " + key_to_hex(key))
    
    // Use key for AES encryption
    crypto.aes_init(crypto.AES256, key.data)
end

Güvenlik: Hardware RNG Usageı software PRNG'den çok daha güvenlidir.

💡 Example 3: Random Number Distribution (Dice Roll)

Problem: Uniform distribution ile dice roll simulation

import hal::rng

// Random in range [min, max]
function random_range(min: sayı, max: sayı) -> sayı do
    variable range = max - min + 1
    variable rand = rng.get_random()
    
    // Avoid modulo bias using rejection sampling
    variable limit = 0xFFFF_FFFF - (0xFFFF_FFFF % range)
    
    loop do
        if rand < limit do
            return min + (rand % range)
        end
        rand = rng.get_random()
    end
end

function dice_roll() -> sayı do
    return random_range(1, 6)
end

function test_distribution() do
    variable histogram: Dizi<sayı> = Dizi.yeni(6, 0)
    constant ROLLS: sayı = 10000
    
    for i içinde 0..ROLLS do
        variable roll = dice_roll()
        histogram[roll - 1] = histogram[roll - 1] + 1
    end
    
    io.yazdir_satır("Dice roll distribution (" + ROLLS.metne() + " rolls):")
    for i içinde 0..6 do
        variable face = i + 1
        variable count = histogram[i]
        variable percent = (count kayıt kesir / ROLLS kayıt kesir) * 100.0
        
        io.yazdir_satır("  [" + face.metne() + "]: " + count.metne() +
                        " (" + percent.metne() + "%)")
    end
end

function ana() do
    rng.init()
    test_distribution()
end

Sonuç: Uniform distribution ~16.67% her yüz for (ideal: 16.666...%).
