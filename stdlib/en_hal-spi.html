🔄 hal::spi

Serial Peripheral Interface - High Speed Communication

394 satır ~16 function Full-Duplex

📖 Genel Bakış

SPI (Serial Peripheral Interface), yüksek hızlı senkron full-duplex haberleşme protokolüdür. SD kart, flash bellek, LCD display and RF Moduleler for kullanılır.

🔑 Temel Özellikler

  • Master/Slave mode
  • Full-duplex communication
  • High speed (up to 18 Mbps)
  • 8/16 bit data frame
  • MSB/LSB first
  • 4 clock modes (CPOL/CPHA)
  • Hardware NSS/CS management
  • DMA support

💡 Example: SD Card Reader

import hal::spi, hal::gpio

constant CMD0 = 0    // GO_IDLE_STATE
constant CMD8 = 8    // SEND_IF_COND
constant CMD17 = 17  // READ_SINGLE_BLOCK

struct SDCard do
    spi: spi.Handle,
    cs: gpio.Pin
end

function sd_send_cmd(sd: SDCard, cmd: sayı, arg: u32) -> u8 do
    // Select card
    gpio.pin_write(sd.cs, gpio.LOW)
    
    // Send command
    spi.transfer_byte(sd.spi, 0x40 | cmd)
    spi.transfer_byte(sd.spi, (arg >> 24) & 0xFF)
    spi.transfer_byte(sd.spi, (arg >> 16) & 0xFF)
    spi.transfer_byte(sd.spi, (arg >> 8) & 0xFF)
    spi.transfer_byte(sd.spi, arg & 0xFF)
    spi.transfer_byte(sd.spi, 0x95)  // CRC (dummy for most commands)
    
    // Wait for response
    variable response = 0xFF
    her i içinde 0..10 for do
        response = spi.transfer_byte(sd.spi, 0xFF)
        if response != 0xFF ise do
            break
        end
    end
    
    // Deselect card
    gpio.pin_write(sd.cs, gpio.HIGH)
    
    return response
end

function sd_init() -> Sonuç[SDCard, Hata] do
    // SPI setup
    variable spi1 = spi.init(spi.SPI1, do
        mode: spi.MODE_MASTER,
        speed: spi.SPEED_400KHZ,  // Start with low speed
        clock_mode: spi.MODE_0
    end)
    
    // CS pin
    variable cs = gpio.pin_init(gpio.PORT_A, 4, do mode: gpio.MODE_OUTPUT end)
    gpio.pin_write(cs, gpio.HIGH)
    
    variable sd = SDCard do spi: spi1, cs: cs end
    
    // Send 80 clock pulses
    her i içinde 0..10 for do
        spi.transfer_byte(sd.spi, 0xFF)
    end
    
    // Send CMD0 (reset)
    if sd_send_cmd(sd, CMD0, 0) != 0x01 ise do
        return Hata("SD card init failed")
    end
    
    // Increase speed
    spi.set_speed(sd.spi, spi.SPEED_18MHZ)
    
    return Tamam(sd)
end

function sd_read_block(sd: SDCard, block: u32, buffer: [u8; 512]) -> Sonuç[Hiçbir, Hata] do
    // Send CMD17 (read single block)
    gpio.pin_write(sd.cs, gpio.LOW)
    
    sd_send_cmd(sd, CMD17, block)
    
    // Wait for data token
    variable token = 0xFF
    her i içinde 0..1000 for do
        token = spi.transfer_byte(sd.spi, 0xFF)
        if token == 0xFE ise do break end
    end
    
    if token != 0xFE ise do
        gpio.pin_write(sd.cs, gpio.HIGH)
        return Hata("SD read timeout")
    end
    
    // Read data
    her i içinde 0..512 for do
        buffer[i] = spi.transfer_byte(sd.spi, 0xFF)
    end
    
    // Read CRC (2 bytes, ignore)
    spi.transfer_byte(sd.spi, 0xFF)
    spi.transfer_byte(sd.spi, 0xFF)
    
    gpio.pin_write(sd.cs, gpio.HIGH)
    return Tamam(Hiçbir)
end

💡 Example 2: NRF24L01+ Wireless Module

import hal::spi, hal::gpio

constant NRF24_CONFIG = 0x00
constant NRF24_EN_AA = 0x01
constant NRF24_RF_CH = 0x05
constant NRF24_RF_SETUP = 0x06
constant NRF24_STATUS = 0x07
constant NRF24_TX_ADDR = 0x10
constant NRF24_RX_ADDR_P0 = 0x0A

struct NRF24 do
    spi: spi.Handle,
    cs: gpio.Pin,
    ce: gpio.Pin
end

function nrf24_write_reg(nrf: NRF24, reg: u8, value: u8) -> Sonuç[Hiçbir, Hata] do
    gpio.pin_write(nrf.cs, gpio.LOW)
    spi.transfer_byte(nrf.spi, 0x20 | reg)  // W_REGISTER command
    spi.transfer_byte(nrf.spi, value)
    gpio.pin_write(nrf.cs, gpio.HIGH)
    return Tamam(Hiçbir)
end

function nrf24_read_reg(nrf: NRF24, reg: u8) -> u8 do
    gpio.pin_write(nrf.cs, gpio.LOW)
    spi.transfer_byte(nrf.spi, reg & 0x1F)  // R_REGISTER command
    variable value = spi.transfer_byte(nrf.spi, 0x00)
    gpio.pin_write(nrf.cs, gpio.HIGH)
    return value
end

function nrf24_init() -> Sonuç[NRF24, Hata] do
    variable spi1 = spi.init(spi.SPI1, do
        mode: spi.MODE_MASTER,
        speed: spi.SPEED_8MHZ,
        clock_mode: spi.MODE_0
    end)
    
    variable cs = gpio.pin_init(gpio.PORT_A, 4, do mode: gpio.MODE_OUTPUT end)
    variable ce = gpio.pin_init(gpio.PORT_A, 3, do mode: gpio.MODE_OUTPUT end)
    
    gpio.pin_write(cs, gpio.HIGH)
    gpio.pin_write(ce, gpio.LOW)
    
    variable nrf = NRF24 do spi: spi1, cs: cs, ce: ce end
    
    core.delay_ms(100)  // Power-on reset delay
    
    // Configure
    nrf24_write_reg(nrf, NRF24_CONFIG, 0x0E)?       // PWR_UP, CRC enabled
    nrf24_write_reg(nrf, NRF24_EN_AA, 0x01)?        // Enable auto-ack pipe 0
    nrf24_write_reg(nrf, NRF24_RF_CH, 76)?          // Channel 76 (2.476 GHz)
    nrf24_write_reg(nrf, NRF24_RF_SETUP, 0x06)?    // 1Mbps, 0dBm
    
    core.delay_ms(5)
    
    return Tamam(nrf)
end

function nrf24_transmit(nrf: NRF24, data: [u8]) -> Sonuç[Hiçbir, Hata] do
    gpio.pin_write(nrf.cs, gpio.LOW)
    spi.transfer_byte(nrf.spi, 0xA0)  // W_TX_PAYLOAD command
    
    her byte içinde data for do
        spi.transfer_byte(nrf.spi, byte)
    end
    
    gpio.pin_write(nrf.cs, gpio.HIGH)
    
    // Pulse CE to start transmission
    gpio.pin_write(nrf.ce, gpio.HIGH)
    core.delay_us(15)
    gpio.pin_write(nrf.ce, gpio.LOW)
    
    // Wait for TX complete (max 1ms)
    her i içinde 0..100 for do
        variable status = nrf24_read_reg(nrf, NRF24_STATUS)
        if (status & 0x20) != 0 ise do  // TX_DS bit
            nrf24_write_reg(nrf, NRF24_STATUS, 0x20)?  // Clear flag
            return Tamam(Hiçbir)
        end
        core.delay_us(10)
    end
    
    return Hata("TX timeout")
end

function ana() do
    variable nrf = nrf24_init()?
    
    loop do
        variable message = "Hello NRF24!".baytlar()
        nrf24_transmit(nrf, message)?
        io.println("Gönderildi")
        core.delay_ms(1000)
    end
end

💡 Example 3: MAX7219 LED Matrix Driver

import hal::spi, hal::gpio

constant MAX7219_NOOP = 0x00
constant MAX7219_DIGIT0 = 0x01
constant MAX7219_DECODE_MODE = 0x09
constant MAX7219_INTENSITY = 0x0A
constant MAX7219_SCAN_LIMIT = 0x0B
constant MAX7219_SHUTDOWN = 0x0C
constant MAX7219_DISPLAY_TEST = 0x0F

struct MAX7219 do
    spi: spi.Handle,
    cs: gpio.Pin
end

function max7219_write(max: MAX7219, reg: u8, data: u8) do
    gpio.pin_write(max.cs, gpio.LOW)
    spi.transfer_byte(max.spi, reg)
    spi.transfer_byte(max.spi, data)
    gpio.pin_write(max.cs, gpio.HIGH)
end

function max7219_init() -> MAX7219 do
    variable spi1 = spi.init(spi.SPI1, do
        mode: spi.MODE_MASTER,
        speed: spi.SPEED_1MHZ,
        clock_mode: spi.MODE_0
    end)
    
    variable cs = gpio.pin_init(gpio.PORT_A, 4, do mode: gpio.MODE_OUTPUT end)
    gpio.pin_write(cs, gpio.HIGH)
    
    variable max = MAX7219 do spi: spi1, cs: cs end
    
    // Initialize display
    max7219_write(max, MAX7219_SHUTDOWN, 0x01)      // Normal operation
    max7219_write(max, MAX7219_DECODE_MODE, 0x00)   // No decode (raw mode)
    max7219_write(max, MAX7219_SCAN_LIMIT, 0x07)    // Display all 8 digits
    max7219_write(max, MAX7219_INTENSITY, 0x08)     // Medium brightness
    max7219_write(max, MAX7219_DISPLAY_TEST, 0x00)  // Normal mode
    
    return max
end

function max7219_clear(max: MAX7219) do
    her i içinde 1..9 for do
        max7219_write(max, i olarak u8, 0x00)
    end
end

function max7219_draw_pattern(max: MAX7219, pattern: [u8; 8]) do
    her i içinde 0..8 for do
        max7219_write(max, (i + 1) olarak u8, pattern[i])
    end
end

function ana() do
    variable max = max7219_init()
    
    // Smiley face pattern
    variable smiley = [
        0b00111100,
        0b01000010,
        0b10100101,
        0b10000001,
        0b10100101,
        0b10011001,
        0b01000010,
        0b00111100
    ]
    
    max7219_draw_pattern(max, smiley)
    
    loop do
        core.delay_ms(1000)
    end
end

💡 Example 4: W25Q128 Flash Memory

import hal::spi, hal::gpio

constant W25Q_WRITE_ENABLE = 0x06
constant W25Q_READ_DATA = 0x03
constant W25Q_PAGE_PROGRAM = 0x02
constant W25Q_SECTOR_ERASE = 0x20
constant W25Q_READ_STATUS = 0x05
constant W25Q_CHIP_ID = 0x9F

struct W25Q do
    spi: spi.Handle,
    cs: gpio.Pin
end

function w25q_wait_busy(w25q: W25Q) do
    loop do
        gpio.pin_write(w25q.cs, gpio.LOW)
        spi.transfer_byte(w25q.spi, W25Q_READ_STATUS)
        variable status = spi.transfer_byte(w25q.spi, 0x00)
        gpio.pin_write(w25q.cs, gpio.HIGH)
        
        if (status & 0x01) == 0 ise do break end  // BUSY bit
        core.delay_ms(1)
    end
end

function w25q_write_enable(w25q: W25Q) do
    gpio.pin_write(w25q.cs, gpio.LOW)
    spi.transfer_byte(w25q.spi, W25Q_WRITE_ENABLE)
    gpio.pin_write(w25q.cs, gpio.HIGH)
end

function w25q_read_id(w25q: W25Q) -> [u8; 3] do
    gpio.pin_write(w25q.cs, gpio.LOW)
    spi.transfer_byte(w25q.spi, W25Q_CHIP_ID)
    
    variable id = [0u8; 3]
    id[0] = spi.transfer_byte(w25q.spi, 0x00)  // Manufacturer
    id[1] = spi.transfer_byte(w25q.spi, 0x00)  // Memory type
    id[2] = spi.transfer_byte(w25q.spi, 0x00)  // Capacity
    
    gpio.pin_write(w25q.cs, gpio.HIGH)
    return id
end

function w25q_read(w25q: W25Q, addr: u32, buffer: &mut [u8]) -> Sonuç[Hiçbir, Hata] do
    w25q_wait_busy(w25q)
    
    gpio.pin_write(w25q.cs, gpio.LOW)
    spi.transfer_byte(w25q.spi, W25Q_READ_DATA)
    spi.transfer_byte(w25q.spi, (addr >> 16) olarak u8)
    spi.transfer_byte(w25q.spi, (addr >> 8) olarak u8)
    spi.transfer_byte(w25q.spi, addr olarak u8)
    
    her i içinde 0..buffer.uzunluk() for do
        buffer[i] = spi.transfer_byte(w25q.spi, 0x00)
    end
    
    gpio.pin_write(w25q.cs, gpio.HIGH)
    return Tamam(Hiçbir)
end

function w25q_write_page(w25q: W25Q, addr: u32, data: [u8]) -> Sonuç[Hiçbir, Hata] do
    w25q_wait_busy(w25q)
    w25q_write_enable(w25q)
    
    gpio.pin_write(w25q.cs, gpio.LOW)
    spi.transfer_byte(w25q.spi, W25Q_PAGE_PROGRAM)
    spi.transfer_byte(w25q.spi, (addr >> 16) olarak u8)
    spi.transfer_byte(w25q.spi, (addr >> 8) olarak u8)
    spi.transfer_byte(w25q.spi, addr olarak u8)
    
    her byte içinde data for do
        spi.transfer_byte(w25q.spi, byte)
    end
    
    gpio.pin_write(w25q.cs, gpio.HIGH)
    w25q_wait_busy(w25q)
    
    return Tamam(Hiçbir)
end

function w25q_erase_sector(w25q: W25Q, addr: u32) -> Sonuç[Hiçbir, Hata] do
    w25q_wait_busy(w25q)
    w25q_write_enable(w25q)
    
    gpio.pin_write(w25q.cs, gpio.LOW)
    spi.transfer_byte(w25q.spi, W25Q_SECTOR_ERASE)
    spi.transfer_byte(w25q.spi, (addr >> 16) olarak u8)
    spi.transfer_byte(w25q.spi, (addr >> 8) olarak u8)
    spi.transfer_byte(w25q.spi, addr olarak u8)
    gpio.pin_write(w25q.cs, gpio.HIGH)
    
    w25q_wait_busy(w25q)  // Sector erase takes ~20-200ms
    return Tamam(Hiçbir)
end

function ana() do
    variable spi1 = spi.init(spi.SPI1, do
        mode: spi.MODE_MASTER,
        speed: spi.SPEED_8MHZ,
        clock_mode: spi.MODE_0
    end)
    
    variable cs = gpio.pin_init(gpio.PORT_A, 4, do mode: gpio.MODE_OUTPUT end)
    gpio.pin_write(cs, gpio.HIGH)
    
    variable w25q = W25Q do spi: spi1, cs: cs end
    
    // Read chip ID
    variable id = w25q_read_id(w25q)
    io.println("Flash ID: {:02X} {:02X} {:02X}".formatla(id[0], id[1], id[2]))
    
    // Erase sector 0
    w25q_erase_sector(w25q, 0x000000)?
    io.println("Sector erased")
    
    // Write data
    variable data = "BERK Flash Test!".baytlar()
    w25q_write_page(w25q, 0x000000, data)?
    io.println("Data written")
    
    // Read back
    variable buffer = [0u8; 32]
    w25q_read(w25q, 0x000000, &mut buffer)?
    io.println("Read: {}", string.utf8_çöz(buffer)?)
end

💡 Example 5: ILI9341 TFT LCD (240x320 Display)

import hal::spi, hal::gpio

struct ILI9341 do
    spi: spi.Handle,
    cs: gpio.Pin,
    dc: gpio.Pin,   // Data/Command
    rst: gpio.Pin   // Reset
end

function ili9341_cmd(lcd: ILI9341, cmd: u8) do
    gpio.pin_write(lcd.dc, gpio.LOW)   // Command mode
    gpio.pin_write(lcd.cs, gpio.LOW)
    spi.transfer_byte(lcd.spi, cmd)
    gpio.pin_write(lcd.cs, gpio.HIGH)
end

function ili9341_data(lcd: ILI9341, data: u8) do
    gpio.pin_write(lcd.dc, gpio.HIGH)  // Data mode
    gpio.pin_write(lcd.cs, gpio.LOW)
    spi.transfer_byte(lcd.spi, data)
    gpio.pin_write(lcd.cs, gpio.HIGH)
end

function ili9341_init() -> ILI9341 do
    variable spi1 = spi.init(spi.SPI1, do
        mode: spi.MODE_MASTER,
        speed: spi.SPEED_18MHZ,  // High speed for display
        clock_mode: spi.MODE_0
    end)
    
    variable cs = gpio.pin_init(gpio.PORT_A, 4, do mode: gpio.MODE_OUTPUT end)
    variable dc = gpio.pin_init(gpio.PORT_A, 3, do mode: gpio.MODE_OUTPUT end)
    variable rst = gpio.pin_init(gpio.PORT_A, 2, do mode: gpio.MODE_OUTPUT end)
    
    variable lcd = ILI9341 do spi: spi1, cs: cs, dc: dc, rst: rst end
    
    // Hardware reset
    gpio.pin_write(lcd.rst, gpio.HIGH)
    core.delay_ms(10)
    gpio.pin_write(lcd.rst, gpio.LOW)
    core.delay_ms(20)
    gpio.pin_write(lcd.rst, gpio.HIGH)
    core.delay_ms(150)
    
    // Software reset
    ili9341_cmd(lcd, 0x01)
    core.delay_ms(150)
    
    // Power control
    ili9341_cmd(lcd, 0xC0)
    ili9341_data(lcd, 0x23)
    
    ili9341_cmd(lcd, 0xC1)
    ili9341_data(lcd, 0x10)
    
    // Memory access control
    ili9341_cmd(lcd, 0x36)
    ili9341_data(lcd, 0x48)  // RGB order
    
    // Pixel format
    ili9341_cmd(lcd, 0x3A)
    ili9341_data(lcd, 0x55)  // 16-bit RGB565
    
    // Display on
    ili9341_cmd(lcd, 0x11)  // Sleep out
    core.delay_ms(120)
    ili9341_cmd(lcd, 0x29)  // Display on
    
    return lcd
end

function ili9341_fill_screen(lcd: ILI9341, color: u16) do
    // Set window (0,0)-(239,319)
    ili9341_cmd(lcd, 0x2A)  // Column address
    ili9341_data(lcd, 0x00)
    ili9341_data(lcd, 0x00)
    ili9341_data(lcd, 0x00)
    ili9341_data(lcd, 0xEF)
    
    ili9341_cmd(lcd, 0x2B)  // Row address
    ili9341_data(lcd, 0x00)
    ili9341_data(lcd, 0x00)
    ili9341_data(lcd, 0x01)
    ili9341_data(lcd, 0x3F)
    
    ili9341_cmd(lcd, 0x2C)  // Write memory
    
    // Fill with color (240 * 320 pixels)
    gpio.pin_write(lcd.dc, gpio.HIGH)
    gpio.pin_write(lcd.cs, gpio.LOW)
    
    her i içinde 0..(240 * 320) for do
        spi.transfer_byte(lcd.spi, (color >> 8) olarak u8)
        spi.transfer_byte(lcd.spi, color olarak u8)
    end
    
    gpio.pin_write(lcd.cs, gpio.HIGH)
end

function ana() do
    variable lcd = ili9341_init()
    
    // RGB565 colors
    constant RED = 0xF800
    constant GREEN = 0x07E0
    constant BLUE = 0x001F
    constant WHITE = 0xFFFF
    
    loop do
        ili9341_fill_screen(lcd, RED)
        core.delay_ms(1000)
        ili9341_fill_screen(lcd, GREEN)
        core.delay_ms(1000)
        ili9341_fill_screen(lcd, BLUE)
        core.delay_ms(1000)
    end
end

📦 Tipler and Enum'lar

// SPI Instance
enum Instance {
    SPI1, SPI2, SPI3, SPI4, SPI5, SPI6
}

// SPI Mode
enum Mode {
    MODE_MASTER,  // Master mode
    MODE_SLAVE    // Slave mode
}

// Clock Mode (CPOL/CPHA)
enum ClockMode {
    MODE_0,  // CPOL=0, CPHA=0 (idle low, sample on rising)
    MODE_1,  // CPOL=0, CPHA=1 (idle low, sample on falling)
    MODE_2,  // CPOL=1, CPHA=0 (idle high, sample on falling)
    MODE_3   // CPOL=1, CPHA=1 (idle high, sample on rising)
}

// Data Size
enum DataSize {
    DATA_8BIT,   // 8-bit data frame
    DATA_16BIT   // 16-bit data frame
}

// Bit Order
enum BitOrder {
    MSB_FIRST,  // Most significant bit first (default)
    LSB_FIRST   // Least significant bit first
}

// Speed (Baud Rate)
enum Speed {
    SPEED_100KHZ,
    SPEED_400KHZ,
    SPEED_1MHZ,
    SPEED_2MHZ,
    SPEED_4MHZ,
    SPEED_8MHZ,
    SPEED_18MHZ,
    SPEED_42MHZ   // Max speed (platform dependent)
}

struct Config do
    mode: Mode,
    speed: Speed,
    clock_mode: ClockMode,
    data_size: DataSize,
    bit_order: BitOrder,
    nss_mode: NSSMode
end

enum NSSMode {
    NSS_SOFT,     // Software NSS management (manual CS)
    NSS_HARD_IN,  // Hardware NSS input (slave mode)
    NSS_HARD_OUT  // Hardware NSS output (master mode)
}

⚡ Performans İpuçları

  • DMA Usageı: Büyük veri transferleri for (>64 byte) DMA kullanın
  • Clock Speed: Kablo uzunluğuna göre ayarlayın (kısa kablo = yüksek hız)
  • CS Control: CS pinini mümkün olduğunca kısa süre LOW tutun
  • Full-Duplex: Aynı anda gönder/al yaparak hızlandırın
  • Burst Transfer: Tek seferde çok byte transfer edin

⚠️ Important Notes

  • CS (Chip Select) pini manuel kontrol edilmelidir (NSS_SOFT mode)
  • MISO hattı pull-up gerektirebilir (open-collector slave'ler for)
  • Kısa kablo kullanın - yüksek hızda sinyal bütünlüğü kritik (<30cm for >10MHz)
  • Multi-slave for her device'ın ayrı CS pini olmalı
  • SCK, MOSI, MISO pinleri tüm slave'ler arasında paylaşılır
  • Full-duplex modda her gönderilen byte bir byte alır
  • DMA kullanırken buffer alignment Attention edin (4-byte boundary)
  • Clock polarity/phase device datasheet'ine göre ayarlayın

🎯 Uygulama Senaryoları

  • Storage: SD/microSD kart, SPI Flash (W25Q, AT25)
  • Display: TFT LCD (ILI9341, ST7735), OLED, E-Paper
  • Wireless: NRF24L01+, RFM69, LoRa (SX1278)
  • Sensors: BME280, MAX6675 (thermocouple), ADXL345
  • ADC/DAC: MCP3008, MCP4922
  • LED Matrix: MAX7219, WS2812 (via DMA)

📚 Platform Desteği

  • STM32 F1: SPI1/SPI2, up to 18 Mbps
  • STM32 F4: SPI1-3, up to 42 Mbps
  • STM32 F7: SPI1-6, up to 54 Mbps
  • STM32 H7: SPI1-6, up to 100 Mbps
  • ESP32: SPI2/SPI3 (HSPI/VSPI), up to 80 MHz
  • GD32VF103: SPI0/SPI1/SPI2 (RISC-V)
