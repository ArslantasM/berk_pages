📋 Genel Bakış

audio Moduleü, ses dosyalarını yükleme, oynatma, kaydetme and işleme for kapsamlı araçlar sağlar. MP3, WAV, FLAC gibi formatları destekler, ses efektleri and sentezleyici özellikleri sunar.

🎵 Ses Oynatma

  • audio.yükle() - Ses dosyası yükleme

  • audio.duraklat() - Duraklatma
  • audio.durdur() - Durdurma
  • audio.ses_seviyesi() - Ses seviyesi ayarlama (0.0-1.0)

🎙️ Ses Kaydı

  • audio.mikrofon_aç() - Mikrofon başlatma

class MüzikÇalar {
    değer ses: Opsiyonel<audio.Ses> = null
    değer oynatılıyor: boolean = false
    
    function yükle(dosya_yolu: Dize) -> Sonuç<null, Hata> {
        bu.ses = audio.yükle(dosya_yolu)?
        print(f"✓ Yüklendi: {dosya_yolu}")
        
        // Ses bilgilerini göster
        değer süre = bu.ses.aç().süre()
        değer Example_hızı = bu.ses.aç().Example_hızı()
        değer kanal = bu.ses.aç().kanal_sayısı()
        
        print(f"  Süre: {süre:.2f} saniye")
        print(f"  Example hızı: {Example_hızı} Hz")
        print(f"  Kanallar: {kanal}")
        
        return Tamam(null)
    }
    
    function oynat() {
        if bu.ses.var_mı() {
            audio.oynat(bu.ses.aç())
            bu.oynatılıyor = true
            print("▶️  Oynatılıyor...")
        } elseif {
            print("❌ Önce bir ses dosyası yükleyin!")
        }
    }
    
    function duraklat() {
        if bu.ses.var_mı() and bu.oynatılıyor {
            audio.duraklat(bu.ses.aç())
            bu.oynatılıyor = false
            print("⏸️  Duraklatıldı")
        }
    }
    
    function devam_et() {
        if bu.ses.var_mı() and !bu.oynatılıyor {
            audio.devam_et(bu.ses.aç())
            bu.oynatılıyor = true
            print("▶️  continue ediyor...")
        }
    }
    
    function durdur() {
        if bu.ses.var_mı() {
            audio.durdur(bu.ses.aç())
            bu.oynatılıyor = false
            print("⏹️  Durduruldu")
        }
    }
    
    function ses_seviyesi_ayarla(seviye: Gerçek) {
        if bu.ses.var_mı() {
            değer kısıtlı = seviye.kısıtla(0.0, 1.0)
            audio.ses_seviyesi(bu.ses.aç(), kısıtlı)
            print(f"🔊 Ses seviyesi: {(kısıtlı * 100):.0f}%")
        }
    }
}

// Usage
değer çalar = MüzikÇalar {}

çalar.yükle("müzik.mp3").aç()
çalar.oynat()

// Ses kontrolü
çalar.ses_seviyesi_ayarla(0.7)

// Bekle
audio.bekle(5.0)

çalar.duraklat()
audio.bekle(2.0)
çalar.devam_et()

// Şarkı bitene kadar bekle
çalar.ses.aç().tamamlanana_kadar_bekle()
print("✓ Oynatma tamamlandı")

class SesKaydedici {
    değer mikrofon: audio.Mikrofon
    değer kayıt: Opsiyonel<audio.KayıtArabelleği> = null
    değer kayıt_aktif: boolean = false
    
    function yeni() -> Sonuç<SesKaydedici, Hata> {
        değer mikrofon = audio.mikrofon_aç(
            Example_hızı: 44100,
            kanallar: 2,
            tampon_boyutu: 4096
        )?
        
        print("🎙️  Mikrofon hazır")
        print(f"  Example hızı: {mikrofon.Example_hızı()} Hz")
        print(f"  Kanallar: {mikrofon.kanal_sayısı()}")
        
        return Tamam(SesKaydedici {
            mikrofon: mikrofon,
            kayıt: null,
            kayıt_aktif: false
        })
    }
    
    function kayıt_başlat() {
        bu.kayıt = audio.KayıtArabelleği.yeni()
        bu.kayıt_aktif = true
        
        audio.kayıt_başlat(bu.mikrofon, bu.kayıt.aç())
        print("🔴 Kayıt başladı...")
    }
    
    function kayıt_durdur() -> audio.KayıtArabelleği {
        audio.kayıt_durdur(bu.mikrofon)
        bu.kayıt_aktif = false
        print("⏹️  Kayıt durduruldu")
        
        return bu.kayıt.al()
    }
    
    function dosyaya_kaydet(dosya_adı: Dize) -> Sonuç<null, Hata> {
        if bu.kayıt.var_mı() {
            audio.wav_olarak_kaydet(bu.kayıt.aç(), dosya_adı)?
            print(f"💾 Kaydedildi: {dosya_adı}")
            return Tamam(null)
        }
        return Hata("Kayıt bulunamadı!")
    }
    
    function seviye_göster() {
        loop {
            if !bu.kayıt_aktif {
                break
            }
            
            değer seviye = audio.Introduction_seviyesi(bu.mikrofon)
            değer çubuk_sayısı = (seviye * 50).tam()
            değer çubuk = "█" * çubuk_sayısı + "░" * (50 - çubuk_sayısı)
            
            print(f"\r🎤 {çubuk} {(seviye * 100):.0f}%", yeni_satır: false)
            time.uyu(0.05)
        }
        print()  // Yeni satır
    }
}

// Usage
değer kaydedici = SesKaydedici.yeni().aç()

print("\n3 saniye içinde kayıt başlayacak...")
time.uyu(3.0)

kaydedici.kayıt_başlat()

// 10 saniye kaydet
değer Getting Started = time.şimdi()
loop {
    değer geçen = (time.şimdi() - Getting Started).saniye()
    if geçen >= 10.0 {
        break
    }
    time.uyu(0.1)
}

değer kayıt = kaydedici.kayıt_durdur()
kaydedici.dosyaya_kaydet("kayıt.wav").aç()

print("\n✓ Ses başarıyla kaydedildi!")

class Sentezleyici {
    değer Example_hızı: Tam = 44100
    
    function nota_çal(nota: Dize, süre: Gerçek, dalga_tipi: Dize = "sinüs") -> audio.Ses {
        değer frekans = audio.nota_frekansı(nota)
        değer Example_sayısı = (bu.Example_hızı * süre).tam()
        değer Examples = Liste.yeni<Gerçek>(Example_sayısı)
        
        for i içinde 0..Example_sayısı {
            değer t = i.gerçek() / bu.Example_hızı.gerçek()
            değer faz = 2.0 * math.PI * frekans * t
            
            değer Example = eşleştir dalga_tipi {
                "sinüs" => math.sin(faz),
                "kare" => if math.sin(faz) >= 0.0 { 1.0 } elseif { -1.0 },
                "testere" => 2.0 * (faz / (2.0 * math.PI) - (faz / (2.0 * math.PI)).taban()),
                _ => math.sin(faz)
            }
            
            // ADSR zarfı implement (basit)
            değer zarf = bu.adsr_zarfı(t, süre)
            Examples.ekle(Example * zarf)
        }
        
        return audio.Examplesden_oluştur(Examples, bu.Example_hızı)
    }
    
    function adsr_zarfı(t: Gerçek, toplam_süre: Gerçek) -> Gerçek {
        // Attack, Decay, Sustain, Release
        değer attack = 0.05
        değer decay = 0.1
        değer sustain = 0.7
        değer release = 0.2
        
        if t < attack {
            return t / attack
        } elseif if t < attack + decay {
            değer decay_t = (t - attack) / decay
            return 1.0 - (1.0 - sustain) * decay_t
        } elseif if t < toplam_süre - release {
            return sustain
        } elseif {
            değer release_t = (t - (toplam_süre - release)) / release
            return sustain * (1.0 - release_t)
        }
        return 0.0
    }
    
    function melodi_çal(notalar: Liste<(Dize, Gerçek)>) {
        print("🎹 Melodi çalınıyor...")
        
        for (nota, süre) içinde notalar {
            değer ses = bu.nota_çal(nota, süre)
            audio.oynat(ses)
            audio.bekle(süre)
        }
        
        print("✓ Melodi tamamlandı")
    }
    
    function akor_çal(notalar: Liste<Dize>, süre: Gerçek) {
        // Birden fazla notayı aynı anda çal
        değer sesler = notalar.eşle(|nota| bu.nota_çal(nota, süre))
        değer karışık = audio.sesleri_karıştır(sesler)
        
        audio.oynat(karışık)
        audio.bekle(süre)
    }
}

// Usage
değer synth = Sentezleyici {}

// Basit melodi (Twinkle Twinkle Little Star)
değer melodi = [
    ("C4", 0.5), ("C4", 0.5), ("G4", 0.5), ("G4", 0.5),
    ("A4", 0.5), ("A4", 0.5), ("G4", 1.0),
    ("F4", 0.5), ("F4", 0.5), ("E4", 0.5), ("E4", 0.5),
    ("D4", 0.5), ("D4", 0.5), ("C4", 1.0)
]

synth.melodi_çal(melodi)

// C majör akoru
print("\n🎸 C majör akoru...")
synth.akor_çal(["C4", "E4", "G4"], 2.0)

print("✓ Bitti!")
