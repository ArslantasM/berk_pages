🚀 Hızlı Getting Started

import io

// Dosya okuma
variable içerik = io.dosya_oku("data.txt")
print(içerik)

// Dosya yazma
io.dosya_yaz("output.txt", "Merhaba BERK!")

// Satır satır okuma
variable satırlar = io.satırları_oku("data.txt")
for satır içinde satırlar do
    print(satır)
end

// Buffered okuma (büyük dosyalar for)
variable dosya = io.aç("büyük_dosya.txt", "r")
variable okuyucu = io.buffered_reader(dosya)
while true do
    variable satır = io.satır_oku(okuyucu)
    if satır.yok_mu() ise break end
    print(satır)
end
io.kapat(dosya)

Örnek 1: Log Dosyası Yazma

import io, time

function log_yaz(mesaj: string) do
    variable zaman = time.şimdi_str()
    variable log_satır = "[" + zaman + "] " + mesaj + "\n"
    io.dosya_ekle("app.log", log_satır)
end

log_yaz("Uygulama başlatıldı")
log_yaz("Kullanıcı Introduction yaptı")
log_yaz("İşlem tamamlandı")

Örnek 2: Büyük Dosya İşleme

import io, string

function büyük_dosya_işle(girdi: string, çıktı: string) do
    variable girdi_dosya = io.aç(girdi, "r")
    variable çıktı_dosya = io.aç(çıktı, "w")
    
    variable okuyucu = io.buffered_reader(girdi_dosya, 8192)
    variable yazıcı = io.buffered_writer(çıktı_dosya, 8192)
    
    while true do
        variable satır = io.satır_oku(okuyucu)
        if satır.yok_mu() ise break end
        
        // İşleme
        variable işlenmiş = string.büyük_harf(satır.değer())
        io.buffer_yaz(yazıcı, işlenmiş + "\n")
    end
    
    io.buffer_flush(yazıcı)
    io.kapat(çıktı_dosya)
    io.kapat(girdi_dosya)
end

büyük_dosya_işle("input.txt", "output.txt")
