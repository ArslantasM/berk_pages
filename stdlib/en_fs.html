🚀 Hızlı Getting Started

import fs

// Dosya işlemleri
fs.dosya_oluştur("test.txt")
fs.dosya_yaz("test.txt", "Merhaba BERK")
variable içerik = fs.dosya_oku("test.txt")
fs.dosya_sil("test.txt")

// Dizin işlemleri
fs.dizin_oluştur("yeni_klasör")
variable dosyalar = fs.dizin_listele(".")
fs.dizin_sil("yeni_klasör")

// Path işlemleri
variable tam_yol = fs.path_join("/home/user", "documents", "file.txt")
variable dizin = fs.path_dirname(tam_yol)  // "/home/user/documents"
variable dosya = fs.path_basename(tam_yol)  // "file.txt"
variable uzantı = fs.path_extension(tam_yol)  // ".txt"

// Metadata
variable meta = fs.metadata("config.json")
print("Boyut: " + meta.boyut.yazıya() + " bytes")
print("Dosya mı: " + meta.dosya_mı.yazıya())
print("Dizin mi: " + meta.dizin_mi.yazıya())

// Var mı kontrol
if fs.var_mı("config.json") ise do
    print("Config dosyası mevcut")
end

Recursive File Search

import fs, string

function dosya_bul(dizin: string, pattern: string) -> Dizi do
    variable sonuçlar = []
    variable entries = fs.dizin_listele(dizin)
    
    her entry içinde entries for do
        variable tam_yol = fs.path_join(dizin, entry)
        variable meta = fs.metadata(tam_yol)
        
        if meta.dizin_mi ise do
            // Recursive arama
            variable alt_sonuçlar = dosya_bul(tam_yol, pattern)
            sonuçlar = sonuçlar + alt_sonuçlar
        end else if string.içerir_mi(entry, pattern) ise do
            sonuçlar.ekle(tam_yol)
        end
    end
    
    return sonuçlar
end

// Tüm .berk dosyalarını bul
variable berk_dosyaları = dosya_bul(".", ".berk")
her dosya içinde berk_dosyaları for do
    print(dosya)
end

Backup Utility

import fs, time

function yedek_al(kaynak_dizin: string, hedef_dizin: string) do
    variable zaman_damgası = time.format(time.şimdi(), "%Y%m%d_%H%M%S")
    variable yedek_adı = "backup_" + zaman_damgası
    variable yedek_yolu = fs.path_join(hedef_dizin, yedek_adı)
    
    fs.dizin_oluştur(yedek_yolu)
    
    function kopyala_recursive(kaynak: string, hedef: string) do
        variable entries = fs.dizin_listele(kaynak)
        
        her entry içinde entries for do
            variable kaynak_yol = fs.path_join(kaynak, entry)
            variable hedef_yol = fs.path_join(hedef, entry)
            variable meta = fs.metadata(kaynak_yol)
            
            if meta.dizin_mi ise do
                fs.dizin_oluştur(hedef_yol)
                kopyala_recursive(kaynak_yol, hedef_yol)
            end else do
                fs.dosya_kopyala(kaynak_yol, hedef_yol)
                print("Kopyalandı: " + kaynak_yol)
            end
        end
    end
    
    kopyala_recursive(kaynak_dizin, yedek_yolu)
    print("Yedek tamamlandı: " + yedek_yolu)
end

yedek_al("./proje", "./yedekler")
