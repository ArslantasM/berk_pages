BERK Programlama Dili

Veri Tipleri

BERK statik tipli bir dildir and çeşitli built-in veri tipleri sunar.

Temel Tipler

Sayı (Number)

integer and ondalıklı sayılar for:

variable tam = 42
variable float = 3.14
variable negatif = -100
variable bilimsel = 1.5e10  // 15000000000

string (String)

Metin değerleri:

variable isim = "Ahmet"
variable mesaj = 'Merhaba Dünya!'
variable çok_satır = """
    Bu
    çok satırlı
    bir metindir
"""

String interpolation:

variable ad = "Ali"
variable yaş = 25
print("Benim adım {ad} and {yaş} yaşındayım")
// Çıktı: Benim adım Ali and 25 yaşındayım

Boolean (boolean)

true/false değerleri:

variable doğru_mu = true  // true
variable yanlış_mı = false  // false

// İngilizce
var is_valid = true
var is_empty = false

Null

Değer yokluğu:

variable değer = null
variable sonuç: Sayı? = null  // Nullable tip

Koleksiyon Tipleri

Liste (List)

Sıralı elemanlar koleksiyonu:

variable sayılar = [1, 2, 3, 4, 5]
variable isimler: Liste<string> = ["Ahmet", "Mehmet", "Ayşe"]
variable karışık = [1, "iki", 3.0, true]  // Mixed types

// Erişim
print(sayılar[0])   // 1
print(isimler[2])   // Ayşe

// Değiştirme
sayılar[0] = 10

// Uzunluk
print(sayılar.uzunluk)  // 5

Liste metodları:

variable liste = [1, 2, 3]

liste.ekle(4)           // [1, 2, 3, 4]
liste.çıkar(1)          // [1, 3, 4]
liste.içerir(3)         // true
liste.temizle()         // []

Demet (Tuple)

constant boyutlu, farklı tipli elemanlar:

variable konum = (41.0082, 28.9784)  // (lat, lon)
variable kişi = ("Ahmet", 30, "İstanbul")

// Destructuring
variable (lat, lon) = konum
variable (ad, yaş, şehir) = kişi

// İndeks erişimi
print(konum.0)  // 41.0082
print(kişi.1)   // 30

Harita (Map/Dictionary)

Anahtar-değer çiftleri:

variable sözlük = {
    "ad": "Ahmet",
    "yaş": 30,
    "şehir": "İstanbul"
}

// Erişim
print(sözlük["ad"])  // Ahmet

// Ekleme/Güncelleme
sözlük["meslek"] = "Mühendis"
sözlük["yaş"] = 31

// Kontrol
if sözlük.içerir("ad") {
    print("Ad bulundu")
}

// Tip belirtimi
variable Notes: Harita<string, Sayı> = {
    "Mathematics": 85,
    "Fizik": 90,
    "Kimya": 78
}

Küme (Set)

Benzersiz elemanlar koleksiyonu:

variable kume = {1, 2, 3, 3, 2, 1}  // {1, 2, 3}

kume.ekle(4)      // {1, 2, 3, 4}
kume.çıkar(2)     // {1, 3, 4}
kume.içerir(3)    // true

// Set işlemleri
variable a = {1, 2, 3}
variable b = {2, 3, 4}

variable birleşim = a | b     // {1, 2, 3, 4}
variable kesişim = a & b      // {2, 3}
variable fark = a - b         // {1}

Struct Tipleri

Özel veri yapıları tanımlama:

tür Kişi = struct {
    ad: string,
    yaş: Sayı,
    email: string
}

// Instance oluşturma
variable kişi = Kişi {
    ad: "Ahmet",
    yaş: 30,
    email: "ahmet@example.com"
}

// Alan erişimi
print(kişi.ad)      // Ahmet
print(kişi.yaş)     // 30

// Güncelleme
kişi.yaş = 31

Enum Tipleri

constant değerler kümesi:

tür Renk = enum {
    Kırmızı,
    Yeşil,
    Mavi
}

variable favori = Renk.Kırmızı

eş favori {
    Renk.Kırmızı => print("Kırmızı seçildi"),
    Renk.Yeşil => print("Yeşil seçildi"),
    Renk.Mavi => print("Mavi seçildi")
}

Enum değerlerle:

tür HttpDurum = enum {
    Tamam = 200,
    Bulunamadı = 404,
    Sunucu Hatası = 500
}

variable durum = HttpDurum.Tamam
print(durum.değer)  // 200

Sonuç Tipi (Result)

Hata yönetimi for:

tür Sonuç<T, E> = enum {
    Tamam(T),
    Hata(E)
}

işlev böl(a: Sayı, b: Sayı) -> Sonuç<Sayı, string> {
    if b == 0 {
        hata return "Sıfıra bölme!"
    }
    tamam return a / b
}

variable sonuç = böl(10, 2)
eş sonuç {
    Tamam(değer) => print("Sonuç: ", değer),
    Hata(mesaj) => print("HATA: ", mesaj)
}

Seçenek Tipi (Option)

Nullable alternatifi:

tür Seçenek<T> = enum {
    Bazı(T),
    Hiç
}

işlev bul(liste: Liste<Sayı>, hedef: Sayı) -> Seçenek<Sayı> {
    loop i 0'dan liste.uzunluk'a kadar {
        if liste[i] == hedef {
            geri return Bazı(i)
        }
    }
    geri return Hiç
}

variable indeks = bul([1, 2, 3, 4], 3)
eş indeks {
    Bazı(i) => print("Bulundu: ", i),
    Hiç => print("Bulunamadı")
}

Tip Dönüşümleri

Açık Dönüşüm (Casting)

variable x = 10
variable y = x as float  // 10.0

variable metin = "42"
variable sayı = metin.parse<Sayı>()  // 42

Tip Kontrol

variable değer: Any = "Hello"

if değer is string {
    print("Bu bir string")
}

if değer is Sayı {
    print("Bu bir sayı")
}

Tip Takma Adları (Type Alias)

tür KullanıcıID = Sayı
tür Email = string
tür Koordinat = (Sayı, Sayı)

variable id: KullanıcıID = 12345
variable email: Email = "user@example.com"
variable konum: Koordinat = (41.0, 29.0)

Boyutlar and Sınırlar

constant SAYI_MIN = -9223372036854775808  // i64 min
constant SAYI_MAX = 9223372036854775807   // i64 max

constant ONDALIK_MIN = -1.7976931348623157e+308  // f64 min
constant ONDALIK_MAX = 1.7976931348623157e+308   // f64 max

constant YAZI_MAX_UZUNLUK = 2147483647  // ~2GB

Tip Çıkarımı

BERK güçlü tip çıkarımı yapar:

// Tip otomatik belirlenir
variable sayı = 42            // Sayı
variable pi = 3.14            // float
variable isim = "Ahmet"       // string
variable doğru_mu = true     // Boolean
variable liste = [1, 2, 3]    // Liste<Sayı>

// Karmaşık çıkarımlar
variable harita = {           // Harita<string, Sayı>
    "a": 1,
    "b": 2
}

variable function = (x) => x * 2  // (Sayı) -> Sayı

Best Practices

✅ Yapılması Gerekenler

// Açık tip belirtimi karmaşık durumlarda
variable kullanıcılar: Liste<Kullanıcı> = []

// Immutable veri for const kullan
constant API_ANAHTARI = "gizli_anahtar"

// Result tipi hata yönetimi for
işlev dosya_oku(yol: string) -> Sonuç<string, Hata> { ... }

❌ Kaçınılması Gerekenler

// Gereksiz tip belirtimi
variable x: Sayı = 10  // "variable x = 10" yeterli

// Any tipini fazla kullanmak
variable karışık: Any = ...  // Mümkünse spesifik tip kullan

// Null yerine Option kullan
variable değer = null  // Option<T> tercih et

Example: Tüm Tipler

işlev ana() {
    // Temel tipler
    variable sayı: Sayı = 42
    variable float: float = 3.14
    variable metin: string = "Hello"
    variable bool: Boolean = true
    
    // Koleksiyonlar
    variable liste: Liste<Sayı> = [1, 2, 3]
    variable demet: (Sayı, string) = (1, "bir")
    variable harita: Harita<string, Sayı> = {"a": 1}
    variable küme: Küme<Sayı> = {1, 2, 3}
    
    // Özel tipler
    tür Nokta = struct { x: Sayı, y: Sayı }
    variable p = Nokta { x: 10, y: 20 }
    
    // Sonuç tipi
    variable sonuç: Sonuç<Sayı, string> = Tamam(42)
}
