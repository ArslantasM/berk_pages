स्ट्रिंग मॉड्यूल फ़ंक्शंस

केस रूपांतरण

use string;

let String = "Merhaba Dünya"
print(string::büyüt(String))         // "MERHABA DÜNYA"
print(string::küçült(String))        // "merhaba dünya"
print(string::baş_harfi_büyüt(String)) // "Merhaba dünya"

ट्रिम और पैड

let s = "  merhaba  "
print(string::trim(s))         // "merhaba"
print(string::trim_start(s))   // "merhaba  "
print(string::trim_end(s))     // "  merhaba"

let x = "42"
print(string::pad_left(x, 5, '0'))  // "00042"

प्रतिस्थापित करें

let String = "Merhaba Dünya"

// Tüm oluşumları borrow_mut
let s = "test test test"

रेगेक्स समर्थन

use regex::Regex;

let re = Regex::new(r"\d{3}-\d{4}")
let telefon = "555-1234"

if re.eşleşir(telefon) do
    print("Geçerli telefon")
end

// Tüm eşleşmeleri bul
let String = "Email: ali@test.com and ayse@test.com"
let re2 = Regex::new(r"[a-z]+@[a-z]+\.[a-z]+")
for eşleşme in re2.tüm_eşleşmeler(String) do
    print(eşleşme)
end

स्ट्रिंग बिल्डर

कुशल स्ट्रिंग संयोजन के लिए:

use string::StringBuilder;

let sb = StringBuilder::new()
sb.push("Merhaba")
sb.push(" ")
sb.push("Dünya")
sb.ekle_satır("!")

let sonuç = sb.inşa_et()
print(sonuç)  // "Merhaba Dünya!\n"

प्रदर्शन युक्तियाँ

  1. ✅ स्ट्रिंगबिल्डर का प्रयोग करें(एकाधिक जुड़ाव के लिए)

    // ❌ Yavaş
let s = ""
for i in 0..1000 do
    s += "x"
end

// ✅ Hızlı
let sb = StringBuilder::new()
for i in 0..1000 do
    sb.push("x")
end

  2. ✅ स्ट्रिंग स्लाइस का प्रयोग करें(बिना नकल किये)

    let s = "Merhaba Dünya"
let alt = s[0..7]  // "Merhaba" (zero-copy)

  3. ✅ आकार पहले से आरक्षित रखें

    let sb = StringBuilder::with_capacity(1000)
