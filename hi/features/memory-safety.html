स्मृति सुरक्षा

बर्क,जंग-स्तर की स्मृति सुरक्षायह एक आधुनिक सिस्टम प्रोग्रामिंग भाषा है जो बीईआरके में संकलन-समय पर सेगमेंटेशन दोष, नल पॉइंटर डीरेफरेंस और बफर ओवरफ्लो को कैप्चर करती है।

बुनियादी सुरक्षा गारंटी

1. अशक्त सुरक्षा

बर्क मेंnullमानSeçenekइसे स्पष्ट रूप से (विकल्प) प्रकार से नियंत्रित किया जाता है:

// ❌ Err: Null olamaz
let isim: String = null  // Derleme hatası!

// ✅ Doğru: Seçenek use
let isim: Seçenek<String> = Hiçbiri

eşleştir isim do
    Bir(değer) => print("İsim: " + değer),
    Hiçbiri => print("İsim yok")
end

2. सीमा नियंत्रण

ऐरे एक्सेस स्वचालित रूप से जांचे जाते हैं:

let sayılar = [1, 2, 3, 4, 5]

// ❌ Runtime hatası: Index out of bounds
let x = sayılar[10]  // Panic!

// ✅ Güvenli erişim
if 10 < sayılar.len() do
    let x = sayılar[10]
yoksa do
    print("Geçersiz index")
end

टाइप 3 सुरक्षा

इसकी मजबूत प्रकार की प्रणाली के साथ, रनटाइम त्रुटियां संकलन-समय पर पकड़ी जाती हैं:

// ❌ Err: Tip uyuşmazlığı
let x: sayı = "merhaba"  // Derleme hatası!

// ❌ Err: İmplicit dönüşüm yok
let y: kesir = 42  // Derleme hatası!

// ✅ Doğru: Açık dönüşüm
let y: kesir = 42.0
// or
let y: kesir = kayıt(kesir, 42)

स्वामित्व प्रणाली

बर्क जंग-प्रेरित स्वामित्व प्रणाली का उपयोग करता है:

बुनियादी नियम

  1. प्रत्येक मान का एक मान होता हैमालिकवहाँ है
  2. इसके साथ हीकेवल एक मालिकयह हो सकता था
  3. मूल्य जब मालिक दायरे से बाहर चला जाता हैस्वचालित रूप से साफ़ किया गया

स्वामित्व स्थानांतरण (स्थानांतरण)

function örnek() do
    let metin1 = "Merhaba"
    let metin2 = move metin1  // Sahiplik transfer edildi
    
    // ❌ Err: metin1 artık geçerli not
    print(metin1)  // Derleme hatası!
    
    // ✅ Doğru: metin2 now sahip
    print(metin2)
end

सन्दर्भ (साझा करें)

function uzunluk_bul(share s: String) -> sayı do
    return s.len()
end

function örnek() do
    let metin1 = "Merhaba"
    let len = uzunluk_bul(share metin1)
    
    // ✅ Doğru: metin1 hala geçerli
    print(metin1)  // Çalışır!
    print("Uzunluk: " + len)
end

विनिमेय संदर्भ

function büyüt(share değiştirilebilir s: String) do
    s += " Dünya!"
end

function örnek() do
    let metin1 = "Merhaba"
    büyüt(share değiştirilebilir metin1)
    print(metin1)  // "Merhaba Dünya!"
end

जीवन काल

बीईआरके कंपाइलर स्वचालित रूप से संदर्भों की वैधता की जांच करता है:

// ❌ Err: Dangling reference
function hatalı() -> share String do
    let yerel = "merhaba"
    return share yerel  // Derleme hatası! yerel scope dışına çıkacak
end

// ✅ Doğru: Sahipliği döndür
function true() -> String do
    let yerel = "merhaba"
    return move yerel  // Sahiplik transfer edildi
end

डेटा रेस रोकथाम

एकल प्रिंटर या एकाधिक रीडर

// ❌ Err: Aynı anda birden fazla değiştirilebilir referans
let x = 5
let ref1 = share değiştirilebilir x
let ref2 = share değiştirilebilir x  // Derleme hatası!

// ✅ Doğru: Çoklu immutable referans
let x = 5
let ref1 = share x
let ref2 = share x  // OK

थ्रेड-सुरक्षित डेटा संरचनाएँ

use thread::Mutex;

let count_occurrences = new Mutex(0)

function artır() do
    let kilit = count_occurrences.kilit()
    *kilit += 1
    // Kilit otomatik serbest bırakılır
end

सामान्य स्मृति त्रुटियाँ और निवारण

1.उपयोग के बाद-मुक्त

सी/सी++ में:

int* ptr = malloc(sizeof(int));
free(ptr);
*ptr = 5;  // ❌ Use-after-free bug!

बर्क में:

let x = new 42
serbest_bırak(x)
print(*x)  // ❌ Derleme hatası! x artık geçerli not

2. डबल फ्री

सी/सी++ में:

int* ptr = malloc(sizeof(int));
free(ptr);
free(ptr);  // ❌ Double free bug!

बर्क में:

let x = new 42
serbest_bırak(x)
serbest_bırak(x)  // ❌ Derleme hatası! x artık geçerli not

3. बफर अतिप्रवाह

सी/सी++ में:

int arr[5];
arr[10] = 42;  // ❌ Buffer overflow!

बर्क में:

let arr = [0; 5]
arr[10] = 42  // ❌ Runtime panic! (debug modda)
              // Optimized modda bounds check yapılır

स्मार्ट पॉइंटर्स

बॉक्स - ढेर आवंटन

let x = Kutu::new(42)
print(*x)  // 42
// Kutu scope dışına çıkınca otomatik temizlenir

आरसी (संदर्भ गणना के साथ)

use bellek::Rc;

let x = Rc::new(42)
let y = x.klon()  // Referans sayısı artar
// Her iki let de aynı veriyi gösterir
// Son referans yok olunca bellek temizlenir

आर्क (परमाणु संदर्भ गिनती के साथ - थ्रेड सुरक्षित)

use bellek::Arc;
use thread;

let veri = Arc::new(vec![1, 2, 3])

let thread1 = thread::spawn(move || do
    print(veri[0])
end)

let thread2 = thread::spawn(move || do
    print(veri[1])
end)
