ऑपरेटर प्राथमिकता

बीईआरके प्रोग्रामिंग भाषा में ऑपरेटरों का प्राथमिकता क्रम और संयोजन निर्देश। जैसे-जैसे प्राथमिकता स्तर बढ़ता है, ऑपरेटरों का मूल्यांकन पहले किया जाता है।

📊 प्राथमिकता तालिका

प्राथमिकता संचालिका स्पष्टीकरण विलयन
1 (सर्वोच्च) () [] . कोष्ठक, सूचकांक, सदस्य पहुंच बाएं से दायां
2 ++ -- ! ~ + - यूनरी ऑपरेटर्स (वृद्धि, कमी, नहीं, बिटवाइज़ नहीं, यूनरी +/-) दाएं से बाएं
3 ** घातांक दाएं से बाएं
4 * / % गुणा, भाग, मॉड बाएं से दायां
5 + - जोड़, घटाव बाएं से दायां
6 << >> बिट शिफ्ट बाएं से दायां
7 < <= > >= संबंधपरक तुलना बाएं से दायां
8 == != समानता की जाँच बाएं से दायां
9 & बिटवाइज़ और बाएं से दायां
10 ^ बिटवाइज़ XOR बाएं से दायां
11 | बिटवाइज़ या बाएं से दायां
12 && ve and तार्किक तथा बाएं से दायां
13 || veya or तार्किक या बाएं से दायां
14 ? : टर्नरी कंडीशन ऑपरेटर दाएं से बाएं
15 = += -= *= /= %= असाइनमेंट ऑपरेटर दाएं से बाएं
16 (न्यूनतम) , अल्पविराम संचालक बाएं से दायां

💡उदाहरण

प्राथमिकता का क्रम

// Çarpma toplama önce gelir
let sonuç = 5 + 3 * 2  // = 5 + 6 = 11 (not 16)

// Parantez en yüksek önceliğe sahip
let sonuç2 = (5 + 3) * 2  // = 8 * 2 = 16

// Üs alma çarpmadan önce
let sonuç3 = 2 * 3 ** 2  // = 2 * 9 = 18

// Mantıksal operatörler
let koşul = true or false and false  // = true (and önce)
let koşul2 = (true or false) and false  // = false

जुड़ने की दिशा

// Soldan sağa (çoğu operatör)
let a = 10 - 5 - 2  // = (10 - 5) - 2 = 3

// Sağdan sola (atama)
let x, y, z
x = y = z = 5  // Sağdan başlar: z=5, y=5, x=5

// Sağdan sola (pow alma)
let b = 2 ** 3 ** 2  // = 2 ** (3 ** 2) = 2 ** 9 = 512

जटिल अभिव्यक्तियाँ

let result = 10 + 5 * 2 - 3 ** 2 / 3
// Adım adım:
// 1. 3 ** 2 = 9 (pow alma)
// 2. 5 * 2 = 10 (çarpma)
// 3. 9 / 3 = 3 (bölme)
// 4. 10 + 10 = 20 (toplama)
// 5. 20 - 3 = 17 (çıkarma)
// Sonuç: 17

// Parantezle netleştirme
let result2 = ((10 + (5 * 2)) - ((3 ** 2) / 3))  // Daha okunabilir

लॉजिकल ऑपरेटर्स

let x = 5
let y = 10
let z = 15

// && (and) || (or) önce gelir
let sonuç = x < y || y > z && z > x
// = (x < y) || ((y > z) && (z > x))
// = true || (false && true)
// = true || false
// = true

// Parantez kullanımı önerilir
let pub = (x < y) || ((y > z) && (z > x))

बिटवाइज़ ऑपरेटर्स

let a = 0b1100  // 12
let b = 0b1010  // 10

// Bitwise operatörler bool operatörlerden önce
let sonuç = a & b | a ^ b
// = (a & b) | (a ^ b)
// = (0b1000) | (0b0110)
// = 0b1110 = 14

⚠️अच्छे आचरण

  • जटिल भावों मेंकोष्ठकउपयोग - पठनीयता में सुधार करता है
  • एक पंक्ति मेंबहुत सारे ऑपरेटरप्रयोग करने से बचें
  • प्राथमिक आदेशयदि अस्पष्ट हो तो कोष्ठकजोड़ना
  • कोड समीक्षा के दौरानऑपरेटर प्राथमिकतागलतियों पर नजर रखें
