छोरों

दोहराए जाने वाले संचालन के लिए लूप संरचनाएं।

लूप के लिए (रेंज)

// 0'from 4'e to (5 hariç)
for i 0'from 5'e {
    print(i)  // 0, 1, 2, 3, 4
}

// 0'from 5'e to (5 dahil)
for i 0'from 6'ya to {
    print(i)  // 0, 1, 2, 3, 4, 5
}

// Tersden sayma
for i 10'from 0'a {
    print(i)  // 10, 9, 8, ..., 1
}

// Adım boyutu
for i 0'from 10'a adım 2 {
    print(i)  // 0, 2, 4, 6, 8
}

अंग्रेजी वाक्यविन्यास:

for i from 0 to 5 {
    print(i)
}

for i from 0 to 5 inclusive {
    print(i)
}

प्रत्येक लूप के लिए

let sayılar = [1, 2, 3, 4, 5]

for sayı sayılarda {
    print(sayı)
}

// İndeks ile
for (indeks, değer) sayılarda {
    print(indeks, ": ", değer)
}

अंग्रेज़ी:

for num in numbers {
    print(num)
}

for (index, value) in numbers {
    print(index, ": ", value)
}

जबकि लूप

let count_occurrences = 0

while count_occurrences < 5 {
    print(count_occurrences)
    count_occurrences += 1
}

अंग्रेज़ी:

var counter = 0

while counter < 5 {
    print(counter)
    counter += 1
}

फाइबोनैचि

function fibonacci(n: number) -> Liste<number> {
    let sonuç = [0, 1]
    
    for i 2'from n'ye {
        let sonraki = sonuç[i-1] + sonuç[i-2]
        sonuç.push(sonraki)
    }
    
    return sonuç
}
