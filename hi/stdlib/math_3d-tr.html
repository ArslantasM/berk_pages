    let position: Vec3
    let target: Vec3
    let up: Vec3
    
    function view_matrix() -> Mat4 do
        return math_3d.mat4_look_at(self.position, self.target, self.up)
    end
    
    function projection_matrix(fov: kesir, aspect: kesir, near: kesir, far: kesir) -> Mat4 do
        return math_3d.mat4_perspective(fov, aspect, near, far)
    end
end
