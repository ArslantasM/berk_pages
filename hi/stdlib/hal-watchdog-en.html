वॉचडॉग टाइमर - सिस्टम विश्वसनीयता और रीसेट प्रबंधन

~280 पंक्तियाँ ~8 कार्य STM32/GD32/ESP32

📖अवलोकन

वॉचडॉग टाइमर स्वचालित रूप से सॉफ़्टवेयर त्रुटियों के कारण होने वाले सिस्टम फ़्रीज़ का पता लगाता है और सिस्टम को रीसेट करता है। यदि माइक्रोकंट्रोलर एक अप्रत्याशित स्थिति (अनंत लूप, गतिरोध) में प्रवेश करता है, तो वॉचडॉग काउंटर टाइमआउट के लिए रीसेट नहीं होता है और सिस्टम पुनरारंभ होता है।

प्रहरी दो प्रकार के होते हैं:आईडब्ल्यूडीजी(स्वतंत्र वॉचडॉग - एलएसआई घड़ी के साथ काम करता है, कम पावर मोड में भी सक्रिय है) औरडब्ल्यूडब्ल्यूडीजी(विंडो वॉचडॉग - विंडो नियंत्रण के साथ अधिक सटीक समय)।

🔑 मुख्य विशेषताएं

  • स्वतंत्र प्रहरी (आईडब्ल्यूडीजी) - एलएसआई घड़ी (32 किलोहर्ट्ज़)
  • विंडो वॉचडॉग (WWDG) - APB1 घड़ी
  • कॉन्फ़िगर करने योग्य टाइमआउट (एमएस - सेकंड)
  • प्रारंभिक चेतावनी व्यवधान (डब्ल्यूडब्ल्यूडीजी)
  • स्रोत का पता लगाना रीसेट करें
  • डिबग मोड फ़्रीज़ समर्थन
  • स्लीप/स्टॉप मोड में काम करना
  • प्रीस्केलर सेटिंग्स (4 - 256)

🚀 त्वरित शुरुआत

import hal::watchdog, hal::core

// Independent watchdog (4 saniye timeout)
let wdg = watchdog.init_iwdg(do
    timeout_ms: 4000,
    window_mode: false
end)

// Ana loop
loop do
    // Normal işlemler
    core.delay_ms(500)
    
    // Watchdog'u besle (refresh)
    watchdog.refresh(wdg)
    
    // Eğer bu satır çalışmazsa sistem 4 saniye in resetlenir
end

📦 प्रकार और एनम

// Watchdog Tipi
enum WatchdogType {
    IWDG,  // Independent watchdog
    WWDG   // Window watchdog
}

// Prescaler değerleri
enum Prescaler {
    DIV_4,
    DIV_8,
    DIV_16,
    DIV_32,
    DIV_64,
    DIV_128,
    DIV_256
}

// Reset Source
enum ResetSource {
    POWER_ON,      // Power-on reset
    PIN_RESET,     // External pin reset
    SOFTWARE,      // Software reset
    IWDG_RESET,    // Independent watchdog
    WWDG_RESET,    // Window watchdog
    LOW_POWER,     // Low-power reset
    BROWNOUT       // Brown-out reset
}

struct WatchdogConfig do
    type: WatchdogType,
    timeout_ms: int,       // Milliseconds
    window_mode: bool,    // Window watchdog only
    prescaler: Prescaler,
    reload_value: int
end

💡 उदाहरण 1: बेसिक आईडब्ल्यूडीजी का उपयोग करना

2-सेकंड वॉचडॉग के साथ सिस्टम सुरक्षा

import hal::watchdog, hal::gpio, hal::core

let led_green = gpio.pin_init(gpio.PORT_C, 13, gpio.MODE_OUTPUT)
let led_red = gpio.pin_init(gpio.PORT_C, 14, gpio.MODE_OUTPUT)

function ana() do
    // Check reset source
    let reset_cause = watchdog.get_reset_source()
    
    if reset_cause == watchdog.IWDG_RESET  do
        // Watchdog reset occurred!
        gpio.pin_write(led_red, gpio.HIGH)
        core.delay_ms(2000)
        gpio.pin_write(led_red, gpio.LOW)
    end
    
    // Initialize watchdog (2 second timeout)
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 2000,
        window_mode: yanlış
    end)
    
    // Main loop
    loop do
        // Normal operation
        gpio.pin_toggle(led_green)
        core.delay_ms(100)
        
        // Feed watchdog every 100ms (well before 2s timeout)
        watchdog.refresh(wdg)
    end
end

💡 उदाहरण 2: टास्क मॉनिटर के साथ वॉचडॉग

मल्टी-टास्क मॉनिटरिंग और वॉचडॉग फीडिंग

import hal::watchdog, hal::time

let son_beslenme_zamani: Zaman
let watchdog_aktif = true

function kritik_gorev_1() do
    // Kritik işlemler
    core.delay_ms(50)
    
    // İşlem başarılı
    watchdog_besleme_istegi()
end

function kritik_gorev_2() do
    // Başka kritik işlemler
    core.delay_ms(75)
    
    // İşlem başarılı
    watchdog_besleme_istegi()
end

function watchdog_besleme_istegi() do
    son_beslenme_zamani = time.now()
end

function watchdog_yonetici(wdg: WatchdogHandle) do
    loop do
        // Check if tasks are healthy
        let elapsed = time.now() - son_beslenme_zamani
        
        if elapsed < time.milliseconds(500)  do
            // Tasks are running OK, feed watchdog
            watchdog.refresh(wdg)
        else do
            // Tasks hung! Let watchdog reset the system
            io.println("UYARI: Görevler askıda! Watchdog resetleyecek...")
        end
        
        core.delay_ms(100)
    end
end

function main() do
    // 1 saniye watchdog timeout
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 1000,
        window_mode: false
    end)
    
    son_beslenme_zamani = time.now()
    
    // Start tasks
    thread.spawn(|| kritik_gorev_1())
    thread.spawn(|| kritik_gorev_2())
    thread.spawn(|| watchdog_yonetici(wdg))
    
    thread.join_all()
end

💡 उदाहरण 3: विंडो वॉचडॉग (डब्ल्यूडब्ल्यूडीजी)

विंडो नियंत्रण के साथ सटीक समय

import hal::watchdog

// WWDG: Hem çok erken hem çok geç beslemeyi engeller
function wwdg_demo() do
    // Window: 50-100ms aralığında beslenmeli
    let wwdg = watchdog.init_wwdg(do
        timeout_ms: 100,
        window_ms: 50,  // Pencere başlangıcı
        early_wakeup_interrupt: doğru
    end)
    
    // Early warning interrupt handler
    watchdog.set_interrupt_handler(wwdg, || do
        io.println("Watchdog erken uyarı!")
        // Burası early wakeup interrupt (zamanında besle)
    end)
    
    let sayac = 0
    loop do
        // İşlemler
        core.delay_ms(30)
        
        // 30ms + 30ms = 60ms (pencere in: 50-100ms ✓)
        core.delay_ms(30)
        
        // Watchdog'u besle
        watchdog.refresh(wwdg)
        
        sayac += 1
        
        // Simulate hang every 10 iterations
        if sayac % 10 == 0  do
            io.println("Simüle edilen gecikme...")
            core.delay_ms(150)  // Too late! WWDG will reset
        end
    end
end

💡 उदाहरण 4: वॉचडॉग + बैकअप रजिस्टर

कारण रिकॉर्डिंग और विश्लेषण रीसेट करें

import hal::watchdog, hal::rtc

struct ResetInfo do
    count: int,
    last_reason: ResetSource,
    timestamp: int
end

function reset_bilgisi_kaydet() do
    let reason = watchdog.get_reset_source()
    
    // Backup register'from önceki bilgiyi input
    let reset_count = rtc.backup_read(0)
    reset_count += 1
    
    // Yeni bilgiyi kaydet
    rtc.backup_write(0, reset_count)
    rtc.backup_write(1, reason.to_int())
    rtc.backup_write(2, time.timestamp())
    
    io.println("Reset sayısı: " + reset_count.yazıya())
    io.println("Reset nedeni: " + reason.to_string())
end

function main() do
    // Initialize RTC for backup registers
    rtc.init()
    
    // Log reset information
    reset_bilgisi_kaydet()
    
    // Check for excessive resets
    let reset_count = rtc.backup_read(0)
    if reset_count > 10  do
        io.println("HATA: Çok fazla reset! Sistem unstable.")
        // Enter safe mode or disable some features
        rtc.backup_write(0, 0)  // Reset counter
    end
    
    // Initialize watchdog
    let wdg = watchdog.init_iwdg(do
        timeout_ms: 3000,
        window_mode: false
    end)
    
    // Normal operation
    loop do
        // Your code here
        watchdog.refresh(wdg)
        core.delay_ms(500)
    end
end

💡 सर्वोत्तम प्रथाएँ

  • समयबाह्य चयन:यह कार्य की अधिकतम अवधि से 2-3 गुना अधिक होनी चाहिए
  • भोजन की आवृत्ति:टाइमआउट के 50-75% के अंतराल पर फ़ीड करें
  • डिबग मोड:विकास के दौरान वॉचडॉग को फ़्रीज़ करें (डीबगर ब्रेकप्वाइंट पर)
  • महत्वपूर्ण अनुभाग:उन स्थानों पर वॉचडॉग टाइमआउट की गणना करें जहां व्यवधान अक्षम है
  • लॉगिंग रीसेट करें:बैकअप रजिस्टर में रीसेट कारणों को स्टोर करें
  • आईडब्ल्यूडीजी बनाम डब्ल्यूडब्ल्यूडीजी:उत्पादन में IWDG (LSI स्वतंत्र), WWDG के लिए महत्वपूर्ण समय
  • पावर मोड:IWDG कम-पावर मोड में काम करता है, WWDG बंद हो जाता है

⚠️ महत्वपूर्ण नोट्स

  • एक बार शुरू होने के बाद, IWDG को रोका नहीं जा सकता (हार्डवेयर प्रतिबंध)
  • डिबगिंग के दौरान वॉचडॉग रीसेट को रोकने के लिए DBGMCU_APB1_FZ रजिस्टर सेट करें
  • एलएसआई घड़ी सहनशीलता 40% तक हो सकती है - टाइमआउट गणना में इसे ध्यान में रखें
  • फ़्लैश प्रोग्रामिंग के दौरान वॉचडॉग को अक्षम करें (मिटाने में लंबा समय लग सकता है)
