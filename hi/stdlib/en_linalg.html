📚एपीआई संदर्भ

मैट्रिक्स निर्माण

linalg_birim_mat(n: tam_sayı) -> [float]

यह आकार n×n का एक पहचान मैट्रिक्स बनाता है।

पैरामीटर:

  • n: मैट्रिक्स आकार

रिटर्न:पहचान मैट्रिक्स (सादे सरणी प्रारूप में)

उदाहरण:

variable I3 = linalg_birim_mat(3)
// I3 = [1, 0, 0,
//       0, 1, 0,
//       0, 0, 1]

linalg_sıfır_mat(m: tam_sayı, n: tam_sayı) -> [float]

यह m×n आकार का एक शून्य मैट्रिक्स बनाता है।

linalg_birler_mat(m: tam_sayı, n: tam_sayı) -> [float]

यह सभी तत्वों 1 के साथ आकार m×n का एक मैट्रिक्स बनाता है।

linalg_diyagonal_mat(d: [float], n: tam_sayı) -> [float]

विकर्ण तत्वdयह एक n×n मैट्रिक्स बनाता है, जो एक सरणी है।

उदाहरण:

variable d = [2.0, 3.0, 5.0]
variable D = linalg_diyagonal_mat(d, 3)
// D = [2, 0, 0,
//      0, 3, 0,
//      0, 0, 5]

मैट्रिक्स अंकगणित

linalg_mat_topla(A: [float], B: [float], m: tam_sayı, n: tam_sayı) -> [float]

दो m×n आव्यूह जोड़ता है: C = A + B

linalg_mat_çıkar(A: [float], B: [float], m: tam_sayı, n: tam_sayı) -> [float]

दो m×n आव्यूहों को घटाता है: C = A - B

linalg_mat_skaler(α: float, A: [float], m: tam_sayı, n: tam_sayı) -> [float]

मैट्रिक्स को अदिश से गुणा करता है: B = α · A

linalg_mat_çarp(A: [float], B: [float], m: tam_sayı, k: tam_sayı, n: tam_sayı) -> [float]

मैट्रिक्स गुणन: C = A × B

  • ए: एम×के
  • बी: के×एन
  • सी: एम×एन

उदाहरण:

// 2×2 matris çarpımı
variable A = [1.0, 2.0,
              3.0, 4.0]
variable B = [5.0, 6.0,
              7.0, 8.0]
variable C = linalg_mat_çarp(A, B, 2, 2, 2)
// C = [19, 22,
//      43, 50]

linalg_mat_hadamard(A: [float], B: [float], m: tam_sayı, n: tam_sayı) -> [float]

तत्व-वार उत्पाद (हैडमार्ड उत्पाद): C[i,j] = A[i,j] × B[i,j]

वेक्टर संचालन

linalg_iç_çarpım(u: [float], v: [float], n: tam_sayı) -> float

डॉट उत्पाद: u · v = Σ(u[i] × v[i])

उदाहरण:

variable u = [1.0, 2.0, 3.0]
variable v = [4.0, 5.0, 6.0]
variable dot = linalg_iç_çarpım(u, v, 3)  // 32.0

linalg_norm2(v: [float], n: tam_sayı) -> float

L2 मानदंड (यूक्लिडियन मानदंड): ||v|| = √(Σv[i]²)

linalg_norm1(v: [float], n: tam_sayı) -> float

L1 मानदंड (मैनहट्टन मानदंड): ||v||₁ = Σ|v[i]|

linalg_norm_inf(v: [float], n: tam_sayı) -> float

L∞ मानदंड (अधिकतम मानदंड): ||v||∞ = अधिकतम(|v[i]|)

linalg_normalize(v: [float], n: tam_sayı) -> [float]

वेक्टर को सामान्यीकृत करता है: v̂ = v / ||v||

linalg_dış_çarpım(u: [float], v: [float], m: tam_sayı, n: tam_sayı) -> [float]

बाहरी उत्पाद: ए[आई,जे] = यू[आई] × वी[जे]

linalg_cross_3d(u: [float], v: [float]) -> [float]

3डी वेक्टर क्रॉस उत्पाद: डब्ल्यू = यू × वी

उदाहरण:

variable u = [1.0, 0.0, 0.0]
variable v = [0.0, 1.0, 0.0]
variable w = linalg_cross_3d(u, v)  // [0, 0, 1]

मैट्रिक्स गुण

linalg_iz(A: [float], n: tam_sayı) -> float

ट्रेस: ​​tr(A) = Σ A[i,i]

linalg_frobenius_norm(A: [float], m: tam_sayı, n: tam_sayı) -> float

फ्रोबेनियस मानदंड: ||A||_F = √(Σ Σ A[i,j]²)

linalg_determinant_2x2(A: [float]) -> float

2×2 मैट्रिक्स निर्धारक: det(A) = a₁₁a₂₂ - a₁₂a₂₁

linalg_determinant_3x3(A: [float]) -> float

3×3 मैट्रिक्स निर्धारक (सारस नियम)

मैट्रिक्स व्युत्क्रम

linalg_inverse_2x2(A: [float]) -> [float]

2×2 मैट्रिक्स व्युत्क्रम की गणना करता है: A⁻¹

उदाहरण:

variable A = [4.0, 7.0,
              2.0, 6.0]
variable A_inv = linalg_inverse_2x2(A)
// det(A) = 10
// A_inv = [0.6, -0.7,
//         -0.2,  0.4]

linalg_inverse_3x3(A: [float]) -> [float]

3×3 मैट्रिक्स व्युत्क्रम (सहकारक विस्तार) की गणना करता है

रैखिक प्रणाली

linalg_gauss_çöz(A: [float], b: [float], n: tam_sayı) -> [float]

गॉस उन्मूलन द्वारा रैखिक प्रणाली Ax = b को हल करें।

विशेषताएँ:

  • आंशिक धुरी
  • आगे उन्मूलन
  • पिछला प्रतिस्थापन

उदाहरण:

// 3x + 2y = 7
// 2x + 5y = 12
variable A = [3.0, 2.0,
              2.0, 5.0]
variable b = [7.0, 12.0]
variable x = linalg_gauss_çöz(A, b, 2)
// x = [1, 2]

मैट्रिक्स अपघटन

linalg_lu_decompose(A: [float], n: tam_sayı) -> [float]

एलयू अपघटन: ए = एल × यू

  • एल: निचला त्रिकोणीय मैट्रिक्स (विकर्ण 1)
  • यू: ऊपरी त्रिकोणीय मैट्रिक्स

उपयोग:रैखिक प्रणालियों को शीघ्रता से हल करना

linalg_qr_gram_schmidt(A: [float], m: tam_sayı, n: tam_sayı) -> ([float], [float])

क्यूआर अपघटन: ए = क्यू × आर

  • प्रश्न: ऑर्थोगोनल मैट्रिक्स
  • आर: ऊपरी त्रिकोणीय मैट्रिक्स

एल्गोरिथम:ग्राम-श्मिट ऑर्थोगोनलाइज़ेशन

रिटर्न:(क्यू,आर) टपल

linalg_cholesky(A: [float], n: tam_sayı) -> [float]

चोलेस्की अपघटन: ए = एल × एलᵀ

ज़रूरत:A एक सकारात्मक निश्चित सममित मैट्रिक्स होना चाहिए

उपयोग:सहप्रसरण मैट्रिक्स, अनुकूलन

आइजेनवैल्यू/आइजेनवेक्टर

linalg_power_iteration(A: [float], n: tam_sayı, max_iter: tam_sayı, tolerans: float) -> ([float], float)

शक्ति पुनरावृत्ति के साथ प्रमुख आइजनवेक्टर ढूँढना।

रिटर्न:(eigenvector, eigenvalue) टुपल

उदाहरण:

variable A = [4.0, 1.0,
              1.0, 3.0]
variable (v, λ) = linalg_power_iteration(A, 2, 100, 1e-6)
// λ ≈ 4.618 (dominant eigenvalue)
// v: karşılık gelen eigenvector

मैट्रिक्स परीक्षण

linalg_simetrik_mi(A: [float], n: tam_sayı, tolerans: float) -> boolean

जाँचता है कि मैट्रिक्स सममित है या नहीं: Aᵢⱼ = Aⱼᵢ

linalg_diagonal_mi(A: [float], n: tam_sayı, tolerans: float) -> boolean

विकर्ण मैट्रिक्स नियंत्रण

linalg_üst_üçgen_mi(A: [float], n: tam_sayı, tolerans: float) -> boolean

ऊपरी त्रिकोणीय मैट्रिक्स नियंत्रण

linalg_alt_üçgen_mi(A: [float], n: tam_sayı, tolerans: float) -> boolean

निचला त्रिकोणीय मैट्रिक्स नियंत्रण

मैट्रिक्स रैंक

linalg_rank(A: [float], m: tam_sayı, n: tam_sayı, tolerans: float) -> tam_sayı

गाऊसी उन्मूलन का उपयोग करके मैट्रिक्स रैंक की गणना करता है।

उदाहरण:

variable A = [1.0, 2.0, 3.0,
              2.0, 4.0, 6.0,
              3.0, 6.0, 9.0]
variable r = linalg_rank(A, 3, 3, 1e-10)  // r = 1

उदाहरण 1: रैखिक प्रतिगमन

use linalg
use stats

// y = ax + b modelini fit et
function linear_fit(X: [float], y: [float], n: tam_sayı): (float, float) do
    // X matrix: [ones, x_values]
    variable A = yeni_dizi(n * 2)
    variable i = 0
    while (i < n) do
        A[i * 2] = 1.0
        A[i * 2 + 1] = X[i]
        i = i + 1
    end
    
    // Normal equations: (A^T A) β = A^T y
    variable params = linalg_least_squares(A, y, n, 2)
    return (params[0], params[1])  // (b, a)
end

उदाहरण 2: पीसीए (प्रमुख घटक विश्लेषण)

use linalg

// Veriyi merkeze receive
function center_data(data: [float], m: tam_sayı, n: tam_sayı): [float] do
    variable centered = linalg_mat_kopyala(data, m, n)
    
    // Her sütun for ortalamayı pop
    variable j = 0
    while (j < n) do
        variable avg = 0.0
        variable i = 0
        while (i < m) do
            avg = avg + linalg_mat_al(data, i, j, n)
            i = i + 1
        end
        avg = avg / m
        
        i = 0
        while (i < m) do
            variable değer = linalg_mat_al(centered, i, j, n) - avg
            linalg_mat_yaz(centered, i, j, n, değer)
            i = i + 1
        end
        
        j = j + 1
    end
    
    return centered
end

// Kovaryans matrisini hesapla
function covariance_matrix(data: [float], m: tam_sayı, n: tam_sayı): [float] do
    variable centered = center_data(data, m, n)
    variable CT = linalg_transpoz(centered, m, n)
    variable cov = linalg_mat_çarp(CT, centered, n, m, n)
    
    // (n-1) ile split
    variable i = 0
    while (i < n * n) do
        cov[i] = cov[i] / (m - 1)
        i = i + 1
    end
    
    return cov
end

उदाहरण 3: मैट्रिक्स नॉर्म और कंडीशन नंबर

use linalg

function analyze_matrix(A: [float], n: tam_sayı): hiçbir do
    // Frobenius norm
    variable norm = linalg_frobenius_norm(A, n, n)
    print("||A||_F = " + norm)
    
    // Determinant
    if (n == 2) then do
        variable det = linalg_determinant_2x2(A)
        print("det(A) = " + det)
        
        // Inverse and condition number
        if (math_mutlak(det) > 1e-10) then do
            variable A_inv = linalg_inverse_2x2(A)
            variable κ = linalg_condition_frobenius(A, A_inv, n)
            print("κ(A) = " + κ)
            
            if (κ > 100.0) then do
                print("Warning: Matris kötü koşullu!")
            end
        end
    end
    
    // Simetri kontrolü
    if (linalg_simetrik_mi(A, n, 1e-10)) then do
        print("Matris simetriktir")
    end
end
