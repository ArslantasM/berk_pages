import string

// Temel işlemler
variable uzunluk = string.uzunluk("Merhaba")  // 7
variable büyük = string.büyük_harf("merhaba") // "MERHABA"
variable küçük = string.küçük_harf("MERHABA") // "merhaba"

// Arama and değiştirme
variable içerir = string.içerir_mi("Hello World", "World") // true
variable değişti = string.değiştir("foo bar", "foo", "baz") // "baz bar"

// Bölme and birleştirme
variable parçalar = string.böl("a,b,c", ",")  // ["a", "b", "c"]
variable birleşik = string.birleştir(["a","b"], "-") // "a-b"

// Whitespace temizleme
variable temiz = string.temizle("  hello  ") // "hello"

import string

function email_geçerli_mi(email: string) -> boolean do
    if string.içerir_mi(email, "@") else return false end
    
    variable parçalar = string.böl(email, "@")
    if string.uzunluk(parçalar) != 2 ise return false end
    
    variable local = parçalar[0]
    variable domain = parçalar[1]
    
    if string.uzunluk(local) == 0 ise return false end
    if string.uzunluk(domain) == 0 ise return false end
    if string.içerir_mi(domain, ".") else return false end
    
    return true
end

print(email_geçerli_mi("user@example.com"))  // true
print(email_geçerli_mi("invalid"))           // false

import string

function csv_parse(satır: string) -> List[string] do
    variable sonuç = []
    variable volatile = ""
    variable tırnak_içinde = false
    
    for char_at in string.chars(satır) do
        if char_at == '"' then do
            tırnak_içinde = !tırnak_içinde
        end else if char_at == ',' and !tırnak_içinde then do
            liste_ekle(sonuç, string.temizle(volatile))
            volatile = ""
        end else do
            volatile += char_at
        end
    end
    
    liste_ekle(sonuç, string.temizle(volatile))
    return sonuç
end

variable veri = 'John,Doe,"123 Main St",30'
print(csv_parse(veri))  // ["John", "Doe", "123 Main St", "30"]
