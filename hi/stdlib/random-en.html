import random, math

function pi_tahmini(numune_sayısı: int) -> float do
    let daire_içi = 0
    
    each i in range(0, numune_sayısı) for do
        let x = random.float()
        let y = random.float()
        
        // Nokta birim daire in mi?
        if (x*x + y*y) < 1.0  do
            daire_içi = daire_içi + 1
        end
    end
    
    // Pi ≈ 4 * (daire içi nokta sayısı / sum nokta sayısı)
    return 4.0 * daire_içi.float() / numune_sayısı.float()
end

let tahmin = pi_tahmini(1000000)
print("Pi tahmini: " + tahmin.yazıya())
print("Gerçek Pi: " + math.PI.yazıya())
print("Error: " + math.abs(tahmin - math.PI).yazıya())

import random

function weighted_choice(items: Dizi, weights: Dizi) -> herhangi do
    // Kümülatif ağırlıklar
    let total = 0.0
    let cumulative = []
    
    each w in weights for do
        total = total + w
        cumulative.push(total)
    end
    
    // Random seçim
    let r = random.float() * total
    
    each i in range(0, cumulative.len()) for do
        if r < cumulative[i]  do
            return items[i]
        end
    end
    
    return items[items.len() - 1]
end
