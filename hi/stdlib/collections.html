📚 डेटा संरचनाएं

  • Vec:गतिशील सरणी, O(1) एक्सेस, O(1) परिशोधित पुश
  • हैशमैप:हैश टेबल, ओ(1) औसत डालें/प्राप्त करें/हटाएं
  • हैशसेट:अद्वितीय तत्वों में O(1) औसत होता है
  • लिंक्डलिस्ट:दोगुनी लिंक्ड सूची, ओ(1) पुश/पॉप स्टार्ट/एंड
  • बीट्रीमैप:अनुक्रमिक मानचित्र, ओ(लॉग एन) संचालन
  • बाइनरीहीप:मैक्स-हीप, ओ(लॉग एन) पुश/पॉप, ओ(1) पीक

import collections, string, io

function word_frequency(text: yazı) -> HashMap do
    let freq = collections.hashmap_new()
    let kelimeler = string.split(text, " ")
    
    her kelime in kelimeler for do
        let küçük = string.to_lowercase(kelime)
        
        if collections.hashmap_contains(freq, küçük) then do
            let count = collections.hashmap_get(freq, küçük)
            collections.hashmap_insert(freq, küçük, count + 1)
        end else do
            collections.hashmap_insert(freq, küçük, 1)
        end
    end
    
    return freq
end

let String = io.fs::read("document.txt")
let frekans = word_frequency(String)

// En çok elapsed 10 kelimeyi göster
let kelimeler = collections.hashmap_keys(frekans)
collections.vec_sort_by(kelimeler, function(a, b) do
    return collections.hashmap_get(frekans, b) - collections.hashmap_get(frekans, a)
end)

her i in range(0, 10) for do
    let k = collections.vec_get(kelimeler, i)
    print(k + ": " + collections.hashmap_get(frekans, k).yazıya())
end

import collections

sınıf LRUCache do
    let kapasite: int
    let cache: HashMap
    let sıra: LinkedList
    
    function new(kap: int) do
        self.kapasite = kap
        self.cache = collections.hashmap_new()
        self.sıra = collections.linkedlist_new()
    end
    
    function get(anahtar: yazı) -> herhangi do
        if collections.hashmap_contains(self.cache, anahtar) else do
            return yok
        end
        
        // Sıraya taşı (LRU güncelle)
        collections.linkedlist_remove(self.sıra, anahtar)
        collections.linkedlist_push_back(self.sıra, anahtar)
        
        return collections.hashmap_get(self.cache, anahtar)
    end
    
    function put(anahtar: yazı, değer: herhangi) do
        if collections.hashmap_contains(self.cache, anahtar) then do
            collections.linkedlist_remove(self.sıra, anahtar)
        end
        
        collections.hashmap_insert(self.cache, anahtar, değer)
        collections.linkedlist_push_back(self.sıra, anahtar)
        
        if collections.hashmap_size(self.cache) > self.kapasite then do
            // En eski elemanı pop
            let eski = collections.linkedlist_pop_front(self.sıra)
            collections.hashmap_remove(self.cache, eski)
        end
    end
end

let cache = LRUCache.new(3)
cache.put("a", 1)
cache.put("b", 2)
cache.put("c", 3)
print(cache.get("a"))  // 1
cache.put("d", 4)  // "b" çıkarılır
print(cache.get("b"))  // yok
