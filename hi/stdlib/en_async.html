🚀 तेजी से शुरुआत करना

import async, http

// Async function
async function veri_getir(url: string) -> Sonuç do
    variable yanıt = await http.get(url)
    
    if yanıt.case != 200 then do
        return Err("HTTP " + yanıt.case.yazıya())
    end
    
    return Ok(yanıt.gövde)
end

// Parallel execution
async function main() do
    variable görevler = [
        veri_getir("https://api.example.com/users"),
        veri_getir("https://api.example.com/posts"),
        veri_getir("https://api.example.com/comments")
    ]
    
    variable sonuçlar = await async.all(görevler)
    
    her sonuç in sonuçlar for do
        print(sonuç)
    end
end

async.run(main())

import async, http

async function kullanıcı_getir(id: integer) -> Sonuç do
    await async.sleep(100)  // Simulate DB query
    return Ok({"id": id, "name": "User " + id.yazıya()})
end

async function main() do
    variable sunucu = http.AsyncServer_new(3000)
    
    sunucu.rotala("GET", "/api/user/:id", async function(istek, yanıt) do
        variable id = istek.params["id"].parse_int()
        variable user = await kullanıcı_getir(id)
        
        eşleştir user do
            Ok(data) => yanıt.json(data),
            Err(e) => yanıt.case(500).json({"error": e})
        end
    end)
    
    print("Async sunucu http://localhost:3000")
    await sunucu.başlat()
end

async.run(main())
