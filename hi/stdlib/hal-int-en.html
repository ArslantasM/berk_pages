⚡ राज्य::int

नेस्टेड वेक्टर्ड इंटरप्ट कंट्रोलर (एनवीआईसी)

366 पंक्तियाँ ~8 कार्य एनवीआईसी

📖अवलोकन

इंटरप्ट नियंत्रक अतुल्यकालिक घटनाओं को संभालता है। यह GPIO, टाइमर, USART जैसी परिधीय इकाइयों के इंटरप्ट का प्रबंधन करता है।

🔑 मुख्य विशेषताएं

  • प्राथमिकता स्तर (0-15)
  • छूट और उप-प्राथमिकता
  • व्यवधान सक्षम/अक्षम करें
  • लंबित ध्वज प्रबंधन
  • नेस्टेड व्यवधान समर्थन

💡 उदाहरण: नेस्टेड इंटरप्ट्स (प्राथमिकता निवारण)

import hal::int, hal::timer, hal::gpio

let high_priority_count = 0
let low_priority_count = 0

function high_priority_isr() do
    // High priority interrupt (timer)
    high_priority_count += 1
    io.println("HIGH priority ISR (count: {})", high_priority_count)
    
    // Simulate work
    core.delay_ms(50)
end

function low_priority_isr() do
    // Low priority interrupt (button)
    low_priority_count += 1
    io.println("  LOW priority ISR (count: {})", low_priority_count)
    
    // Simulate work
    core.delay_ms(100)
end

function nested_interrupt_demo() do
    // Set priority grouping (4 bits preemption, 0 bits sub)
    int.set_priority_grouping(int.PRIORITY_GROUP_4)
    
    // Configure Timer2 interrupt (high priority)
    timer.init(timer.TIM2, do
        frequency: 10,  // 10Hz
        callback: high_priority_isr
    end)
    
    int.set_priority(int.IRQ_TIM2, do
        preempt: 0,  // Highest priority
        sub: 0
    end)
    int.enable(int.IRQ_TIM2)
    
    // Configure GPIO interrupt (low priority)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_INPUT,
        interrupt: gpio.INT_RISING,
        callback: low_priority_isr
    end)
    
    int.set_priority(int.IRQ_EXTI0, do
        preempt: 5,  // Lower priority
        sub: 0
    end)
    int.enable(int.IRQ_EXTI0)
    
    io.println("Nested interrupt demo")
    io.println("Timer interrupt can preempt button interrupt")
    
    timer.start(timer.TIM2)
    
    loop do
        core.delay_ms(1000)
    end
end

function ana() do
    nested_interrupt_demo()
end

💡 उदाहरण: बाहरी व्यवधान (EXTI)

import hal::int, hal::gpio

let button1_pressed = 0
let button2_pressed = 0

function button1_callback() do
    button1_pressed += 1
    io.println("Button 1 pressed (total: {})", button1_pressed)
end

function button2_callback() do
    button2_pressed += 1
    io.println("Button 2 pressed (total: {})", button2_pressed)
end

function multi_button_exti() do
    // Button 1 on PA0 (EXTI0)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_INPUT,
        pull: gpio.PULL_DOWN,
        interrupt: gpio.INT_RISING
    end)
    
    gpio.set_callback(gpio.PORT_A, 0, button1_callback)
    
    int.set_priority(int.IRQ_EXTI0, do preempt: 5, sub: 0 end)
    int.enable(int.IRQ_EXTI0)
    
    // Button 2 on PB1 (EXTI1)
    gpio.clock_enable(gpio.PORT_B)
    gpio.pin_init(gpio.PORT_B, 1, do
        mode: gpio.MODE_INPUT,
        pull: gpio.PULL_DOWN,
        interrupt: gpio.INT_RISING
    end)
    
    gpio.set_callback(gpio.PORT_B, 1, button2_callback)
    
    int.set_priority(int.IRQ_EXTI1, do preempt: 5, sub: 0 end)
    int.enable(int.IRQ_EXTI1)
    
    io.println("EXTI demo: Press buttons")
    
    loop do
        core.delay_ms(100)
    end
end

function ana() do
    multi_button_exti()
end

💡 उदाहरण: विलंबता माप को बाधित करें

import hal::int, hal::timer, hal::gpio

let interrupt_timestamp = 0
let latency_us = 0

function fast_isr() do
    // Measure interrupt latency
    let now = core.uptime_us()
    latency_us = now - interrupt_timestamp
    
    // Toggle LED as visual indicator
    gpio.pin_toggle(gpio.PORT_C, 13)
end

function latency_test() do
    // Configure GPIO interrupt on PA0
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_INPUT,
        interrupt: gpio.INT_RISING,
        callback: fast_isr
    end)
    
    // Set highest priority
    int.set_priority_grouping(int.PRIORITY_GROUP_4)
    int.set_priority(int.IRQ_EXTI0, do preempt: 0, sub: 0 end)
    int.enable(int.IRQ_EXTI0)
    
    // LED output
    gpio.pin_init(gpio.PORT_C, 13, do mode: gpio.MODE_OUTPUT end)
    
    io.println("Interrupt latency test")
    io.println("Trigger PA0 rising edge")
    
    loop do
        // Wait for interrupt
        if latency_us > 0 ise do
            io.println("Interrupt latency: {}μs", latency_us)
            latency_us = 0
        end
        
        core.delay_ms(100)
    end
end

function ana() do
    latency_test()
end

import hal::int

let software_int_count = 0

function software_isr() do
    software_int_count += 1
    io.println("Software interrupt triggered! ({})", software_int_count)
end

function software_interrupt_demo() do
    // Configure software interrupt
    int.set_callback(int.IRQ_PendSV, software_isr)
    int.set_priority(int.IRQ_PendSV, do preempt: 15, sub: 0 end)
    int.enable(int.IRQ_PendSV)
    
    io.println("Software interrupt demo")
    
    loop do
        io.println("Triggering software interrupt...")
        
        // Trigger PendSV interrupt manually
        int.set_pending(int.IRQ_PendSV)
        
        core.delay_ms(2000)
    end
end

function ana() do
    software_interrupt_demo()
end

import hal::int

let shared_counter = 0

function critical_section_demo() do
    // Unsafe increment (race condition possible)
    function unsafe_increment() do
        let temp = shared_counter
        core.delay_us(10)  // Simulate work
        shared_counter = temp + 1
    end
    
    // Safe increment (atomic with interrupts disabled)
    function safe_increment() do
        // Enter critical section
        int.global_disable()
        
        let temp = shared_counter
        core.delay_us(10)
        shared_counter = temp + 1
        
        // Exit critical section
        int.global_enable()
    end
    
    // Better: use RAII-style guard
    function best_increment() do
        let guard = int.critical_section()  // Auto disable
        
        let temp = shared_counter
        core.delay_us(10)
        shared_counter = temp + 1
        
        // Auto enable on scope exit
    end
    
    // Test
    each i forde 0..1000 for do
        safe_increment()
    end
    
    io.println("Counter: {}", shared_counter)  // Should be 1000
end

function ana() do
    critical_section_demo()
end

⚙️ एनवीआईसी संरचना

// Priority Configuration
struct PriorityConfig do
    preempt: int,  // Preemption priority (0-15)
    sub: int       // Sub-priority (0-15)
end

// Priority Grouping
enum PriorityGroup {
    PRIORITY_GROUP_0,  // 0:4 (0 preemption, 16 sub)
    PRIORITY_GROUP_1,  // 1:3 (2 preemption, 8 sub)
    PRIORITY_GROUP_2,  // 2:2 (4 preemption, 4 sub)
    PRIORITY_GROUP_3,  // 3:1 (8 preemption, 2 sub)
    PRIORITY_GROUP_4   // 4:0 (16 preemption, 0 sub)
}

// Common IRQ Numbers (STM32F4)
enum IRQ {
    IRQ_WWDG          = 0,   // Window watchdog
    IRQ_EXTI0         = 6,   // External line 0
    IRQ_EXTI1         = 7,   // External line 1
    IRQ_DMA1_STREAM0  = 11,  // DMA1 stream 0
    IRQ_TIM2          = 28,  // Timer 2
    IRQ_USART1        = 37,  // USART1
    IRQ_EXTI15_10     = 40,  // External lines 10-15
    IRQ_SPI1          = 35,  // SPI1
    IRQ_PendSV        = -2,  // Pendable request
    IRQ_SysTick       = -1   // System tick
}

📊 प्राथमिकता स्तर के सुझाव

प्राथमिकता उदाहरण उदाहरण
0-2 (सर्वोच्च) समय-महत्वपूर्ण, वास्तविक समय मोटर एनकोडर, हाई-स्पीड एडीसी
3-7 (उच्च) तेज़ परिधीय USART, SPI, I2C, टाइमर
8-11 (मध्यम) सामान्य परिधीय जीपीआईओ, एडीसी, आरटीसी अलार्म
12-15 (कम) नेपथ्य कार्य सॉफ़्टवेयर व्यवधान, कम-प्राथमिकता वाला IO

⚡अच्छे आचरण

  • आईएसआर लघु: इंटरप्ट हैंडलर को छोटा रखें (<10μs आदर्श)
  • कार्य टालें: मुख्य लूप (ध्वज सेट) में भारी प्रसंस्करण में देरी करें
  • परिवर्तनशील: आइए ISR से पढ़ें अस्थिर होना चाहिए
  • प्राथमिकता योजना: समय-महत्वपूर्ण नौकरियां उच्च प्राथमिकता हैं
  • घोंसला करने की क्रिया: ग्रुप_4 का उपयोग करें (प्रीएम्प्शन लेवल 16)
  • महत्वपूर्ण अनुभाग: साझा डेटा एक्सेस में व्यवधान अक्षम करें

⚠️ महत्वपूर्ण नोट्स

  • प्राथमिकता संख्या: कम संख्या = उच्च प्राथमिकता (0 उच्चतम है)
  • हथियाना: भिन्न प्रीएम्प्शन प्राथमिकता → नेस्टिंग संभव
  • उप-प्राथमिकता: निर्धारित करता है कि समान प्रीएम्पशन में पहले किस पर कार्रवाई की जाएगी
  • आईएसआर विलंबता: विशिष्ट 10-20 चक्र (~12 चक्र पुश/पॉप रजिस्टर)
  • पूँछ की चेनिंग: बैक-टू-बैक इंटरप्ट में पुश/पॉप स्किप (6 चक्र बचाए गए)।
  • बेसपरी: मास्किंग एक निश्चित प्राथमिकता के तहत बाधित होती है
  • दोष संचालक: हार्डफॉल्ट, मेममैनेज, बसफॉल्ट (प्राथमिकता -1, -2, -3)
  • सिस्टिक: विशिष्ट प्राथमिकता 15 (न्यूनतम, समय-महत्वपूर्ण नहीं)

🔗संबंधित मॉड्यूल

  • hal::gpio- EXTI बाहरी व्यवधान
  • hal::timer- टाइमर व्यवधान (अतिप्रवाह, कैप्चर)
  • hal::usart- USART TX/RX बाधित होता है
  • hal::dma- डीएमए स्थानांतरण पूर्ण व्यवधान
  • hal::rtc- आरटीसी अलार्म बाधित होता है
  • hal::core- स्लीप मोड वेकअप
