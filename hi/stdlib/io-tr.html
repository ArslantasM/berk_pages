उदाहरण 2: बड़ी फ़ाइल प्रोसेसिंग

import io, string

function büyük_dosya_işle(girdi: yazı, çıktı: yazı) do
    let girdi_dosya = io.aç(girdi, "r")
    let çıktı_dosya = io.aç(çıktı, "w")
    
    let reader = io.buffered_reader(girdi_dosya, 8192)
    let writer = io.buffered_writer(çıktı_dosya, 8192)
    
    while true do
        let satır = io.satır_oku(reader)
        if satır.yok_mu() then kır end
        
        // İşleme
        let işlenmiş = string.to_uppercase(satır.değer())
        io.buffer_yaz(writer, işlenmiş + "\n")
    end
    
    io.buffer_flush(writer)
    io.kapat(çıktı_dosya)
    io.kapat(girdi_dosya)
end

büyük_dosya_işle("input.txt", "output.txt")
