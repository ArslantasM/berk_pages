📖कैसे उपयोग करें?

1. मॉड्यूल आयात करना

// Single module
import math

// Multiple modules
import math, stats, linalg

// Using alias
import math as m
import http as h

2. फंक्शन कॉलिंग

// Full name
let result = math.sqrt(16.0)

// With alias
import math as m
let result = m.sqrt(16.0)

3. स्थिरांक का उपयोग करना

import math

print(math.PI)        // 3.14159...
print(math.E)         // 2.71828...
print(math.TAU)       // 6.28318... (2π)
