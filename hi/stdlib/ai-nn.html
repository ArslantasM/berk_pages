    एनएन मॉड्यूल आधुनिक तंत्रिका नेटवर्क आर्किटेक्चर बनाने के लिए आवश्यक सभी घटक प्रदान करता है। यह PyTorch nn.Module API के समान डिज़ाइन के साथ आसान उपयोग प्रदान करता है।

    परत के प्रकार

    रैखिक (पूरी तरह से जुड़ा हुआ)

    पूरी तरह से जुड़ी हुई परत: y = xW^T + b

    use ai/nn;

// 784 giriş → 128 çıkış
let layer = Linear::new(784, 128);

// Forward pass
let x = Tensor::random([32, 784]);  // batch_size=32
let y = layer.ileri(x);               // [32, 128]

    Conv2d (2D कनवल्शन)

    2डी कनवल्शन परत

    use ai/nn;

// 3 kanal → 64 kanal, 3×3 kernel
let conv = Conv2d::new(
    in_channels: 3,
    out_channels: 64,
    kernel_size: 3,
    stride: 1,
    padding: 1
);

let x = Tensor::random([32, 3, 224, 224]);  // NCHW format
let y = conv.ileri(x);  // [32, 64, 224, 224]

    LSTM (दीर्घकालिक अल्पकालिक मेमोरी)

    आवर्तक तंत्रिका नेटवर्क परत

    use ai/nn;

// 100 girdi → 256 hidden units
let lstm = LSTM::new(
    input_size: 100,
    hidden_size: 256,
    num_layers: 2,
    dropout: 0.2
);

// Sequence: [batch, seq_len, features]
let x = Tensor::random([32, 50, 100]);
let (output, (h_n, c_n)) = lstm.ileri(x);
// output: [32, 50, 256]

    कस्टम मॉड्यूल

    अपना खुद का मॉड्यूल बनाएं

    use ai/nn;
use ai/tensor;

sınıf ResBlock {
    let conv1: Conv2d;
    let bn1: BatchNorm;
    let conv2: Conv2d;
    let bn2: BatchNorm;
    let relu: ReLU;
    
    function new(channels: int) -> ResBlock do
        ResBlock {
            conv1: Conv2d::new(channels, channels, 3, 1, 1),
            bn1: BatchNorm::new(channels),
            conv2: Conv2d::new(channels, channels, 3, 1, 1),
            bn2: BatchNorm::new(channels),
            relu: ReLU::new()
        }
    end
    
    function ileri(let kendi, x: Tensor) -> Tensor do
        let identity = x;
        
        let out = kendi.conv1.ileri(x);
        out = kendi.bn1.ileri(out);
        out = kendi.relu.ileri(out);
        
        out = kendi.conv2.ileri(out);
        out = kendi.bn2.ileri(out);
        
        out = out.topla(identity);  // Residual connection
        out = kendi.relu.ileri(out);
        
        return out;
    end
}

    एपीआई संदर्भ

    परतें

    • रैखिक:पूरी तरह से जुड़ी हुई परत
    • Conv2d:2डी कनवल्शन
    • Conv3d:3डी कनवल्शन
    • एलएसटीएम:दीर्घकालिक अल्पकालिक स्मृति
    • जीआरयू:गेटेड आवर्ती इकाई
    • ट्रांसफार्मर:ध्यान परत

    सक्रियण

    • ReLU:अधिकतम(0, x)
    • लीकरीएलयू:अधिकतम(0.01x, x)
    • सिग्मॉइड:1/(1+e^-x)
    • तन:(ई^एक्स - ई^-एक्स)/(ई^एक्स + ई^-एक्स)
    • सॉफ्टमैक्स:ई^xi / Σe^xj
    • गेलु:गाऊसी त्रुटि रैखिक इकाई
