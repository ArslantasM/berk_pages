⏱️ राज्य::सिस्टिक

सिस्टम टिक टाइमर - आरटीओएस और देरी के लिए मिलीसेकंड टाइमबेस

~240 पंक्तियाँ ~10 कार्य एआरएम कॉर्टेक्स-एम

📖अवलोकन

SysTick एक 24-बिट सिस्टम टाइमर है जो ARM Cortex-M आर्किटेक्चर पर उपलब्ध है। इसका उपयोग आम तौर पर 1ms टिक इंटरप्ट उत्पन्न करने के लिए किया जाता है और RTOS (रियल-टाइम ऑपरेटिंग सिस्टम) शेड्यूलर का आधार बनता है। एमएस-आधारित विलंब कार्यों और अपटाइम काउंटरों के लिए भी उपयोग किया जाता है।

🔑 मुख्य विशेषताएं

  • 24-बिट डाउन काउंटर (0xFFFFFF)
  • कॉन्फ़िगर करने योग्य टिक दर (आमतौर पर 1ms)
  • अपवाद/व्यवधान उत्पन्न करना
  • सीपीयू घड़ी या बाहरी घड़ी से प्राप्त
  • कम ओवरहेड टाइमिंग
  • आरटीओएस शेड्यूलर टाइमबेस
  • मिलीसेकंड विलंब कार्य
  • अपटाइम काउंटर

🚀 त्वरित शुरुआत

import hal::systick, hal::core

let g_tick_count: u64 = 0

// SysTick interrupt handler
@interrupt
function SysTick_Handler() do
    g_tick_count += 1
end

function main() do
    // 72 MHz CPU, 1ms tick
    systick.init(72000000 / 1000)  // 1ms period
    systick.enable_interrupt()
    systick.enable()
    
    // Now delay_ms() works
    for do
        io.println("Uptime: " + g_tick_count.yazıya() + " ms")
        systick.delay_ms(1000)
    end
end

💡 उदाहरण 1: मिलीसेकंड विलंब कार्यान्वयन

ब्लॉकिंग विलंब और नॉन-ब्लॉकिंग टाइमआउट

import hal::systick

let g_ticks: sayacı64 = 0

@interrupt
function SysTick_Handler() do
    g_ticks += 1
end

// Blocking delay
function delay_ms(ms: sayı) do
    let start = g_ticks
    for (g_ticks - start) < ms.to_u64() for do
        // Wait
    end
end

// Non-blocking timeout
struct Timeout do
    start: sayı64,
    duration: sayı64
end

function timeout_start(ms: sayı) -> Timeout do
    return Timeout do
        start: g_ticks,
        duration: ms.to_u64()
    end
end

function timeout_expired(timeout: Timeout) -> mantık do
    return (g_ticks - timeout.start) >= timeout.duration
end

// Kullanım örneği
function main_loop() do
    let timeout = timeout_start(5000)  // 5 saniye
    
    for do
        // Non-blocking işlemler
        process_data()
        
        if timeout_expired(timeout) then do
            io.println("5 saniye geçti!")
            timeout = timeout_start(5000)  // Reset
        end
    end
end

💡 उदाहरण 2: सरल आरटीओएस शेड्यूलर

SysTick के साथ सहकारी मल्टीटास्किंग

import hal::systick

struct Task do
    function: Fonksiyon(),
    period_ms: sayı,
    son_calistirma: sayı64
end

let g_ticks: sayı64 = 0
let g_tasks: Dizi[Task] = []

@interrupt
function SysTick_Handler() do
    g_ticks += 1
end

function task_register(func: Fonksiyon(), period_ms: sayı) do
    g_tasks.push(Task do
        function: func,
        period_ms: period_ms,
        son_calistirma: 0
    end)
end

function scheduler_run() do
    for do
        her task in g_tasks for do
            let elapsed = g_ticks - task.son_calistirma
            
            if elapsed >= task.period_ms.to_u64() then do
                // Task zamanı geldi
                task.function()
                task.son_calistirma = g_ticks
            end
        end
    end
end

// Örnek tasklar
function led_blink_task() do
    gpio.pin_toggle(LED_PIN)
end

function sensor_read_task() do
    let temp = sensor.read_temperature()
    io.println("Sıcaklık: " + temp.yazıya())
end

function main() do
    systick.init(72000000 / 1000)  // 1ms
    systick.enable_interrupt()
    systick.enable()
    
    // Register tasks
    task_register(led_blink_task, 500)      // 500ms period
    task_register(sensor_read_task, 1000)   // 1000ms period
    
    // Run scheduler
    scheduler_run()
end

💡 उदाहरण 3: प्रदर्शन प्रोफ़ाइलिंग

फ़ंक्शन निष्पादन समय माप

import hal::systick

function micros() -> sayı64 do
    // SysTick counts down from reload value
    let reload = systick.get_reload()
    let current = systick.get_value()
    
    // Calculate microseconds
    let ticks_elapsed = reload - current
    let us_per_tick = 1000 / reload  // Assuming 1ms period
    
    return (g_ticks * 1000) + (ticks_elapsed * us_per_tick).to_u64()
end

function profile_start() -> sayı64 do
    return micros()
end

function profile_end(start: sayı64) -> sayı64 do
    return micros() - start
end

// Kullanım
function slow_function() do
    for i in 0..1000 for do
        math.sqrt(i.kesir())
    end
end

function main() do
    systick.init(72000)  // 1ms
    systick.enable()
    
    let start = profile_start()
    slow_function()
    let duration = profile_end(start)
    
    io.println("Süre: " + duration.yazıya() + " μs")
end

💡 सर्वोत्तम प्रथाएँ

  • टिक दर:आरटीओएस के लिए 1ms उद्योग मानक
  • व्यवधान प्राथमिकता:SysTick आमतौर पर सबसे कम प्राथमिकता है (15)
  • हैंडलर अवधि:इंटरप्ट हैंडलर को न्यूनतम रखा जाना चाहिए (<10μs)
  • अतिप्रवाह प्रबंधन:64-बिट काउंटर का उपयोग करके अतिप्रवाह को रोकें
  • शक्ति अनुकूलन:टिकलेस आइडल मोड (आरटीओएस) पर विचार करें
  • डीबग फ़्रीज़:आप डिबग के दौरान SysTick को रोक सकते हैं

⚠️ महत्वपूर्ण नोट्स

  • 24-बिट काउंटर: अधिकतम पुनः लोड मान 0xFFFFFF (16,777,215)
  • 72 मेगाहर्ट्ज पर 1 एमएस के लिए पुनः लोड = 72000
  • इंटरप्ट हैंडलर में लंबी प्रोसेसिंग (संदर्भ स्विच ओवरहेड)
  • परमाणु संचालन के लिए इंटरप्ट डिसेबल की आवश्यकता हो सकती है
