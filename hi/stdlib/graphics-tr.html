import graphics

// Window oluştur
let pencere = graphics.window_create("BERK Graphics", 800, 600)

// OpenGL context
let gl = graphics.gl_context_create(pencere)
graphics.gl_make_current(gl)

// Shader program
let vertex_shader = graphics.shader_compile(VERTEX_SHADER, vertex_code)
let fragment_shader = graphics.shader_compile(FRAGMENT_SHADER, fragment_code)
let program = graphics.program_link(vertex_shader, fragment_shader)

// Vertex data
let vertices = [
    -0.5, -0.5, 0.0,  // sol alt
     0.5, -0.5, 0.0,  // sağ alt
     0.0,  0.5, 0.0   // üst
]

let vbo = graphics.buffer_create()
graphics.buffer_data(vbo, vertices)

// Render loop
for do
    graphics.clear(0.2, 0.3, 0.3, 1.0)
    
    graphics.program_use(program)
    graphics.draw_arrays(TRIANGLES, 0, 3)
    
    graphics.window_swap_buffers(pencere)
    graphics.poll_events()
    
    if graphics.window_should_close(pencere) then do
        kır
    end
end

import graphics, math, image

// Cube vertices with texture coordinates
let cube_vertices = [
    // position        // texcoord
    -0.5, -0.5, -0.5,  0.0, 0.0,
     0.5, -0.5, -0.5,  1.0, 0.0,
     0.5,  0.5, -0.5,  1.0, 1.0,
    // ... (diğer yüzler)
]

// Load texture
let texture = graphics.texture_create()
let img = image.load("crate.png")
graphics.texture_data(texture, img.genişlik, img.yükseklik, img.veri)
graphics.texture_set_filter(texture, LINEAR, LINEAR_MIPMAP_LINEAR)
graphics.texture_generate_mipmap(texture)

// Camera
let kamera = do
    pozisyon: [0.0, 0.0, 3.0],
    hedef: [0.0, 0.0, 0.0],
    yukarı: [0.0, 1.0, 0.0]
end

// Transformation matrices
let model = math.mat4_rotate_y(graphics.time() * 50.0)
let view = math.mat4_look_at(kamera.pozisyon, kamera.hedef, kamera.yukarı)
let projection = math.mat4_perspective(45.0, 800.0/600.0, 0.1, 100.0)

// Render
graphics.program_use(program)
graphics.program_set_mat4(program, "model", model)
graphics.program_set_mat4(program, "view", view)
graphics.program_set_mat4(program, "projection", projection)
graphics.texture_bind(texture, 0)
graphics.draw_arrays(TRIANGLES, 0, 36)

import graphics

// Framebuffer oluştur
let fbo = graphics.framebuffer_create()
let color_texture = graphics.texture_create()
graphics.texture_data(color_texture, 800, 600, yok)
graphics.framebuffer_attach_texture(fbo, color_texture)

// Post-processing shader (grayscale)
let post_shader = graphics.program_create("
    #version 330
    in vec2 TexCoord;
    out vec4 FragColor;
    uniform sampler2D screenTexture;
    
    void main() {
        vec3 color = texture(screenTexture, TexCoord).rgb;
        float gray = 0.299 * color.r + 0.587 * color.g + 0.114 * color.b;
        FragColor = vec4(gray, gray, gray, 1.0);
    }
")

// Render loop
for do
    // 1. Render sahneyi framebuffer'a
    graphics.framebuffer_bind(fbo)
    graphics.clear(0.1, 0.1, 0.1, 1.0)
    render_scene()
    
    // 2. Post-processing uygula
    graphics.framebuffer_bind_default()
    graphics.clear(0.0, 0.0, 0.0, 1.0)
    graphics.program_use(post_shader)
    graphics.texture_bind(color_texture, 0)
    graphics.draw_fullscreen_quad()
    
    graphics.window_swap_buffers(pencere)
end
