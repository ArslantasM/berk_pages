🔌 राज्य::gpio

सामान्य प्रयोजन इनपुट/आउटपुट नियंत्रण - GPIO पिन प्रबंधन

381 पंक्तियाँ ~30 कार्य STM32/GD32/ESP32

📖अवलोकन

GPIO (सामान्य प्रयोजन इनपुट/आउटपुट) मॉड्यूल डिजिटल इनपुट/आउटपुट के रूप में माइक्रोकंट्रोलर पिन का नियंत्रण प्रदान करता है। एलईडी नियंत्रण, बटन रीडिंग, सेंसर इंटरफेस और डिजिटल सिग्नल के लिए उपयोग किया जाता है।

🔑 मुख्य विशेषताएं

  • डिजिटल इनपुट/आउटपुट मोड
  • पुश-पुल/ओपन-ड्रेन आउटपुट
  • पुल-अप/पुल-डाउन प्रतिरोधक
  • पिन गति विन्यास
  • व्यवधान समर्थन (EXTI)
  • वैकल्पिक फ़ंक्शन मैपिंग
  • परमाणु बिट संचालन
  • पोर्ट लॉकिंग

🚀 त्वरित शुरुआत

import hal::gpio

// GPIO clock enable
gpio.clock_enable(gpio.PORT_C)

// LED pin setup (output)
let led = gpio.pin_init(gpio.PORT_C, 13, do
    mode: gpio.MODE_OUTPUT,
    speed: gpio.SPEED_LOW,
    pull: gpio.PULL_NONE
end)

// LED control
gpio.pin_write(led, gpio.HIGH)  // Turn on
gpio.pin_toggle(led)  // Toggle
gpio.pin_write(led, gpio.LOW)  // Turn off

📦 प्रकार और एनम

// GPIO Portları
enum Port {
    A, B, C, D, E, F, G, H, I
}

// Pin Modları
enum Mode {
    INPUT,        // Digital input
    OUTPUT,       // Push-pull output
    OUTPUT_OD,    // Open-drain output
    AF,           // Alternate function push-pull
    AF_OD,        // Alternate function open-drain
    ANALOG        // Analog mode
}

// Hız Seviyesi
enum Speed {
    LOW,       // Low speed (2 MHz)
    MEDIUM,    // Medium speed (25 MHz)
    HIGH,      // High speed (50 MHz)
    VERY_HIGH  // Very high speed (100 MHz)
}

// Pull Direnci
enum Pull {
    NONE,  // No pull resistor
    UP,    // Pull-up
    DOWN   // Pull-down
}

// Logic Level
enum Level {
    LOW,   // Logic 0 (0V)
    HIGH   // Logic 1 (3.3V/5V)
}

// Interrupt Tetikleme
enum Trigger {
    RISING,   // Rising edge
    FALLING,  // Falling edge
    BOTH      // Both edges
}

💡 उदाहरण 1: एलईडी ब्लिंक

import hal::gpio, hal::core

function main() do
    // System init
    core.system_init()
    core.clock_config(72_000_000)  // 72MHz
    
    // LED setup (PC13 on STM32 Blue Pill)
    gpio.clock_enable(gpio.PORT_C)
    let led = gpio.pin_init(gpio.PORT_C, 13, do
        mode: gpio.MODE_OUTPUT,
        speed: gpio.SPEED_LOW,
        pull: gpio.PULL_NONE
    end)
    
    // Blink loop
    for do
        gpio.pin_toggle(led)
        core.delay_ms(500)
    end
end

💡 उदाहरण 2: इंटरप्ट के साथ बटन इनपुट

import hal::gpio, hal::int

let button_pressed = false

function button_handler() do
    button_pressed = true
    io.println("Button pressed!")
end

function main() do
    // LED setup (output)
    gpio.clock_enable(gpio.PORT_C)
    let led = gpio.pin_init(gpio.PORT_C, 13, do
        mode: gpio.MODE_OUTPUT,
        speed: gpio.SPEED_LOW,
        pull: gpio.PULL_NONE
    end)
    
    // Button setup (input with pull-up)
    gpio.clock_enable(gpio.PORT_A)
    let button = gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_INPUT,
        pull: gpio.PULL_UP,
        speed: gpio.SPEED_LOW
    end)
    
    // Enable button interrupt
    gpio.interrupt_enable(button, gpio.TRIGGER_FALLING, button_handler)
    
    // Main loop
    for do
        if button_pressed then do
            button_pressed = false
            gpio.pin_toggle(led)
        end
        core.delay_ms(10)
    end
end

💡 उदाहरण 3: मल्टी-एलईडी नियंत्रण (पोर्ट संचालन)

import hal::gpio, hal::core

function main() do
    core.system_init()
    
    // Setup 8 LEDs on Port B (PB0-PB7)
    gpio.clock_enable(gpio.PORT_B)
    
    // Initialize all 8 pins
    her i in 0..8 for do
        gpio.pin_init(gpio.PORT_B, i, do
            mode: gpio.MODE_OUTPUT,
            speed: gpio.SPEED_MEDIUM,
            pull: gpio.PULL_NONE
        end)
    end
    
    // LED patterns
    let patterns = [
        0b00000001,  // Single LED
        0b00000011,  // 2 LEDs
        0b00000111,  // 3 LEDs
        0b00001111,  // 4 LEDs
        0b00011111,  // 5 LEDs
        0b00111111,  // 6 LEDs
        0b01111111,  // 7 LEDs
        0b11111111   // All LEDs
    ]
    
    for do
        her pattern in patterns for do
            gpio.port_write(gpio.PORT_B, pattern)
            core.delay_ms(200)
        end
        
        // Reverse
        her pattern in patterns.reverse() for do
            gpio.port_write(gpio.PORT_B, pattern)
            core.delay_ms(200)
        end
    end
end

💡 उदाहरण 4: ट्रैफिक लाइट नियंत्रक

import hal::gpio, hal::core

struct TrafficLight do
    red: gpio.Pin,
    yellow: gpio.Pin,
    green: gpio.Pin
end

function traffic_light_init() -> TrafficLight do
    gpio.clock_enable(gpio.PORT_A)
    
    let tl = TrafficLight do
        red: gpio.pin_init(gpio.PORT_A, 0, do
            mode: gpio.MODE_OUTPUT,
            speed: gpio.SPEED_LOW,
            pull: gpio.PULL_NONE
        end),
        yellow: gpio.pin_init(gpio.PORT_A, 1, do
            mode: gpio.MODE_OUTPUT,
            speed: gpio.SPEED_LOW,
            pull: gpio.PULL_NONE
        end),
        green: gpio.pin_init(gpio.PORT_A, 2, do
            mode: gpio.MODE_OUTPUT,
            speed: gpio.SPEED_LOW,
            pull: gpio.PULL_NONE
        end)
    end
    
    return tl
end

function set_state(tl: TrafficLight, red: bool, yellow: bool, green: bool) do
    gpio.pin_write(tl.red, if red then gpio.HIGH yoksa gpio.LOW)
    gpio.pin_write(tl.yellow, if yellow then gpio.HIGH yoksa gpio.LOW)
    gpio.pin_write(tl.green, if green then gpio.HIGH yoksa gpio.LOW)
end

function main() do
    core.system_init()
    let tl = traffic_light_init()
    
    for do
        // Green light
        set_state(tl, false, false, true)
        core.delay_ms(5000)
        
        // Yellow light
        set_state(tl, false, true, false)
        core.delay_ms(2000)
        
        // Red light
        set_state(tl, true, false, false)
        core.delay_ms(5000)
    end
end

⚠️ महत्वपूर्ण नोट्स

  • पिन का उपयोग करने से पहले आपको पोर्ट क्लॉक को सक्षम करना होगा।
  • ओपन-ड्रेन आउटलेट्स को बाहरी पुल-अप की आवश्यकता होती है
  • 5V सहिष्णु पिन के लिए डेटाशीट की जाँच करें
  • इंटरप्ट का उपयोग करते समय एनवीआईसी कॉन्फ़िगरेशन की आवश्यकता होती है
  • अधिकतम पिन वर्तमान सीमा (~25mA) पर ध्यान दें
  • एनालॉग मोड का उपयोग करते समय डिजिटल बफ़र अक्षम हो जाते हैं
