    चैट समापन

    use ai/llm;

let client = OpenAI::new(
    api_key: env::var("OPENAI_API_KEY")?
);

let response = client.chat(
    model: "gpt-4",
    messages: [
        Message { role: "system", content: "Sen yardımcı bir asistansın." },
        Message { role: "user", content: "Rust ile Berk arasındaki farklar neler?" }
    ],
    temperature: 0.7,
    max_tokens: 1000
).await?;

print("{}", response.content);

    फ़ंक्शन कॉलिंग

    use ai/llm;

let functions = [
    Function {
        name: "get_weather",
        description: "Hava durumu bilgisini receive",
        parameters: json!({
            "type": "object",
            "properties": {
                "location": { "type": "string" },
                "unit": { "type": "string", "enum": ["celsius", "fahrenheit"] }
            }
        })
    }
];

let response = client.chat(
    model: "gpt-4",
    messages: messages,
    functions: functions,
    function_call: "auto"
).await?;

if response.function_call.is_some() then do
    let func = response.function_call.unwrap();
    print("Fonksiyon çağrısı: {}({})", func.name, func.arguments);
end

    एकीकृत एपीआई

    प्रदाता-अज्ञेयवादी कोड

    use ai/llm;

// Tüm provider'lar for aynı arayüz
function chat(provider: LLMProvider, prompt: dizgi) -> dizgi do
    let response = provider.chat(
        messages: [Message::user(prompt)],
        temperature: 0.7
    ).await?;
    
    return response.content;
end

// Kullanım
let openai = OpenAI::new(openai_key);
let claude = Anthropic::new(anthropic_key);

let r1 = chat(openai, "Merhaba")?;
let r2 = chat(claude, "Merhaba")?;

    बातचीत का इतिहास

    use ai/llm;

let conversation = Conversation::new();

conversation.add_system("Sen bir BERK uzmanısın.");
conversation.add_user("BERK'te vektör nasıl tanımlanır?");

let response = client.chat(
    messages: conversation.messages(),
    temperature: 0.7
).await?;

conversation.add_assistant(response.content);

// Devam eden konuşma
conversation.add_user("Peki matris nasıl?");
let response2 = client.chat(
    messages: conversation.messages()

    पुनः प्रयास करें और त्रुटि प्रबंधन

    use ai/llm;

let client = OpenAI::new(api_key)
    .with_retry(max_attempts: 3, backoff: 2.0)
    .with_timeout(30.0);

deneme do
    let response = client.chat(messages).await?;
    print("{}", response.content);
catch hata for do
    match hata {
        LLMError::RateLimit => print!("Rate limit aşıldı, bekleyin..."),
        LLMError::InvalidKey => print!("API key geçersiz"),
        LLMError::Timeout => print!("Zaman aşımı"),
        _ => print!("Err: {}", hata)
    }
end

    एपीआई संदर्भ

    प्रदाता विशेषता

    • chat(messages, options) -> Response- चैट पूरा होना
    • chat_stream(messages) -> Stream- स्ट्रीमिंग चैट
    • count_tokens(messages) -> usize- टोकन की संख्या
    • list_models() -> Vec<String>- उपलब्ध मॉडल

    संदेश संरचना

    • role: "system" | "user" | "assistant"
    • content: dizgi- संदेश की सामग्री
    • name: Option<dizgi>- वैकल्पिक नाम
