वैज्ञानिक कंप्यूटिंग

import math

function fourier_series(x: float, n: integer) -> float do
    // Fourier serisi ile kare dalga yaklaşımı
    variable sonuç = 0.0
    
    her k in range(1, n + 1) for do
        variable terim = math.sin((2.0 * k - 1.0) * x) / (2.0 * k - 1.0)
        sonuç = sonuç + terim
    end
    
    return (4.0 / math.PI) * sonuç
end

// Test
her x in range(0, 100) for do
    variable rad = (x * math.PI) / 50.0
    variable değer = fourier_series(rad, 10)
    print(değer.yazıya())
end

संख्यात्मक एकीकरण

import math

function simpson_integration(
    f: function,
    a: float, 
    b: float, 
    n: integer
) -> float do
    // Simpson's rule ile sayısal integrasyon
    variable h = (b - a) / n
    variable sum = f(a) + f(b)
    
    her i in range(1, n) for do
        variable x = a + i * h
        if i % 2 == 0 then do
            sum = sum + 2.0 * f(x)
        elseif do
            sum = sum + 4.0 * f(x)
        end
    end
    
    return (h / 3.0) * sum
end

// π hesaplama: ∫₀¹ 4/(1+x²) dx = π
variable pi_approx = simpson_integration(
    function(x) do return 4.0 / (1.0 + x * x) end,
    0.0,
    1.0,
    1000
)
print("π ≈ " + pi_approx.yazıya())
