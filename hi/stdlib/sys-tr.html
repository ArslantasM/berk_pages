🚀 त्वरित शुरुआत

import sys

// Command line arguments
let args = sys.args()
print("Program: " + args[0])

// Environment variables
let path = sys.env("PATH")
sys.set_env("MY_VAR", "value")

// Process execution
let sonuç = sys.exec("ls", ["-la"])
print("Çıktı: " + sonuç.stdout)
print("Exit code: " + sonuç.exit_code.yazıya())

// Platform info
print("OS: " + sys.os())  // "linux", "windows", "macos"
print("Arch: " + sys.arch())  // "x86_64", "aarch64"
