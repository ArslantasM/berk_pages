📋अवलोकन

xmlमॉड्यूल XML (एक्स्टेंसिबल मार्कअप लैंग्वेज) फ़ाइलों को पढ़ने, लिखने, पार्सिंग और संपादन के लिए व्यापक उपकरण प्रदान करता है। DOM और SAX पार्सर्स, XPath क्वेरीज़ और XML स्कीमा सत्यापन का समर्थन करता है।

🔍एक्सएमएल पार्सिंग

  • xml.parse()- XML ​​टेक्स्ट को DOM ट्री में कनवर्ट करना
  • parse_from_xml.file()- एक्सएमएल फ़ाइल पढ़ना
  • xml.sax_parse()- SAX आधारित स्ट्रीम पार्सिंग
  • xml.मान्य करें()- XML ​​स्कीमा के विरुद्ध सत्यापन

📝 XML बनाना

  • xml.दस्तावेज़()- नया XML दस्तावेज़ बनाना
  • xml.तत्व()- एक XML तत्व बनाना
  • xml.पाठ()- टेक्स्ट नोड जोड़ें
  • xml.टिप्पणी()- टिप्पणी करें
  • xml.cdata()- सीडीएटीए विभाजन जोड़ें

🔎 XPath क्वेरीज़

  • xml.xpath()- एक XPath अभिव्यक्ति निष्पादित करना
  • xml.खोजें()- पहला मिलान तत्व ढूँढना
  • xml.find_all()- सभी मेल खाने वाले तत्वों को ढूँढना
  • xml.get_attribute()- तत्व विशेषता प्राप्त करना

✏️ एक्सएमएल संपादन

  • xml.alt_add()- बाल तत्व जोड़ना
  • xml.निकालें()-तत्व हटाना

💾एक्सएमएल क्रमांकन

XML को पार्स करना और पढ़ना

use xml

// XML metnini ayrıştırma
değer xml_metin = """
<kitaplar>
    <kitap id="1">
        <başlık>BERK Programlama</başlık>
        <yazar>Ahmet Yılmaz</yazar>
        <fiyat>49.90</fiyat>
    </kitap>
    <kitap id="2">
        <başlık>Web Geliştirme</başlık>
        <yazar>Ayşe Demir</yazar>
        <fiyat>59.90</fiyat>
    </kitap>
</kitaplar>
"""

değer belge = xml.ayrıştır(xml_metin)

// XPath ile sorgulama
değer kitaplar = belge.xpath("//kitap")
for kitap in kitaplar {
    değer başlık = kitap.bul("başlık").String
    değer yazar = kitap.bul("yazar").String
    değer fiyat = kitap.bul("fiyat").String
    değer id = kitap.öznitelik_al("id")
    
    print(f"Kitap #{id}: {başlık} - {yazar} ({fiyat} TL)")
}

// Çıktı:
// Kitap #1: BERK Programlama - Ahmet Yılmaz (49.90 TL)
// Kitap #2: Web Geliştirme - Ayşe Demir (59.90 TL)

एक्सएमएल बनाना

use xml

// Yeni XML belgesi oluşturma
değer belge = xml.Belge("1.0", "UTF-8")
değer kök = belge.kök_element_oluştur("katalog")

// Ürünler ekleme
değer ürünler = [
    {"id": "P001", "ad": "Laptop", "fiyat": 5500, "stok": 15},
    {"id": "P002", "ad": "Mouse", "fiyat": 75, "stok": 120},
    {"id": "P003", "ad": "Klavye", "fiyat": 250, "stok": 45}
]

for ürün in ürünler {
    değer elem = xml.Element("ürün")
    elem.öznitelik_ayarla("id", ürün["id"])
    elem.öznitelik_ayarla("stok", ürün["stok"].dizgeye())
    
    değer ad = xml.Element("ad")
    ad.metin_ayarla(ürün["ad"])
    elem.alt_ekle(ad)
    
    değer fiyat = xml.Element("fiyat")
    fiyat.metin_ayarla(ürün["fiyat"].dizgeye())
    fiyat.öznitelik_ayarla("para_birimi", "TRY")
    elem.alt_ekle(fiyat)
    
    kök.alt_ekle(elem)
}

// XML'i dosyaya kaydetme
xml.dosyaya_yaz(belge, "katalog.xml", girintili: true)
print("XML belgesi oluşturuldu!")

कॉन्फ़िगरेशन फ़ाइल प्रबंधक

use xml
use fs

sınıf YapılandırmaYöneticisi {
    değer dosya_yolu: Dize
    değer belge: xml.Belge
    
    function new(dosya_yolu: Dize) -> YapılandırmaYöneticisi {
        değer örnek = YapılandırmaYöneticisi {
            dosya_yolu: dosya_yolu,
            belge: null
        }
        
        if fs.dosya_var_mı(dosya_yolu) {
            örnek.belge = xml.dosyadan_ayrıştır(dosya_yolu)
        } yoksa {
            örnek.belge = xml.Belge("1.0", "UTF-8")
            örnek.belge.kök_element_oluştur("yapılandırma")
        }
        
        return örnek
    }
    
    function ayar_al(bölüm: Dize, anahtar: Dize) -> Sonuç<Dize, Err> {
        değer xpath = f"//bölüm[@ad='{bölüm}']/{anahtar}"
        değer bulunan = bu.belge.xpath(xpath)
        
        if bulunan.len > 0 {
            return Ok(bulunan[0].String)
        }
        return Err(f"Ayar bulunamadı: {bölüm}.{anahtar}")
    }
    
    function ayar_ata(bölüm: Dize, anahtar: Dize, değer: Dize) {
        değer kök = bu.belge.kök()
        değer bölüm_xpath = f"//bölüm[@ad='{bölüm}']"
        değer bölüm_elem = kök.xpath(bölüm_xpath)
        
        if bölüm_elem.len == 0 {
            bölüm_elem = xml.Element("bölüm")
            bölüm_elem.öznitelik_ayarla("ad", bölüm)
            kök.alt_ekle(bölüm_elem)
        } yoksa {
            bölüm_elem = bölüm_elem[0]
        }
        
        değer ayar_elem = xml.Element(anahtar)
        ayar_elem.metin_ayarla(değer)
        bölüm_elem.alt_ekle(ayar_elem)
    }
    
    function kaydet() {
        xml.dosyaya_yaz(bu.belge, bu.dosya_yolu, girintili: true)
    }
}

// Kullanım
değer yapılandırma = YapılandırmaYöneticisi.new("uygulama.xml")

yapılandırma.ayar_ata("veritabanı", "sunucu", "localhost")
yapılandırma.ayar_ata("veritabanı", "port", "5432")
yapılandırma.ayar_ata("uygulama", "dil", "tr")
yapılandırma.ayar_ata("uygulama", "tema", "koyu")

yapılandırma.kaydet()

// Ayarları okuma
değer sunucu = yapılandırma.ayar_al("veritabanı", "sunucu").aç()
değer dil = yapılandırma.ayar_al("uygulama", "dil").aç()
print(f"Sunucu: {sunucu}, Dil: {dil}")
