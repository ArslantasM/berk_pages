🕐 राज्य::rtc

वास्तविक समय घड़ी - दिनांक/समय प्रबंधन

379 पंक्तियाँ ~12 कार्य कम बिजली

📖अवलोकन

आरटीसी (रियल-टाइम क्लॉक) मॉड्यूल कम बिजली की खपत के साथ दिनांक और समय रखने का कार्य प्रदान करता है। यह बैटरी चालित संचालन, अलार्म और कैलेंडर सुविधाएँ प्रदान करता है।

🔑 मुख्य विशेषताएं

  • दिनांक/समय रखना
  • लीप वर्ष वाला कैलेंडर
  • 2 प्रोग्रामयोग्य अलार्म
  • जागने का टाइमर
  • बैकअप रजिस्टर (80 बाइट्स)
  • बैटरी समर्थित ऑपरेशन
  • टाइमस्टैम्प घटना
  • उप-सेकंड परिशुद्धता

💡 उदाहरण: डिजिटल क्लॉक डिस्प्ले

import hal::rtc, hal::core

struct DateTime do
    year: sayı,
    month: sayı,
    day: sayı,
    hours: sayı,
    minutes: sayı,
    seconds: sayı,
    weekday: sayı
end

function weekday_name(day: sayı) -> yazı do
    return eşle day do
        1 => "Pazartesi",
        2 => "Salı",
        3 => "Çarşamba",
        4 => "Perşembe",
        5 => "Cuma",
        6 => "Cumartesi",
        7 => "Pazar",
        _ => "Bilinmeyen"
    end
end

function main() do
    core.system_init()
    
    // Initialize RTC
    let rtc_handle = rtc.init(do
        hour_format: rtc.FORMAT_24H
    end)
    
    // Set initial time (2024-12-25 14:30:00 Wednesday)
    rtc.set_date(rtc_handle, do
        year: 2024,
        month: 12,
        day: 25,
        weekday: rtc.WEDNESDAY
    end)
    
    rtc.set_time(rtc_handle, do
        hours: 14,
        minutes: 30,
        seconds: 0
    end)
    
    for do
        // Read current time
        let time = rtc.get_time(rtc_handle)
        let date = rtc.get_date(rtc_handle)
        
        // Display
        io.print("\r")  // Carriage return
        io.print("{:02}/{:02}/{} {} {:02}:{:02}:{:02}".formatla(
            date.day,
            date.month,
            date.year,
            weekday_name(date.weekday),
            time.hours,
            time.minutes,
            time.seconds
        ))
        
        core.delay_ms(1000)
    end
end

💡 उदाहरण: अलार्म घड़ी

import hal::rtc

let alarm_triggered = false

function alarm_callback() do
    alarm_triggered = true
    io.println("\n🔔 ALARM! Wake up!")
end

function set_alarm(rtc_handle: rtc.Handle, hours: sayı, minutes: sayı) do
    rtc.set_alarm(rtc_handle, rtc.ALARM_A, do
        hours: hours,
        minutes: minutes,
        seconds: 0,
        alarm_mask: rtc.MASK_SECONDS,  // Ignore seconds
        callback: alarm_callback
    end)
    
    io.println("Alarm set for {:02}:{:02}".formatla(hours, minutes))
end

function main() do
    let rtc_handle = rtc.init(do
        hour_format: rtc.FORMAT_24H
    end)
    
    // Set current time: 07:58:00
    rtc.set_time(rtc_handle, do
        hours: 7,
        minutes: 58,
        seconds: 0
    end)
    
    // Set alarm for 08:00
    set_alarm(rtc_handle, 8, 0)
    
    for do
        let time = rtc.get_time(rtc_handle)
        io.print("\r{:02}:{:02}:{:02}".formatla(
            time.hours, time.minutes, time.seconds
        ))
        
        if alarm_triggered then do
            alarm_triggered = false
            // Snooze for 5 minutes
            set_alarm(rtc_handle, time.hours, time.minutes + 5)
        end
        
        core.delay_ms(1000)
    end
end

💡 उदाहरण: वेकअप टाइमर (स्टैंडबाय से ऑटो-वेकअप)

import hal::rtc, hal::core

function periodic_wakeup() do
    let rtc_handle = rtc.init(do
        hour_format: rtc.FORMAT_24H
    end)
    
    // Configure wakeup every 10 seconds
    rtc.set_wakeup(rtc_handle, do
        prescaler: rtc.WAKEUP_DIV_16,  // 16Hz clock
        counter: 160  // 160 / 16Hz = 10 seconds
    end)
    
    io.println("Entering standby, wakeup in 10s...")
    core.delay_ms(100)  // Flush UART
    
    // Enter standby mode (lowest power)
    core.standby()
    
    // After wakeup, system resets
    io.println("Woke up from standby!")
    
    // Check wakeup flag
    if rtc.was_wakeup_flag_set() then do
        io.println("Wakeup timer triggered")
        rtc.clear_wakeup_flag()
    end
end

function main() do
    // Count wakeups using backup register
    let wakeup_count = rtc.read_backup(rtc_handle, 0)
    wakeup_count += 1
    rtc.write_backup(rtc_handle, 0, wakeup_count)
    
    io.println("Wakeup count: {}", wakeup_count)
    
    periodic_wakeup()
end

💡 उदाहरण: बैटरी-समर्थित डेटा लॉगर

import hal::rtc

// Store sensor readings in backup registers (survives power loss)
struct LogEntry do
    timestamp: sayı,  // Unix timestamp
    temp: u16,        // Temperature x100
    humidity: u16     // Humidity x100
end

const MAX_ENTRIES = 20  // 80 bytes / 4 = 20 entries

function log_data(temp: kesir, humidity: kesir) do
    let rtc_handle = rtc.init(do hour_format: rtc.FORMAT_24H end)
    
    // Get current index
    let index = rtc.read_backup(rtc_handle, 0)
    if index >= MAX_ENTRIES then index = 0
    
    // Get timestamp
    let time = rtc.get_time(rtc_handle)
    let date = rtc.get_date(rtc_handle)
    let timestamp = date_to_unix(date, time)
    
    // Store in backup registers (4 bytes per entry)
    let base = 1 + (index * 2)
    rtc.write_backup(rtc_handle, base, timestamp)
    rtc.write_backup(rtc_handle, base + 1, 
        ((temp * 100.0).sayı() << 16) | (humidity * 100.0).sayı()
    )
    
    // Update index
    rtc.write_backup(rtc_handle, 0, index + 1)
    
    io.println("Logged: {}°C, {}%RH", temp, humidity)
end

function print_log() do
    let rtc_handle = rtc.init(do hour_format: rtc.FORMAT_24H end)
    let count = rtc.read_backup(rtc_handle, 0).min(MAX_ENTRIES)
    
    io.println("\n=== Data Log ({} entries) ===", count)
    
    her i in 0..count for do
        let base = 1 + (i * 2)
        let timestamp = rtc.read_backup(rtc_handle, base)
        let data = rtc.read_backup(rtc_handle, base + 1)
        
        let temp = ((data >> 16) & 0xFFFF).kesir() / 100.0
        let hum = (data & 0xFFFF).kesir() / 100.0
        
        io.println("{}: {:.1f}°C, {:.1f}%", 
            unix_to_string(timestamp), temp, hum)
    end
end

function date_to_unix(date: rtc.Date, time: rtc.Time) -> sayı do
    // Simplified unix timestamp calculation
    return (date.year - 1970) * 31536000 + 
           date.month * 2592000 + 
           date.day * 86400 + 
           time.hours * 3600 + 
           time.minutes * 60 + 
           time.seconds
end

function main() do
    // Simulate sensor readings
    log_data(23.5, 65.2)
    core.delay_ms(5000)
    
    log_data(24.1, 63.8)
    core.delay_ms(5000)
    
    // Print all logged data
    print_log()
end

💡 उदाहरण: टाइमस्टैम्प इवेंट (इनपुट कैप्चर)

import hal::rtc, hal::gpio

let event_count = 0

function timestamp_callback() do
    let rtc_handle = rtc.init(do hour_format: rtc.FORMAT_24H end)
    
    // Get timestamp
    let time = rtc.get_timestamp_time(rtc_handle)
    let date = rtc.get_timestamp_date(rtc_handle)
    
    event_count += 1
    
    io.println("Event #{}: {:02}:{:02}:{:02}.{:03}", 
        event_count,
        time.hours, 
        time.minutes, 
        time.seconds,
        time.subseconds
    )
end

function timestamp_input_capture() do
    // Configure PA0 (WKUP pin) as timestamp trigger
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_INPUT,
        pull: gpio.PULL_DOWN
    end)
    
    let rtc_handle = rtc.init(do
        hour_format: rtc.FORMAT_24H
    end)
    
    // Enable timestamp on rising edge
    rtc.enable_timestamp(rtc_handle, do
        edge: rtc.TIMESTAMP_RISING_EDGE,
        pin: rtc.TIMESTAMP_PIN_0,
        callback: timestamp_callback
    end)
    
    io.println("Press button on PA0 to timestamp events...")
    
    for do
        core.delay_ms(100)
    end
end

function main() do
    timestamp_input_capture()
end

import hal::rtc

let tasks = []

function task_morning() do
    io.println("🌅 Good morning! Time to wake up!")
end

function task_lunch() do
    io.println("🍽️ Lunch break!")
end

function task_evening() do
    io.println("🌙 Evening reminder: Review tasks")
end

function schedule_daily_tasks() do
    let rtc_handle = rtc.init(do
        hour_format: rtc.FORMAT_24H
    end)
    
    // Set current time (for testing: 06:58)
    rtc.set_time(rtc_handle, do
        hours: 6,
        minutes: 58,
        seconds: 0
    end)
    
    // Alarm A: Morning wakeup (07:00)
    rtc.set_alarm(rtc_handle, rtc.ALARM_A, do
        hours: 7,
        minutes: 0,
        seconds: 0,
        alarm_mask: rtc.MASK_DATEWEEKDAY,  // Daily
        callback: task_morning
    end)
    
    // Alarm B: Lunch (12:00)
    rtc.set_alarm(rtc_handle, rtc.ALARM_B, do
        hours: 12,
        minutes: 0,
        seconds: 0,
        alarm_mask: rtc.MASK_DATEWEEKDAY,
        callback: task_lunch
    end)
    
    io.println("Scheduler started. Waiting for alarms...")
    
    for do
        let time = rtc.get_time(rtc_handle)
        io.print("\r{:02}:{:02}:{:02}", 
            time.hours, time.minutes, time.seconds)
        
        core.delay_ms(1000)
    end
end

function main() do
    schedule_daily_tasks()
end

⚙️ आरटीसी कॉन्फ़िगरेशन

// RTC Configuration
struct RTCConfig do
    hour_format: HourFormat,    // 12H or 24H
    async_prediv: sayı,         // Asynchronous prescaler (7-bit)
    sync_prediv: sayı,          // Synchronous prescaler (15-bit)
    clock_source: ClockSource   // LSE, LSI, HSE
end

// Hour Format
enum HourFormat {
    FORMAT_12H,  // AM/PM
    FORMAT_24H   // 00:00-23:59
}

// Clock Source
enum ClockSource {
    LSE,  // 32.768kHz external crystal (most accurate)
    LSI,  // 32kHz internal RC (less accurate, ±5%)
    HSE   // High-speed external (rarely used for RTC)
}

// Alarm Configuration
struct AlarmConfig do
    hours: sayı,
    minutes: sayı,
    seconds: sayı,
    alarm_mask: AlarmMask,      // What to match
    callback: fn()              // Interrupt callback
end

// Alarm Mask (what to ignore)
enum AlarmMask {
    MASK_NONE,           // Match all (specific time)
    MASK_DATEWEEKDAY,    // Ignore date (daily alarm)
    MASK_HOURS,          // Every hour
    MASK_MINUTES,        // Every minute
    MASK_SECONDS         // Every second
}

// Wakeup Configuration
struct WakeupConfig do
    prescaler: WakeupPrescaler,
    counter: u16  // 0-65535
end

enum WakeupPrescaler {
    WAKEUP_DIV_16,    // RTC/16 (~2048Hz for 32kHz LSE)
    WAKEUP_DIV_8,     // RTC/8
    WAKEUP_DIV_4,     // RTC/4
    WAKEUP_DIV_2      // RTC/2
}

📊 आरटीसी उपयोग परिदृश्य

परिदृश्य विशेषता बिजली की खपत स्पष्टीकरण
डिजिटल घड़ी समय/दिनांक सक्रिय वास्तविक समय प्रदर्शन
अलार्म घड़ी अलार्म ए/बी नींद + व्यवधान जागने का अलार्म
आवधिक जागरण जागने का टाइमर स्टैंडबाय (~1μA) अल्ट्रा-लो-पावर सेंसर रीडिंग
इवेंट टाइमस्टैम्प TIMESTAMP सक्रिय सटीक ईवेंट लॉगिंग
डेटा लॉकर बैकअप रेग्ज़ वीबीएटी (~1μA) बिजली हानि से बचे

⚡ प्रदर्शन युक्तियाँ

  • एलएसई क्रिस्टल: उच्चतम सटीकता के लिए उपयोग करें (±20ppm)
  • वीबीएटी: CR2032 कॉइन सेल (~1μA) के साथ वर्षों तक RTC चला सकते हैं
  • जागने का टाइमर: आवधिक सेंसर रीडिंग के लिए आदर्श (स्टैंडबाय मोड)
  • बैकअप रजिस्टर: कॉन्फ़िगरेशन डेटा के लिए उपयोग करें (80 बाइट्स)
  • अंशांकन: तापमान क्षतिपूर्ति के लिए सुचारू अंशांकन

🖥️ प्लेटफार्म समर्थन

  • STM32F1: बेसिक आरटीसी (1 अलार्म, बैकअप रेज)
  • STM32F4/L4: उन्नत आरटीसी (2 अलार्म, वेकअप, टाइमस्टैम्प, 80 बैकअप बाइट्स)
  • STM32H7: उप-सेकंड रिज़ॉल्यूशन के साथ उच्च परिशुद्धता आरटीसी
  • ईएसपी32: यूएलपी कोप्रोसेसर समर्थन के साथ आरटीसी

⚠️ महत्वपूर्ण नोट्स

  • घड़ी स्रोत: LSE (32.768kHz क्रिस्टल, ±20ppm) सबसे सटीक, LSI (32kHz RC, ±5%) बैकअप
  • वीबीएटी पिन: RTC CR2032 बैटरी (3V) के साथ पावर-डाउन में काम करना जारी रखता है
  • सुरक्षा लिखें: आरटीसी रजिस्टर राइट-प्रोटेक्टेड हैं, अनलॉक आवश्यक है
  • अलार्म बीच में आता है: EXTI लाइन 17 (अलार्म ए), 18 (अलार्म बी) का उपयोग करता है
  • स्टैंडबाय से जागो: वेकअप टाइमर के साथ अल्ट्रा-लो-पावर आवधिक संचालन
  • बैकअप डोमेन: आरटीसी + बैकअप रेग अलग पावर डोमेन (वीबीएटी)
  • कैलेंडर: लीप वर्ष की गणना स्वचालित रूप से की जाती है, 2000-2099 रेंज
  • उप-दूसरे: सिंक्रोनस प्रीस्केलर के साथ एमएस रिज़ॉल्यूशन संभव
