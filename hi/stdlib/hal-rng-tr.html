🎲 राज्य::आरएनजी

हार्डवेयर ट्रू रैंडम नंबर जेनरेटर

~380 पंक्तियाँ ~14 कार्य STM32F4/F7/H7

📖अवलोकन

आरएनजी मॉड्यूल हार्डवेयर-आधारित वास्तविक यादृच्छिक संख्याएँ उत्पन्न करता है। यह एनालॉग शोर स्रोत का उपयोग करके क्रिप्टोग्राफ़िक गुणवत्ता वाले यादृच्छिक मान प्रदान करता है। यूयूआईडी पीढ़ी, क्रिप्टोग्राफ़िक कुंजी पीढ़ी और मोंटे कार्लो सिमुलेशन के लिए IoT सुरक्षा महत्वपूर्ण है।

🔑 मुख्य विशेषताएं

  • सच्चा यादृच्छिक:एनालॉग शोर स्रोत, छद्म-यादृच्छिक नहीं
  • 32-बिट आउटपुट:प्रत्येक रीड पर पूर्ण 32-बिट यादृच्छिक मान
  • FIPS 140-2 अनुपालक:क्रिप्टोग्राफ़िक मानकों के अनुरूप
  • एकल चक्र:40 सीपीयू चक्रों में नया यादृच्छिक मान
  • घड़ी त्रुटि का पता लगाना:घड़ी की समस्या का पता चला है
  • बीज त्रुटि का पता लगाना:एन्ट्रापी त्रुटि का पता चला है
  • व्यवधान समर्थन:डेटा तैयार व्यवधान
  • डीएमए संगत:निरंतर यादृच्छिक धारा

💡 उदाहरण 1: यूयूआईडी जनरेशन (आरएफसी 4122)

समस्या: विशिष्ट पहचानकर्ता बनाना (IoT डिवाइस आईडी)

import hal::rng

struct UUID do
    data: Dizi<sayı8> = Dizi.new(16)
end

function uuid_v4_generate() -> UUID do
    let uuid = UUID do end
    
    // Generate 128 bits (16 bytes) of random data
    for i in 0..4 do
        let rand32 = rng.get_random()
        
        // Split into 4 bytes
        uuid.data[i*4 + 0] = (rand32 >> 24) & 0xFF kayıt sayı8
        uuid.data[i*4 + 1] = (rand32 >> 16) & 0xFF kayıt sayı8
        uuid.data[i*4 + 2] = (rand32 >> 8) & 0xFF kayıt sayı8
        uuid.data[i*4 + 3] = rand32 & 0xFF kayıt sayı8
    end
    
    // Set version (4) and variant (RFC 4122)
    uuid.data[6] = (uuid.data[6] & 0x0F) | 0x40  // Version 4
    uuid.data[8] = (uuid.data[8] & 0x3F) | 0x80  // Variant RFC4122
    
    return uuid
end

function uuid_to_string(uuid: UUID) -> Metin do
    // Format: xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx
    let hex = ""
    for i in 0..16 do
        hex = hex + uuid.data[i].onaltılıya(2)
        if (i == 3 or i == 5 or i == 7 or i == 9) do
            hex = hex + "-"
        end
    end
    return hex
end

function main() do
    rng.init()
    
    for i in 0..5 do
        let uuid = uuid_v4_generate()
        io.yazdir_satır("UUID: " + uuid_to_string(uuid))
    end
end

💡 उदाहरण 2: क्रिप्टोग्राफ़िक कुंजी जनरेशन

समस्या: AES-256 के लिए 256-बिट यादृच्छिक कुंजी उत्पन्न करना

import hal::rng, hal::crypto

const AES256_KEY_SIZE: sayı = 32  // 256 bit = 32 bytes

struct CryptoKey do
    data: Dizi<sayı8>,
    size: sayı
end

function generate_aes256_key() -> CryptoKey do
    let key = CryptoKey do
        data: Dizi.new(AES256_KEY_SIZE),
        size: AES256_KEY_SIZE
    end
    
    // Generate 256 bits of cryptographic random data
    for i in 0..8 do  // 8 * 32 bits = 256 bits
        let rand = rng.get_random()
        
        if rng.get_flag(rng.FLAG_CECS) or rng.get_flag(rng.FLAG_SECS) do
            io.yazdir_satır("RNG ERROR: Clock or seed error!")
            return key  // Invalid key
        end
        
        key.data[i*4 + 0] = (rand >> 24) & 0xFF kayıt sayı8
        key.data[i*4 + 1] = (rand >> 16) & 0xFF kayıt sayı8
        key.data[i*4 + 2] = (rand >> 8) & 0xFF kayıt sayı8
        key.data[i*4 + 3] = rand & 0xFF kayıt sayı8
    end
    
    return key
end

function key_to_hex(key: CryptoKey) -> Metin do
    let hex = ""
    for byte in key.data do
        hex = hex + byte.onaltılıya(2)
    end
    return hex
end

function main() do
    rng.init()
    
    io.yazdir_satır("Generating AES-256 key...")
    let key = generate_aes256_key()
    io.yazdir_satır("Key: " + key_to_hex(key))
    
    // Use key for AES encryption
    crypto.aes_init(crypto.AES256, key.data)
end

सुरक्षा:सॉफ्टवेयर पीआरएनजी की तुलना में हार्डवेयर आरएनजी का उपयोग करना अधिक सुरक्षित है।

💡 उदाहरण 3: यादृच्छिक संख्या वितरण (पासा रोल)

समस्या: समान वितरण के साथ पासा रोल सिमुलेशन

import hal::rng

// Random in range [min, max]
function random_range(min: sayı, max: sayı) -> sayı do
    let range = max - min + 1
    let rand = rng.get_random()
    
    // Avoid modulo bias using rejection sampling
    let limit = 0xFFFF_FFFF - (0xFFFF_FFFF % range)
    
    for do
        if rand < limit do
            return min + (rand % range)
        end
        rand = rng.get_random()
    end
end

function dice_roll() -> sayı do
    return random_range(1, 6)
end

function test_distribution() do
    let histogram: Dizi<sayı> = Dizi.new(6, 0)
    const ROLLS: sayı = 10000
    
    for i in 0..ROLLS do
        let roll = dice_roll()
        histogram[roll - 1] = histogram[roll - 1] + 1
    end
    
    io.yazdir_satır("Dice roll distribution (" + ROLLS.metne() + " rolls):")
    for i in 0..6 do
        let face = i + 1
        let count = histogram[i]
        let percent = (count kayıt kesir / ROLLS kayıt kesir) * 100.0
        
        io.yazdir_satır("  [" + face.metne() + "]: " + count.metne() +
                        " (" + percent.metne() + "%)")
    end
end

function main() do
    rng.init()
    test_distribution()
end

निष्कर्ष:समान वितरण ~16.67% प्रति चेहरा (आदर्श: 16.666...%)।

⚠️ महत्वपूर्ण नोट्स:
  • घड़ी आवश्यक:आरएनजी के लिए आवश्यक 48 मेगाहर्ट्ज घड़ी (पीएलएल से)

  • मोडुलो पूर्वाग्रह:रेंज ऑपरेशंस में रिजेक्शन सैंपलिंग का उपयोग करें
