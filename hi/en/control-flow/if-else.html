यदि-अन्यथा शर्तें

सशर्त कथन प्रोग्राम प्रवाह को नियंत्रित करते हैं।

बुनियादी अगर

if x > 0 {
    print("Pozitif sayı")
}

अंग्रेजी वाक्यविन्यास:

if x > 0 {
    print("Positive number")
}

यदि-अन्यथा

if yaş >= 18 {
    print("Yetişkin")
} else {
    print("Çocuk")
}

वरना अगर

if not >= 90 {
    print("Pekiyi")
} else if not >= 80 {
    print("İyi")
} else if not >= 70 {
    print("Orta")
} else {
    print("Zayıf")
}

नेस्टेड इफ़

if kullanıcı != null {
    if kullanıcı.aktif {
        if kullanıcı.doğrulanmış {

        }
    }
}

टर्नरी ऑपरेटर

let case = yaş >= 18 ? "Yetişkin" : "Çocuk"

यदि अभिव्यक्ति

बर्क में यदि अभिव्यक्ति के रूप में उपयोग किया जा सकता है:

let maksimum = if a > b { a } else { b }

let mesaj = if hata {
    "Err oluştu"
} else {
    "Başarılı"
}

रक्षक उपवाक्य

शीघ्र वापसी पैटर्न:

function işle(değer: number?) -> Sonuç<number, string> {
    if değer == null {
        return Err "Değer null olamaz"
    }
    
    if değer < 0 {
        return Err "Değer negatif olamaz"
    }
    
    return Ok değer * 2
}

सर्वोत्तम प्रथाएं

✅ करने योग्य कार्य

// Anlamlı değişkenler
if kullanıcı_doğrulandı && oturum_geçerli {
    izin_ver()
}

// Guard clauses kullanın
if !geçerli_mi {
    return
}

// Pozitif koşullar tercih edin
if aktif {
    işle()
}

❌ परहेज करने योग्य बातें

// Çok uzun if-else zincirleri
if x == 1 {
    ...
} else if x == 2 {
    ...
} // 20 tane else if
// Bunun yerine match kullanın!

// Gereksiz iç içe if
if a {
    if b {
        if c {
            // 5 seviye
        }
    }
}
