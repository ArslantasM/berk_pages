पैटर्न मिलान

उन्नत मिलान और मूल्य विनाश।

मूल मिलान

variable sayı = 2

match sayı {
    1 => print("bir"),
    2 => print("iki"),
    3 => print("üç"),
    _ => print("diğer")
}

अंग्रेज़ी:

match number {
    1 => print("one"),
    2 => print("two"),
    _ => print("other")
}

मिलान अभिव्यक्ति

यह एक मान लौटा सकता है:

variable mesaj = match sayı {
    1 => "bir",
    2 => "iki",
    3 => "üç",
    _ => "bilinmeyen"
}

परिणाम मिलान

function split(a: number, b: number) -> Sonuç<number, string> {
    if b == 0 {
        hata return "Sıfıra bölme"
    }
    tamam return a / b
}

variable sonuç = split(10, 2)

match sonuç {
    Ok(değer) => print("Sonuç: ", değer),
    Err(mesaj) => print("HATA: ", mesaj)
}

सूची पैटर्न

variable liste = [1, 2, 3]

eş liste {
    [] => print("null liste"),
    [x] => print("Tek eleman: ", x),
    [x, y] => print("İki eleman: ", x, ", ", y),
    [ilk, ...geri_kalan] => {
        print("İlk: ", ilk)
        print("Geri kalan: ", geri_kalan)
    }
}

संरचना पैटर्न

type Kişi = struct {
    ad: string,
    yaş: number
}

variable kişi = Kişi { ad: "Ahmet", yaş: 30 }

match kişi {
    Kişi { ad: "Ahmet", yaş } => {
        print("Ahmet, ", yaş, " yaşında")
    },
    Kişi { ad, yaş: 18..25 } => {
        print(ad, " genç")
    },
    _ => print("Diğer")
}

नेस्टेड पैटर्न

type Mesaj = enum {
    Metin(string),
    number(number),
    Çoklu(Liste<Mesaj>)
}

variable mesaj = Çoklu([Metin("hello"), number(42)])

match mesaj {
    Metin(s) => print("Metin: ", s),
    number(n) => print("number: ", n),
    Çoklu([Metin(s), number(n)]) => {
        print("Metin and sayı: ", s, ", ", n)
    },
    _ => print("Diğer")
}

वाइल्डकार्ड पैटर्न

// Tam wildcard
match değer {
    _ => print("Herhangi bir şey")
}

// Kısmi wildcard
match tuple {
    (_, 0) => print("Y sıfır"),
    (0, _) => print("X sıfır"),
    (_, _) => print("Diğer")
}

सर्वोत्तम प्रथाएं

✅ करने योग्य कार्य

// Exhaustive matching (tüm durumlar)
match sonuç {
    Ok(x) => işle(x),
    Err(e) => hata_işle(e)
    // _ gerekli not, tüm durumlar kapsıyor
}

// Anlamlı variable isimleri
match kullanıcı {
    Kişi { ad: kullanıcı_adı, yaş: kullanıcı_yaşı } => {
        // ...
    }
}

❌ परहेज करने योग्य बातें

// Çok karmaşık iç içe pattern
match değer {
    A(B(C(D(x)))) => ...  // Okunabilir not
}

// Gereksiz wildcard
match x {
    _ => ...  // Tüm durumları yok sayıyor
}

उदाहरण

JSON पार्सर

type Json = enum {
    Null,
    Bool(Boolean),
    Number(number),
    String(string),
    Array(Liste<Json>),
    Object(Harita<string, Json>)
}

function stringify(json: Json) -> string {
    match json {
        Null => "null",
        Bool(b) => b ? "true" : "false",
        Number(n) => n.to_string(),
        String(s) => "\""+s+"\"",
        Array(arr) => "[" + arr.map(stringify).join(",") + "]",
        Object(obj) => "{...}"
    }
}
