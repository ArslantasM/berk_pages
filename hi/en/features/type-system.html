प्रकारप्रणाली

बर्क,मजबूत स्थैतिक टाइपिंगसुरक्षित और के साथअनुमान प्रकारयह एक व्यावहारिक भाषा प्रदान करता है. रस्ट और टाइपस्क्रिप्ट से प्रेरित एक आधुनिक प्रकार की प्रणाली।

अनुमान टाइप करें

बर्क कंपाइलर स्वचालित रूप से प्रकार निकाल सकता है:

// Tip belirtilmemiş ama derleyici çıkarıyor
let x = 42          // sayı
let y = 3.14        // kesir
let isim = "Ali"    // String
let List = [1,2,3] // Dizi<sayı>

संग्रह के प्रकार

सरणियों

// Sabit boyutlu Array
let sayılar: [sayı; 5] = [1, 2, 3, 4, 5]

// Dinamik Array
let List: Dizi<sayı> = Dizi::new()
List.push(42)

वैक्टर

let vec = Vec::from([1, 2, 3, 4])
vec.push(5)
print(vec.len())  // 5

हैश मानचित्र

use koleksiyonlar::HashMap;

let yaşlar = HashMap::new()
yaşlar.push("Ali", 25)
yaşlar.push("Ayşe", 30)

struct Nokta do

    y: kesir
end

// Kullanım
let p = Nokta { x: 1.0, y: 2.0 }

विकल्प और परिणाम प्रकार

पसंद(विकल्प)

function split(a: kesir, b: kesir) -> Seçenek<kesir> do
    if b == 0.0 do
        return Hiçbiri
    end
    return Bir(a / b)
end

निष्कर्ष(परिणाम)

function fs::read(yol: String) -> Sonuç<String, Err> do
    if yol == "" do
        return Err("Boş yol")
    end
    return Ok(içerik)
end

सामान्य प्रकार

struct Kutu<T> do
    değer: T
end

function yeni_kutu<T>(değer: T) -> Kutu<T> do
    return Kutu { değer: değer }
end

// Kullanım
let sayı_kutusu = yeni_kutu(42)
let metin_kutusu = yeni_kutu("merhaba")

विशेषता प्रणाली (इंटरफ़ेस)

trait Yazdırılabilir do
    function print(self)
end

struct Kişi do
    isim: String
end

uygula Yazdırılabilir for Kişi do
    function print(self) do
        print("Kişi: " + self.isim)
    end
end

रूपांतरण टाइप करें

कास्टिंग खोलें

let x: sayı = 42
let y: kesir = kayıt(kesir, x)
let z: sayı8 = kayıt(sayı8, x)

सुरक्षित रूपांतरण

let metin1 = "42"
let sayı1 = metin1.sayıya_çevir()  // Sonuç<sayı, Err>

eşleştir sayı1 do
    Ok(n) => print(n),
    Err(e) => print("Err: " + e)
end
