💡 नमूना उपयोग

// Türkçe syntax
function sum(a: tam, b: tam) -> tam {
    if a > 0 and b > 0 {
        döndür a + b
    } else {
        döndür 0
    }
}

// İngilizce syntax (aynı kod)
fn sum(a: int, b: int) -> int {
    if a > 0 and b > 0 {
        return a + b
    } else {
        return 0
    }
}

// Karışık kullanım da mümkün
let sonuç = sum(5, 10)

📝नोट
