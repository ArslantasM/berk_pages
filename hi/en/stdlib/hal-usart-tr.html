📡 राज्य::usart

यूनिवर्सल सिंक्रोनस/एसिंक्रोनस रिसीवर/ट्रांसमीटर

407 पंक्तियाँ ~20 कार्य यूएआरटी/यूएसएआरटी

📖अवलोकन

USART/UART मॉड्यूल एक सीरियल संचार प्रोटोकॉल है। इसका उपयोग जीपीएस, जीएसएम मॉड्यूल, सेंसर और पीसी के साथ संचार के लिए किया जाता है।

🔑 मुख्य विशेषताएं

  • एसिंक्रोनस (यूएआरटी) मोड
  • सिंक्रोनस (यूएसएआरटी) मोड
  • विन्यास योग्य बॉड दरें (1200-12Mbps)
  • 7/8/9 बिट डेटा
  • समता: कोई नहीं/सम/विषम
  • 1/2 स्टॉप बिट्स
  • हार्डवेयर प्रवाह नियंत्रण (आरटीएस/सीटीएस)
  • डीएमए समर्थन
  • व्यवधान-संचालित TX/RX

🚀 त्वरित शुरुआत

import hal::usart, hal::gpio

// GPIO setup for USART1
gpio.clock_enable(gpio.PORT_A)
gpio.pin_init(gpio.PORT_A, 9, do   // TX
    mode: gpio.MODE_AF,
    af: 7  // AF7 = USART1
end)
gpio.pin_init(gpio.PORT_A, 10, do  // RX
    mode: gpio.MODE_AF,
    pull: gpio.PULL_UP,
    af: 7
end)

// USART init
let uart = usart.init(usart.USART1, do
    baud_rate: 115200,
    data_bits: usart.DATA_8BIT,
    parity: usart.PARITY_NONE,
    stop_bits: usart.STOP_1BIT
end)

// Send data
usart.write(uart, "Hello World!\n")

// Receive data
let data = usart.read(uart, 100)  // Read up to 100 bytes

💡 उदाहरण 1: सीरियल कंसोल

import hal::usart, hal::gpio, hal::core

function uart_print(uart: usart.Handle, msg: yazı) do
    usart.write(uart, msg)
end

function uart_println(uart: usart.Handle, msg: yazı) do
    usart.write(uart, msg + "\r\n")
end

function main() do
    core.system_init()
    core.clock_config(72_000_000)
    
    // Setup USART1 GPIO (PA9=TX, PA10=RX)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 9, do
        mode: gpio.MODE_AF,
        speed: gpio.SPEED_HIGH,
        af: 7
    end)
    gpio.pin_init(gpio.PORT_A, 10, do
        mode: gpio.MODE_AF,
        pull: gpio.PULL_UP,
        af: 7
    end)
    
    // Initialize USART
    usart.clock_enable(usart.USART1)
    let uart = usart.init(usart.USART1, do
        baud_rate: 115200,
        data_bits: usart.DATA_8BIT,
        parity: usart.PARITY_NONE,
        stop_bits: usart.STOP_1BIT,
        flow_control: usart.FLOW_NONE
    end)
    
    uart_println(uart, "BERK v1.0.0 Console")
    uart_println(uart, "Type 'help' for commands")
    
    let buffer = ""
    
    for do
        // Check if data available
        if usart.available(uart) > 0 then do
            let ch = usart.read_byte(uart)
            
            if ch == '\r' or ch == '\n' then do
                if buffer.len() > 0 then do
                    uart_println(uart, "")
                    process_command(uart, buffer)
                    buffer = ""
                    uart_print(uart, "> ")
                end
            else if ch == '\b' then do  // Backspace
                if buffer.len() > 0 then do
                    buffer = buffer[0..buffer.len()-1]
                    uart_print(uart, "\b \b")  // Erase character
                end
            else do
                buffer += ch
                usart.write_byte(uart, ch)  // Echo
            end
        end
    end
end

function process_command(uart: usart.Handle, cmd: yazı) do
    if cmd == "help" then do
        uart_println(uart, "Available commands:")
        uart_println(uart, "  help    - Show this message")
        uart_println(uart, "  status  - System status")
        uart_println(uart, "  reset   - Reset system")
        uart_println(uart, "  info    - System info")
    else if cmd == "status" then do
        let uptime = core.uptime_ms() / 1000
        uart_println(uart, "System Status:")
        uart_println(uart, "  Uptime: " + uptime.yazıya() + "s")
        uart_println(uart, "  Clock: 72 MHz")
        uart_println(uart, "  Free RAM: 64 KB")
    else if cmd == "reset" then do
        uart_println(uart, "Resetting...")
        core.delay_ms(100)
        core.system_reset()
    else if cmd == "info" then do
        uart_println(uart, "BERK Embedded System")
        uart_println(uart, "Version: 1.0.0")
        uart_println(uart, "Platform: STM32F103")
    else do
        uart_println(uart, "Unknown command: " + cmd)
    end
end

💡 उदाहरण 2: जीपीएस पार्सर (एनएमईए)

import hal::usart, string

struct GPSData do
    latitude: kesir,
    longitude: kesir,
    altitude: kesir,
    speed: kesir,
    satellites: sayı,
    valid: bool
end

function parse_nmea(sentence: yazı) -> İsteğe_Bağlı[GPSData] do
    // Parse GPGGA sentence: $GPGGA,time,lat,N,lon,E,quality,sats,hdop,alt,M,...
    if sentence.starts_with("$GPGGA") else do
        return None
    end
    
    let parts = sentence.split(",")
    if parts.len() < 15 then do
        return None
    end
    
    let quality = parts[6].parse_int()?
    if quality == 0 then do  // Invalid fix
        return None
    end
    
    // Parse latitude (DDMM.MMMM format)
    let lat_str = parts[2]
    let lat_deg = lat_str[0..2].parse_int()?
    let lat_min = lat_str[2..].parse_float()?
    let latitude = lat_deg.kesir() + lat_min / 60.0
    if parts[3] == "S" then do
        latitude = -latitude
    end
    
    // Parse longitude (DDDMM.MMMM format)
    let lon_str = parts[4]
    let lon_deg = lon_str[0..3].parse_int()?
    let lon_min = lon_str[3..].parse_float()?
    let longitude = lon_deg.kesir() + lon_min / 60.0
    if parts[5] == "W" then do
        longitude = -longitude
    end
    
    return Some(GPSData do
        latitude: latitude,
        longitude: longitude,
        altitude: parts[9].parse_float()?,
        speed: 0.0,
        satellites: parts[7].parse_int()?,
        valid: true
    end)
end

function main() do
    // Setup USART for GPS (9600 baud, typical for GPS)
    let gps_uart = usart.init(usart.USART2, do
        baud_rate: 9600,
        data_bits: usart.DATA_8BIT,
        parity: usart.PARITY_NONE,
        stop_bits: usart.STOP_1BIT
    end)
    
    let buffer = ""
    
    for do
        if usart.available(gps_uart) > 0 then do
            let ch = usart.read_byte(gps_uart)
            
            if ch == '\n' then do
                // Process complete sentence
                if gps_data = parse_nmea(buffer) then do
                    io.println("GPS Fix:")
                    io.println("  Lat: {:.6f}".formatla(gps_data.latitude))
                    io.println("  Lon: {:.6f}".formatla(gps_data.longitude))
                    io.println("  Alt: {:.1f}m".formatla(gps_data.altitude))
                    io.println("  Sats: {}".formatla(gps_data.satellites))
                end
                buffer = ""
            else if ch != '\r' then do
                buffer += ch
            end
        end
        
        core.delay_ms(10)
    end
end

💡 उदाहरण 3: एटी कमांड इंटरफ़ेस (जीएसएम)

import hal::usart, hal::core

struct GSMModem do
    uart: usart.Handle
end

function send_at_command(modem: GSMModem, cmd: yazı, timeout_ms: sayı) -> Sonuç[yazı, Err] do
    // Send command
    usart.write(modem.uart, cmd + "\r\n")
    
    // Wait for response
    let response = ""
    let start_time = core.millis()
    
    for do
        if usart.available(modem.uart) > 0 then do
            let ch = usart.read_byte(modem.uart)
            response += ch
            
            // Check for termination
            if response.ends_with("OK\r\n") or response.ends_with("ERROR\r\n") then do
                return Ok(response)
            end
        end
        
        // Timeout check
        if core.millis() - start_time > timeout_ms then do
            return Err("AT command timeout")
        end
        
        core.delay_ms(10)
    end
end

function gsm_init() -> Sonuç[GSMModem, Err] do
    let uart = usart.init(usart.USART3, do
        baud_rate: 115200,
        data_bits: usart.DATA_8BIT,
        parity: usart.PARITY_NONE,
        stop_bits: usart.STOP_1BIT
    end)
    
    let modem = GSMModem do uart: uart end
    
    // Test communication
    send_at_command(modem, "AT", 1000)?
    
    // Disable echo
    send_at_command(modem, "ATE0", 1000)?
    
    // Check SIM card
    let sim_status = send_at_command(modem, "AT+CPIN?", 1000)?
    if sim_status.contains("READY") else do
        return Err("SIM card not ready")
    end
    
    return Ok(modem)
end

function send_sms(modem: GSMModem, phone: yazı, message: yazı) -> Sonuç[None, Err] do
    // Set text mode
    send_at_command(modem, "AT+CMGF=1", 1000)?
    
    // Set phone number
    send_at_command(modem, "AT+CMGS=\"" + phone + "\"", 1000)?
    
    // Send message
    usart.write(modem.uart, message)
    usart.write_byte(modem.uart, 0x1A)  // Ctrl+Z
    
    let response = send_at_command(modem, "", 10000)?
    if response.contains("OK") else do
        return Err("Failed to send SMS")
    end
    
    return Ok(None)
end

function main() do
    let modem = gsm_init()?
    
    io.println("GSM Modem ready!")
    
    // Send alert SMS
    send_sms(modem, "+905551234567", "BERK System Alert: Temperature exceeded 50°C")?
    
    io.println("SMS sent successfully!")
end

⚠️ महत्वपूर्ण नोट्स

  • वैकल्पिक फ़ंक्शन को सही करने के लिए TX/RX पिन सेट करें
  • बॉड दर की गणना घड़ी की आवृत्ति पर निर्भर करती है
  • लंबी केबलों पर कम बॉड दर का उपयोग करें
  • आरएक्स बफर ओवरफ़्लो से बचें (इंटरप्ट/डीएमए का उपयोग करें)
  • 3.3V-5V लेवल शिफ्टिंग की आवश्यकता हो सकती है

🔗संबंधित मॉड्यूल

  • hal::gpio- TX/RX पिन कॉन्फ़िगरेशन
  • hal::dma- डीएमए स्थानान्तरण
  • hal::int- यूएसएआरटी बीच में आता है
