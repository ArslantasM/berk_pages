import thread

// Thread spawn
let handle = thread.spawn(function() do
    print("Thread'from merhaba!")
    return 42
end)

let sonuç = thread.join(handle)
print("Sonuç: " + sonuç.yazıya())

// Mutex (mutual exclusion)
let count_occurrences = thread.mutex_new(0)

let threads = []
her i in range(0, 10) for do
    threads.push(thread.spawn(function() do
        her j in range(0, 100) for do
            thread.mutex_lock(count_occurrences, function(değer) do
                return değer + 1
            end)
        end
    end))
end

her t in threads for do
    thread.join(t)
end

let sum = thread.mutex_get(count_occurrences)
print("Toplam: " + sum.yazıya())  // 1000

// Channel (message passing)
let (tx, rx) = thread.channel_create()

thread.spawn(function() do
    thread.channel_send(tx, "Mesaj 1")
    thread.channel_send(tx, "Mesaj 2")
end)

print(thread.channel_receive(rx))  // "Mesaj 1"
print(thread.channel_receive(rx))  // "Mesaj 2"

📚सिंक करें

  • म्यूटेक्स:पारस्परिक बहिष्करण लॉक
  • RwLock:पढ़ें-लिखें लॉक (एकाधिक पाठक, एकल लेखक)
  • चैनल:संदेश भेजना (एमपीएससी, एमपीएमसी)
  • परमाणु:लॉक-मुक्त परमाणु संचालन
  • आर्क:परमाणु संदर्भ गिनती
  • रुकावट:सभी थ्रेड्स की प्रतीक्षा करें
  • सेमाफोर:संसाधन गिनती

import thread

function parallel_map(veri: Dizi, f: Fonksiyon, thread_sayısı: int) -> Dizi do
    let chunk_boyutu = (veri.len() + thread_sayısı - 1) / thread_sayısı
    let handles = []
    
    her i in range(0, thread_sayısı) for do
        let başlangıç = i * chunk_boyutu
        let bitiş = math.min(başlangıç + chunk_boyutu, veri.len())
        
        if başlangıç >= veri.len() then do
            devam
        end
        
        let chunk = veri.dilimle(başlangıç, bitiş)
        
        let handle = thread.spawn(function() do
            return chunk.map(f)
        end)
        
        handles.push(handle)
    end
    
    // Sonuçları join
    let sonuç = []
    her handle in handles for do
        let chunk_result = thread.join(handle)
        sonuç = sonuç + chunk_result
    end
    
    return sonuç
end

function parallel_reduce(veri: Dizi, f: Fonksiyon, başlangıç: herhangi) -> herhangi do
    let partial_results = parallel_map(veri, function(x) do return x end, 4)
    
    // Local reduce'lar
    let handles = []
    let chunk_boyutu = (partial_results.len() + 4 - 1) / 4
    
    her i in range(0, 4) for do
        let başlangıç_idx = i * chunk_boyutu
        let bitiş_idx = math.min(başlangıç_idx + chunk_boyutu, partial_results.len())
        
        if başlangıç_idx >= partial_results.len() then do
            devam
        end
        
        let chunk = partial_results.dilimle(başlangıç_idx, bitiş_idx)
        
        handles.push(thread.spawn(function() do
            let acc = başlangıç
            her x in chunk for do
                acc = f(acc, x)
            end
            return acc
        end))
    end
    
    // Final reduce
    let sonuç = başlangıç
    her handle in handles for do
        sonuç = f(sonuç, thread.join(handle))
    end
    
    return sonuç
end

// Kullanım: 1'from 1_000_000'a to topla
let sayılar = range(1, 1_000_001).collect()
let sum = parallel_reduce(sayılar, function(a, b) do return a + b end, 0)
print("Toplam: " + sum.yazıya())  // 500000500000
