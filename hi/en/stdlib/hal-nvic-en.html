नेस्टेड वेक्टर्ड इंटरप्ट कंट्रोलर - एआरएम कॉर्टेक्स-एम

~380 पंक्तियाँ ~18 कार्य एआरएम कॉर्टेक्स-एम

सिंहावलोकन

एनवीआईसी मॉड्यूल एआरएम कॉर्टेक्स-एम श्रृंखला प्रोसेसर पर इंटरप्ट प्रबंधन प्रदान करता है। यह 16 प्राथमिकता स्तरों के साथ 240+ इंटरप्ट स्रोतों का प्रबंधन करता है, नेस्टेड इंटरप्ट का समर्थन करता है और महत्वपूर्ण क्षेत्र सुरक्षा तंत्र प्रदान करता है। यह वास्तविक समय एम्बेडेड सिस्टम की रीढ़ है।

प्रमुख विशेषताऐं

  • 240+ व्यवधान:STM32F4 में 82 IRQ, F7 में 97, H7 में 150+ IRQ
  • 16 प्राथमिकता स्तर:4-बिट प्राथमिकता (0=उच्चतम)
  • निवारण:उच्च प्राथमिकता वाला व्यवधान निम्न प्राथमिकता में प्रवेश कर सकता है
  • उप-प्राथमिकता:कतार को समान प्राथमिकता के साथ क्रमबद्ध करना
  • मास्किंग:प्राइमास्क, फॉल्टमास्क, बेसप्री के साथ शटडाउन को बाधित करें
  • पूँछ जंजीर:व्यवधान संक्रमण के दौरान कोई स्थिति सहेजी/पुनर्स्थापित नहीं की गई
  • देरी से आगमन:यदि स्टैक प्रक्रिया में कोई उच्च प्राथमिकता वाला व्यवधान आता है, तो यह तुरंत चलता है।
  • वेक्टर तालिका:फ्लैश/रैम से स्थानांतरित करने योग्य इंटरप्ट वैक्टर

// Priority Grouping
enum PriorityGroup {
    PRIGROUP_0_4,  // 0 bit preempt, 4 bit sub
    PRIGROUP_1_3,  // 1 bit preempt, 3 bit sub
    PRIGROUP_2_2,  // 2 bit preempt, 2 bit sub
    PRIGROUP_3_1,  // 3 bit preempt, 1 bit sub
    PRIGROUP_4_0   // 4 bit preempt, 0 bit sub (default)
}

// IRQ Numbers (platform-specific)
enum IRQn {
    // Cortex-M core exceptions
    NMI_IRQn = -14,

    BusFault_IRQn = -11,
    UsageFault_IRQn = -10,
    SVCall_IRQn = -5,

    RTC_WKUP_IRQn = 3,

उदाहरण 1: प्राथमिकता समूहन और प्रीएम्प्शन

समस्या: USART और टाइमर का प्राथमिकता प्रबंधन बाधित होता है

import hal::nvic, hal::usart, hal::timer

function interrupt_sistemi_init() do
    // Set priority grouping: 3-bit preempt, 1-bit sub
    nvic.set_priority_grouping(nvic.PRIGROUP_3_1)
    
    // USART1 (high priority, preempt=1, sub=0)
    nvic.enable_irq(nvic.USART1_IRQn, do
        priority: 1,      // Preemption priority
        sub_priority: 0
    end)
    
    // Timer2 (medium priority, preempt=3, sub=0)
    nvic.enable_irq(nvic.TIM2_IRQn, do
        priority: 3,
        sub_priority: 0
    end)
    
    // ADC (low priority, preempt=5, sub=1)
    nvic.enable_irq(nvic.ADC_IRQn, do
        priority: 5,
        sub_priority: 1
    end)
end

// USART handler (can preempt TIM2 and ADC)
function USART1_IRQHandler() do
    if usart.get_flag(usart.USART1, usart.FLAG_RXNE) do
        let data = usart.receive_data(usart.USART1)
        buffer_yaz(data)
        usart.clear_flag(usart.USART1, usart.FLAG_RXNE)
    end
end

// Timer handler (can preempt ADC, but NOT USART)
function TIM2_IRQHandler() do
    if timer.get_flag(timer.TIM2, timer.FLAG_UPDATE) do
        // Timer tick processing
        systick_emulate()
        timer.clear_flag(timer.TIM2, timer.FLAG_UPDATE)
    end
end

// ADC handler (lowest priority, cannot preempt anyone)
function ADC_IRQHandler() do
    if adc.get_flag(adc.ADC1, adc.FLAG_EOC) do
        let value = adc.read_value(adc.ADC1)
        sensor_buffer[buffer_idx] = value
        buffer_idx = (buffer_idx + 1) % 32
        adc.clear_flag(adc.ADC1, adc.FLAG_EOC)
    end
end

आउटपुट:USART व्यवधान TIM2 को रोक सकता है, लेकिन TIM2 USART में प्रवेश नहीं कर सकता। एडीसी को दोनों द्वारा छूट दी जा सकती है।

उदाहरण 2: महत्वपूर्ण अनुभाग (सभी व्यवधान अक्षम करें)

समस्या: साझा संदेश की सुरक्षा (थ्रेड-सुरक्षित पहुंच)

import hal::nvic

let paylasilanOlan sayac: int = 0

function kritik_islem() do
    // Disable all interrupts (PRIMASK = 1)
    nvic.disable_all_interrupts()
    
    // Critical section (atomic operation)
    sayac = sayac + 1
    let local_sayac = sayac
    
    // Re-enable interrupts
    nvic.enable_all_interrupts()

end

    // Critical section

उदाहरण 3: वेक्टर तालिका स्थानांतरण (बूटलोडर)

समस्या: बूटलोडर से एप्लिकेशन पर स्विच करना

import hal::nvic, hal::flash

const BOOTLOADER_BASE: int = 0x0800_0000
const APP_BASE: int = 0x0800_8000  // 32 KB offset

function jump_to_application() do
    // Read application's stack pointer
    let app_sp = *(APP_BASE cast int*)
    
    // Read application's reset handler
    let app_reset = *((APP_BASE + 4) cast int*)
    
    io.yazdir_satır('Jumping to application at 0x' + APP_BASE.onaltılıya())
    
    // Disable all interrupts
    nvic.disable_all_interrupts()
    
    // Relocate vector table to application address
    nvic.set_vector_table(APP_BASE)
    
    // Set stack pointer
    asm do
        'MSR MSP, r0' // Set Main Stack Pointer
    end with app_sp
    
    // Jump to application reset handler
    let app_func = app_reset cast function()
    app_func()
end

// Bootloader mode check
function bootloader_mode() -> bool do
    // Check if user button pressed during reset
    gpio.pin_init(gpio.PORT_A, 0, do mode: gpio.MODE_INPUT end)
    
    let buton = gpio.pin_read(gpio.PORT_A, 0)
    
    if (buton == 1) do
        io.yazdir_satır('Bootloader mode (button pressed)')
        return true
    end
    
    // Check if application is valid (stack pointer in RAM)
    let app_sp = *(APP_BASE cast int*)
    if (app_sp >= 0x2000_0000 and app_sp < 0x2002_0000) do
        io.yazdir_satır('Valid application found')
        return false  // Jump to app
    end
    
    io.yazdir_satır('No valid application, staying in bootloader')
    return true
end

आउटपुट:बूटलोडर 0x0800_0000 से चलता है, एप्लिकेशन 0x0800_8000 पर लोड होता है। रीसेट एंड्रा ट्रू वेक्टर तालिका का उपयोग किया जाता है।

महत्वपूर्ण टिप्पणियाँ:
  • प्राथमिकता उलटा:यदि कम प्राथमिकता वाले व्यवधान में अधिक समय लगता है, तो उच्च प्राथमिकता प्रतीक्षा करती है
  • साझा परिधीय:समान परिधीय का उपयोग करने वाले इंटरप्ट की प्राथमिकता को सावधानीपूर्वक समायोजित किया जाना चाहिए।
  • स्टैक ओवरफ़्लो:यदि नेस्टेड व्यवधान बहुत गहरा हो जाता है, तो स्टैक ओवरफ़्लो हो सकता है
  • वेक्टर तालिका:RAM में स्थानांतरित करते समय संरेखण (0x200) आवश्यक है

सर्वोत्तम प्रथाएं

  • प्राथमिकता डिज़ाइन:महत्वपूर्ण ऑपरेशन 0-3, सामान्य 4-10, पृष्ठभूमि 11-15
  • शॉर्टहैंडलर:आईएसआर में न्यूनतम प्रसंस्करण, ध्वज सेट + मुख्य में प्रक्रिया
  • उप-प्राथमिकता का उपयोग करें:व्यवधानों को समान प्राथमिकता से क्रमबद्ध करें
  • लंबित स्पष्ट:हैंडलर एंड पेंडिंग बिट को साफ़ किया जाना चाहिए
  • फ्लोट का उपयोग करना:Cortex-M4F पर FPU आलसी स्टैकिंग महंगी है
  • लंबा महत्वपूर्ण क्षेत्र:disable_all_interpts() न्यूनतम समय
