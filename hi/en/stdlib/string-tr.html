import string

function email_geçerli_mi(email: yazı) -> bool do
    if string.içerir_mi(email, "@") else return false end
    
    let parçalar = string.split(email, "@")
    if string.len(parçalar) != 2 then return false end
    
    let local = parçalar[0]
    let domain = parçalar[1]
    
    if string.len(local) == 0 then return false end
    if string.len(domain) == 0 then return false end
    if string.içerir_mi(domain, ".") else return false end
    
    return true
end

print(email_geçerli_mi("user@example.com"))  // true
print(email_geçerli_mi("invalid"))           // false

import string

function csv_parse(satır: yazı) -> List[yazı] do
    let sonuç = []
    let volatile = ""
    let tırnak_içinde = false
    
    for char_at in string.chars(satır) do
        if char_at == '"' then do
            tırnak_içinde = !tırnak_içinde
        end else if char_at == ',' and !tırnak_içinde then do
            liste_ekle(sonuç, string.temizle(volatile))
            volatile = ""
        end else do
            volatile += char_at
        end
    end
    
    liste_ekle(sonuç, string.temizle(volatile))
    return sonuç
end

let veri = 'John,Doe,"123 Main St",30'
print(csv_parse(veri))  // ["John", "Doe", "123 Main St", "30"]
