💡 उदाहरण: STM32 LED ब्लिंक

import hal::core, hal::gpio, hal::timer

function ana() do
    // System initialization
    core.system_init()
    core.clock_config(72_000_000)  // 72MHz
    
    // GPIO setup
    gpio.clock_enable(gpio.PORT_C)
    variable led = gpio.pin_init(gpio.PORT_C, 13, do
        mode: gpio.MODE_OUTPUT,
        speed: gpio.SPEED_LOW,
        pull: gpio.PULL_NONE
    end)
    
    // Blink loop
    loop do
        gpio.pin_toggle(led)
        core.delay_ms(500)
    end
end

💡 उदाहरण: एडीसी रीडिंग

import hal::adc, hal::gpio, hal::core

function sıcaklık_oku() -> kesir do
    // ADC setup
    adc.clock_enable()
    variable adc1 = adc.init(adc.ADC1, do
        resolution: adc.RESOLUTION_12BIT,
        align: adc.ALIGN_RIGHT
    end)
    
    // GPIO for analog input
    variable pin = gpio.pin_init(gpio.PORT_A, 0, gpio.MODE_ANALOG)
    
    // Configure channel
    adc.channel_config(adc1, adc.CHANNEL_0, do
        sampling_time: adc.SAMPLETIME_480CYCLES
    end)
    
    // Read value
    adc.start(adc1)
    variable raw = adc.read(adc1)
    
    // Convert to temperature (example)
    variable voltage = raw.kesir() * 3.3 / 4095.0
    variable temp = (voltage - 0.76) / 0.0025 + 25.0
    
    return temp
end

function main() do
    core.system_init()
    
    loop do
        variable temp = sıcaklık_oku()
        io.println("Sıcaklık: " + temp.yazıya() + "°C")
        core.delay_ms(1000)
    end
end

💡 उदाहरण: I2C OLED डिस्प्ले

import hal::i2c, hal::gpio

constant OLED_ADDR = 0x3C

function oled_init(i2c: I2CHandle) do
    // Initialization sequence
    variable init_cmds = [
        0xAE,  // Display off
        0xD5, 0x80,  // Clock divide
        0xA8, 0x3F,  // Multiplex
        0xD3, 0x00,  // Display offset
        0x40,  // Start line
        0x8D, 0x14,  // Charge pump
        0x20, 0x00,  // Memory mode
        0xA1,  // Segment remap
        0xC8,  // COM scan direction
        0xDA, 0x12,  // COM pins
        0x81, 0xCF,  // Contrast
        0xD9, 0xF1,  // Pre-charge
        0xDB, 0x40,  // VCOMH deselect
        0xA4,  // Display resume
        0xA6,  // Normal display
        0xAF   // Display on
    ]
    
    her cmd in init_cmds for do
        i2c.write_byte(i2c, OLED_ADDR, 0x00, cmd)
    end
end

function main() do
    // I2C setup
    gpio.pin_init(gpio.PORT_B, 6, gpio.MODE_AF_OD)  // SCL
    gpio.pin_init(gpio.PORT_B, 7, gpio.MODE_AF_OD)  // SDA
    
    variable i2c1 = i2c.init(i2c.I2C1, do
        speed: i2c.SPEED_100KHZ,
        mode: i2c.MODE_I2C
    end)
    
    oled_init(i2c1)
    
    // Display text
    oled_print(i2c1, "BERK v1.0.0")
    oled_print(i2c1, "Embedded!")
end
