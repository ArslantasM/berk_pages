import yaml

// Parse YAML
let yaml_str = "
name: BERK
version: 1.0
features:
  - fast
  - safe
  - modern
"

let config = yaml.parse(yaml_str)
print(config["name"])  // "BERK"
print(config["features"][0])  // "fast"

// Dump to YAML
let veri = do
    "database": do
        "host": "localhost",
        "port": 5432
    end,
    "enabled": true
end

let yaml_output = yaml.dump(veri)
print(yaml_output)

import yaml, io

function config_load(dosya: yazı) -> Sonuç do
    try do
        let içerik = io.fs::read(dosya)
        let config = yaml.parse(içerik)
        return Ok(config)
    end catch hata do
        return Err("Config yüklenemedi: " + hata.mesaj())
    end
end

function config_save(dosya: yazı, config: herhangi) do
    let yaml_str = yaml.dump(config)
    io.dosya_yaz(dosya, yaml_str)
end

let cfg = config_load("app.yaml")
if cfg.tamam_mı() then do
    let config = cfg.pop()
    config["last_run"] = time.now().yazıya()
    config_save("app.yaml", config)
end
