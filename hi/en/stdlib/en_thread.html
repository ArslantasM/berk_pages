🚀 तेजी से शुरुआत करना

import thread

// Thread spawn
variable handle = thread.spawn(function() do
    print("Thread'from merhaba!")
    return 42
end)

variable sonuç = thread.join(handle)
print("Sonuç: " + sonuç.yazıya())

// Mutex (mutual exclusion)
variable count_occurrences = thread.mutex_new(0)

variable threads = []
her i in range(0, 10) for do
    threads.push(thread.spawn(function() do
        her j in range(0, 100) for do
            thread.mutex_lock(count_occurrences, function(değer) do
                return değer + 1
            end)
        end
    end))
end

her t in threads for do
    thread.join(t)
end

variable sum = thread.mutex_get(count_occurrences)
print("Toplam: " + sum.yazıya())  // 1000

// Channel (message passing)
variable (tx, rx) = thread.channel_create()

thread.spawn(function() do
    thread.channel_send(tx, "Mesaj 1")
    thread.channel_send(tx, "Mesaj 2")
end)

print(thread.channel_receive(rx))  // "Mesaj 1"
print(thread.channel_receive(rx))  // "Mesaj 2"

📚 तुल्यकालन

  • म्यूटेक्स:पारस्परिक बहिष्करण लॉक
  • RwLock:पढ़ें-लिखें लॉक (एकाधिक पाठक, एकल लेखक)
  • चैनल:संदेश भेजना (एमपीएससी, एमपीएमसी)
  • परमाणु:लॉक-मुक्त परमाणु संचालन
  • आर्क:परमाणु संदर्भ गिनती
  • रुकावट:सभी थ्रेड्स की प्रतीक्षा करें
  • सेमाफोर:संसाधन गिनती

निर्माता-उपभोक्ता पैटर्न

import thread, collections

variable (tx, rx) = thread.channel_create()

// Producers
her i in range(0, 3) for do
    variable sender = tx.clone()
    thread.spawn(function() do
        her j in range(0, 5) for do
            variable mesaj = "Producer " + i.yazıya() + " - Item " + j.yazıya()
            thread.channel_send(sender, mesaj)
            thread.sleep(100)  // 100ms
        end
    end)
end

// Consumer
variable consumer = thread.spawn(function() do
    variable items = []
    
    loop do
        variable sonuç = thread.channel_receive_timeout(rx, 1000)
        
        if sonuç.tamam_mı() then do
            variable mesaj = sonuç.pop()
            print("Tüketildi: " + mesaj)
            items.push(mesaj)
        end else do
            // Timeout - tüm producer'lar bitmiş
            break
        end
    end
    
    return items.len()
end)

variable sum = thread.join(consumer)
print("Toplam item: " + sum.yazıya())  // 15
