📚 सिंहावलोकन

mathमॉड्यूल बीईआरके की वैज्ञानिक कंप्यूटिंग क्षमताओं की नींव है। इसे जूलिया मैथ, न्यूमपी, साइपी और जीएनयू साइंटिफिक लाइब्रेरी (जीएसएल) मानकों से प्रेरित होकर डिजाइन किया गया था।

प्रेरणा के स्रोत

  • जूलिया:गणित + फास्टमैथ + स्पेशलफंक्शन
  • पायथन:NumPy / SciPy वैज्ञानिक कंप्यूटिंग
  • जंग:नाल्जेब्रा, एनडीरे, स्टैटर्स, आर्ग्मिन
  • ए:जीएनयू वैज्ञानिक पुस्तकालय (जीएसएल)
  • मानक:ब्लास/लैपैक/लिनपैक

विशेषताएँ

  • ✓ त्रिकोणमितीय फलन (रेडियन + डिग्री)
  • ✓ अतिशयोक्तिपूर्ण कार्य
  • ✓ घातीय और लघुगणकीय कार्य
  • ✓ विशेष कार्य (गामा, बीटा, बेसेल)
  • ✓ बहुपद मूल्यांकन (हॉर्नर विधि)
  • ✓ जड़ें ढूँढना (न्यूटन-राफसन, बिसेक्शन)
  • ✓ संख्यात्मक एकीकरण (सिम्पसन, गॉस)
  • ✓ इंटरपोलेशन (रैखिक, घन तख़्ता)
  • ✓ कॉम्प्लेक्स इंट सपोर्ट
  • ✓ एलएलवीएम तेज़-गणित अनुकूलन

import math

function main() do
    // Temel işlemler
    let x = math.sqrt(16.0)        // Karekök
    let y = math.pow(2.0, 8.0)     // Üs alma
    let z = math.abs(-42.0)        // Mutlak değer
    
    print("√16 = " + x.yazıya())       // √16 = 4
    print("2^8 = " + y.yazıya())       // 2^8 = 256
    print("|-42| = " + z.yazıya())     // |-42| = 42
    
    // Trigonometri (radyan)
    let sin_pi_2 = math.sin(math.PI / 2.0)
    print("sin(π/2) = " + sin_pi_2.yazıya())  // sin(π/2) = 1.0
    
    // Logaritma
    let ln_e = math.log(math.E)
    print("ln(e) = " + ln_e.yazıya())        // ln(e) = 1.0
end

1. बुनियादी अंकगणित

पेट(x: दशमलव) -> दशमलवabs_int(x: पूर्णांक) -> पूर्णांक

निरपेक्ष मान की गणना करता है. एलएलवीएमfabsआंतरिक उपयोगकर्ता.

संकेत(x: दशमलव) -> दशमलवसाइन_इंट(x: पूर्णांक) -> पूर्णांक

साइन फ़ंक्शन: -1 (नकारात्मक), 0 (शून्य), +1 (सकारात्मक)

अधिकतम(ए, बी: दशमलव) -> दशमलवमिन(ए, बी: दशमलव) -> दशमलव

अधिकतम और न्यूनतम मूल्य. NaN हैंडलिंग के साथ सुरक्षित।

क्लैंप(x, min_val, max_val: दशमलव) -> दशमलव

मान को [min_val, max_val] श्रेणी तक सीमित करता है।

उदाहरण: बुनियादी अंकगणित

import math

function main() do
    // Mutlak değer
    print(math.abs(-15.7))           // 15.7
    print(math.abs_int(-42))         // 42
    
    // İşaret
    print(math.sign(-5.5))           // -1.0
    print(math.sign(0.0))            // 0.0
    print(math.sign(3.14))           // 1.0
    
    // Max/Min
    print(math.max(10.5, 20.3))      // 20.3
    print(math.min(-5, 5))           // -5
    
    // Clamp (sınırlama)
    print(math.clamp(15, 0, 10))     // 10 (üst sınır)
    print(math.clamp(-5, 0, 10))     // 0 (alt sınır)
    print(math.clamp(5, 0, 10))      // 5 (range in)
end

2. घातांक और मूल फलन

पाउ(आधार, घातांक: दशमलव) -> दशमलवpow_int(आधार, प्रतिपादक: पूर्णांक) -> पूर्णांक

घातांक।pow_intद्विआधारी घातांक के साथ ओ(लॉग एन) गति।

sqrt(x: दशमलव) -> दशमलवसीबीआरटी(x: दशमलव) -> दशमलवnth_root(x: दशमलव, n: पूर्णांक) -> दशमलव

वर्गमूल, घनमूल और n-वाँ मूल। न्यूटन-रेफसन या एलएलवीएम आंतरिक।

वर्ग(x: दशमलव) -> दशमलवघनक्षेत्र(x: दशमलव) -> दशमलव

वर्ग और घन. विशेष कार्यों के लिए अनुकूलन.

हाइपोट(ए, बी: दशमलव) -> दशमलवहाइपोट3(ए, बी, सी: दशमलव) -> दशमलव

कर्ण: √(a² + b²). अतिप्रवाह-सुरक्षित कार्यान्वयन।

उदाहरण: घातांक और मूल

import math

function main() do
    // Üs alma
    print(math.pow(2.0, 10.0))           // 1024.0
    print(math.pow_int(2, 20))           // 1048576 (hızlı)
    
    // Kökler
    print(math.sqrt(144.0))              // 12.0
    print(math.cbrt(27.0))               // 3.0
    print(math.nth_root(256.0, 4))       // 4.0 (4. kök)
    
    // Kare and küp
    print(math.kare(5.0))                // 25.0
    print(math.küp(3.0))                 // 27.0
    
    // Pisagor teoremi
    let a = 3.0
    let b = 4.0
    let c = math.hypot(a, b)
    print("Hipotenüs: " + c.yazıya())    // 5.0
    
    // 3D mesafe
    let dist = math.hypot3(1.0, 2.0, 2.0)
    print("3D mesafe: " + dist.yazıya()) // 3.0
end

3. त्रिकोणमितीय फलन

पाप(x: दशमलव) -> दशमलवओल(x: दशमलव) -> दशमलवभोर(x: दशमलव) -> दशमलव

ज्या, कोज्या, स्पर्शज्या. इनपुट रेडियन में है.

सिंध(x: दशमलव) -> दशमलवकॉसडी(x: दशमलव) -> दशमलवtand(x: दशमलव) -> दशमलव

डिग्री में त्रिकोणमितीय कार्य.

के रूप में(x: दशमलव) -> दशमलवएकोस(x: दशमलव) -> दशमलवपिटाई(x: दशमलव) -> दशमलवatan2(y, x: दशमलव) -> दशमलव

व्युत्क्रम त्रिकोणमितीय फलन.atan2सभी डायल का समर्थन करता है.

उदाहरण: त्रिकोणमिति

import math

function main() do
    // Radyan cinsinden
    print(math.sin(0.0))                    // 0.0
    print(math.cos(0.0))                    // 1.0
    print(math.tan(math.PI / 4.0))          // 1.0
    
    // Derece cinsinden (daha sezgisel)
    print(math.sind(0.0))                   // 0.0
    print(math.sind(30.0))                  // 0.5
    print(math.sind(90.0))                  // 1.0
    print(math.cosd(60.0))                  // 0.5
    print(math.tand(45.0))                  // 1.0
    
    // Ters functionlar
    let açı = math.asin(0.5)
    print("arcsin(0.5) = " + açı.yazıya()) // π/6 radyan
    
    // atan2 ile açı hesaplama (x,y koordinatından)
    let açı2 = math.atan2(1.0, 1.0)
    print("atan2(1,1) = " + açı2.yazıya()) // π/4 (45°)
end

4. लघुगणक और घातांकीय फलन

ऍक्स्प(x: दशमलव) -> दशमलवexp2(x: दशमलव) -> दशमलवexp10(x: दशमलव) -> दशमलव

घातीय फलन: e^x, 2^x, 10^x. एलएलवीएम आंतरिक के साथ अनुकूलित।

लकड़ी का लट्ठा(x: दशमलव) -> दशमलवलॉग2(x: दशमलव) -> दशमलवलॉग10(x: दशमलव) -> दशमलव

लघुगणक: ln(x), log₂(x), log₁₀(x)।

expm1(x: दशमलव) -> दशमलवलॉग1पी(x: दशमलव) -> दशमलव

परिशुद्धता के लिए छोटे मान: e^x-1, ln(1+x).

उदाहरण: लघुगणक और घातांक

import math

function main() do
    // Üstel functionlar
    print(math.exp(1.0))              // e¹ = 2.71828...
    print(math.exp2(10.0))            // 2¹⁰ = 1024
    print(math.exp10(3.0))            // 10³ = 1000
    
    // Logaritma
    print(math.log(math.E))           // ln(e) = 1.0
    print(math.log2(1024.0))          // log₂(1024) = 10.0
    print(math.log10(1000.0))         // log₁₀(1000) = 3.0
    
    // Küçük değerlerde hassasiyet
    let küçük = 1e-10
    print(math.log1p(küçük))          // ln(1 + 10⁻¹⁰)
    print(math.expm1(küçük))          // e^(10⁻¹⁰) - 1
    
    // pH hesaplama örneği
    let H_konsantrasyon = 1e-7    // mol/L
    let pH = -math.log10(H_konsantrasyon)
    print("pH = " + pH.yazıya())      // pH = 7.0 (nötr)
end

🎓उन्नत उदाहरण और ट्यूटोरियल

ट्यूटोरियल 1: संख्यात्मक व्युत्पन्न गणना

किसी फ़ंक्शन के संख्यात्मक व्युत्पन्न (केंद्रीय अंतर विधि) की गणना करें।

import math

// f'(x) ≈ [f(x+h) - f(x-h)] / (2h)
function sayısal_türev(f: function(float)->float, x: float, h: float) -> float do
    let fx_plus_h = f(x + h)
    let fx_minus_h = f(x - h)
    return (fx_plus_h - fx_minus_h) / (2.0 * h)
end

// Test functionu: f(x) = x²
function f(x: float) -> float do
    return x * x
end

function main() do
    // f'(x) = 2x for x=3 noktasında türev
    let x = 3.0
    let h = 1e-5           // Küçük adım
    let türev = sayısal_türev(f, x, h)
    
    print("f'(3) sayısal = " + türev.yazıya())  // ~6.0
    print("f'(3) analitik = " + (2.0 * x).yazıya())  // 6.0
    
    // Error analizi
    let Error = math.abs(türev - 2.0 * x)
    print("Error: " + Error.yazıya())  // ~10⁻¹⁰
end
f'(3) संख्यात्मक = 6.0000001 एफ'(3) विश्लेषण = 6.0 त्रुटि: 0.0000001

ट्यूटोरियल 2: न्यूटन-रेफसन जड़ें ढूँढना

समीकरण का मूल ज्ञात कीजिए: f(x) = x² - 2 = 0 (अर्थात् √2)

import math

// Newton-Raphson: x_{n+1} = x_n - f(x_n) / f'(x_n)
function newton_raphson(
    f: function(float)->float,
    df: function(float)->float,
    x0: float,
    tolerans: float,
    max_iter: int
) -> float do
    let x = x0
    
    for i = 0; i < max_iter; i += 1 do
        let fx = f(x)
        let dfx = df(x)
        
        if math.abs(fx) < tolerans  do
            print("Yakınsadı: " + i.yazıya() + " iterasyon")
            return x
        end
        
        x = x - fx / dfx
    end
    
    print("Maksimum iterasyon aşıldı!")
    return x
end

// f(x) = x² - 2
function f(x: float) -> float do
    return x * x - 2.0
end

// f'(x) = 2x
function df(x: float) -> float do
    return 2.0 * x
end

function main() do
    let kök = newton_raphson(f, df, 1.0, 1e-10, 100)
    
    print("Bulunan kök: " + kök.yazıya())
    print("√2 (referans): " + math.SQRT2.yazıya())
    print("Error: " + math.abs(kök - math.SQRT2).yazıya())
end
एकत्रित: 4 पुनरावृत्तियाँ मिला रूट: 1.414213562373095 √2 (संदर्भ): 1.414213562373095 त्रुटि: 0.0000000000000

ट्यूटोरियल 3: मोंटे कार्लो पाई गणना

यादृच्छिक संख्याओं का उपयोग करके π के मान का अनुमान लगाएं।

import math, random

function monte_carlo_pi(n: int) -> float do
    let içeride = 0
    
    for i = 0; i < n; i += 1 do
        // [-1, 1] aralığında random nokta
        let x = random.uniform(-1.0, 1.0)
        let y = random.uniform(-1.0, 1.0)
        
        // Birim çember in mi?
        if x*x + y*y <= 1.0  do
            içeride += 1
        end
    end
    
    // π ≈ 4 * (içerideki noktalar / sum noktalar)
    return 4.0 * içeride.float() / n.float()
end

function main() do
    // Farklı Examplelem boyutları
    let boyutlar = [100, 1000, 10000, 100000, 1000000]
    
    print("Gerçek π = " + math.PI.yazıya())
    print("\nMonte Carlo Tahminleri:")
    print("------------------------")
    
    for len in boyutlar do
        let tahmin = monte_carlo_pi(len)
        let Error = math.abs(tahmin - math.PI)
        let Error_yüzde = (Error / math.PI) * 100.0
        
        print("n=" + len.yazıya() + ": π≈" + tahmin.yazıya() + 
               " (Error: %" + Error_yüzde.yazıya() + ")")
    end
end
वास्तविक π = 3.141592653589793 मोंटे कार्लो भविष्यवाणियाँ: ---------------------- n=100: π≈3.16 (त्रुटि: 0.59%) n=1000: π≈3.148 (त्रुटि: 0.20%) n=10000: π≈3.1432 (त्रुटि: 0.05%) n=100000: π≈3.14204 (त्रुटि: 0.014%) n=1000000: π≈3.141456 (त्रुटि: 0.004%)

ट्यूटोरियल 4: फूरियर श्रृंखला सन्निकटन

फूरियर श्रृंखला के साथ वर्गाकार तरंग फलन की लगभग गणना करें।

import math

// Kare dalga Fourier serisi: f(x) = (4/π) Σ [sin((2k-1)x) / (2k-1)]
function fourier_kare_dalga(x: float, terimler: int) -> float do
    let sum = 0.0
    
    for k = 1; k <= terimler; k += 1 do
        let n = 2 * k - 1
        let terim = math.sin(n.float() * x) / n.float()
        sum += terim
    end
    
    return (4.0 / math.PI) * sum
end

function main() do
    print("Fourier Kare Dalga Yaklaşımı")
    print("================================\n")
    
    // x = π/4 noktasında (yükselen kenar)
    let x = math.PI / 4.0
    
    for n_terim = 1; n_terim <= 50; n_terim *= 5 do
        let Result = fourier_kare_dalga(x, n_terim)
        print(n_terim.yazıya() + " terim: f(π/4) = " + Result.yazıya())
    end
    
    // Görselleştirme (basit ASCII)
    print("\nGrafik (0 ile 2π arası, 10 terim):")
    print("-----------------------------------")
    
    for i = 0; i < 40; i += 1 do
        let x_nokta = (i.float() / 40.0) * 2.0 * math.PI
        let y = fourier_kare_dalga(x_nokta, 10)
        
        // Y değerini karaktere dönüştür (basitleştirilmiş)
        let seviye = ((y + 1.0) / 2.0 * 20.0).int()
        let çizgi = ""
        
        for j = 0; j < seviye; j += 1 do
            çizgi += " "
        end
        çizgi += "*"
        
        print(çizgi)
    end
end
