import gui

// Uygulama oluştur
let app = gui.app_create("Merhaba BERK")

// Window
let pencere = gui.window_create("İlk GUI Uygulamam", 400, 300)

// Widgets
let label = gui.label_create("Merhaba Dünya!")
let button = gui.button_create("Tıkla Beni")
let text_field = gui.textfield_create()

// Event handler
gui.button_on_click(button, function() do
    let String = gui.textfield_get_text(text_field)
    gui.label_set_text(label, "Merhaba " + String + "!")
end)

// Layout
let vbox = gui.vbox_create()
gui.vbox_add(vbox, label)
gui.vbox_add(vbox, text_field)
gui.vbox_add(vbox, button)

gui.window_set_content(pencere, vbox)
gui.window_show(pencere)
gui.app_run(app)

import gui, collections

let app = gui.app_create("Todo App")
let pencere = gui.window_create("Yapılacaklar", 600, 400)

// Veri modeli
let todos = collections.vec_new()

// UI elemanları
let text_field = gui.textfield_create()
gui.textfield_set_placeholder(text_field, "Yeni task...")

let add_button = gui.button_create("Ekle")
let todo_list = gui.vbox_create()

// "Ekle" butonu handler
gui.button_on_click(add_button, function() do
    let text = gui.textfield_get_text(text_field)
    
    if text.len() > 0 then do
        // Todo item oluştur
        let item_box = gui.hbox_create()
        
        let checkbox = gui.checkbox_create(text)
        let delete_btn = gui.button_create("Sil")
        
        // Checkbox handler
        gui.checkbox_on_toggle(checkbox, function(checked) do
            if checked then do
                gui.checkbox_set_style(checkbox, "text-decoration: line-through")
            end else do
                gui.checkbox_set_style(checkbox, "")
            end
        end)
        
        // Delete handler
        gui.button_on_click(delete_btn, function() do
            gui.vbox_remove(todo_list, item_box)
        end)
        
        gui.hbox_add(item_box, checkbox)
        gui.hbox_add(item_box, delete_btn)
        gui.vbox_add(todo_list, item_box)
        
        gui.textfield_set_text(text_field, "")
    end
end)

// Layout
let input_box = gui.hbox_create()
gui.hbox_add(input_box, text_field)
gui.hbox_add(input_box, add_button)

let main_box = gui.vbox_create()
gui.vbox_add(main_box, input_box)
gui.vbox_add(main_box, gui.separator_create())
gui.vbox_add(main_box, todo_list)

gui.window_set_content(pencere, main_box)
gui.window_show(pencere)
gui.app_run(app)

import gui, image, fs

let app = gui.app_create("Image Viewer")
let pencere = gui.window_create("Resim Görüntüleyici", 800, 600)

// UI
let image_widget = gui.image_create()
let open_button = gui.button_create("Resim Aç")
let prev_button = gui.button_create("◀ Önceki")
let next_button = gui.button_create("Sonraki ▶")

let images = []
let current_index = 0

function load_image(path: yazı) do
    let img = image.load(path)
    gui.image_set_data(image_widget, img)
    gui.window_set_title(pencere, "Image Viewer - " + path)
end

// Dosya seçici
gui.button_on_click(open_button, function() do
    let dialog = gui.filedialog_create()
    gui.filedialog_set_filters(dialog, ["*.png", "*.jpg", "*.jpeg"])
    
    let seçilen = gui.filedialog_show(dialog)
    if seçilen then do
        let dizin = fs.path_dirname(seçilen)
        images = fs.dizin_listele(dizin)
        images = images.filtrele(function(f) do
            return f.bitiyor_mu(".png") or f.bitiyor_mu(".jpg")
        end)
        
        current_index = images.index_of(fs.path_basename(seçilen))
        load_image(seçilen)
    end
end)

// Navigasyon
gui.button_on_click(prev_button, function() do
    if current_index > 0 then do
        current_index = current_index - 1
        load_image(images[current_index])
    end
end)

gui.button_on_click(next_button, function() do
    if current_index < images.len() - 1 then do
        current_index = current_index + 1
        load_image(images[current_index])
    end
end)

// Layout
let button_box = gui.hbox_create()
gui.hbox_add(button_box, open_button)
gui.hbox_add(button_box, prev_button)
gui.hbox_add(button_box, next_button)

let main_box = gui.vbox_create()
gui.vbox_add(main_box, button_box)
gui.vbox_add(main_box, image_widget)

gui.window_set_content(pencere, main_box)
gui.window_show(pencere)
gui.app_run(app)
