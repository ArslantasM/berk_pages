//  [FN, TN]]

    let precision_i = cm[i][i] / cm[i].topla();

end

    कस्टम मेट्रिक्स

    अपनी खुद की मीट्रिक बनाएं

    use ai/metrics;

function custom_accuracy(y_true: Tensor, y_pred: Tensor) -> float do
    let threshold = 0.7;
    let high_conf_mask = y_pred.gt(threshold);
    let correct = y_true.eq(y_pred.round()).and(high_conf_mask);
    
    return correct.topla() as float / y_true.len() as float;
end

// Kullanım
let acc = custom_accuracy(y_true, y_pred);
print("High-confidence accuracy: {:.2f}%", acc * 100.0);

    बैच मूल्यांकन

    डेटालोडर के साथ बैच मूल्यांकन

    use ai/metrics;
use ai/train;

function evaluate(model: Model, test_loader: DataLoader) -> Metrics do
    model.değerlendirme();
    
    let tüm_preds = [];
    let tüm_targets = [];
    
    her (x, y) in test_loader for do
        let pred = model.ileri(x);
        tüm_preds.extend(pred.argmax(dim: 1));
        tüm_targets.extend(y);
    end
    
    return Metrics {
        accuracy: accuracy(tüm_targets, tüm_preds),
        f1: f1_score(tüm_targets, tüm_preds, average: "weighted"),
        precision: precision_score(tüm_targets, tüm_preds, average: "weighted"),
        recall: recall_score(tüm_targets, tüm_preds, average: "weighted")
    };
end

    मीट्रिकसमारोहउपयोग
    शुद्धताaccuracy(y_true, y_pred)समग्र शुद्धता
    शुद्धताprecision_score(y_true, y_pred)सकारात्मक भविष्यवाणी सटीकता
    याद करनाrecall_score(y_true, y_pred)सकारात्मक कैप्चर दर
    एफ1 स्कोरf1_score(y_true, y_pred)परिशुद्धता-स्मरण संतुलन
    आरओसी-एयूसीroc_auc_score(y_true, y_score)वर्गीकरण गुणवत्ता
    एमएसईmean_squared_error(y_true, y_pred)प्रतिगमन त्रुटि
    मॅईmean_absolute_error(y_true, y_pred)पूर्ण माध्य त्रुटि
    आर² स्कोरr2_score(y_true, y_pred)समझाया गया विचरण
