🧠 राज्य::कोर

सिस्टम कोर - घड़ी, रीसेट, पावर प्रबंधन

367 पंक्तियाँ ~15 कार्य आरसीसी

📖अवलोकन

कोर मॉड्यूल सिस्टम इनिशियलाइज़ेशन, क्लॉक कॉन्फ़िगरेशन, रीसेट और पावर प्रबंधन फ़ंक्शन प्रदान करता है।

🔑 मुख्य विशेषताएं

  • सिस्टम आरंभीकरण
  • घड़ी विन्यास (168 मेगाहर्ट्ज तक)
  • विलंब कार्य (एमएस/यूएस)
  • सिस्टम रीसेट
  • पावर मोड (स्लीप/स्टॉप/स्टैंडबाय)
  • अपटाइम काउंटर

import hal::core

function ana() do
    // System initialization
    core.system_init()
    
    // Configure system clock to 72MHz
    core.clock_config(72_000_000)
    
    // Delays
    core.delay_ms(1000)  // 1 second
    core.delay_us(500)   // 500 microseconds
    
    // Get uptime
    variable uptime = core.uptime_ms()
    io.println("Uptime: {}ms", uptime)
    
    // System reset
    core.system_reset()
end

💡 उदाहरण: सिस्टम क्लॉक कॉन्फ़िगरेशन (STM32F4 168MHz)

import hal::core

function configure_168mhz() do
    // Configure PLL for 168MHz from 8MHz HSE
    core.pll_config(do
        source: core.PLL_HSE,        // External crystal
        pllm: 8,                     // VCO input: 8MHz / 8 = 1MHz
        plln: 336,                   // VCO output: 1MHz * 336 = 336MHz
        pllp: core.PLLP_DIV2,        // System clock: 336MHz / 2 = 168MHz
        pllq: 7                      // USB clock: 336MHz / 7 = 48MHz
    end)
    
    // Configure bus clocks
    core.bus_config(do
        ahb_prescaler: core.AHB_DIV1,    // AHB = 168MHz
        apb1_prescaler: core.APB_DIV4,   // APB1 = 42MHz (max 42MHz)
        apb2_prescaler: core.APB_DIV2    // APB2 = 84MHz (max 84MHz)
    end)
    
    // Flash latency for 168MHz @ 3.3V
    core.flash_latency(5)  // 5 wait states
    
    // Switch to PLL
    core.system_clock_source(core.SYSCLK_PLL)
    
    // Wait for PLL ready
    loop core.get_clock_source() != core.SYSCLK_PLL else do
        // Wait
    end
    
    io.println("System clock: {}MHz", core.get_sysclk_freq() / 1_000_000)
end

function ana() do
    configure_168mhz()
    
    // Verify clocks
    io.println("SYSCLK: {}MHz", core.get_sysclk_freq() / 1_000_000)
    io.println("AHB: {}MHz", core.get_hclk_freq() / 1_000_000)
    io.println("APB1: {}MHz", core.get_pclk1_freq() / 1_000_000)
    io.println("APB2: {}MHz", core.get_pclk2_freq() / 1_000_000)
end

💡 उदाहरण: लो-पावर स्लीप मोड

import hal::core, hal::int, hal::gpio

variable wakeup_flag = false

function button_interrupt() do
    wakeup_flag = true
end

function sleep_until_button() do
    // Configure button interrupt (PA0)
    gpio.clock_enable(gpio.PORT_A)
    gpio.pin_init(gpio.PORT_A, 0, do
        mode: gpio.MODE_INPUT,
        pull: gpio.PULL_DOWN,
        interrupt: gpio.INT_RISING
    end)
    
    gpio.set_callback(gpio.PORT_A, 0, button_interrupt)
    
    io.println("Going to sleep... (press button to wake)")
    core.delay_ms(100)  // Flush UART
    
    // Enter sleep mode
    core.sleep()
    
    // Wakeup on any interrupt
    io.println("Woke up from sleep!")
end

function ana() do
    loop do
        sleep_until_button()
        core.delay_ms(1000)
    end
end

💡 उदाहरण: वॉचडॉग टाइमर (IWDG)

import hal::core

function watchdog_test() do
    // Initialize Independent Watchdog (4 second timeout)
    core.iwdg_init(do
        prescaler: core.IWDG_PRESCALER_64,  // LSI/64
        reload: 2000  // (32kHz/64) * 2000 = 4 seconds
    end)
    
    io.println("Watchdog started (4s timeout)")
    
    // Normal operation: feed watchdog regularly
    her i in 0..10 for do
        io.println("Working... {}", i)
        core.delay_ms(500)
        
        core.iwdg_refresh()  // "Feed" watchdog
    end
    
    // Simulate hang (return't feed watchdog)
    io.println("Simulating hang...")
    loop do
        core.delay_ms(1000)
        // Watchdog will reset system after 4s
    end
end

function main() do
    // Check if last reset was watchdog
    if core.get_reset_reason() == core.RESET_IWDG then do
        io.println("⚠️ System reset by watchdog!")
        core.clear_reset_flags()
    end
    
    watchdog_test()
end

💡 उदाहरण: ब्राउनआउट डिटेक्शन और रीसेट हैंडलिंग

import hal::core

enum ResetReason {
    POWER_ON,
    RESET_PIN,
    IWDG,
    WWDG,
    SOFTWARE,
    BROWNOUT,
    UNKNOWN
}

function get_reset_reason_detailed() -> ResetReason do
    variable flags = core.get_reset_flags()
    
    variable reason = eşle flags do
        _ if (flags & core.FLAG_LPWR) != 0 => ResetReason::BROWNOUT,
        _ if (flags & core.FLAG_WWDG) != 0 => ResetReason::WWDG,
        _ if (flags & core.FLAG_IWDG) != 0 => ResetReason::IWDG,
        _ if (flags & core.FLAG_SFT) != 0 => ResetReason::SOFTWARE,
        _ if (flags & core.FLAG_PIN) != 0 => ResetReason::RESET_PIN,
        _ if (flags & core.FLAG_POR) != 0 => ResetReason::POWER_ON,
        _ => ResetReason::UNKNOWN
    end
    
    core.clear_reset_flags()
    return reason
end

function reset_handler() do
    variable reason = get_reset_reason_detailed()
    
    io.println("\n=== System Boot ===")
    io.println("Reset reason: {}", reason)
    
    eşle reason do
        ResetReason::POWER_ON => do
            io.println("Fresh power-on, initializing...")
            // First boot initialization
        end,
        
        ResetReason::BROWNOUT => do
            io.println("⚠️ Brownout detected! Check power supply.")
            // Save diagnostics
        end,
        
        ResetReason::IWDG => do
            io.println("⚠️ Watchdog reset! System hang detected.")
            // Crash recovery
        end,
        
        ResetReason::SOFTWARE => do
            io.println("Software reset (firmware update?)")
        end,
        
        _ => do
            io.println("Unexpected reset")
        end
    end
end

function main() do
    reset_handler()
    
    // Normal program
    loop do
        io.println("Running...")
        core.delay_ms(1000)
    end
end

💡 उदाहरण: सीपीयू लोड मॉनिटरिंग

import hal::core

variable idle_counter = 0
variable total_counter = 0

function calculate_cpu_load() -> kesir do
    variable idle = idle_counter
    variable total = total_counter
    
    // Reset counters
    idle_counter = 0
    total_counter = 0
    
    variable cpu_load = 100.0 - (idle.kesir() / total.kesir() * 100.0)
    return cpu_load
end

function idle_task() do
    // Count idle cycles
    idle_counter += 1
    total_counter += 1
end

function work_task() do
    // Simulate work
    variable sum = 0
    her i in 0..1000 for do
        sum += i
    end
    total_counter += 1000
end

function main() do
    loop do
        // Do some work
        work_task()
        
        // Idle time
        idle_task()
        
        // Report every second
        if core.uptime_ms() % 1000 == 0 then do
            variable cpu_load = calculate_cpu_load()
            io.println("CPU load: {:.1f}%", cpu_load)
        end
    end
end

📊 पावर मोड तुलना

तरीका शक्ति जागने का समय वेकअप स्रोत
दौड़ना ~30एमए - -
नींद ~15एमए तुरंत कोई रुकावट
रुकना ~10μA ~5μs EXTI, RTC, वॉचडॉग
समर्थन करना ~1μA ~50μs WKUP पिन, RTC अलार्म

⚡ प्रदर्शन युक्तियाँ

  • फ़्लैश विलंबता: प्रतीक्षा स्थिति के लिए उच्च आवृत्ति जोड़ें (168 मेगाहर्ट्ज = 5 डब्ल्यूएस)
  • पीएलएल कॉन्फिग: VCO को 192-432MHz रेंज में रखें (इष्टतम)
  • एपीबी1: अधिकतम 42 मेगाहर्ट्ज (STM32F4), टाइमर घड़ी x2
  • स्लीप मोड: त्वरित बिजली बचत, तुरंत वेकअप
  • देखो कुत्ता: (हैंग रिकवरी) के लिए सिस्टम विश्वसनीयता का उपयोग करें

⚠️ महत्वपूर्ण नोट्स

  • फ़्लैश विलंबता: उच्च आवृत्ति स्वचालित रूप से समायोजित हो जाती है (168MHz @ 3.3V = 5 WS)
  • पीएलएल लॉक टाइम: ~200μs तक PLL स्थिरीकरण की प्रतीक्षा करें
  • आधार रीति: बाहर निकलने पर सिस्टम रीसेट होता है, बैकअप डोमेन संरक्षित रहता है
  • वॉचडॉग प्रारंभ: IWDG को एक बार शुरू करने के बाद रोका नहीं जा सकता
  • एलएसई स्टार्टअप: ~1s स्टार्टअप समय के लिए बाहरी 32kHz क्रिस्टल
  • एचएसई बाईपास: क्रिस्टल के लिए बाहरी घड़ी स्रोत के बजाय

🔗संबंधित मॉड्यूल

  • hal::rtc- आरटीसी घड़ी विन्यास, वेकअप टाइमर
  • hal::int- प्राथमिकताओं में बाधा, स्लीप मोड वेकअप
  • hal::gpio- WKUP पिन कॉन्फ़िगरेशन
