import async, http

// async function tanımı
async function veri_getir(url: yazı) -> Sonuç do
    let yanıt = await http.get(url)
    if yanıt.case == 200 then do
        return Ok(yanıt.gövde)
    end else do
        return Err("HTTP " + yanıt.case.yazıya())
    end
end

// Paralel çalıştırma
async function paralel_istekler() do
    let görevler = [
        veri_getir("https://api.example.com/users"),
        veri_getir("https://api.example.com/posts"),
        veri_getir("https://api.example.com/comments")
    ]
    
    let sonuçlar = await async.all(görevler)
    
    her sonuç in sonuçlar for do
        print("Veri alındı: " + sonuç.len().yazıya() + " byte")
    end
end

// Ana async runtime
async.run(paralel_istekler())

import async, http, json

async function handle_request(istek: HttpIstek) -> HttpYanit do
    if istek.yol == "/api/users" then do
        // Simulate DB query
        await async.sleep(100)
        
        let users = [
            {"id": 1, "name": "Ali"},
            {"id": 2, "name": "Ayşe"}
        ]
        
        return http.yanıt(200, json.stringify(users))
    end else do
        return http.yanıt(404, "Not Found")
    end
end

async function sunucu_başlat() do
    let sunucu = http.sunucu("127.0.0.1:8080")
    print("Server listening on http://127.0.0.1:8080")
    
    for do
        let istek = await sunucu.sleep()
        
        // Her isteği ayrı task'ta işle
        async.spawn(async do
            let yanıt = await handle_request(istek)
            await sunucu.gönder(yanıt)
        end)
    end
end

async.run(sunucu_başlat())

समवर्ती कार्य प्रबंधन

import async

async function download_file(url: yazı, dosya: yazı) -> Sonuç do
    print("İndirme başladı: " + dosya)
    
    // Simulate download
    await async.sleep(random.range(1000, 3000))
    
    print("İndirme tamamlandı: " + dosya)
    return Ok(dosya)
end

async function download_with_timeout(url: yazı, dosya: yazı, timeout_ms: int) -> Sonuç do
    let sonuç = await async.timeout(
        download_file(url, dosya),
        timeout_ms
    )
    
    if sonuç.hata_mı() then do
        print("Timeout: " + dosya)
    end
    
    return sonuç
end

async function batch_download() do
    let dosyalar = [
        ("https://example.com/file1.pdf", "file1.pdf"),
        ("https://example.com/file2.pdf", "file2.pdf"),
        ("https://example.com/file3.pdf", "file3.pdf")
    ]
    
    // Maksimum 2 eşzamanlı indirme
    let semaphore = async.semaphore_new(2)
    let görevler = []
    
    her (url, dosya) in dosyalar for do
        let task = async.spawn(async do
            let izin = await semaphore.acquire()
            let sonuç = await download_with_timeout(url, dosya, 5000)
            semaphore.release(izin)
            return sonuç
        end)
        görevler.push(task)
    end
    
    let sonuçlar = await async.all(görevler)
    print("Tüm indirmeler tamamlandı: " + sonuçlar.len().yazıya())
end

async.run(batch_download())
