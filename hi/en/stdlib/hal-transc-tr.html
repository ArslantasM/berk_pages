🚀 उपयोग

import hal::transc

// Initialize nRF24L01
let nrf = transc.nrf24_init(do
    ce_pin: gpio.PA0,
    csn_pin: gpio.PA1,
    spi: spi1,
    channel: 76,
    data_rate: transc.DR_2MBPS
end)

// Send packet
transc.nrf24_send(nrf, [0x01, 0x02, 0x03])

// Receive packet
if data = transc.nrf24_receive(nrf, 100) then do
    io.println("Received: {:?}", data)
end

// LoRa module
let lora = transc.lora_init(do
    frequency: 915_000_000,  // 915 MHz
    bandwidth: transc.BW_125KHZ,
    spreading_factor: 7
end)

transc.lora_send(lora, "Hello LoRa!")
