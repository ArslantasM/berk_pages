फ्लैश मेमोरी प्रोग्रामिंग - प्रोग्राम स्टोरेज और डेटा पर्सिस्टेंस

~340 पंक्तियाँ ~15 कार्य एसटीएम32/जीडी32

📖अवलोकन

फ्लैश मेमोरी मॉड्यूल माइक्रोकंट्रोलर की आंतरिक फ्लैश मेमोरी में प्रोग्राम कोड लिखने, पढ़ने और हटाने की अनुमति देता है। इसका उपयोग लगातार डेटा भंडारण, फर्मवेयर अपडेट (ओटीए), कॉन्फ़िगरेशन सेटिंग्स और ईईपीरोम इम्यूलेशन के लिए किया जाता है।

🔑 मुख्य विशेषताएं

  • प्रोग्राम मेमोरी पढ़ना/लिखना/मिटाना
  • पृष्ठ/सेक्टर आधारित मिटाने की कार्रवाई
  • बड़े पैमाने पर समर्थन मिटाएँ
  • सुरक्षा तंत्र लिखें
  • विकल्प बाइट्स प्रोग्रामिंग
  • डुअल-बैंक फ़्लैश (F4/F7)
  • ईसीसी त्रुटि का पता लगाना
  • फ़्लैश मेमोरी लॉकिंग

import hal::flash, hal::crc

const APP_START = 0x08010000  // Application start
const APP_SIZE = 0x00030000   // 192 KB
const UPDATE_START = 0x08040000  // Update area

struct FirmwareHeader do
    magic: sayı,        // 0x42455247 (BERK)
    version: sayı,
    size: sayı,
    crc32: sayı
end

function firmware_validate(addr: sayı) -> mantık do
    // Read header
    let header = flash.read_struct::(addr)
    
    if header.magic != 0x42455247 then do
        return false
    end
    
    // Calculate CRC
    let calculated_crc = crc.calculate(
        addr + sizeofreşitli, 
        header.size
    )
    
    return calculated_crc == header.crc32
end

function firmware_copy(src: sayı, dest: sayı, size: sayı) do
    flash.unlock()
    
    // Erase destination
    let pages = (size + 2047) / 2048
    for i in 0..pages for do
        flash.erase_page(dest + i * 2048)
    end
    
    // Copy data
    let offset = 0
    for offset < size for do
        let data = flash.read_word(src + offset)
        flash.write_word(dest + offset, data)
        offset += 4
    end
    
    flash.lock()
end

function bootloader_ana() do
    io.println("Bootloader başlatılıyor...")
    
    // Check for update
    if firmware_validate(UPDATE_START) then do
        io.println("Yeni firmware bulundu, güncelleniyor...")
        
        let header = flash.read_struct::(UPDATE_START)
        
        // Copy new firmware to application area
        firmware_copy(
            UPDATE_START,
            APP_START,
            header.size
        )
        
        // Verify
        if firmware_validate(APP_START) then do
            io.println("Güncelleme başarılı!")
            
            // Clear update area
            flash.unlock()
            flash.erase_sector(UPDATE_START)
            flash.lock()
        yoksa do
            io.println("Güncelleme başarısız!")
            return
        end
    end
    
    // Jump to application
    io.println("Uygulamaya atlanıyor...")
    jump_to_application(APP_START)
end

function jump_to_application(addr: sayı) do
    // Read stack pointer and reset vector
    let sp = flash.read_word(addr)
    let reset_handler = flash.read_word(addr + 4)
    
    // Disable interrupts
    asm!("cpsid i")
    
    // Set stack pointer
    asm!("msr msp, {}", in(reg) sp)
    
    // Jump to reset handler
    let jump: function() = transmute(reset_handler)
    jump()
end

💡 उदाहरण 3: विकल्प बाइट्स प्रोग्रामिंग

सुरक्षा पढ़ें और सुरक्षा सेटिंग लिखें

import hal::flash

enum ReadProtection {
    LEVEL_0,  // No protection
    LEVEL_1,  // Memory read protection
    LEVEL_2   // Permanent protection (irreversible!)
}

function set_read_protection(level: ReadProtection) do
    flash.unlock()
    flash.unlock_option_bytes()
    
    if level == ReadProtection.LEVEL_1 then do
        flash.set_option_byte(flash.OB_RDP, 0x00)
        io.println("UYARI: Read protection Level 1 etkinleştirildi")
    yoksa if level == ReadProtection.LEVEL_2 then do
        io.println("UYARI: Level 2 GERİ ALINAMAZ! Devam edilsin mi? (y/n)")
        // Level 2 is permanent - chip cannot be debugged anymore!
    end
    
    flash.lock_option_bytes()
    flash.lock()
end

function protect_bootloader() do
    // Protect first 16KB (bootloader area)
    flash.unlock()
    flash.unlock_option_bytes()
    
    flash.set_write_protection(do
        sectors: [flash.SECTOR_0, flash.SECTOR_1],
        enable: true
    end)
    
    flash.lock_option_bytes()
    flash.lock()
    
    io.println("Bootloader koruması etkinleştirildi")
end

💡 सर्वोत्तम प्रथाएँ

  • संरेखण लिखें:डेटा को हमेशा निर्दिष्ट संरेखण के अनुरूप होना चाहिए
  • लिखने से पहले मिटाएँ:फ़्लैश केवल 1 → 0 लिख सकता है, पहले मिटाना (0xFF) आवश्यक है
  • समतलन पुराना होना:एक ही पेज पर बार-बार नहीं लिखना चाहिए (100K चक्र सीमा)
  • बिजली की हानि:लेखन के दौरान बिजली हानि की स्थिति में डेटा भ्रष्टाचार हो सकता है
  • व्यवधान:फ़्लैश प्रोग्रामिंग करते समय इंटरप्ट का उपयोग सावधानी से करें
  • पढ़ें-समय-लिखें:कुछ MCU लिखते समय पढ़ नहीं सकते

⚠️ महत्वपूर्ण नोट्स

  • फ्लैश प्रोग्रामिंग सीपीयू को ब्लॉक कर देती है (~20-50ms प्रति पेज)
  • विकल्प बाइट्स स्तर 2 सुरक्षा पढ़ेंअपरिवर्तनीय!
  • फ्लैश सहनशक्ति विशिष्ट 10K-100K मिटा चक्र
  • डेटा प्रतिधारण: 10+ वर्ष (25°C पर)
