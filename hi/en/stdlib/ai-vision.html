    use ai/vision;
use ai/tensor;

// Pre-trained ResNet-50 yükle
let model = ResNet::new(
    architecture: "resnet50",
    pretrained: true,
    num_classes: 1000  // ImageNet
);

// Görüntü yükle and preprocess
let img = vision::load_image("cat.jpg")?;
let img_tensor = vision::preprocess(
    img,
    size: [224, 224],
    normalize: true
);

// Inference
model.değerlendirme();
let logits = model.ileri(img_tensor);
let probs = softmax(logits);
let top5 = probs.topk(5);

her (idx, prob) in top5 for do
    let label = IMAGENET_LABELS[idx];
    print("{}: {:.2f}%", label, prob * 100.0);
end

// YOLOv8 medium model

    version: "v8m",
    pretrained: true,
    conf_threshold: 0.25,
    iou_threshold: 0.45
);

let detections = detector.detect(img)?;

print("{} nesne tespit edildi", detections.len());

her det in detections for do
    print!("Sınıf: {}, Güven: {:.2f}%, ", 
            COCO_LABELS[det.class_id], 
            det.confidence * 100.0);
    print!("Bbox: ({}, {}, {}, {})", 
            det.x, det.y, det.width, det.height);
end

// Bounding box çizimi
let img_with_boxes = vision::draw_boxes(img, detections);
vision::save_image(img_with_boxes, "output.jpg")?;

// UNet model
let model = UNet::new(
    in_channels: 3,
    num_classes: 21,  // Pascal VOC
    pretrained: true
);

let img = vision::load_image("scene.jpg")?;
let img_tensor = vision::preprocess(img, size: [512, 512]);

// Segmentation
model.değerlendirme();
let mask = model.ileri(img_tensor);
let predicted_mask = mask.argmax(dim: 1);

// Renklendirme
let colored_mask = vision::colorize_mask(
    predicted_mask,
    palette: PASCAL_VOC_COLORS
);

// Overlay
let overlay = vision::overlay(img, colored_mask, alpha: 0.5);
vision::save_image(overlay, "segmented.jpg")?;

    कस्टम विज़न मॉडल

    अपना खुद का मॉडल बनाएं

    use ai/vision;
use ai/nn;

sınıf CustomClassifier {
    let backbone: ResNet;
    let classifier: Sequential;
    
    function new(num_classes: int) -> CustomClassifier do
        let backbone = ResNet::new("resnet34", pretrained: true);
        
        // Backbone'u dondur
        her param in backbone.parametreler() for do
            param.requires_grad = false;
        end
        
        // Custom classifier
        let classifier = Sequential::new([
            Linear::new(512, 256),
            ReLU::new(),
            Dropout::new(0.5),
            Linear::new(256, num_classes)
        ]);
        
        return CustomClassifier { backbone, classifier };
    end
    
    function ileri(let kendi, x: Tensor) -> Tensor do
        let features = kendi.backbone.extract_features(x);
        let pooled = features.adaptive_avg_pool2d([1, 1]);
        let flattened = pooled.flatten(1);
        return kendi.classifier.ileri(flattened);
    end
}

    एपीआई संदर्भ

    विज़न मॉड्यूल

    • load_image(path: dizgi) -> Image- छवियाँ अपलोड करें
    • save_image(img: Image, path: dizgi)- चित्र को सेव करें
    • preprocess(img: Image, ...) -> Tensor- प्रीप्रोसेसिंग
    • draw_boxes(img: Image, boxes: [Box]) -> Image- बॉक्स ड्रा करें
    • overlay(img1: Image, img2: Image, alpha: ondalık) -> Image

    रेसनेट क्लास

    • yeni(architecture: dizgi, pretrained: mantıksal, ...) -> ResNet
    • ileri(kendi, x: Tensor) -> Tensor
    • extract_features(kendi, x: Tensor) -> Tensor

    योलो क्लास

    • yeni(version: dizgi, ...) -> YOLO
    • detect(kendi, img: Image) -> [Detection]
    • detect_batch(kendi, imgs: [Image]) -> [[Detection]]
