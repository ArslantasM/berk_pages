परिणाम प्रकार

त्रुटि प्रबंधन के लिए:

type Sonuç<T, E> = enum {
    Ok(T),
    Err(E)
}

function split(a: number, b: number) -> Sonuç<number, string> {
    if b == 0 {
        return Err "Sıfıra bölme!"
    }
    return Ok a / b
}

let sonuç = split(10, 2)
match sonuç {
    Ok(değer) => print("Sonuç: ", değer),
    Err(mesaj) => print("HATA: ", mesaj)
}

विकल्प प्रकार (विकल्प)

निरर्थक विकल्प:

type Seçenek<T> = enum {
    Some(T),
    Hiç
}

function bul(List: Liste<number>, hedef: number) -> Seçenek<number> {
    for i 0'from List.len'a to {
        if List[i] == hedef {
            return Some(i)
        }
    }
    return Hiç
}

let indeks = bul([1, 2, 3, 4], 3)
match indeks {
    Some(i) => print("Bulundu: ", i),
    Hiç => print("Bulunamadı")
}

अनुमान टाइप करें

बर्क मजबूत प्रकार का अनुमान लगाता है:

// Tip otomatik belirlenir
let sayı = 42            // number
let pi = 3.14            // Ondalık
let isim = "Ahmet"       // string
let doğru_mu = true     // Boolean
let List = [1, 2, 3]    // Liste<number>

// Karmaşık çıkarımlar
let harita = {           // Harita<string, number>
    "a": 1,
    "b": 2
}

let function = (x) => x * 2  // (number) -> number
