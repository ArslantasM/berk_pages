चर और स्थिरांक

बर्क में मूल्यों को संग्रहीत करने के लिएचर(चर) औरस्थिरांक(स्थिरांक) का प्रयोग किया जाता है।

परिवर्तनीय परिभाषा

तुर्की सिंटैक्स

let x = 10
let isim = "Ahmet"
let pi = 3.14159

अंग्रेजी सिंटेक्स

var x = 10
var name = "John"
var pi = 3.14159

निर्दिष्ट प्रकार

बर्क एक स्थिर टाइप की गई भाषा है। प्रकार का अनुमान समर्थित है:

// Tip çıkarımı (önerilen)
let sayı = 42          // number tipi
let String = "Hello"    // string tipi
let gerçek = 3.14      // number tipi (float)

// Açık type belirtimi
let sayı: number = 42
let String: string = "Hello"
let List: Liste<number> = [1, 2, 3]

परिवर्तनीय प्रतिस्थापन

एक बार वेरिएबल परिभाषित हो जाने के बाद, उनके मान बदले जा सकते हैं:

let count_occurrences = 0
print(count_occurrences)  // 0

count_occurrences = count_occurrences + 1
print(count_occurrences)  // 1

count_occurrences += 5
print(count_occurrences)  // 6

स्थिरांक

एक बार असाइन किए जाने के बाद स्थिरांकों को बदला नहीं जा सकता:

const PI = 3.14159
const MAX_KULLANICI = 100
const UYGULAMA_ADI = "BERK App"

// Err! Sabit değiştirilemez
// PI = 3.14  // Derleme hatası

अंग्रेजी सिंटेक्स

const PI = 3.14159
const MAX_USERS = 100

दायरा

सार्वत्रिक चर

फ़ंक्शन के बाहर परिभाषित चर वैश्विक हैं:

let global_sayaç = 0  // Global

function artır() {
    global_sayaç += 1
}

function main() {
    print(global_sayaç)  // 0
    artır()
    print(global_sayaç)  // 1
}

स्थानीय चर

फ़ंक्शन के भीतर परिभाषित चर स्थानीय हैं:

function hesapla() {
    let yerel = 10  // Sadece bu fonksiyonda erişilebilir
    print(yerel)
}

function main() {
    hesapla()
    // print(yerel)  // Err! yerel burada tanımlı not
}

ब्लॉक का दायरा

प्रत्येक ब्लॉक अपना स्वयं का दायरा बनाता है:

function main() {
    let x = 10
    
    if x > 5 {
        let y = 20  // Bu blokta geçerli
        print(x, y)     // 10, 20
    }
    
    print(x)     // 10
    // print(y)  // Err! y bu kapsamda yok
}

पीछा

समान नाम वाले एक वेरिएबल को आंतरिक दायरे में परिभाषित किया जा सकता है:

function main() {
    let x = 5
    print(x)  // 5
    
    {
        let x = 10  // Farklı let
        print(x)        // 10
    }
    
    print(x)  // 5 (dış kapsam)
}

विध्वंसक

टुपल्स और सूचियों से मान निकालना:

// Tuple destructuring
let (x, y) = (10, 20)
print(x)  // 10
print(y)  // 20

// Liste destructuring
let [first, ikinci, ...geri_kalan] = [1, 2, 3, 4, 5]
print(first)         // 1
print(ikinci)      // 2
print(geri_kalan)  // [3, 4, 5]

एकाधिक असाइनमेंट

एक साथ अनेक चर परिभाषित करना:

let a, b, c = 1, 2, 3
print(a, b, c)  // 1 2 3

// Değer takası (swap)
let x = 10
let y = 20
x, y = y, x
print(x, y)  // 20 10

अशक्त चर

चर शून्य हो सकते हैं:

let değer: number? = null  // Nullable type

if değer != null {
    print("Değer: ", değer)
} else {
    print("Değer yok")
}

// Varsayılan değer ile
let sonuç = değer ?: 0  // Eğer null then 0 use

नमूना कार्यक्रम

import math

// Global sabitler
const YERÇEKIMI = 9.81  // m/s²
const PI = 3.14159

function dairesel_hareket(yarıçap: number, hız: number) -> number {
    // Merkezcil ivme hesaplama
    let ivme = (hız * hız) / yarıçap
    return ivme
}

function serbest_düşüş(yükseklik: number) -> number {
    // v = sqrt(2 * g * h)
    let hız = math.sqrt(2.0 * YERÇEKIMI * yükseklik)
    return hız
}

function main() {
    // Değişkenler
    let yarıçap = 5.0   // metre
    let hız = 10.0      // m/s
    
    let ivme = dairesel_hareket(yarıçap, hız)
    print("Merkezcil ivme: ", ivme, " m/s²")
    
    // Sabit değer
    const YÜKSEKLİK = 50.0  // metre
    let düşüş_hızı = serbest_düşüş(YÜKSEKLİK)
    print("Düşüş hızı: ", düşüş_hızı, " m/s")
}
