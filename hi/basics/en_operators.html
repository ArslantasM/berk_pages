ऑपरेटरों

गणितीय संचालन, तुलना और बूलियन संचालन के लिए बीईआरके में विभिन्न ऑपरेटरों का उपयोग किया जाता है।

अंकगणित संचालक

बुनियादी संचालन

variable a = 10
variable b = 3

print(a + b)   // 13 - Toplama
print(a - b)   // 7  - Çıkarma
print(a * b)   // 30 - Çarpma
print(a / b)   // 3  - Bölme (integer)
print(a % b)   // 1  - Mod (kalan)
print(a ** b)  // 1000 - Üs alma

फ्लोट डिवीजन

variable x = 10.0
variable y = 3.0

print(x / y)   // 3.333333 - float bölme

यूनरी ऑपरेटर्स

variable n = 5

print(+n)  // 5  - Pozitif işareti
print(-n)  // -5 - Negatif işareti
print(++n) // 6  - Ön artırma
print(n++) // 6  - end artırma (7 olur)
print(--n) // 6  - Ön azaltma
print(n--) // 6  - end azaltma (5 olur)

तुलना संचालक

variable a = 10
variable b = 5

print(a == b)  // false - Eşit mi
print(a != b)  // true  - Eşit not mi
print(a > b)   // true  - Büyük mü
print(a < b)   // false - Küçük mü
print(a >= b)  // true  - Büyük eşit mi
print(a <= b)  // false - Küçük eşit mi

बूलियन ऑपरेटर्स

variable x = true
variable y = false

print(x && y)  // false - and (AND)
print(x || y)  // true  - or (OR)
print(!x)      // false - not (NOT)

शॉर्ट-सर्किट मूल्यांकन

// İkinci ifade değerlendirilmez
if (x || çağır_fonksiyon()) {
    // x true then çağır_fonksiyon() çalıştırılmaz
}

// İkinci ifade değerlendirilir
if (x && çağır_fonksiyon()) {
    // x false then çağır_fonksiyon() çalıştırılmaz
}

असाइनमेंट संचालक

सरल असाइनमेंट

variable x = 10
x = 20  // Yeni değer atama

यौगिक असाइनमेंट

variable n = 10

n += 5   // n = n + 5  (15)
n -= 3   // n = n - 3  (12)
n *= 2   // n = n * 2  (24)
n /= 4   // n = n / 4  (6)
n %= 5   // n = n % 5  (1)
n **= 3  // n = n ** 3 (1)

एकाधिक असाइनमेंट

variable a, b, c = 1, 2, 3
print(a, b, c)  // 1 2 3

// Değer takası (swap)
a, b = b, a
print(a, b)  // 2 1

बिटवाइज़ ऑपरेटर्स

variable a = 0b1010  // 10
variable b = 0b1100  // 12

print(a & b)   // 0b1000 (8)  - AND
print(a | b)   // 0b1110 (14) - OR
print(a ^ b)   // 0b0110 (6)  - XOR
print(~a)      // 0b0101 (-11) - NOT
print(a << 1)  // 0b10100 (20) - Sol kaydırma
print(a >> 1)  // 0b0101 (5)   - Sağ kaydırma

स्ट्रिंग ऑपरेटर्स

मर्ज

variable ad = "Ahmet"

दोहराना

तुलना करने के लिए

print("abc" == "abc")   // true
print("abc" < "abd")    // true (leksikografik)
print("a" + "b" == "ab") // true

रेंज ऑपरेटर

// Inclusive range (end dahil)
loop i 1'from 5'e to {
    print(i)  // 1, 2, 3, 4, 5
}

// Exclusive range (end hariç)
loop i 1'from 5'e {
    print(i)  // 1, 2, 3, 4
}

// Liste oluşturma
variable sayılar = [1..10]  // [1, 2, 3, ..., 10]
variable harfler = [to'..'z']

एल्विस ऑपरेटर (नल कोलेसिंग)

variable değer: number? = null

// if null then Default Value use
variable sonuç = değer ?: 0
print(sonuç)  // 0

değer = 42
sonuç = değer ?: 0
print(sonuç)  // 42

सुरक्षित नेविगेशन ऑपरेटर

variable kullanıcı: Kullanıcı? = null

// Null kontrolü olmadan güvenli erişim
variable email = kullanıcı?.email
// if kullanıcı null ise email de null olur

// Zincirleme
variable şehir = kullanıcı?.adres?.şehir?.ad

पाइपलाइन ऑपरेटर

// function zincirleme
variable sonuç = [1, 2, 3, 4, 5]
    |> filtrele(x => x % 2 == 0)
    |> eşle(x => x * 2)
    |> azalt((a, b) => a + b)

print(sonuç)  // 12 (2*2 + 4*2)

स्प्रेड ऑपरेटर

// Liste spread
variable liste1 = [1, 2, 3]
variable liste2 = [4, 5, 6]
variable birleşik = [...liste1, ...liste2]
print(birleşik)  // [1, 2, 3, 4, 5, 6]

// function argümanı
function topla(a, b, c) -> number {
    geri return a + b + c
}

variable args = [1, 2, 3]
print(topla(...args))  // 6

टर्नरी ऑपरेटर

variable yaş = 18
variable durum = yaş >= 18 ? "Yetişkin" : "Çocuk"
print(durum)  // Yetişkin

// İç içe ternary
variable not = 85
variable harf = not >= 90 ? "A" :
                not >= 80 ? "B" :
                not >= 70 ? "C" : "F"
print(harf)  // B

चेक ऑपरेटर टाइप करें

    print("Bu bir string")
}

if değer !is number {
    print("Bu bir sayı not")
}

// Casting ile
variable String = değer as string

ऑपरेटर में

// Liste kontrol
variable sayılar = [1, 2, 3, 4, 5]
if 3 in sayılar {
    print("3 listede var")
}

// String kontrol
variable String = "Merhaba Dünya"
if "Dünya" in String {
    print("'Dünya' bulundu")
}

// Range kontrol
if 5 in 1..10 {
    print("5 aralıkta")
}

// Map kontrol
variable sözlük = {"ad": "Ahmet", "yaş": 30}
if "ad" in sözlük {
    print("'ad' anahtarı var")
}

कोष्ठकों के साथ प्राथमिकता बदलना

variable b = (2 + 3) * 4    // 20 (toplama önce)

variable c = 10 / 2 + 3     // 8
variable d = 10 / (2 + 3)   // 2

// Short-circuit evaluation'from faydalanın
if kullanıcı != null && kullanıcı.aktif {
    // ...
}

// Null-safe operatörler use

// Operatör etrafında boşluk

❌ परहेज करने योग्य बातें

// Çok uzun ternary zincirleri
variable x = a ? b ? c ? d : e : f : g  // Okunması zor

// Gereksiz karmaşık bitwise işlemler
variable y = (a & b) | (~c ^ d)  // Ne yaptığı belli not

// Yan etki yaratan operatörler
variable z = x++ + ++x + x++  // Kafa karıştırıcı

function main() {
    // Aritmetik
    variable sum = 10 + 20

    variable product = 5 * 6
    variable Section = 20 / 4
