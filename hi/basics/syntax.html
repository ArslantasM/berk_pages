BERK Programlama Dili

सिंटैक्स और संरचना

BERK प्रोग्रामिंग भाषा के बुनियादी वाक्यविन्यास नियम और कोड संरचना।

बुनियादी संरचना

प्रत्येक बर्क कार्यक्रम एक हैana()(याmain()) फ़ंक्शन से प्रारंभ होता है:

function main() {
    print("Merhaba Dünya!")
}

अंग्रेजी वाक्यविन्यास:

function main() {
    print("Hello World!")
}

अभिव्यक्ति टर्मिनेटर

बर्क में बयानअर्धविराम की आवश्यकता नहीं है. प्रत्येक पंक्ति एक अभिव्यक्ति का मूल्यांकन करती है:

let x = 10
let y = 20
let sum = x + y
print(sum)

बहु-पंक्ति अभिव्यक्तियों मेंबैकस्लैशआप उपयोग कर सकते हैं:

let uzunMetin = "Bu çok uzun bir yazı \
                      birden fazla satıra yayılıyor"

ब्लॉक संरचनाएँ

ब्लाकोंमध्यम कोष्ठकइसे इसके द्वारा परिभाषित किया गया है:

function hesapla() {
    let a = 5
    let b = 3
    
    if a > b {
        print("a büyük")
    } else {
        print("b büyük or eşit")
    }
}

श्वेत रिक्ति

ऑपरेटरों के आसपास रिक्त स्थान का उपयोग करने से पठनीयता में सुधार होता है:

// Tercih edilen
let sonuç = a + b * c

// Geçerli ama okunabilir not
let sonuç=a+b*c

let çokUzunDeğişken = birinciFonksiyon(parametere1, parametre2)

द्विभाषी सिंटेक्स

बर्क की सबसे महत्वपूर्ण विशेषताद्विभाषी वाक्यविन्याससमर्थन है. आप तुर्की और अंग्रेजी कीवर्ड को मिला सकते हैं:

// Tamamen Türkçe
function hesapla(x: number) -> number {
    if x < 0 {
        return Err "Negatif sayı!"
    }
    return x * x
}

// Tamamen İngilizce
function calculate(x: Number) -> Number {
    if x < 0 {
        error return "Negative number!"
    }
    return x * x
}

// Karışık (geçerli ama önerilmez)
function hesapla(x: number) -> Number {
    if x < 0 {
        return Err "Negative!"
    }
    return x * x
}

सर्वश्रेष्ठ प्रणालियां: अपने प्रोजेक्ट में एक भाषा चुनें (तुर्की या अंग्रेजी)।

मॉड्यूल

प्रत्येक फ़ाइल को एक मॉड्यूल माना जाता है:

// matematik.berk
export function topla(a: number, b: number) -> number {
    return a + b
}

export const PI = 3.14159

// main.berk
import matematik

function main() {
    print(matematik.topla(5, 3))
    print(matematik.PI)
}

उदाहरण: पूर्ण कार्यक्रम

// Fibonacci sayı dizisi hesaplama

import math

// Recursive versiyonu
function fib_recursive(n: number) -> number {
    if n <= 1 {
        return n
    }
    return fib_recursive(n - 1) + fib_recursive(n - 2)
}

// Iterative versiyonu (daha verimli)
function fib_iterative(n: number) -> number {
    if n <= 1 {
        return n
    }
    
    let a = 0
    let b = 1
    
    for i 2'from n+1'e to {
        let temp = a + b
        a = b
        b = temp
    }
    
    return b
}

function main() {
    const N = 10
    
    print("İlk ", N, " Fibonacci sayısı:")
    
    for i 0'from N'ye to {
        print("fib(", i, ") = ", fib_iterative(i))
    }
}
